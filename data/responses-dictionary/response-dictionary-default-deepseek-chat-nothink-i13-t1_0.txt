```python
import collections.abc
from collections import defaultdict

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    def __init__(self):
        self.children = {}
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a case-insensitive trie index from the dictionary for efficient phrase lookup.
    """
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize phrase to lowercase and split into words
        normalized = phrase.lower()
        words = normalized.split()
        current = trie
        
        for word in words:
            if word not in current.children:
                current.children[word] = TrieNode()
            current = current.children[word]
        current.entry_ids.add(entry_id)
    
    return trie

def is_word_token(token: str) -> bool:
    """Check if token is a word (not separator/punctuation)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with dictionary entries using the trie index.
    """
    tokens = list(tokens)  # Need random access
    n = len(tokens)
    annotations = defaultdict(set)
    
    # First pass: find all phrase matches
    for i in range(n):
        if not is_word_token(tokens[i]):
            continue
            
        current = dictionary_index
        j = i
        matched = True
        
        while j < n and matched:
            token = tokens[j]
            if not is_word_token(token):
                j += 1
                continue
                
            word = token.lower()
            if word in current.children:
                current = current.children[word]
                # Add all entries for phrases ending here
                for entry_id in current.entry_ids:
                    # Annotate all tokens in this phrase (i to j)
                    for k in range(i, j + 1):
                        if k == i or k == j or is_word_token(tokens[k]):
                            annotations[k].add(entry_id)
                j += 1
            else:
                matched = False
    
    # Second pass: add single-word matches and build result
    result = []
    for i, token in enumerate(tokens):
        entry_ids = set(annotations[i])
        
        # Add single-word match if applicable
        if is_word_token(token):
            word = token.lower()
            current = dictionary_index
            if word in current.children:
                entry_ids.update(current.children[word].entry_ids)
        
        result.append((token, entry_ids))
    
    return result
```
