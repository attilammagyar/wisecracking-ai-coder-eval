## Algorithm Design

### Dictionary Index Structure

I'll use a **trie (prefix tree)** to store normalized dictionary entries. Each node represents a character, and nodes marking the end of a dictionary entry store the associated IDs. This enables efficient longest-match searches and handles overlapping phrases naturally.

The normalization process will:
- Convert to lowercase for case-insensitive matching
- Split compound phrases into words
- Store the word sequence in the trie

### Annotation Algorithm

1. **Token Classification**: First, classify each token as either a "word" token (contains alphabetic characters) or a "separator" token.

2. **Word Sequence Extraction**: Group consecutive word tokens together, tracking their positions and any separators between them.

3. **Trie Matching**: For each starting position in the word sequence:
   - Traverse the trie to find all matching phrases
   - For compound words (consecutive tokens without separators), also check their concatenation
   - Collect all matches with their token spans

4. **Annotation Assignment**: For each match found, annotate all tokens in its span (including inner separators for phrases).

```python
import collections.abc

class TrieNode:
    def __init__(self):
        self.children = {}
        self.entry_ids = set()

class DictionaryIndex:
    def __init__(self):
        self.root = TrieNode()
    
    def add_entry(self, normalized_words, entry_id):
        """Add a normalized word sequence to the trie."""
        node = self.root
        for word in normalized_words:
            if word not in node.children:
                node.children[word] = TrieNode()
            node = node.children[word]
        node.entry_ids.add(entry_id)
    
    def find_matches(self, words, start_idx):
        """Find all matches starting at start_idx in the word list."""
        matches = []
        node = self.root
        
        for i in range(start_idx, len(words)):
            word = words[i]
            if word not in node.children:
                break
            node = node.children[word]
            if node.entry_ids:
                matches.append((i - start_idx + 1, node.entry_ids.copy()))
        
        return matches

def normalize_phrase(phrase):
    """Normalize a phrase by converting to lowercase and extracting words."""
    words = []
    current_word = []
    
    for char in phrase:
        if char.isalpha():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def is_word_token(token):
    """Check if a token contains alphabetic characters."""
    return any(c.isalpha() for c in token)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = DictionaryIndex()
    
    for phrase, entry_id in dictionary.items():
        normalized_words = normalize_phrase(phrase)
        if normalized_words:
            index.add_entry(normalized_words, entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    if not tokens:
        return []
    
    # Build word sequence with position mapping
    words = []
    word_positions = []  # Maps word index to token index
    token_to_word = {}   # Maps token index to word index (if it's a word token)
    
    for i, token in enumerate(tokens):
        if is_word_token(token):
            normalized = token.lower()
            word_idx = len(words)
            words.append(normalized)
            word_positions.append(i)
            token_to_word[i] = word_idx
    
    # Find all matches in the word sequence
    for start_word_idx in range(len(words)):
        matches = dictionary_index.find_matches(words, start_word_idx)
        
        for match_length, entry_ids in matches:
            # Calculate token span for this match
            start_token_idx = word_positions[start_word_idx]
            end_token_idx = word_positions[start_word_idx + match_length - 1]
            
            # Annotate all tokens in the span
            for token_idx in range(start_token_idx, end_token_idx + 1):
                annotations[token_idx].update(entry_ids)
    
    # Handle compound words (consecutive word tokens without separators)
    i = 0
    while i < len(tokens):
        if is_word_token(tokens[i]):
            # Find consecutive word tokens
            j = i
            while j < len(tokens) and is_word_token(tokens[j]):
                j += 1
            
            if j > i + 1:  # Multiple consecutive word tokens

                concatenated = ''.join(tokens[i:j]).lower()

                if concat_matches:
                    for _, entry_ids in concat_matches:
                        for k in range(i, j):
                            annotations[k].update(entry_ids)
            
            i = j

    # Create result
    annotated_tokens = []
    for token, annotation_set in zip(tokens, annotations):
        annotated_tokens.append((token, annotation_set))
    
    return annotated_tokens
```
