def extract_words(text):
    """Extract word tokens from text, ignoring separators."""
    words = []
    current = []
    
    for char in text:
        if char.isalnum():
            current.append(char.lower())
        else:
            if current:
                words.append(''.join(current))
                current = []
    
    if current:
        words.append(''.join(current))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for phrase, entry_id in dictionary.items():
        words = extract_words(phrase)
        
        if words:
            key = tuple(words)
            if key not in index:
                index[key] = set()
            index[key].add(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    token_list = list(tokens)
    n = len(token_list)
    annotations = [set() for _ in range(n)]
    
    # Process each starting position
    for start in range(n):
        if not is_word_token(token_list[start]):
            continue
        
        # Track words found and positions involved
        words_found = []
        positions = []
        
        # Also track consecutive words for compound matching
        consecutive_words = []
        consecutive_positions = []
        
        for i in range(start, n):
            token = token_list[i]
            
            if is_word_token(token):
                word = normalize_word(token)
                words_found.append(word)
                positions = list(range(start, i + 1))
                
                # Track consecutive words
                if i == start or (consecutive_positions and i == consecutive_positions[-1] + 1):
                    consecutive_words.append(word)
                    consecutive_positions.append(i)
                else:
                    # Reset consecutive tracking
                    consecutive_words = [word]
                    consecutive_positions = [i]
                
                # Check for phrase match
                key = tuple(words_found)
                if key in dictionary_index:
                    for entry_id in dictionary_index[key]:
                        for pos in positions:
                            annotations[pos].add(entry_id)
                
                # Check for compound word match
                if len(consecutive_words) > 1:
                    compound_key = tuple([''.join(consecutive_words)])
                    if compound_key in dictionary_index:
                        for entry_id in dictionary_index[compound_key]:
                            for pos in consecutive_positions:
                                annotations[pos].add(entry_id)
            else:
                # Separator
                if not words_found:
                    break
                # Reset consecutive words
                consecutive_words = []
                consecutive_positions = []
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(token_list):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
