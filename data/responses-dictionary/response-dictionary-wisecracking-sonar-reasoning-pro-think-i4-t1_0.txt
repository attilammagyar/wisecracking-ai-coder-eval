```python
import re
from collections.abc import Iterable, Mapping

def build_dictionary_index(dictionary: Mapping) -> dict:
    def normalize(phrase):
        return re.sub(r'\W+', ' ', phrase).strip().lower()
    return {normalize(phrase): id for phrase, id in dictionary.items()}

def is_word(token):
    return re.fullmatch(r'\w+', token) is not None

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[tuple[str, set]]:
    is_separator = []
    for token in tokens:
        is_separator.append(not is_word(token))
    
    annotated = [(token, set()) for token in tokens]
    
    for i in range(len(tokens)):
        non_seps = []
        j = i
        while j >= 0:
            if is_separator[j]:
                j -= 1
                continue
            non_seps.append({'token': tokens[j], 'pos': j})
            j -= 1
        non_seps = non_seps[::-1]  # Reverse to get [earliest, ..., latest]
        
        for phrase_length in range(1, len(non_seps) + 1):
            candidate = non_seps[:phrase_length]
            normalized = ' '.join([t['token'].lower() for t in candidate]).strip()
            if normalized in dictionary_index:
                entry_id = dictionary_index[normalized]
                start = candidate[0]['pos']
                end = candidate[-1]['pos']
                for pos in range(start, end + 1):
                    if pos < len(tokens):
                        annotated[pos][1].add(entry_id)
    
    return annotated
```
