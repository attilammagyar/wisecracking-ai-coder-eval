import collections.abc

# ---------------- Helpers ----------------

def _norm(s: str) -> str:
    # Case-insensitive normalization
    return s.casefold()

def _is_word_token(tok: str) -> bool:
    # Define "word" as strictly alphanumeric-only token
    return tok.isalnum()

def _split_dict_key_into_words(key: str) -> list[str]:
    # Split dictionary key into alphanumeric word chunks, ignoring separators
    words = []
    cur = []
    for ch in key:
        if ch.isalnum():
            cur.append(ch)
        else:
            if cur:
                words.append(_norm("".join(cur)))
                cur = []
    if cur:
        words.append(_norm("".join(cur)))
    return words

def _make_node():
    # Trie node: dict with 'children' dict and 'ids' set
    return {"children": {}, "ids": set()}

def _insert_char_trie(root, word: str, entry_id):
    node = root
    for ch in word:
        node = node["children"].setdefault(ch, _make_node())
    node["ids"].add(entry_id)

def _insert_word_trie(root, words: list[str], entry_id):
    node = root
    for w in words:
        node = node["children"].setdefault(w, _make_node())
    node["ids"].add(entry_id)

# ---------------- Public API ----------------

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    single_word_map: dict[str, set] = {}      # normalized single-word -> set(ids)
    char_trie_root = _make_node()             # for glued-word matches across word tokens
    phrase_trie_root = _make_node()           # for multi-word phrase matches (2+ words)

    for key, entry_id in dictionary.items():
        words = _split_dict_key_into_words(key)
        if not words:
            continue
        if len(words) == 1:
            w = words[0]
            single_word_map.setdefault(w, set()).add(entry_id)
            # Also index in char trie to support concatenation across word tokens
            _insert_char_trie(char_trie_root, w, entry_id)
        else:
            _insert_word_trie(phrase_trie_root, words, entry_id)

    return {
        "single": single_word_map,
        "char_trie": char_trie_root,
        "phrase_trie": phrase_trie_root,
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)
    n = len(tokens)

    single = dictionary_index["single"]
    char_trie = dictionary_index["char_trie"]
    phrase_trie = dictionary_index["phrase_trie"]

    norm_tokens = [_norm(t) for t in tokens]
    is_word = [_is_word_token(t) for t in tokens]
    annotations = [set() for _ in range(n)]

    # 1) Direct single-word matches
    for i in range(n):
        if is_word[i]:
            ids = single.get(norm_tokens[i])
            if ids:
                annotations[i].update(ids)

        while k < n and is_word[k] and not dead:

            # Consume the whole token's characters
            for ch in tok:
                nxt = node["children"].get(ch)
                if nxt is None:
                    dead = True
                    break
                node = nxt
            if dead:

            # If a word ends here in the trie, accept it (token-end boundary)
            if node["ids"]:
                for t in range(i, k + 1):
                    annotations[t].update(node["ids"])
            k += 1

    # 3) Multi-word phrase matches (separator-insensitive), annotate words and inner separators

                for t in range(start_tok_idx, end_tok_idx + 1):
                    annotations[t].update(node["ids"])
            e_idx += 1
