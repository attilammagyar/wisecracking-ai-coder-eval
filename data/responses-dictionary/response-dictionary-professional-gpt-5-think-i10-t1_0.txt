Main ideas

- Normalize the dictionary into sequences of “words.” A dictionary key like "AAA, BBB" is split into words ["aaa", "bbb"] by removing non-alphanumeric separators and case-folding to ensure case-insensitive matching. The precise separator characters don’t matter; they only indicate that between these words there must be one or more non-word tokens in the text.
- Build an index keyed by the first word of each dictionary entry. Each index entry stores the full word sequence and the dictionary value (identifier). Also store the maximum length of all first words (in characters) to limit how far we try to concatenate tokens when matching a first word.
- Classify input tokens as “word” tokens if and only if the entire token is alphanumeric (str.isalnum()). Non-word tokens (spaces, punctuation, markdown symbols, etc.) are separators.
- Matching:
  - Start matches only at word tokens (so no leading separators are annotated).
  - To match the first dictionary word, concatenate successive word tokens (with no separator in between) until the concatenated lowercase string equals a first word in the index, or exceeds the maximum length. This naturally supports tokens split finer than dictionary words (e.g., "AAA", "BBB" matching "AAABBB") while prohibiting mid-token matches (e.g., "AA" inside "AAA") because we never split tokens.
  - For multi-word entries, after matching the first word, require at least one separator token, then match the next dictionary word in the same way (again concatenating adjacent word tokens, with no non-word tokens in between). Repeat for all words.
  - When a full entry matches, annotate all word tokens covered by the match and all inner separator tokens (those between the words). Leading/trailing separators outside the matched words are not annotated.
- Overlaps and nesting are handled by attempting matches starting at every word-token position and collecting all matches; annotations are unions of all applicable entries.

```python
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Helper: split a dictionary key into a tuple of "words"
    # A "word" is a consecutive sequence of alphanumeric characters.
    # Lookup is case-insensitive (use casefold()).
    def split_into_words(text: str) -> tuple[str, ...]:
        words = []
        buf = []
        for ch in text:
            if ch.isalnum():
                buf.append(ch)
            else:
                if buf:
                    words.append(''.join(buf).casefold())
                    buf = []
        if buf:
            words.append(''.join(buf).casefold())
        return tuple(words)

    first_index = {}  # first_word -> list of (words_tuple, entry_id)
    max_first_len = 0

    for key, entry_id in dictionary.items():
        words = split_into_words(key)
        if not words:
            continue  # ignore empty after normalization
        first = words[0]
        max_first_len = max(max_first_len, len(first))
        first_index.setdefault(first, []).append((words, entry_id))

    # The index object is a simple dict with what's needed by annotate()
    return {
        "first_index": first_index,
        "max_first_len": max_first_len,
    }


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)

    first_index = dictionary_index.get("first_index", {})
    max_first_len = dictionary_index.get("max_first_len", 0)

    # Classify tokens: a "word token" must be entirely alphanumeric.
    is_word = [t.isalnum() for t in tokens]
    # Normalized form for word tokens (case-insensitive).
    norm = [t.casefold() if is_word[i] else "" for i, t in enumerate(tokens)]

    anns = [set() for _ in range(n)]

    if not first_index or n == 0:
        return [(tokens[i], anns[i]) for i in range(n)]

    # Helper: match a dictionary "word" (string) at token index `start`,
    # allowing concatenation of consecutive word tokens (no separators),
    # and requiring the match to end exactly at a token boundary.
    def match_dic_word(start: int, dic_word: str):
        if start >= n or not is_word[start]:
            return start, False
        s = ""
        pos = start
        # Concatenate whole word tokens until length >= len(dic_word) or we hit a separator
        while pos < n and is_word[pos] and len(s) < len(dic_word):
            s += norm[pos]
            pos += 1
        return (pos, s == dic_word)

    # Try to match every entry starting at each word-token position.
    for i in range(n):
        if not is_word[i]:
            continue

        # Build candidate first words by concatenating adjacent word tokens.
        s = ""
        pos = i
        while pos < n and is_word[pos]:
            # Append next word token
            s += norm[pos]
            # If longer than any first word, no further concatenation can help
            if len(s) > max_first_len:
                break

            # If s matches some dictionary first words, try to complete those entries
            entries = first_index.get(s)
            if entries:
                for words_tuple, entry_id in entries:
                    # words_tuple[0] == s is already matched from i..pos
                    cur_pos = pos + 1  # first index AFTER the matched first word
                    ok = True
                    sep_ranges = []

                    # Match the rest of the words (if any)
                    for next_word in words_tuple[1:]:
                        # Require at least one separator token
                        if cur_pos >= n or is_word[cur_pos]:
                            ok = False
                            break
                        sep_start = cur_pos
                        # Skip one or more separator tokens
                        while cur_pos < n and not is_word[cur_pos]:
                            cur_pos += 1
                        sep_ranges.append((sep_start, cur_pos))

                        # Now match the next dictionary word starting at cur_pos
                        next_pos, matched = match_dic_word(cur_pos, next_word)
                        if not matched:
                            ok = False
                            break
                        cur_pos = next_pos

                    if ok:
                        # Annotate all matched word tokens and inner separators
                        # Word tokens: from i .. cur_pos-1 that are word tokens
                        for t in range(i, cur_pos):
                            if is_word[t]:
                                anns[t].add(entry_id)
                        # Separators only inside the phrase (between words)
                        for a, b in sep_ranges:
                            for t in range(a, b):
                                anns[t].add(entry_id)

            # Move to try a longer concatenation for the first word
            pos += 1

    return [(tokens[i], anns[i]) for i in range(n)]
```
