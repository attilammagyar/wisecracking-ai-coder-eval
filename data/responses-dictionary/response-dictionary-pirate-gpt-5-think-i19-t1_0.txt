import collections.abc

# ----------------- Helpers -----------------

def _is_word_token(tok: str) -> bool:
    # Word tokens consist entirely of alphanumerics (letters/digits).
    # This keeps matching clean and avoids mid-token splits.
    return bool(tok) and all(ch.isalnum() for ch in tok)

def _split_to_words(s: str) -> list[str]:
    # Split a dictionary key into case-insensitive "words" (runs of alnum chars).
    words = []
    cur = []
    for ch in s:
        if ch.isalnum():
            cur.append(ch.lower())
        else:
            if cur:
                words.append(''.join(cur))
                cur = []
    if cur:
        words.append(''.join(cur))
    return words

def _make_node():
    return {'children': {}, 'end_ids': set()}

# ----------------- Public API -----------------

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized trie index for fast lookup of words and compound phrases.
    """
    root = _make_node()
    for key, ident in dictionary.items():
        words = _split_to_words(key)
        if not words:
            continue
        node = root
        for w in words:
            child = node['children'].get(w)
            if child is None:
                child = _make_node()
                node['children'][w] = child
            node = child
        node['end_ids'].add(ident)
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary (words and compound phrases).
    """
    toks = list(tokens)
    n = len(toks)
    if n == 0:
        return []

    # Precompute helpers.
    lc = [t.lower() for t in toks]
    is_word = [_is_word_token(t) for t in toks]
    # next_word_from[k] = smallest idx >= k that is a word token, else n
    next_word_from = [n] * (n + 1)
    next_word_from[n] = n
    for i in range(n - 1, -1, -1):
        next_word_from[i] = i if is_word[i] else next_word_from[i + 1]

    anns = [set() for _ in range(n)]
    root = dictionary_index  # trie root

    def add_range(i_start: int, j_end: int, idset):
        if not idset:
            return
        for p in range(i_start, j_end + 1):
            anns[p].update(idset)

    for i in range(n):
        if not is_word[i]:
            continue

        # Try all possible concatenations of consecutive word tokens for the first dict word.
        s = ""
        j = i
        while j < n and is_word[j]:
            s += lc[j]
            first_node = root['children'].get(s)
            if first_node is not None:
                # Single-word match ends here (even if this dict "word" spans multiple tokens).
                if first_node['end_ids']:
                    add_range(i, j, first_node['end_ids'])

                # Try to extend to multi-word phrases via separators.
                stack = [(first_node, j)]
                while stack:
                    node_cur, end_j = stack.pop()
                    # Require at least one non-word token between dict words.
                    m = next_word_from[end_j + 1]  # first word token after some separators
                    if m >= n:
                        continue
                    if m == end_j + 1:
                        # No separators between words; cannot extend multi-word phrase.
                        continue
                    # Now match next dictionary word starting at m, allowing it to span multiple word tokens.
                    s2 = ""
                    j2 = m
                    while j2 < n and is_word[j2]:
                        s2 += lc[j2]
                        node_next = node_cur['children'].get(s2)
                        if node_next is not None:
                            if node_next['end_ids']:
                                # Annotate from the start i to current end j2 (includes inner separators).
                                add_range(i, j2, node_next['end_ids'])
                            # Continue extending further words.
                            stack.append((node_next, j2))
                        j2 += 1

            j += 1

    # Wrap it up as requested: list of (token, set_of_ids)
    return [(toks[k], anns[k]) for k in range(n)]
