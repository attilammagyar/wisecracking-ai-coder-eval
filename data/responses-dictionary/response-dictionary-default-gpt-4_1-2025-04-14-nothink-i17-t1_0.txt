import collections.abc

def is_word(token):
    # Heuristic: at least one alphanumeric character, not punctuation, not markdown.
    # This is a simplification; adjust as needed for your tokenizer.
    return any(c.isalnum() for c in token)

def normalize(token):
    return token.lower()

class TrieNode:
    __slots__ = ['children', 'entries', 'is_word']
    def __init__(self):
        # {normalized word: TrieNode}
        self.children = dict()
        # set of entry ids (could be multiple if exact same phrase used for different entries)
        self.entries = set()
        self.is_word = None  # Only for debugging/clarity; not required

class DictionaryIndex:
    __slots__ = ['word_entries', 'phrase_trie']
    def __init__(self):
        self.word_entries = dict()  # normalized_word -> set of dict_entry_ids
        self.phrase_trie = TrieNode()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    index = DictionaryIndex()
    # 1. Single words: for O(1) individual token lookup
    for key, entry in dictionary.items():
        # Decompose phrase into words
        tokens = []
        word = ''
        for c in key:
            if c.isalnum():
                word += c
            else:
                if word:
                    tokens.append(word)
                    word = ''
                # treat space or punctuation as separator (ignore)
        if word:
            tokens.append(word)
        # Remove empty tokens
        tokens = [w.lower() for w in tokens if w]
        if len(tokens) == 1:
            norm_word = tokens[0]
            index.word_entries.setdefault(norm_word, set()).add(entry)

    # 2. Phrases: trie for compound/nested matching
    for key, entry in dictionary.items():
        # Split key into tokens: keep only word tokens for trie path
        tokens = []
        word = ''
        for c in key:
            if c.isalnum():
                word += c
            else:
                if word:
                    tokens.append(normalize(word))
                    word = ''
                # separators are skipped in dictionary parsing
        if word:
            tokens.append(normalize(word))
        # Remove empty
        tokens = [w for w in tokens if w]
        if not tokens:
            continue
        # Insert into trie
        node = index.phrase_trie
        for w in tokens:
            if w not in node.children:
                node.children[w] = TrieNode()
            node = node.children[w]
        node.entries.add(entry)
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    tokens = list(tokens)
    N = len(tokens)
    annotation_sets = [set() for _ in range(N)]

    # For individual tokens (single words)
    for i, token in enumerate(tokens):
        if is_word(token):
            norm_token = normalize(token)
            if norm_token in dictionary_index.word_entries:
                annotation_sets[i].update(dictionary_index.word_entries[norm_token])

    # For multi-word phrases using the trie
    # To handle overlapping/nested phrases, do full scan from each possible start
    for start in range(N):
        if not is_word(tokens[start]):
            continue  # Can't start phrase on a separator
        node = dictionary_index.phrase_trie
        idxs = []  # list of token idxs (only word tokens)
        positions = []  # The actual positions in tokens, for tracking
        cur = start
        while cur < N:
            # Seek next word token at/after cur
            if is_word(tokens[cur]):
                norm_token = normalize(tokens[cur])
                if norm_token in node.children:
                    node = node.children[norm_token]
                    positions.append(cur)
                    idxs.append(cur)
                    # We found a phrase (could be nested), annotate all tokens from start to cur (see separator rules)
                    if node.entries:
                        # The participant range is from the first word of the phrase to the last, but may be interleaved with separators
                        # Backtrack to get start/end inclusive span:
                        phrase_start = idxs[0]
                        phrase_end = cur
                        # All word tokens in span [phrase_start, phrase_end] inclusive participate
                        # Inner separators: all in (phrase_start+1, phrase_end-1) (if any!)
                        # Find positions of separators within [phrase_start, phrase_end]:
                        # Leading/trailing separators (if any) outside this span!
                        for j in range(phrase_start, phrase_end + 1):
                            # All word tokens: always annotate
                            if is_word(tokens[j]):
                                annotation_sets[j].update(node.entries)
                        # Inner separators:
                        for j in range(phrase_start + 1, phrase_end):
                            if not is_word(tokens[j]):
                                annotation_sets[j].update(node.entries)
                    cur += 1
                else:
                    break
            else:
                # If at a separator, skip it for matching (allowed between words)
                cur += 1

    return [(tok, anns) for tok, anns in zip(tokens, annotation_sets)]
