```python
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    word_pattern = re.compile(r'\w+')
    
    def is_word_token(token):
        return bool(word_pattern.fullmatch(token))
    
    # Process each starting position
    for start_pos in range(n):
        if not is_word_token(tokens[start_pos]):
            continue
            
        # 1. Check single word
        word = tokens[start_pos].lower()
        if word in dictionary_index['single_words']:
            annotations[start_pos].update(dictionary_index['single_words'][word])
        
        # 2. Check compound phrases (words with flexible separators)
        collected_words = []
        word_positions = []
        
        for pos in range(start_pos, n):
            if is_word_token(tokens[pos]):
                collected_words.append(tokens[pos].lower())
                word_positions.append(pos)
                
                # Check if this sequence of words forms a phrase
                word_tuple = tuple(collected_words)
                if word_tuple in dictionary_index['compound_phrases']:
                    # Annotate all tokens from first word to last word
                    for i in range(word_positions[0], word_positions[-1] + 1):
                        annotations[i].update(dictionary_index['compound_phrases'][word_tuple])
        
        # 3. Check compound words (consecutive word tokens without separators)
        compound = ""
        for pos in range(start_pos, n):
            if is_word_token(tokens[pos]):
                compound += tokens[pos].lower()
                
                if compound in dictionary_index['compound_words']:
                    # Annotate all tokens from start_pos to pos
                    for i in range(start_pos, pos + 1):
                        annotations[i].update(dictionary_index['compound_words'][compound])
            else:
                # Stop at first non-word token
                break
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
```

The beauty of this approach be that it finds all possible matches - overlapping phrases, nested phrases, and even those tricky compound words that be formed from multiple tokens. And it does all this while respectin' the rules about not annotatin' leading and trailing separators!
