Main ideas

- Normalize the dictionary once. Split each entry into “words” as maximal runs of alphanumeric characters, lowercased.
  - If an entry has exactly one word (no separators among alphanumerics), store it in a direct lookup map: word -> set(ids).
  - If an entry has two or more words (there were non-alphanumeric separators in the original key), index it in a trie over words, ignoring the exact separators. This lets us match phrases like “AAA, BBB” against any tokens “AAA … BBB” with any inner separators.

- At annotation time:
  - Classify each input token as a word token if it is entirely alphanumeric (token.isalnum()). This matches the tests and gives simple, deterministic behavior.
  - For direct word matches, look up each word token (lowercased) in the direct map and add its ids.
  - For multi-word phrases (with separators in the dictionary key), scan the sequence of word tokens only (ignoring all separator tokens) while walking the trie, starting from every word-position. Whenever a trie node is terminal, annotate all tokens from the first matched word token through the last matched word token inclusive. This marks the inner separators but never leading/trailing ones.
  - For compound words (dictionary entry is a single word but the text may be tokenized into multiple adjacent word tokens), for each maximal run of consecutive word tokens (no separators between them), concatenate lowercased tokens incrementally and check the direct map. When a concatenation equals a dictionary single-word entry, annotate the whole span. This does not cross separators, so entries like “AAABBB” do not match “AAA ␣ BBB”.

- All matches are case-insensitive and must align to token boundaries (no mid-token matches). Overlapping and nested matches are all included by unioning into sets.

```python
import collections.abc

class _TrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: dict[str, _TrieNode] = {}
        self.ids: set = set()

def _split_words_lower(s: str) -> list[str]:
    """
    Split a string into lowercase 'words' as maximal alphanumeric runs.
    """
    words = []
    cur = []
    for ch in s:
        if ch.isalnum():
            cur.append(ch.lower())
        else:
            if cur:
                words.append("".join(cur))
                cur = []
    if cur:
        words.append("".join(cur))
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index:
      - 'word_ids': map of single-word entries -> set(ids)
      - 'phrase_trie': trie over sequences of words for multi-word entries
    """
    word_ids: dict[str, set] = {}
    trie_root = _TrieNode()

    for key, entry_id in dictionary.items():
        words = _split_words_lower(key)
        if not words:
            continue
        if len(words) == 1:
            w = words[0]
            s = word_ids.get(w)
            if s is None:
                s = set()
                word_ids[w] = s
            s.add(entry_id)
        else:
            node = trie_root
            for w in words:
                nxt = node.children.get(w)
                if nxt is None:
                    nxt = _TrieNode()
                    node.children[w] = nxt
                node = nxt
            node.ids.add(entry_id)

    return {"word_ids": word_ids, "phrase_trie": trie_root}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens using the built index.
    """
    idx = dictionary_index
    word_ids: dict[str, set] = idx["word_ids"]
    trie_root: _TrieNode = idx["phrase_trie"]

    toks = list(tokens)
    n = len(toks)
    toks_lower = [t.lower() for t in toks]
    is_word = [t.isalnum() for t in toks]
    annotations = [set() for _ in range(n)]

    # Direct single-token word matches
    for i, isw in enumerate(is_word):
        if not isw:
            continue
        ids = word_ids.get(toks_lower[i])
        if ids:
            annotations[i].update(ids)

    # Precompute positions of word tokens (ignore separators)
    word_positions = [i for i, isw in enumerate(is_word) if isw]

    # Multi-word phrase matches via trie (separator-insensitive)
    W = len(word_positions)
    for start_idx in range(W):
        node = trie_root
        # Walk forward over word tokens
        for step_idx in range(start_idx, W):
            wi = word_positions[step_idx]
            w = toks_lower[wi]
            node = node.children.get(w)
            if node is None:
                break
            if node.ids:
                si = word_positions[start_idx]
                ei = wi  # inclusive
                # Annotate all tokens from first to last matched word, including inner separators
                for k in range(si, ei + 1):
                    annotations[k].update(node.ids)

    # Compound single-word entries formed by adjacent word tokens (no separators)
    # Scan maximal runs of consecutive word tokens
    i = 0
    while i < n:
        if not is_word[i]:
            i += 1
            continue
        # start of a run
        run_start = i
        j = i + 1
        while j < n and is_word[j]:
            j += 1
        # run is [run_start, j)
        # For each start in the run, build concatenations to the right
        for s in range(run_start, j):
            concat = ""
            for e in range(s, j):
                concat += toks_lower[e]
                ids = word_ids.get(concat)
                if ids:
                    for k in range(s, e + 1):
                        annotations[k].update(ids)
        i = j

    # Produce output pairs
    return [(toks[i], annotations[i]) for i in range(n)]
```
