```python
import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    index = {}
    
    for phrase, entry_id in dictionary.items():
        # Extract words from the phrase (ignore separators)
        words = re.findall(r'\w+', phrase.lower())
        if not words:
            continue
            
        # Store as tuple of normalized words
        word_tuple = tuple(words)
        
        # Add to index (handle multiple entries with same normalized form)
        if word_tuple not in index:
            index[word_tuple] = set()
        index[word_tuple].add(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Helper function to check if a token is a word
    def is_word_token(token):
        return bool(re.search(r'\w', token))
    
    # Try to match phrases starting at each position
    for start_idx in range(n):
        # Skip if starting with a non-word token
        if not is_word_token(tokens[start_idx]):
            continue
        
        # Collect words starting from this position
        words_collected = []
        token_indices = []
        
        for end_idx in range(start_idx, n):
            if is_word_token(tokens[end_idx]):
                words_collected.append(tokens[end_idx].lower())
                token_indices.append(end_idx)
                
                # Check if current word sequence matches anything in dictionary
                word_tuple = tuple(words_collected)
                if word_tuple in dictionary_index:
                    # Found a match! Annotate all tokens in the range
                    for entry_id in dictionary_index[word_tuple]:
                        # Annotate word tokens
                        for idx in token_indices:
                            annotations[idx].add(entry_id)
                        
                        # Annotate separators between first and last word token
                        if len(token_indices) > 1:
                            first_word_idx = token_indices[0]
                            last_word_idx = token_indices[-1]
                            for idx in range(first_word_idx + 1, last_word_idx):
                                if not is_word_token(tokens[idx]):
                                    annotations[idx].add(entry_id)
    
    # Create annotated tokens
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
```
