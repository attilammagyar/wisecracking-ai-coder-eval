        self.single_words = {}             # word(str) -> set(ids)

def _split_words_lower(s: str) -> list[str]:

    for ch in s:

            cur.append(ch)
        else:
            if cur:
                out.append("".join(cur))
                cur = []
    if cur:
        out.append("".join(cur))
