- Inner separators of phrases should be annotated, but not leading/trailing ones
- We need to try matching at every position to find all overlapping phrases

Here's my solution approach:

1. **Index Building**: Parse dictionary entries into normalized word sequences. Distinguish between single words and phrases (phrases have separators between words).

2. **Matching Algorithm**: For each starting position, try to match dictionary entries:
   - For single-word entries: concatenate consecutive word tokens and check for exact matches
   - For phrase entries: match word sequences with flexible separators between them

3. **Optimization**: Index entries by their first word for faster lookup.

```python
import collections.abc

def is_word_token(token):
    """Check if a token contains any alphabetic characters."""
    return any(c.isalpha() for c in token)

def extract_words(text):
    """Extract words from text, returning list of lowercase words."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalpha():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries_by_first_word': {},  # Map from first word to list of (words, value) tuples
        'entries_by_length': {}  # Map from word count to list of entries (for optimization)
    }
    
    for key, value in dictionary.items():
        words = extract_words(key)
        if not words:
            continue
        
        # Index by first word
        first_word = words[0]
        if first_word not in index['entries_by_first_word']:
            index['entries_by_first_word'][first_word] = []
        index['entries_by_first_word'][first_word].append((words, value))
        
        # Index by length
        length = len(words)
        if length not in index['entries_by_length']:
            index['entries_by_length'][length] = []
        index['entries_by_length'][length].append((words, value))
    
    return index

def try_match_single_word(tokens, start_pos, target_word):
    """
    Try to match a single dictionary word by concatenating consecutive word tokens.
    Returns (matched, indices) where indices are the token positions that form the word.
    """
    if start_pos >= len(tokens) or not is_word_token(tokens[start_pos]):
        return False, []
    
    current_word = ""
    indices = []
    pos = start_pos
    
    # Concatenate consecutive word tokens
    while pos < len(tokens) and is_word_token(tokens[pos]):
        current_word += tokens[pos].lower()
        indices.append(pos)
        
        if current_word == target_word:
            return True, indices
        elif len(current_word) >= len(target_word):
            # No point continuing if we've already exceeded the target length
            break
        
        pos += 1
    
    return False, []

def try_match_phrase(tokens, start_pos, phrase_words):
    """
    Try to match a multi-word phrase with flexible separators.
    Returns (matched, indices) where indices are the token positions that are part of the phrase.
    """
    indices = []
    token_pos = start_pos
    word_idx = 0
    
    while token_pos < len(tokens) and word_idx < len(phrase_words):
        if is_word_token(tokens[token_pos]):
            # Extract and match the next word
            word_start = token_pos
            word_parts = []
            
            while token_pos < len(tokens) and is_word_token(tokens[token_pos]):
                word_parts.append(tokens[token_pos])
                indices.append(token_pos)
                token_pos += 1
            
            word = ''.join(word_parts).lower()
            if word != phrase_words[word_idx]:
                return False, []
            
            word_idx += 1
        else:
            # Separator token
            if word_idx > 0 and word_idx < len(phrase_words):
                # Inner separator - include it
                indices.append(token_pos)
                token_pos += 1
            else:
                # Leading or trailing separator - don't include
                break
    
    # Check if we matched all words
    if word_idx == len(phrase_words):
        return True, indices
    else:
        return False, []

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    # Try to match entries starting at each position
    for start_pos in range(len(tokens)):
        if not is_word_token(tokens[start_pos]):
            continue
        
        # Get the first word at this position for index lookup
        first_word = ""
        pos = start_pos
        while pos < len(tokens) and is_word_token(tokens[pos]):
            first_word += tokens[pos].lower()
            pos += 1
        
        # Try all entries that could start with this word
        for prefix_len in range(1, len(first_word) + 1):
            prefix = first_word[:prefix_len]
            if prefix in dictionary_index['entries_by_first_word']:
                for entry_words, value in dictionary_index['entries_by_first_word'][prefix]:
                    if len(entry_words) == 1:
                        # Single word entry
                        matched, indices = try_match_single_word(tokens, start_pos, entry_words[0])
                    else:
                        # Multi-word phrase
                        matched, indices = try_match_phrase(tokens, start_pos, entry_words)
                    
                    if matched:
                        for idx in indices:
                            annotations[idx].add(value)
    
    # Build result
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(len(tokens))]
    return annotated_tokens
```
