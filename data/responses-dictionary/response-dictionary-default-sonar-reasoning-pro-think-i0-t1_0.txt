import collections
from typing import Iterable

def build_dictionary_index(dictionary: collections.abc.Mapping) -> dict:
    """Preprocess dictionary into normalized string structures for fast lookup"""
    normalized = collections.defaultdict(set)
    concatenated = collections.defaultdict(set)
    
    for key, value in dictionary.items():
        tokens = []
        current = []
        for c in key.lower():
            if not c.isalnum():
                if current:
                    tokens.append(''.join(current))
                    current = []
            else:
                current.append(c)
        if current:
            tokens.append(''.join(current))
        
        # Create normalized (with spaces) and concatenated versions
        normalized_phrase = ' '.join(tokens) if tokens else ''
        concatenated_phrase = ''.join(tokens) if tokens else ''
        
        if normalized_phrase:
            normalized[normalized_phrase].add(value)
        if concatenated_phrase:
            concatenated[concatenated_phrase].add(value)
    
    return {'normalized': normalized, 'concatenated': concatenated}

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[tuple[str, collections.abc.Set]]:
    tokens = list(tokens)
    word_mask = [False]*len(tokens)
    
    # Precompute word tokens and separators
    for i, token in enumerate(tokens):
        is_word = True
        for c in token:
            if not c.isalnum() and not c.isalpha():
                is_word = False
        word_mask[i] = is_word
    
    groups = collections.defaultdict(set)
    
    # Sliding window approach for phrase detection
    for s in range(len(tokens)):
        current_norm = []
        current_concat = []
        for e in range(s, len(tokens)):
            if word_mask[e]:
                current_norm.append(tokens[e].lower())
                current_concat.append(tokens[e].lower())
            else:
                current_norm.append('')
                current_concat.append('')
            
            norm_phrase = ' '.join(current_norm)
            concat_phrase = ''.join(current_concat)
            
            # Check both normalized and concatenated versions
            possible_entries = set()
            if norm_phrase in dictionary_index['normalized']:
                possible_entries.update(dictionary_index['normalized'][norm_phrase])
            if concat_phrase in dictionary_index['concatenated']:
                possible_entries.update(dictionary_index['concatenated'][concat_phrase])
            
            if possible_entries:
                groups[(s, e)] = possible_entries.copy()
    
    result = [(token, set()) for token in tokens]
    
    # Annotate based on detected groups
    for (s, e), entries in groups.items():
        for i in range(s, e+1):
            if i > s and i < e:  # Inner elements only
                if word_mask[i]:
                    result[i][1].update(entries)
                else:
                    result[i][1].update(entries)
    
    # Add individual word annotations
    normalized = dictionary_index['normalized']
    concatenated = dictionary_index['concatenated']
    for i, token in enumerate(tokens):
        if word_mask[i]:
            lower_token = token.lower()
            if lower_token:
                if lower_token in normalized:
                    result[i][1].update(normalized[lower_token])
                if lower_token in concatenated:
                    result[i][1].update(concatenated[lower_token])
    
    return result
