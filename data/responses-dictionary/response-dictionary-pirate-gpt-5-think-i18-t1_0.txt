import collections.abc

class _DictEntry:
    __slots__ = ("words", "value")
    def __init__(self, words, value):
        self.words = tuple(words)  # tuple[str], lowercase
        self.value = value         # dictionary value/id

class _DictIndex:
    __slots__ = ("by_first", "first_prefixes", "max_first_len")
    def __init__(self):
        self.by_first = {}          # dict[str, list[_DictEntry]]
        self.first_prefixes = set() # set[str]
        self.max_first_len = 0      # int

def _split_units(s: str):
    # Yield (is_word: bool, chunk: str) as runs of isalnum vs non-alnum
    if not s:
        return
    cur = []
    cur_is_word = s[0].isalnum()
    for ch in s:
        iw = ch.isalnum()
        if iw == cur_is_word:
            cur.append(ch)
        else:
            yield cur_is_word, "".join(cur)
            cur = [ch]
            cur_is_word = iw
    if cur:
        yield cur_is_word, "".join(cur)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    idx = _DictIndex()
    first_words = set()

    for key, val in dictionary.items():
        # Normalize: split into units, keep lowercase word runs only.
        units = list(_split_units(key))
        # Extract words (ignore leading/trailing/middle separators entirely)
        words = [chunk.lower() for is_word, chunk in units if is_word and chunk]
        if not words:
            continue  # nothing to match
        entry = _DictEntry(words, val)
        first = words[0]
        first_words.add(first)
        idx.by_first.setdefault(first, []).append(entry)

    # Build prefix set for early stopping, and remember max first-word length
    maxlen = 0
    prefixes = set()
    for fw in first_words:
        maxlen = max(maxlen, len(fw))
        # all non-empty prefixes
        for i in range(1, len(fw) + 1):
            prefixes.add(fw[:i])

    idx.first_prefixes = prefixes
    idx.max_first_len = maxlen
    return idx

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    toks = list(tokens)
    n = len(toks)
    annos = [set() for _ in range(n)]

    if n == 0:
        return []

    idx = dictionary_index
    by_first = idx.by_first
    first_prefixes = idx.first_prefixes
    max_first_len = idx.max_first_len

    # Token classification and lowercase cache
    is_word = [t.isalnum() for t in toks]
    lower_tok = [t.lower() if w else t for t, w in zip(toks, is_word)]

    def consume_word(dict_word: str, start_pos: int):
        """
        Consume whole word tokens starting at start_pos so that their
        concatenation equals dict_word. Return end_pos or None on failure.
        """
        if start_pos >= n or not is_word[start_pos]:
            return None
        consumed_len = 0
        pos = start_pos
        L = len(dict_word)
        while pos < n and is_word[pos] and consumed_len < L:
            piece = lower_tok[pos]
            lp = len(piece)
            if consumed_len + lp > L:
                return None  # would end mid-token
            if dict_word[consumed_len:consumed_len + lp] != piece:
                return None
            consumed_len += lp
            end_pos = pos
            pos += 1
            if consumed_len == L:
                return end_pos
        return None

    for i in range(n):
        if not is_word[i]:
            continue

        # Build possible first-word candidates by concatenating contiguous word tokens
        s = ""
        j = i
        while j < n and is_word[j]:
            piece = lower_tok[j]
            if len(s) + len(piece) > max_first_len:
                break  # can't exceed any first-word length
            s += piece
            if s not in first_prefixes:
                break  # no dictionary first word starts with this
            # Try candidates whose first word equals s
            candidates = by_first.get(s)
            if candidates:
                for entry in candidates:
                    # entry.words[0] is s; attempt to match the remainder
                    pos = j + 1
                    end_word_idx = j
                    ok = True
                    for w in entry.words[1:]:
                        # Need at least one separator
                        if pos >= n or is_word[pos]:
                            ok = False
                            break
                        # Skip all separators (they're all equivalent)
                        while pos < n and not is_word[pos]:
                            pos += 1
                        if pos >= n:
                            ok = False
                            break
                        # Consume next dictionary word across whole tokens only
                        end_pos = consume_word(w, pos)
                        if end_pos is None:
                            ok = False
                            break
                        end_word_idx = end_pos
                        pos = end_pos + 1
                    if ok:

                        for k in range(i, end_word_idx + 1):

            j += 1

    # Return (token, annotations) for each token
    annotated_tokens = [(toks[k], annos[k]) for k in range(n)]
