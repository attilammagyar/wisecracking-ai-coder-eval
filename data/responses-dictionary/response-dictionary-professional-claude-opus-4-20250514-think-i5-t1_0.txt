import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},     # normalized word -> set of entry IDs
        'word_sequences': {},   # tuple of normalized words -> list of (entry_id, is_compound_word)
    }
    
    for text, entry_id in dictionary.items():
        # Extract words (alphanumeric sequences)
        words = re.findall(r'\w+', text)
        normalized_words = [w.lower() for w in words]
        
        if len(words) == 1:
            # Single word entry
            word = normalized_words[0]
            if word not in index['single_words']:
                index['single_words'][word] = set()
            index['single_words'][word].add(entry_id)
        else:
            # Multi-word entry
            # Check if it's a compound word (no separators between words)
            is_compound = re.match(r'^\w+$', text) is not None
            
            word_tuple = tuple(normalized_words)
            if word_tuple not in index['word_sequences']:
                index['word_sequences'][word_tuple] = []
            index['word_sequences'][word_tuple].append((entry_id, is_compound))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token position
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        """Check if token contains any word characters"""
        return bool(re.search(r'\w', token))
    
    # Process each starting position
    for start in range(n):
        # Collect word tokens and their positions starting from 'start'
        words = []
        word_positions = []
        
        for pos in range(start, n):
            if is_word_token(tokens[pos]):
                words.append(tokens[pos].lower())
                word_positions.append(pos)
                
                # Try to match collected words
                if len(words) == 1 and len(word_positions) == 1:
                    # Single word match
                    word = words[0]
                    if word in dictionary_index['single_words']:
                        for entry_id in dictionary_index['single_words'][word]:
                            annotations[word_positions[0]].add(entry_id)
                
                # Multi-word sequence match
                if len(words) > 1:
                    word_tuple = tuple(words)
                    
                    # Check for phrase matches
                    if word_tuple in dictionary_index['word_sequences']:
                        for entry_id, is_compound in dictionary_index['word_sequences'][word_tuple]:
                            if is_compound:
                                # Compound word: annotate only word positions
                                for wp in word_positions:
                                    annotations[wp].add(entry_id)
                            else:
                                # Phrase: annotate words and inner separators
                                for i in range(word_positions[0], word_positions[-1] + 1):
                                    if i in word_positions:
                                        # Word position
                                        annotations[i].add(entry_id)
                                    elif word_positions[0] < i < word_positions[-1]:
                                        # Inner separator
                                        annotations[i].add(entry_id)
                    
                    # Also check if concatenated words form a compound word
                    concatenated = ''.join(words)
                    if concatenated in dictionary_index['single_words']:
                        for entry_id in dictionary_index['single_words'][concatenated]:
                            for wp in word_positions:
                                annotations[wp].add(entry_id)
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
