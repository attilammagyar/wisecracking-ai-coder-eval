import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    def _split_into_key_words(text: str) -> list[str]:
        """Splits a string into a list of alphanumeric words, lowercased."""
        key_words = []
        current_word = []
        for char in text:
            if char.isalnum():
                current_word.append(char.lower())
            else:
                if current_word:
                    key_words.append("".join(current_word))
                    current_word = []
        if current_word:
            key_words.append("".join(current_word))
        return key_words

    # The trie is represented by a nested dictionary.
    # 'values' holds a set of dictionary entry IDs for phrases ending at this node.
    # 'children' maps the next key-word to a subsequent trie node.
    trie_root = {'children': {}, 'values': set()}

    for phrase, value in dictionary.items():
        key_words = _split_into_key_words(phrase)
        if not key_words:
            continue

        node = trie_root
        for key_word in key_words:
            if key_word not in node['children']:
                node['children'][key_word] = {'children': {}, 'values': set()}
            node = node['children'][key_word]
        
        node['values'].add(value)

    return trie_root

    if n == 0:

    annotated_tokens = [(token, set()) for token in tokens_list]

        return any(c.isalnum() for c in token)

        - token_cursor: The index in `tokens_list` from where to start searching for the next key-word.
        - trie_node: The current node in the dictionary_index trie.
        - matched_indices: The list of token indices that form the phrase prefix found so far.
        """

        # in the dictionary, we must record the annotation.

            for i in matched_indices:

                    last_word_idx = i

                word_start_cursor = i
                break
        
        if word_start_cursor == -1:

        # A key-word can be formed by concatenating multiple tokens.
        concatenated_word = ""

                # It can only exist between key-words.
                pass

    # Main loop: Try to start a phrase match from every token.
    for i in range(n):
        token = tokens_list[i]
        if not _is_word_token(token):
            continue
        
        # Try to match the first key-word of a phrase.
        concatenated_word = ""
        for j in range(i, n):
            current_token = tokens_list[j]
            if _is_word_token(current_token):
                concatenated_word += current_token.lower()
                if concatenated_word in dictionary_index['children']:
                    next_node = dictionary_index['children'][concatenated_word]
                    indices = list(range(i, j + 1))
                    search_from(j + 1, next_node, indices)
            else:
                # First key-word is formed by contiguous word tokens.
                # A separator means we can't extend this key-word further.
                break
    
    return annotated_tokens
