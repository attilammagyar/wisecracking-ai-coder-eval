```python
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},  # normalized_word -> set of entry IDs
        'phrases': {},       # tuple of normalized words -> set of entry IDs
        'entries': {}        # original dictionary for compound word detection
    }
    
    for key, value in dictionary.items():
        normalized_key = key.lower()
        index['entries'][normalized_key] = value
        
        # Split into tokens, keeping only word tokens
        tokens = []
        current_word = []
        for char in key:
            if char.isalnum():
                current_word.append(char.lower())
            else:
                if current_word:
                    tokens.append(''.join(current_word))
                    current_word = []
        if current_word:
            tokens.append(''.join(current_word))
        
        if len(tokens) == 1:
            # Single word
            word = tokens[0]
            if word not in index['single_words']:
                index['single_words'][word] = set()
            index['single_words'][word].add(value)
        elif len(tokens) > 1:
            # Phrase
            phrase_key = tuple(tokens)
            if phrase_key not in index['phrases']:
                index['phrases'][phrase_key] = set()
            index['phrases'][phrase_key].add(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []
    
    # Helper function to check if a token is a word token
    def is_word_token(token):
        return any(c.isalnum() for c in token)
    
    # Helper function to normalize a token
    def normalize_token(token):
        return token.lower()
    
    # Find all phrases that match at each position
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
            
        # Try to match phrases starting at this position
        word_tokens = []
        word_positions = []
        
        for i in range(start, n):
            if is_word_token(tokens[i]):
                word_tokens.append(normalize_token(tokens[i]))
                word_positions.append(i)
                
                # Check if current word sequence matches a phrase
                phrase_key = tuple(word_tokens)
                if phrase_key in dictionary_index['phrases']:
                    # Annotate all tokens in the phrase
                    for entry_id in dictionary_index['phrases'][phrase_key]:
                        # Annotate word tokens
                        for pos in word_positions:
                            annotations[pos].add(entry_id)
                        # Annotate separators between words (but not leading/trailing)
                        if len(word_positions) > 1:
                            for j in range(word_positions[0] + 1, word_positions[-1]):
                                if j not in word_positions:
                                    annotations[j].add(entry_id)
    
    # Check for single words and compound words
    i = 0
    while i < n:
        if is_word_token(tokens[i]):
            # Check single word
            normalized = normalize_token(tokens[i])
            if normalized in dictionary_index['single_words']:
                annotations[i].update(dictionary_index['single_words'][normalized])
            
            # Try to form compound words by concatenating adjacent word tokens
            compound = normalized
            j = i + 1
            while j < n and is_word_token(tokens[j]):
                compound += normalize_token(tokens[j])
                if compound in dictionary_index['entries']:
                    entry_id = dictionary_index['entries'][compound]
                    # Annotate all tokens that form this compound word
                    for k in range(i, j + 1):
                        annotations[k].add(entry_id)
                j += 1
        i += 1
    
    # Create result
    annotated_tokens = []
    for i in range(n):
        annotated_tokens.append((tokens[i], annotations[i]))
    
    return annotated_tokens
```
