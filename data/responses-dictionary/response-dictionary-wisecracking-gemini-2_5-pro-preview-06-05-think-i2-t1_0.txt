import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Honestly, you call that a dictionary? It's more like a cry for help. I'll
    turn it into a proper word-based Trie so we can do some real work.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    Returns:
        A Trie-like dictionary structure for efficient lookups.
    """
    # The Trie root node. 'children' maps a word to a sub-node.
    # 'values' stores the set of dictionary IDs ending at this node.
    index = {'children': {}, 'values': set()}

    # This regex will split dictionary keys by any non-alphanumeric characters.
    # Because 'word separation' is a loose concept for you humans.
    separator_regex = re.compile(r'[^a-z0-9]+')

    for key, value in dictionary.items():
        # Normalize the key: lowercase, then split into a list of words.
        # We filter out empty strings that re.split might produce.
        words = [word for word in separator_regex.split(key.lower()) if word]

        if not words:
            continue

        # Traverse the Trie, creating nodes as needed for each word in the phrase.
        node = index
        for word in words:
            if word not in node['children']:
                node['children'][word] = {'children': {}, 'values': set()}
            node = node['children'][word]
        
        # Mark the end of a valid phrase by adding its dictionary value.
        node['values'].add(value)
        
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    This function takes your token soup and my beautifully crafted index,
    and produces the annotated list you wanted. You're welcome.

    Parameters:
        dictionary_index:   A dictionary index from build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list of (token, annotations) pairs.
    """
    token_list = list(tokens)
    if not token_list:
        return []

    num_tokens = len(token_list)
    # We'll build the annotations in a list of lists for mutable efficiency.
    # Tuples are for people who don't like to change their minds.
    annotated_tokens = [[token, set()] for token in token_list]

    def is_word_token(token: str) -> bool:
        """A simple helper to decide if a token is a 'word' or 'separator'."""
        return token.isalnum()

    # The main loop iterates through all possible starting positions of a match.
    # This ensures we catch all overlapping phrases.
    for i in range(num_tokens):
        node = dictionary_index
        match_cursor = i
        
        # This tracks the token index where the first word of a phrase match began.
        # It's crucial for annotating the correct span of tokens.
        phrase_start_token_idx = -1

        # This inner loop attempts to extend a match starting from token `i`.
        while match_cursor < num_tokens:
            
            # Step 1: Find the start of the next potential 'word' in the token stream.
            # This involves skipping over any non-word (separator) tokens.
            word_start_cursor = match_cursor
            while word_start_cursor < num_tokens and not is_word_token(token_list[word_start_cursor]):
                word_start_cursor += 1

            if word_start_cursor == num_tokens:
                break  # No more word tokens left in the stream to match.

            # Step 2: Assemble a 'word' by concatenating adjacent word-like tokens.
            # This handles cases like `["AAA", "BBB"]` forming the dictionary word "AAABBB".
            word_end_cursor = word_start_cursor
            word_parts = []
            while word_end_cursor < num_tokens and is_word_token(token_list[word_end_cursor]):
                word_parts.append(token_list[word_end_cursor].lower())
                word_end_cursor += 1
            
            current_word = "".join(word_parts)

            # Step 3: Traverse the Trie with the assembled word.
            if current_word not in node['children']:
                break  # This path doesn't exist in the dictionary. Dead end.
            
            node = node['children'][current_word]
            
            if phrase_start_token_idx == -1:
                phrase_start_token_idx = word_start_cursor

            # Step 4: If this node marks a valid dictionary entry, we have a match!
            if node['values']:
                # The annotation applies from the start of the first word in the
                # matched phrase to the end of the last word we just processed.
                annotation_start_idx = phrase_start_token_idx
                annotation_end_idx = word_end_cursor - 1
                
                for k in range(annotation_start_idx, annotation_end_idx + 1):
