import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    word_pattern = re.compile(r'\w+')
    
    for phrase, entry_id in dictionary.items():
        # Split phrase into tokens (words and non-words)
        tokens = []
        pos = 0
        while pos < len(phrase):
            word_match = word_pattern.match(phrase, pos)
            if word_match:
                tokens.append(word_match.group().lower())
                pos = word_match.end()
            else:
                tokens.append(phrase[pos].lower())
                pos += 1
        
        # Store all possible subphrases (1 to N tokens)
        for start in range(len(tokens)):
            for end in range(start + 1, len(tokens) + 1):
                subphrase = tuple(tokens[start:end])
                if subphrase not in index:
                    index[subphrase] = set()
                index[subphrase].add(entry_id)
    
    return index

def is_word_token(token: str) -> bool:
    """Check if token is a word (contains at least one letter/digit)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    if not isinstance(dictionary_index, dict):
        raise ValueError("Invalid dictionary index")
    
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    max_phrase_length = max(len(phrase) for phrase in dictionary_index.keys()) if dictionary_index else 0
    
    for i in range(len(tokens)):
        current_token = tokens[i].lower()
        
        # Check individual word token
        if is_word_token(tokens[i]):
            single_token_phrase = (current_token,)
            if single_token_phrase in dictionary_index:
                annotations[i].update(dictionary_index[single_token_phrase])
        
        # Look ahead for compound phrases
        if is_word_token(tokens[i]):
            current_phrase = []
            j = i
            while j < len(tokens) and len(current_phrase) < max_phrase_length:
                if is_word_token(tokens[j]):
                    current_phrase.append(tokens[j].lower())
                else:
                    current_phrase.append(tokens[j].lower())
                
                phrase_tuple = tuple(current_phrase)
                if phrase_tuple in dictionary_index:
                    # Annotate all tokens in the phrase
                    for k in range(i, j + 1):
                        annotations[k].update(dictionary_index[phrase_tuple])
                j += 1
    
    return list(zip(tokens, annotations))
