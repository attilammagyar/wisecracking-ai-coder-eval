      def build_dictionary_index(dictionary: dict) -> object:
          # Normalize each entry to lowercase, split into words
          normalized_dict = {}
          max_phrase_length = 0
          for phrase, entry in dictionary.items():
              normalized_phrase = ' '.join(phrase.lower().split())
              if normalized_phrase not in normalized_dict:
                  normalized_dict[normalized_phrase] = set()
              normalized_dict[normalized_phrase].add(entry)
              # Track max phrase length
              word_count = len(normalized_phrase.split())
              if word_count > max_phrase_length:
                  max_phrase_length = word_count
          # Store the normalized dict and max length
          return {
              'normalized_dict': normalized_dict,
              'max_phrase_length': max_phrase_length
          }

def build_dictionary_index(dictionary):
    entries = {}
    max_length = 0
    for phrase, entry_id in dictionary.items():
        lowercase = phrase.lower()
        words = lowercase.split()
        normalized = ' '.join(words)
        # Update max phrase length
        if len(words) > max_length:
            max_length = len(words)
        # Add to entries
        if normalized in entries:
            entries[normalized].add(entry_id)
        else:
            entries[normalized] = {entry_id}
    return {'entries': entries, 'max_length': max_length}

def annotate(tokens, dictionary_index):
    token_info = []
    for token in tokens:
        is_word = bool(re.fullmatch(r'\w+', token))
        lower = token.lower() if is_word else None
        token_info.append({'is_word': is_word, 'lower': lower})

    annotations = [set() for _ in tokens]

    # Individual word handling
    for i, info in enumerate(token_info):
        if info['is_word']:
            lower = info['lower']
            if lower in dictionary_index['entries']:
                annotations[i].update(dictionary_index['entries'][lower])

    # Compound phrase handling
    for i in range(len(tokens)):
        for j in range(i, len(tokens)):
            words = []
            has_non_word = False
            for k in range(i, j+1):
                if token_info[k]['is_word']:
                    words.append(token_info[k]['lower'])
                else:
                    has_non_word = True
            if len(words) >= 2 and has_non_word:
                phrase = ' '.join(words)
                if phrase in dictionary_index['entries']:
                    phrase_entries = dictionary_index['entries'][phrase]
                    for k in range(i, j+1):
                        annotations[k].update(phrase_entries)
