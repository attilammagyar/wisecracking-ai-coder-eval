The text annotation problem is solved using an efficient dictionary index and a recursive search algorithm. The solution is broken down into two main parts: building the index and performing the annotation.

#### 1. Dictionary Indexing (`build_dictionary_index`)

This handles variations in separators.
*   **Trie Construction**: The normalized parts are inserted into the Trie. Each node in the Trie represents a part of a phrase. A special marker in each node indicates if it represents the end of a complete dictionary entry, and stores the corresponding entry ID(s). This structure allows for efficient searching of phrases part by part.

*   **Iteration**: The algorithm iterates through each token in the input sequence. *   **Recursive Search**: A recursive helper function, `_search_matches`, performs the core matching logic. It explores all possible continuations of a match from a given point in the token sequence and a given node in the Trie. This search handles three types of matches seamlessly:

```python
import collections.abc

# Helper class for the Trie data structure. Each node represents a
# potential part of a dictionary phrase.
class _TrieNode:
    """A node in the dictionary Trie."""
    def __init__(self):
        # Maps a word part (str) to the next _TrieNode
        self.children: dict[str, _TrieNode] = {}
        # Stores dictionary entry IDs if this node marks the end of a phrase
        self.values: set = set()

def _is_word_token(token: str) -> bool:
    """
    Checks if a token is a word token.
    A token is considered a word token if it is alphanumeric. This is used
    to distinguish words from separators and punctuation.
    """
    return token.isalnum()

def _get_parts_from_phrase(phrase: str) -> list[str]:
    """
    Normalizes a dictionary phrase into a list of word parts.
    For example, "black, swan" becomes ['black', 'swan'].
    This handles various separators and is case-insensitive.
    """
    parts = []
    current_part = ""
    for char in phrase.lower():
        if char.isalnum():
            current_part += char
        else:
            if current_part:
                parts.append(current_part)
                current_part = ""
    if current_part:
        parts.append(current_part)
    return parts

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    This implementation uses a Trie data structure.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        The root node of the Trie.
    """
    root = _TrieNode()
    for phrase, value in dictionary.items():
        parts = _get_parts_from_phrase(phrase)
        if not parts:
            continue

        node = root
        for part in parts:
            if part not in node.children:
                node.children[part] = _TrieNode()
            node = node.children[part]
        node.values.add(value)
    return root

def _search_matches(
    match_start_idx: int,
    cursor: int,
    node: _TrieNode,
    tokens: list[str],
    annotations: list[set]
):
    """
    Recursively searches for dictionary matches starting from `cursor`.

    Args:
        match_start_idx: The token index where the overall match started.
        cursor: The current token index to start consuming from.
        node: The current node in the dictionary Trie.
        tokens: The list of all input tokens.
        annotations: The list of sets to populate with annotations.
    """
    part_buffer = ""
    for k in range(cursor, len(tokens)):
        token = tokens[k]

        if not _is_word_token(token):
            # A part cannot be formed across separators (e.g., 'AAA BBB'
            # for dict key 'AAABBB' is not a match).
            break

        part_buffer += token.lower()

        if part_buffer in node.children:
            child_node = node.children[part_buffer]

            # If the child node has values, we've found a complete match.
            if child_node.values:
                # Annotate all tokens from the start of the phrase match
                # up to the end of the current part. This includes inner separators.
                for i in range(match_start_idx, k + 1):
                    annotations[i].update(child_node.values)

            # This part might also be a prefix of a longer phrase.
            # We search for the next part, which must be separated by
            # one or more non-word tokens.
            next_part_cursor = k + 1
            while next_part_cursor < len(tokens) and \
                  not _is_word_token(tokens[next_part_cursor]):
                next_part_cursor += 1

            if next_part_cursor < len(tokens):
                _search_matches(
                    match_start_idx,
                    next_part_cursor,
                    child_node,
                    tokens,
                    annotations
                )

def annotate(
    tokens: collections.abc.Iterable[str],
    dictionary_index: object
) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index: A dictionary index created by build_dictionary_index().
        tokens: The tokens to be annotated.

    Return:
        annotated_tokens: A list containing (token, annotations) pairs for each token in tokens.
    """
    # Ensure dictionary_index is of the expected type.
    if not isinstance(dictionary_index, _TrieNode):
        raise TypeError("dictionary_index must be a _TrieNode object.")
        
    # Convert iterable to list for indexed access.
    token_list = list(tokens)
    n = len(token_list)
    if n == 0:
        return []

    annotations = [set() for _ in range(n)]

    # A match can only start with a word token. This handles the requirement
    # that leading separators are not part of a phrase match.
    for i in range(n):
        if _is_word_token(token_list[i]):
            _search_matches(i, i, dictionary_index, token_list, annotations)

    return list(zip(token_list, annotations))
```
