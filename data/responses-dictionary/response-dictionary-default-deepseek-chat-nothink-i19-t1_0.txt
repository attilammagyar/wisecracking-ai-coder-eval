```python
import collections.abc
from collections import defaultdict

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # Structure to hold both individual words and phrase trie
    index = {
        'words': set(),  # Case-normalized individual words
        'phrases': defaultdict(dict),  # Trie structure for phrases
        'max_phrase_length': 0  # Maximum number of words in any phrase
    }
    
    for phrase, entry_id in dictionary.items():
        # Normalize case and split into words
        normalized = phrase.lower()
        words = normalized.split()
        
        if len(words) == 1:
            # Single word entry
            index['words'].add(normalized)
        else:
            # Multi-word phrase - add to trie
            current_level = index['phrases']
            for word in words:
                current_level = current_level.setdefault(word, {})
            current_level['__entry_ids__'] = current_level.get('__entry_ids__', set()) | {entry_id}
            index['max_phrase_length'] = max(index['max_phrase_length'], len(words))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)  # Need random access
    annotated_tokens = []
    n = len(tokens)
    
    # First pass: find all phrase matches and their locations
    phrase_matches = defaultdict(set)
    
    for i in range(n):
        if not tokens[i].strip():  # Skip pure whitespace tokens
            continue
            
        current_level = dictionary_index['phrases']
        current_phrase_length = 0
        j = i
        
        while j < n and current_phrase_length < dictionary_index['max_phrase_length']:
            token = tokens[j]
            normalized = token.lower()
            
            if not token.strip():  # Separator
                j += 1
                continue
                
            if normalized not in current_level:
                break
                
            current_level = current_level[normalized]
            current_phrase_length += 1
            j += 1
            
            if '__entry_ids__' in current_level:
                # Found a complete phrase match from i to j-1
                entry_ids = current_level['__entry_ids__']
                # Include all tokens between i and j-1 (inclusive)
                # but skip leading/trailing separators
                start = i
                end = j - 1
                
                # Find first non-separator from start
                while start < n and not tokens[start].strip():
                    start += 1
                
                # Find last non-separator before end
                while end >= 0 and not tokens[end].strip():
                    end -= 1
                
                if start > end:
                    continue  # No actual words in this phrase
                
                # Annotate all tokens in the range
                for k in range(start, end + 1):
                    phrase_matches[k].update(entry_ids)
    
    # Second pass: create annotations combining individual words and phrases
    for i, token in enumerate(tokens):
        annotations = set()
        
        # Check if token is a word (not just whitespace)
        if token.strip():
            normalized = token.lower()
            # Individual word match
            if normalized in dictionary_index['words']:
                # Find all dictionary entries that match this word
                for phrase, entry_id in dictionary_index['dictionary'].items():
                    if phrase.lower() == normalized:
                        annotations.add(entry_id)
        
        # Add any phrase matches
        annotations.update(phrase_matches.get(i, set()))
        
        annotated_tokens.append((token, annotations))
    
    return annotated_tokens
```
