import re
from collections import defaultdict
from typing import Iterable, Tuple, Dict, Set, Any

def build_dictionary_index(dictionary: Dict[str, Any]) -> Dict[str, Set[Any]]:
    """Build a normalized index of dictionary entries for efficient lookup."""
    index = defaultdict(set)
    regex = re.compile(r'\W+')
    for key, value in dictionary.items():
        normalized = ''.join(regex.split(key)).lower()
        if normalized:  # Ignore empty strings
            index[normalized].add(value)
    return dict(index)

def annotate(tokens: Iterable[str], dictionary_index: Dict[str, Set[Any]]) -> Iterable[Tuple[str, Set[Any]]]:
    """Annotate tokens using the normalized dictionary index."""
    annotated = [set() for _ in tokens]
    tokens_list = list(tokens)
    n = len(tokens_list)
    
    for i in range(n):
        word_tokens = []
        for j in range(i, n):
            token = tokens_list[j]
            # Check if token contains letter characters (word token heuristic)
            if any(c.isalpha() for c in token):
                word_tokens.append(token)
            normalized = ''.join(word_tokens).lower()
            if normalized in dictionary_index:
                ids = dictionary_index[normalized]
                for k in range(i, j + 1):
                    annotated[k].update(ids)
    
    return [(token, set(ids)) for token, ids in zip(tokens_list, annotated)]
