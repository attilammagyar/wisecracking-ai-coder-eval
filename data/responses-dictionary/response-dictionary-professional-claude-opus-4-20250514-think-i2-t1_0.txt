import collections.abc

def is_word_token(token):
    """Check if a token contains alphanumeric characters (is a word)."""
    return any(c.isalnum() for c in token)

def extract_words(text):
    """Extract word tokens from a text string."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'exact': {},      # exact normalized string -> set of entry IDs
        'words': {}       # tuple of normalized words -> set of entry IDs
    }
    
    for entry, entry_id in dictionary.items():
        # Store exact match (normalized to lowercase)
        normalized = entry.lower()
        if normalized not in index['exact']:
            index['exact'][normalized] = set()
        index['exact'][normalized].add(entry_id)
        
        # Store by word sequence
        words = tuple(w.lower() for w in extract_words(entry))
        if words:
            if words not in index['words']:
                index['words'][words] = set()
            index['words'][words].add(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Convert to list for indexing
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try all possible subsequences
    for start in range(n):
        for end in range(start + 1, n + 1):
            subseq = tokens[start:end]
            
            # Check exact concatenation match
            concatenated = ''.join(subseq).lower()
            if concatenated in dictionary_index['exact']:
                # All tokens in the subsequence get annotated
                for entry_id in dictionary_index['exact'][concatenated]:
                    for i in range(start, end):
                        annotations[i].add(entry_id)
            
            # Check word sequence match
            word_indices = []
            words = []
            for i, token in enumerate(subseq):
                if is_word_token(token):
                    word_indices.append(start + i)
                    words.append(token.lower())
            
            if words:
                word_tuple = tuple(words)
                if word_tuple in dictionary_index['words']:
                    # Only annotate from first word to last word (inclusive)
                    if len(word_indices) >= 1:
                        first_word_idx = word_indices[0]
                        last_word_idx = word_indices[-1]
                        for entry_id in dictionary_index['words'][word_tuple]:
                            for i in range(first_word_idx, last_word_idx + 1):
                                annotations[i].add(entry_id)
    
    annotated_tokens = [(token, annot) for token, annot in zip(tokens, annotations)]
