import collections.abc
from typing import Dict, Set, List, Tuple

class TrieNode:
    __slots__ = ('children', 'ids')
    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.ids: Set[int] = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    root = TrieNode()
    
    for phrase, phrase_id in dictionary.items():
        # Normalize phrase to lowercase and split into tokens
        normalized_phrase = phrase.lower()
        current_node = root
        
        for char in normalized_phrase:
            if char not in current_node.children:
                current_node.children[char] = TrieNode()
            current_node = current_node.children[char]
        
        current_node.ids.add(phrase_id)
    
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)  # Convert to list for random access
    n = len(tokens)
    annotated_tokens = []
    annotations_per_token = [set() for _ in range(n)]
    
    # First pass: find all individual word matches
    for i, token in enumerate(tokens):
        normalized_token = token.lower()
        current_node = dictionary_index
        
        for char in normalized_token:
            if char not in current_node.children:
                break
            current_node = current_node.children[char]
        else:
            annotations_per_token[i].update(current_node.ids)
    
    # Second pass: find all compound phrases
    for i in range(n):
        current_node = dictionary_index
        j = i
        phrase_chars = []
        
        while j < n:
            token = tokens[j]
            normalized_token = token.lower()
            
            for char in normalized_token:
                if char not in current_node.children:
                    break
                current_node = current_node.children[char]
                phrase_chars.append(char)
            else:
                j += 1
                if current_node.ids:
                    # Found a matching phrase from i to j-1
                    phrase_ids = current_node.ids
                    start = i
                    end = j - 1
                    
                    # Skip leading and trailing separators

                        start += 1
                    while end >= start and not tokens[end].strip():
                        end -= 1

                        continue
                    
                    # Annotate all tokens in the phrase (including inner separators)
