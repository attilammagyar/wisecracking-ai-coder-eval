## Main Ideas

### Data Structure

### Normalization

- Identify word vs non-word tokens (separators)

### Lookup Algorithm
1. For each starting position in the token list, try to match dictionary entries
2. Walk the trie following the normalized tokens
3. Skip separators when matching compound phrases (except between words)
4. Collect all matches (single words and phrases of various lengths)

import collections.abc

class TrieNode:
    def __init__(self):
        self.children = {}
        self.entry_ids = set()

def is_word_token(token):
    """Check if a token contains any alphanumeric characters."""
    return any(c.isalnum() for c in token)

def normalize_token(token):
    """Normalize a token for case-insensitive matching."""
    return token.lower()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Split phrase into tokens, preserving separators
        tokens = []
        current = []
        
        for char in phrase:
            if char.isalnum():
                current.append(char)
            else:
                if current:
                    tokens.append(''.join(current))
                    current = []
                if char not in ' \t\n\r':  # Keep non-whitespace separators
                    tokens.append(char)
        
        if current:
            tokens.append(''.join(current))
        
        # Filter to only word tokens for trie storage
        word_tokens = [normalize_token(t) for t in tokens if is_word_token(t)]
        
        if word_tokens:
            # Insert into trie
            node = root
            for token in word_tokens:
                if token not in node.children:
                    node.children[token] = TrieNode()
                node = node.children[token]
            node.entry_ids.add(entry_id)
    
    return root

def find_matches_from_position(tokens, start, trie_root):
    """Find all dictionary matches starting from a given position."""
    matches = []
    node = trie_root
    word_positions = []
    i = start
    
    # Skip leading separators
    while i < len(tokens) and not is_word_token(tokens[i]):
        i += 1
    
    if i >= len(tokens):
        return matches
    
    while i < len(tokens):
        token = tokens[i]
        
        if is_word_token(token):
            normalized = normalize_token(token)
            if normalized in node.children:
                node = node.children[normalized]
                word_positions.append(i)
                
                if node.entry_ids:
                    # Found a match, calculate the span
                    end = i
                    # Include trailing separators within the phrase
                    while end + 1 < len(tokens) and not is_word_token(tokens[end + 1]):
                        end += 1
                    
                    matches.append((start, end, node.entry_ids))
            else:
                break
        
        i += 1
    
    return matches

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    # Find all matches starting from each position
    for start in range(len(tokens)):
        matches = find_matches_from_position(tokens, start, dictionary_index)
        
        for match_start, match_end, entry_ids in matches:
            # Determine the actual range to annotate
            first_word = match_start
            while first_word <= match_end and not is_word_token(tokens[first_word]):
                first_word += 1
            
            last_word = match_end
            while last_word >= first_word and not is_word_token(tokens[last_word]):
                last_word -= 1
            
            # Annotate tokens in the range
            for i in range(first_word, last_word + 1):
                annotations[i].update(entry_ids)
    
    annotated_tokens = [(token, annot) for token, annot in zip(tokens, annotations)]
    return annotated_tokens
