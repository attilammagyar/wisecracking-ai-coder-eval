import collections.abc
import re

def is_word_token(token):
    """Check if a token is a word (contains only alphabetic characters)."""
    return bool(re.match(r'^[a-zA-Z]+$', token))

def extract_words(text):
    """Extract words from text, preserving order."""
    return re.findall(r'[a-zA-Z]+', text)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'direct': {},  # normalized full text -> entry IDs
        'phrases': {}  # normalized words tuple -> entry IDs
    }
    
    for key, entry_id in dictionary.items():
        # Direct lookup (case-insensitive)
        normalized_key = key.lower()
        if normalized_key not in index['direct']:
            index['direct'][normalized_key] = []
        index['direct'][normalized_key].append(entry_id)
        
        # For phrases, extract and normalize words
        words = extract_words(key)
        if len(words) > 1:
            normalized_words = tuple(w.lower() for w in words)
            if normalized_words not in index['phrases']:
                index['phrases'][normalized_words] = []
            index['phrases'][normalized_words].append(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Convert to list for indexing
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Try all possible spans
    for start in range(n):
        for end in range(start + 1, n + 1):
            span_tokens = tokens[start:end]
            
            # Single token lookup
            if end - start == 1:
                token = span_tokens[0]
                normalized = token.lower()
                if normalized in dictionary_index['direct']:
                    annotations[start].update(dictionary_index['direct'][normalized])
            
            # Multi-token spans
            else:
                # Skip if span starts or ends with non-word token
                if not is_word_token(span_tokens[0]) or not is_word_token(span_tokens[-1]):
                    continue
                
                # Extract words from span
                words_in_span = []
                for token in span_tokens:
                    if is_word_token(token):
                        words_in_span.append(token.lower())
                
                # Try as compound word (concatenated)
                concatenated = ''.join(span_tokens).lower()
                if concatenated in dictionary_index['direct']:
                    for i in range(start, end):
                        annotations[i].update(dictionary_index['direct'][concatenated])
                
                # Try as compound phrase
                if len(words_in_span) > 1:
                    words_tuple = tuple(words_in_span)
                    if words_tuple in dictionary_index['phrases']:
                        for i in range(start, end):
                            annotations[i].update(dictionary_index['phrases'][words_tuple])
    
    # Return annotated tokens
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
    return annotated_tokens
