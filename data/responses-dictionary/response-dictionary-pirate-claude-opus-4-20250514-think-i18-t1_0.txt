                # Try as a compound word (concatenate all tokens)
                compound = ''.join(tokens[start:end])
                compound_words = extract_words_from_text(compound)
                if compound_words:
                    key = normalize_for_lookup(compound_words)
                    if key in dictionary_index:
                        entry_ids = dictionary_index[key]
                        # Annotate all tokens in the range
                        for i in range(start, end):
                            annotations[i].update(entry_ids)
            
            # Also try as a phrase (with possible separators)
            words = extract_words_from_tokens(tokens, start, end)
            if words:
                key = normalize_for_lookup(words)
                if key in dictionary_index:
                    entry_ids = dictionary_index[key]
                    
                    # Find first and last word-containing token positions
                    first_word_pos = None
                    last_word_pos = None
                    for i in range(start, end):
                        if is_word_token(tokens[i]):
                            if first_word_pos is None:
                                first_word_pos = i
                            last_word_pos = i
                    
                    # Annotate from first word to last word (inclusive)
                    if first_word_pos is not None:
                        for i in range(first_word_pos, last_word_pos + 1):
                            annotations[i].update(entry_ids)
    
    return [(tokens[i], annotations[i]) for i in range(n)]
