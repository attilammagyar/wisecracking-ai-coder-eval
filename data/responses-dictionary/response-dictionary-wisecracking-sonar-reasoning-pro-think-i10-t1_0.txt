**Main Ideas:**

1. **Dictionary Indexing**  
   Convert each dictionary phrase into a lowercase sequence of words (ignoring separators like spaces/commas) stored in a trie. This allows case-insensitive lookup and quick identification of word sequences.

2. **Sliding Window for Phrases**  
   For each token, examine all possible starting positions up to the maximum phrase length. This handles overlapping/nested phrases by checking all valid word sequences ending at the current token.

3. **Efficient Annotation**  
   Track word sequences dynamically while scanning tokens. For each potential phrase match, annotate all tokens in its span—ensuring inner separators and words are tagged.

**Implementation:**

```python
import re
from collections import defaultdict

def build_dictionary_index(dictionary: dict) -> dict:
    normalized = defaultdict(set)
    max_length = 0
    
    for phrase, phrase_id in dictionary.items():
        words = re.split(r'\W+', phrase.lower())
        words = [word for word in words if word]
        if not words: continue
        word_seq = tuple(words)
        normalized[word_seq].add(phrase_id)
        if len(word_seq) > max_length:
            max_length = len(word_seq)
    
    return {'normalized': normalized, 'max_length': max_length}

def annotate(tokens: list[str], dictionary_index: dict) -> list[tuple[str, set]]:
    normal_dict, max_length = dictionary_index['normalized'], dictionary_index['max_length']
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    for i in range(n):
        for j in range(max(0, i - (max_length - 1)), i + 1):
            word_seq = []
            for k in range(j, i + 1):
                token = tokens[k].lower()
                parts = re.split(r'\W+', token)
                valid_parts = [p for p in parts if p]
                word_seq.extend(valid_parts)
                if len(word_seq) > max_length:
                    break
            seq_lengths_to_check = range(1, min(max_length, len(word_seq)) + 1)
            for k in seq_lengths_to_check:
                current_seq = tuple(word_seq[-k:]) if word_seq else None
                if current_seq in normal_dict:
                    for idx in range(j, i + 1):
                        annotations[idx].update(normal_dict[current_seq])
    
    return [(token, ann) for token, ann in zip(tokens, annotations)]
```
