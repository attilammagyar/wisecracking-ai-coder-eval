import collections.abc
from collections import defaultdict

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    
    def __init__(self):
        self.children = defaultdict(TrieNode)
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie-based index from a dictionary for fast lookup.
    Normalizes keys to lowercase and splits compound phrases into token sequences.
    """
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize case and split into tokens
        normalized = phrase.lower()
        tokens = normalized.split()
        current = trie
        
        # Build trie path for this phrase
        for token in tokens:
            current = current.children[token]
        current.entry_ids.add(entry_id)
    
    return trie

def is_separator(token: str) -> bool:
    """Check if token is a separator (non-word character)."""
    return not token.strip() or not token.isalnum()

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with dictionary entries using the trie index.
    Handles both individual words and compound phrases.
    """
    tokens = list(tokens)  # Need random access
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # First pass: individual word matches
    for i, token in enumerate(tokens):
        normalized = token.lower()
        # Check if current token is a complete entry
        if normalized in dictionary_index.children:
            node = dictionary_index.children[normalized]
            if node.entry_ids:
                annotations[i].update(node.entry_ids)
    
    # Second pass: compound phrase matches
    for i in range(n):
        if is_separator(tokens[i]):
            continue  # Skip leading separators
            
        current = dictionary_index
        j = i
        matched = True
        
        while j < n and matched:
            token = tokens[j]
            normalized = token.lower()
            
            # Check if we can continue the current phrase
            if normalized in current.children:
                current = current.children[normalized]
                # If this completes a phrase, annotate all participating tokens
                if current.entry_ids:
                    for k in range(i, j + 1):
                        if not (k == i and is_separator(tokens[k])) and \
                           not (k == j and is_separator(tokens[k])):
                            annotations[k].update(current.entry_ids)
                j += 1
            else:
                matched = False
                
            # Skip separators between words in phrase
            while j < n and is_separator(tokens[j]):
                if current.entry_ids:
                    annotations[j].update(current.entry_ids)
                j += 1
    
    return [(token, annotations[i]) for i, token in enumerate(tokens)]
