import collections.abc
import re

def _is_word(token: str) -> bool:
    """
    Checks if a token is a "word" by determining if it consists solely of
    alphanumeric characters. This distinction is crucial for identifying the
    boundaries of potential phrases and ignoring leading/trailing separators.
    """
    return token.isalnum()

def _normalize(text: str) -> str:
    """
    Creates a canonical string representation for case-insensitive and
    separator-agnostic matching.

    The normalization process involves:
    1. Converting to lowercase.
    2. Replacing all sequences of non-alphanumeric characters with a single space.
    3. Stripping leading/trailing whitespace.
    """
    lowered = text.lower()
    # Replace any sequence of one or more non-alphanumeric characters with a single space.
    normalized = re.sub(r'[^a-z0-9]+', ' ', lowered)
    return normalized.strip()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index is a Python dictionary that maps a normalized, canonical form
    of a dictionary entry to a set of its original IDs. This handles cases
    where different dictionary keys (e.g., "black swan", "Black-Swan")
    normalize to the same canonical form.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        A dictionary-based index for the annotation function.
    """
    index = {}
    for key, value in dictionary.items():
        normalized_key = _normalize(key)
        # We only add entries that result in a non-empty normalized key.
        if normalized_key:
            index.setdefault(normalized_key, set()).add(value)
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    The algorithm iterates through all possible contiguous sub-sequences of the
    input tokens. For a sub-sequence to be considered a potential match, it must
    start and end with a "word" token. This prevents matching leading or
    trailing separators of a phrase.

    Each valid sub-sequence is normalized and looked up in the dictionary index.
    If a match is found, all tokens within that sub-sequence (including any
    internal separators) are annotated with the corresponding dictionary entry IDs.
    This exhaustive search ensures that all single words, compound phrases,
    overlapping phrases, and nested phrases are correctly identified and annotated.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each
                            token in tokens.
    """
    # Cast dictionary_index to its actual type for clarity and type hinting.
    index: dict[str, set] = dictionary_index

    # Convert iterable to a list to allow multiple iterations and indexing.
    token_list = list(tokens)
    num_tokens = len(token_list)
    annotated_tokens = [(token, set()) for token in token_list]

    for i in range(num_tokens):
        # A valid phrase must start with a word, not a separator.
        if not _is_word(token_list[i]):
            continue

        for j in range(i, num_tokens):
            # A valid phrase must also end with a word.
            if not _is_word(token_list[j]):
                continue
            
            # Form a slice and normalize it to create a lookup key.
            token_slice = token_list[i : j + 1]

            candidate_key = _normalize("".join(token_slice))

                # If a match is found, update annotations for all tokens in the slice.
                for k in range(i, j + 1):
                    annotated_tokens[k][1].update(entry_ids)
                    
    return annotated_tokens
