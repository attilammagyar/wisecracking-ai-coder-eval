#### **Approach:**

- **Tokenization**: Split each dictionary entry into word tokens and separator tokens. This allows us to reconstruct how multi-word entries align to varying token splits.
- **Separator-insensitivity**:
    - For matching, treat all non-word (i.e., separator) tokens in the dictionary key as *wildcards*, which can align to any sequence of non-word tokens in the input.

#### **Trie**
- Use a **trie** (prefix tree) for all dictionary entries (after normalization and tokenization), where:
    - Each node represents a word token.
    - Separator "edges" can match any contiguous sequence of separator tokens in the input (but don't require exact matches).
    - The trie enables efficient scanning for long multi-token matches, supporting overlapping and nested matches.

### 2. Annotating the Token Sequence

import re

# Helper: determine if token is word
def is_word(token):
    # Accept tokens containing at least one alphanumeric character
    # (Assume tokenization is reasonable for the language.)
    return bool(re.search(r'[A-Za-z0-9]', token))

# Helper: normalize token for case-insensitive matching
def normalize_token(token):
    return token.lower()

class TrieNode:
    __slots__ = ('children', 'entries')
    def __init__(self):
        self.children = {}  # mapping normalized word token -> TrieNode
        self.entries = set()  # set of entry ids completed at this node

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    # The index: a root trie for phrases; a mapping for single-word-entries (for fast access)
    root = TrieNode()
    
    for key, entry_id in dictionary.items():
        # Split key into "word"/"sep" tokens
        # Tokenize with regex to keep words and separators
        # E.g. "AAA, BBB" --> ["AAA", ",", "BBB"]
        # We'll use a very lenient tokenizer: split on word boundaries
        tokens = []
        for match in re.finditer(r'\w+|[^\w\s]+|\s+', key):
            tokens.append(match.group(0))
        # Build normalized word token sequence, skipping separators
        node = root
        i = 0
        while i < len(tokens):
            tok = tokens[i]
            if is_word(tok):
                norm = normalize_token(tok)
                if norm not in node.children:
                    node.children[norm] = TrieNode()
                node = node.children[norm]
            # Skip separators (not part of trie branch, matching is handled in run)
            i += 1
        node.entries.add(entry_id)
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    # Prepare: build a list from the tokens to allow indexing
    tokens = list(tokens)
    n = len(tokens)
    # Map from token index to set of entry ids that annotate it
    annotations = [set() for _ in range(n)]
    # For efficiency, precompute "is_word" and "norm_token" for each token
    token_info = []
    for t in tokens:
        word = is_word(t)
        token_info.append({
            "is_word": word,
            "norm": normalize_token(t) if word else None
        })
    
    root = dictionary_index
    # For every possible start position
    i = 0
    while i < n:
        index = i
        node = root

        # Inner separator positions under compound phrase (between words, not leading/trailing)
        sep_indices = []
        last_word_idx = None
        # For each position starting at i, walk through trie
        k = index
        first_word = True
        while k < n:
            # Skip separators (but record them for later possible annotation)
            while k < n and not token_info[k]["is_word"]:
                if not first_word:
                    sep_indices.append(k)
                k += 1
            if k >= n:
                break
            # Now k is at a word token
            norm = token_info[k]["norm"]
            if norm not in node.children:
                break
            node = node.children[norm]
            path_indices.append(k)
            last_word_idx = k

                j = k

                first = path_indices[0]

                for idx in path_indices:
                    annotations[idx].update(node.entries)
                # Annotate inner separator tokens only

            child = node.children.get(norm)
