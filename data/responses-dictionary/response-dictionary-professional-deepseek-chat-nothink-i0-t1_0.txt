import collections.abc
from collections import defaultdict
from typing import Set, Dict, List, Tuple

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    
    def __init__(self):
        self.children = defaultdict(TrieNode)
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie-based index from a dictionary for fast lookup of words and
    compound phrases. The index is case-insensitive.
    """
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize to lowercase and split into words
        words = phrase.lower().split()
        current = trie
        
        for word in words:
            current = current.children[word]
        
        current.entry_ids.add(entry_id)
    
    return trie

def is_word_token(token: str) -> bool:
    """Check if token is a word (contains at least one letter/digit)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[Tuple[str, Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)  # Convert to list for random access
    n = len(tokens)
    annotations = [set() for _ in range(n)]  # Initialize empty annotations
    
    # First pass: find all individual word matches
    for i, token in enumerate(tokens):
        if not is_word_token(token):
            continue
            
        # Look up individual word
        normalized = token.lower()
        current = dictionary_index
        if normalized in current.children:
            node = current.children[normalized]
            annotations[i].update(node.entry_ids)
    
    # Second pass: find all compound phrases
    for i in range(n):
        if not is_word_token(tokens[i]):
            continue
            
        current = dictionary_index
        j = i
        matched = True
        
        while j < n and matched:
            token = tokens[j]
            
            if is_word_token(token):
                normalized = token.lower()
                if normalized in current.children:
                    current = current.children[normalized]
                    # Propagate the compound phrase's entry IDs to all tokens in the phrase
                    for k in range(i, j + 1):
                        annotations[k].update(current.entry_ids)
                else:
                    matched = False
            else:
                # For separators, continue only if we're in the middle of a phrase
                if j > i:
                    pass  # Separators are included in the phrase
                else:
                    matched = False
            
            j += 1
    
    # Combine tokens with their annotations
    return list(zip(tokens, annotations))
