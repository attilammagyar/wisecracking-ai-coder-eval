def _get_norm_tokens(text: str) -> list[str]:
    """
    Splits text into a list of lowercased alphanumeric words.
    This is basically a re-implementation of re.findall(r'\w+', text.lower())
    to adhere to the "built-in libraries only" rule. How quaint.
    """
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append("".join(current_word))
                current_word = []
    if current_word:
        words.append("".join(current_word))
    return words

    """

    character. This helps distinguish words from pure separators like " " or "*".
    """
    return any(c.isalnum() for c in token)

    compound phrases.
