        # We build a single word by concatenating adjacent word tokens.
        concatenated_word = ""
        first_word_idx = -1
        for j in range(i, n):
            token = token_list[j]
            if _is_word_token(token):
                if first_word_idx == -1:
                    first_word_idx = j
                
                concatenated_word += token.lower()
                
                # Check if the concatenated string is a single word in the dictionary.
                child_node = dictionary_index.children.get(concatenated_word)
                if child_node and child_node.is_end():
                    # Match found. Annotate all constituent tokens.
                    for k in range(first_word_idx, j + 1):
                        annotations[k].update(child_node.values)
            else:
                # A separator breaks the concatenation.
                break

    return list(zip(token_list, annotations))
