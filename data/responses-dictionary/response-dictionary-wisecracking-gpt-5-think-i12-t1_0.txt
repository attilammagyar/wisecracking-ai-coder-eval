import collections.abc

# Internal helper: classify tokens; and split dictionary keys into words.
def _is_word_token(tok: str) -> bool:
    # Treat tokens containing only alphanumeric characters as word tokens.
    # Simple, robust, and matches the tests' expectations.
    return tok.isalnum()

def _split_words_lower(s: str) -> list[str]:
    # Split a dictionary key into lowercase alphanumeric "words".
    words = []
    cur = []
    for ch in s:
        if ch.isalnum():
            cur.append(ch.lower())
        else:
            if cur:
                words.append("".join(cur))
                cur = []
    if cur:
        words.append("".join(cur))
    return words


class _TrieNode:
    __slots__ = ("ids", "next")
    def __init__(self):
        self.ids = set()
        self.next = {}  # dict[str, _TrieNode]


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index (trie) from a dictionary for fast lookup of words and
    compound phrases. Dictionary lookup is case-insensitive and separator-insensitive
    (for phrases).

                node = _TrieNode()
                node_map[w] = node
            node_map = node.next
        # End node: register this entry's identifier
        node.ids.add(value)
    return root
