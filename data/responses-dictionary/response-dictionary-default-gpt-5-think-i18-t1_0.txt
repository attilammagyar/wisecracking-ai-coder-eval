Main ideas

- Normalize everything with casefold() for case-insensitive lookup.
- Split dictionary entries into “words” using the regex \w+ (letters, digits, underscore). This makes multi-word phrases separator-insensitive: only the word sequence matters.
- Build two tries and one direct map:
  - A word-sequence trie for phrases with at least two words. This matches sequences of word tokens in the text, ignoring any separator tokens between them. When a phrase matches, we annotate all the word tokens and all inner separators between the first and last word (never separators outside).
  - A character trie for single-word dictionary entries. This enables matching entries like “AAABBB” across multiple adjacent word tokens such as ["AAA","BBB"], but never across separators. Matches must start and end at token boundaries.
  - A direct single-word map (normalized string -> set of ids) for O(1) exact single-token matches.
- Annotation algorithm:
  1) Exact single-token matches via the map.
  2) Concatenated single-word matches via the character trie across adjacent word tokens only. Record matches only at the end of a token (no mid-token matches).
  3) Multi-word phrase matches via the word-sequence trie across word tokens (any separators allowed). Annotate inner separators too by annotating the token range from first to last word inclusive.
- Tokens are considered “word tokens” if they fully match \w+; everything else is a separator.

```python
import collections.abc
import re


# Internal helpers and data structures

_WORD_RE = re.compile(r"\w+")


class _CharTrieNode:
    __slots__ = ("children", "ends")

    def __init__(self):
        self.children: dict[str, _CharTrieNode] = {}
        self.ends: set = set()


class _WordTrieNode:
    __slots__ = ("children", "ends")

    def __init__(self):
        self.children: dict[str, _WordTrieNode] = {}
        self.ends: set = set()


class _DictionaryIndex:
    __slots__ = ("single_word_map", "char_root", "phrase_root")

    def __init__(self):
        self.single_word_map: dict[str, set] = {}
        self.char_root = _CharTrieNode()     # for single-word entries (characters)
        self.phrase_root = _WordTrieNode()   # for multi-word phrases (word units)


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = _DictionaryIndex()

    for key, value in dictionary.items():
        # Normalize and split into "words" (separator-insensitive)
        words = _WORD_RE.findall(key.casefold())
        if not words:
            continue

        if len(words) == 1:
            w = words[0]
            # Exact single-token matches
            s = index.single_word_map.get(w)
            if s is None:
                s = set()
                index.single_word_map[w] = s
            s.add(value)

            # Also enable spanning across adjacent word tokens via character trie
            node = index.char_root
            for ch in w:
                nxt = node.children.get(ch)
                if nxt is None:
                    nxt = _CharTrieNode()
                    node.children[ch] = nxt
                node = nxt
            node.ends.add(value)
        else:
            # Multi-word phrase trie
            node = index.phrase_root
            for w in words:
                nxt = node.children.get(w)
                if nxt is None:
                    nxt = _WordTrieNode()
                    node.children[w] = nxt
                node = nxt
            node.ends.add(value)

    return index


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    # Convert to list for indexing
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]

    if n == 0:
        return []

    # Determine which tokens are "word" tokens
    is_word = [bool(_WORD_RE.fullmatch(tok)) for tok in tokens]
    word_positions = [i for i, flag in enumerate(is_word) if flag]
    word_lowers = [tokens[i].casefold() for i in word_positions]

    # Fast paths to the index components
    idx = dictionary_index
    single_word_map = idx.single_word_map
    char_root = idx.char_root
    phrase_root = idx.phrase_root

    # 1) Exact single-token matches
    for w_idx, tok_idx in enumerate(word_positions):
        w = word_lowers[w_idx]
        s = single_word_map.get(w)
        if s:
            annotations[tok_idx].update(s)

    # 2) Concatenated single-word matches across adjacent word tokens using the character trie.
    #    Matches must start and end at token boundaries, and only span adjacent word tokens
    #    (i.e., no separators between).
    m = len(word_positions)
    for start_w in range(m):
        node = char_root
        k = start_w
        while True:
            # Feed characters of the current word token
            s = word_lowers[k]
            ok = True
            for ch in s:
                nxt = node.children.get(ch)
                if nxt is None:
                    ok = False
                    break
                node = nxt
            if not ok:
                break

            # Record matches that end exactly at the end of this token
            if node.ends:
                start_tok = word_positions[start_w]
                end_tok = word_positions[k]
                entryset = node.ends
                for ti in range(start_tok, end_tok + 1):
                    annotations[ti].update(entryset)

            # Proceed only if the next word token is immediately adjacent (no separators)
            k += 1
            if k >= m:
                break
            if word_positions[k] != word_positions[k - 1] + 1:
                break

    # 3) Multi-word phrase matches using the word-sequence trie.
    #    Separators between words are allowed and must be annotated if inside the phrase.
    for start_w in range(m):
        node = phrase_root
        k = start_w
        while True:
            w = word_lowers[k]
            nxt = node.children.get(w)
            if nxt is None:
                break
            node = nxt

            if node.ends:
                # Annotate words and all inner separators:
                start_tok = word_positions[start_w]
                end_tok = word_positions[k]
                entryset = node.ends
                for ti in range(start_tok, end_tok + 1):
                    annotations[ti].update(entryset)

            k += 1
            if k >= m:
                break

    # Produce the final list of (token, set) pairs
    return [(tokens[i], annotations[i]) for i in range(n)]
```
