import re
from collections.abc import Mapping

class TrieNode:
    def __init__(self):
        self.children = {}
        self.entries = set()

def build_dictionary_index(dictionary: Mapping) -> TrieNode:
    root = TrieNode()
    for entry, entry_id in dictionary.items():
        # Preprocess entry to lowercase words, ignoring non-alphanumeric separators
        words = re.findall(r'\w+', entry.lower())
        current = root
        for word in words:
            # Create node if missing, then move to child node
            current = current.children.setdefault(word, TrieNode())
        current.entries.add(entry_id)
    return root

def annotate(tokens: list, dictionary_index: TrieNode) -> list:
    normalized = [t.lower() for t in tokens]
    annotations = [set() for _ in tokens]
    
    for i in range(len(tokens)):
        current_token = normalized[i]
        if not current_token.isalpha():
            continue
        
        # Collect all possible sequences starting at token i
        current_sequence = []
        word_indices = []
        j = i
        while j < len(tokens):
            if normalized[j].isalpha():
                current_sequence.append(normalized[j])
                word_indices.append(j)
            j += 1
        
        # Check sequences from length 1 to max possible
        for k in range(1, len(current_sequence) + 1):
            sequence = current_sequence[:k]
            current_node = dictionary_index
            valid_seq = True
            for word in sequence:
                if word not in current_node.children:
                    valid_seq = False
                    break
                current_node = current_node.children[word]
            if valid_seq and current_node.entries:
                start = word_indices[0]
                end = word_indices[k-1]
                # Annotate all tokens from start to end inclusive
                for idx in range(start, end + 1):
                    annotations[idx].update(current_node.entries)
        
        # Check single-word entry for token i
        current_node = dictionary_index
        if current_token in current_node.children and current_node.children[current_token].entries:
            annotations[i].update(current_node.children[current_token].entries)
    
    return [(token, ann) for token, ann in zip(tokens, annotations)]
