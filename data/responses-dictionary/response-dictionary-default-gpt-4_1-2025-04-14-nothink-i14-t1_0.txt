import collections.abc
import re

def _is_word(token):
    "Return True iff token is to be considered a wordish token."
    # Unicode aware word test (letter/digit/underscore), at least one char.
    return re.match(r"^\w+$", token, re.UNICODE) is not None

def _tokenize_dict_phrase(phrase):
    """
    Tokenize a dictionary phrase into a list of tokens (words and separators),
    using a pattern that splits at word boundaries.
    """
    # Split into word and non-word sequences, preserving them.
    tokens = []
    # \w+ matches word, \W+ matches non-word (at least one char)
    for m in re.finditer(r"\w+|\W+", phrase, re.UNICODE):
        tokens.append(m.group(0))
    return tokens

class DictionaryTrieNode:
    __slots__ = ('children', 'entry_ids')
    def __init__(self):
        self.children = {}  # token -> DictionaryTrieNode
        self.entry_ids = set()  # entry ids for a phrase ending here

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    # Build trie and word entry map (normalize case to lower)
    trie_root = DictionaryTrieNode()
    word_entries = {}  # normalized word -> set of entry id(s)
    max_phrase_tokens = 1

    for key, entry_id in dictionary.items():
        # Normalize key: lower-case for case insensitivity
        norm_phrase = key.lower()
        tokens = _tokenize_dict_phrase(norm_phrase)
        if len(tokens) == 1 and _is_word(tokens[0]):
            # Store simple word entries
            word = tokens[0]
            word_entries.setdefault(word, set()).add(entry_id)
        # Build trie for all (also for single token; safe)
        node = trie_root
        for token in tokens:
            token_key = token.lower()
            if token_key not in node.children:
                node.children[token_key] = DictionaryTrieNode()
            node = node.children[token_key]
        node.entry_ids.add(entry_id)
        if len(tokens) > max_phrase_tokens:
            max_phrase_tokens = len(tokens)

    index = {
        'trie_root': trie_root,
        'word_entries': word_entries,
        'max_phrase_tokens': max_phrase_tokens,
    }
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    # Prepare tokens as list for indexing
    tokens_list = list(tokens)
    N = len(tokens_list)
    annotated_entries = [set() for _ in range(N)]
    # Lowercase tokens for ALL lookups, for case-insensitive match
    norm_tokens = [t.lower() for t in tokens_list]
    max_phrase_tokens = dictionary_index['max_phrase_tokens']
    trie_root = dictionary_index['trie_root']
    word_entries = dictionary_index['word_entries']

    # Store word/separator for each token
    is_word = [_is_word(t) for t in tokens_list]

    # Phrase matching (windowed scan of token ranges)
    for start in range(N):
        node = trie_root
        # We'll allow phrase lengths up to as many tokens remain.
        for end in range(start, min(N, start + max_phrase_tokens)):
            token_key = norm_tokens[end]
            child = node.children.get(token_key)
            if child is None:
                break
            node = child
            if node.entry_ids:
                # Found a phrase match for tokens[start:end+1]
                # Determine which tokens should be annotated for this match:
                # All word tokens, and all _internal_ separators (not leading or trailing)
                span = range(start, end + 1)
                # Identify leading/trailing non-word tokens
                # Internal separator: non-word tokens not at the start/end of the phrase
                # First, find indices of first and last word
                word_indices = [i for i in span if is_word[i]]
                if not word_indices:
                    continue  # skip phrases that do not contain words
                first_word = word_indices[0]
                last_word = word_indices[-1]
                # Annotate: all word tokens in span, and any separator tokens (not at start/end)
                for i in span:
                    if is_word[i]:
                        annotated_entries[i].update(node.entry_ids)
                # Annotate internal separators: non-word tokens which are not at the
                # leading/trailing position -- i.e., falling strictly between first_word and last_word
                for i in span:
                    if not is_word[i] and first_word < i < last_word:
                        annotated_entries[i].update(node.entry_ids)

    # Individual word matches
    for idx, (token, word_flag) in enumerate(zip(norm_tokens, is_word)):
        if word_flag:
            ids = word_entries.get(token)
            if ids:
                annotated_entries[idx].update(ids)

    # Output assembled
    return [(tokens_list[i], annotated_entries[i]) for i in range(N)]

- **Multiple Annotations:** Tokens can be annotated with entries from overlapping phrases and/or as stand-alone words.
