Main idea

- Normalize the dictionary into a case-insensitive index keyed by sequences of word tokens (where “word” means any sequence of alphanumerics/underscore). Non-word characters in dictionary entries (spaces, commas, punctuation) are ignored and only the sequence of words matters. The index is a trie (prefix tree) over these normalized words. Each trie node holds the IDs of dictionary entries that end at that node.
- While annotating, we walk the token stream and:
  - Distinguish word tokens from separators using a simple regex. Matching against the dictionary is done only starting at word-token positions.
  - Use the trie to find all dictionary matches starting at each word-token position. Between dictionary words, any number of separator tokens is allowed (and those inner separators are included in the match span and get annotated). Leading/trailing separators are not included.
  - A dictionary “word” may be matched by concatenating multiple consecutive word tokens (no separators in between). This allows “AAABBB” in the dictionary to match tokens ["AAA", "BBB"]. Concatenation is deterministic: we keep concatenating adjacent word tokens until we exactly match the dictionary word or fail (no mid-token partial matches).
  - We collect every possible match (to support overlaps and nesting), and add its entry ID to the annotation set of each token within the match span (words and inner separators).
- Lookup is case-insensitive throughout. Mid-token matches are impossible because dictionary words must equal either one token or an exact concatenation of whole tokens.

```python
import re
import collections.abc

_WORD_RE = re.compile(r"\w+$")


def _is_word_token(tok: str) -> bool:
    return bool(_WORD_RE.match(tok))


class _TrieNode:
    __slots__ = ("children", "entries")

    def __init__(self):
        self.children: dict[str, _TrieNode] = {}
        self.entries: set = set()


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie over sequences of normalized words (case-insensitive).
    Non-word characters in dictionary keys are ignored (act as separators).
    Multiple keys that normalize to the same word sequence are all stored.
    """
    root = _TrieNode()
    for key, value in dictionary.items():
        # Normalize key into a sequence of words (ignore separators), case-insensitive.
        words = re.findall(r"\w+", key.lower())
        if not words:
            continue  # No word content; cannot be matched against tokens.
        node = root
        for w in words:
            node = node.children.setdefault(w, _TrieNode())
        node.entries.add(value)
    return root


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with dictionary entries using the trie built above.
    """
    tokens = list(tokens)
    n = len(tokens)
    if n == 0:
        return []

    root: _TrieNode = dictionary_index  # type: ignore[assignment]
    is_word = [_is_word_token(t) for t in tokens]
    tokens_low = [t.lower() for t in tokens]

    # Result container: one set per token index.
    ann = [set() for _ in range(n)]

    # Helper: generate candidate concatenations of consecutive word tokens starting at pos.
    # Returns list of (end_index_exclusive, concatenated_string).
    def _concat_candidates_from(pos: int):
        out = []
        if pos >= n or not is_word[pos]:
            return out
        s = tokens_low[pos]
        out.append((pos + 1, s))
        j = pos + 1
        while j < n and is_word[j]:
            s += tokens_low[j]
            out.append((j + 1, s))
            j += 1
        return out

    # DFS to extend matches from a given starting word-token position.
    def _extend_from(node: _TrieNode, start_pos: int, covered: list[int]):
        # Try to match the next dictionary word starting exactly at start_pos (must be word token).
        for end_pos, cand in _concat_candidates_from(start_pos):
            child = node.children.get(cand)
            if child is None:
                continue

            # Word span [start_pos, end_pos)
            word_span = list(range(start_pos, end_pos))
            new_covered = covered + word_span

            # Any entries ending here form a complete match; annotate now.
            if child.entries:
                for entry in child.entries:
                    for idx in new_covered:
                        ann[idx].add(entry)

            # Prepare to match the next dictionary word: skip separators (inner seps are included).
            j = end_pos
            while j < n and not is_word[j]:
                j += 1
            if j >= n:
                continue  # No next word available.

            # Include the separators between words as inner separators.
            seps = list(range(end_pos, j))
            _extend_from(child, j, new_covered + seps)

    # Start matches at every word-token position.
    for i in range(n):
        if not is_word[i]:
            continue
        # Quick check: if no child matches any concatenation at i, skip.
        # This avoids descending when nothing can match.
        any_possible = False
        for _, cand in _concat_candidates_from(i):
            if cand in root.children:
                any_possible = True
                break
        if not any_possible:
            # Still need to add single-word annotations if present? The check above covers that,
            # but if dictionary is empty or no match, there is nothing to do.
            continue

        _extend_from(root, i, [])

    # Return annotated list of (original_token, set_of_entries)
    return [(tokens[i], ann[i]) for i in range(n)]
```
