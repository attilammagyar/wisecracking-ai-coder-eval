   - Separator tokens: contain only non-alphanumeric characters

3. **Annotation Algorithm**:
   - Try all possible contiguous subsequences of tokens
   - For each subsequence, extract word tokens and try two lookup strategies:
     - **Phrase lookup**: Join words with spaces (handles "AAA BBB" style entries)
     - **Compound word lookup**: Concatenate adjacent word tokens (handles "AAABBB" style entries)
   - When a match is found, annotate:
     - All word tokens in the match
     - Inner separators (but not leading/trailing ones) for phrase matches

## Implementation

```python
import re
import collections.abc

def is_word_token(token):
    """Check if token contains any word characters."""
    return bool(re.search(r'\w', token))

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for key, value in dictionary.items():
        # Extract words from the key
        words = re.findall(r'\w+', key.lower())
        if not words:
            continue
        
        # Determine how to store the key
        # If there are non-word characters between words, it's a phrase
        non_word_chars = re.sub(r'\w+', '', key).strip()
        
        if non_word_chars:
            # Has separators between words - store as phrase
            normalized = ' '.join(words)
        else:
            # No separators - store as compound word (or single word)
            normalized = ''.join(words)
        
        if normalized not in index:
            index[normalized] = set()
        index[normalized].add(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try all possible contiguous subsequences of tokens
    for start in range(n):
        for end in range(start, n):
            # Extract the subsequence
            subseq = tokens[start:end+1]
            
            # Find word tokens and their positions
            words = []
            word_positions = []
            for i, token in enumerate(subseq):
                if is_word_token(token):
                    words.append(token.lower())
                    word_positions.append(start + i)
            
            if not words:
                continue
            
            # Try looking up as a phrase (words separated by spaces)
            phrase_form = ' '.join(words)
            if phrase_form in dictionary_index:
                entries = dictionary_index[phrase_form]
                # Annotate all word tokens
                for pos in word_positions:
                    annotations[pos].update(entries)
                # Annotate inner separators (between first and last word)
                if len(word_positions) > 1:
                    first_word = word_positions[0]
                    last_word = word_positions[-1]
                    for pos in range(first_word + 1, last_word):
                        if pos not in word_positions:
                            annotations[pos].update(entries)
            
            # Try looking up as a compound word (concatenated)
            # Only if the subsequence consists entirely of word tokens
            if len(words) == len(subseq):
                compound_form = ''.join(words)
                if compound_form in dictionary_index:
                    entries = dictionary_index[compound_form]
                    for pos in word_positions:
                        annotations[pos].update(entries)
    
    annotated_tokens = [(token, annotations[i]) for i, token in enumerate(tokens)]
    return annotated_tokens
```
