```python
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    class TrieNode:
        def __init__(self):
            self.children = {}
            self.entries = set()  # Dictionary entries that end at this node
    
    class DictionaryIndex:
        def __init__(self):
            self.root = TrieNode()
        
        def add_phrase(self, phrase, entry_id):
            # Normalize the phrase: lowercase and split into tokens
            tokens = []
            current_word = []
            
            for char in phrase:
                if char.isalnum() or char == "'":  # Part of a word
                    current_word.append(char.lower())
                else:  # Separator
                    if current_word:
                        tokens.append(''.join(current_word))
                        current_word = []
            
            if current_word:  # Don't forget the last word
                tokens.append(''.join(current_word))
            
            if not tokens:
                return
            
            # Add to trie
            node = self.root
            for token in tokens:
                if token not in node.children:
                    node.children[token] = TrieNode()
                node = node.children[token]
            node.entries.add(entry_id)
        
        def find_matches(self, tokens, start_idx):
            """Find all phrases that start at start_idx in tokens."""
            matches = []  # List of (end_idx_exclusive, entry_ids)
            node = self.root
            word_count = 0
            
            for i in range(start_idx, len(tokens)):
                token = tokens[i]
                normalized = self._normalize_token(token)
                
                if self._is_word_token(token):
                    if normalized in node.children:
                        node = node.children[normalized]
                        word_count += 1
                        if node.entries:
                            matches.append((i + 1, node.entries))
                    else:
                        break
                else:  # Separator
                    # For compound phrases, separators are allowed between words
                    if word_count > 0:
                        continue
                    else:
                        break
            
            return matches
        
        def _normalize_token(self, token):
            return token.lower()
        
        def _is_word_token(self, token):
            # A token is a word if it contains at least one alphanumeric character
            return any(c.isalnum() for c in token)
    
    # Build the index
    index = DictionaryIndex()
    for phrase, entry_id in dictionary.items():
        index.add_phrase(phrase, entry_id)
    
    return index


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens_list = list(tokens)
    n = len(tokens_list)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Try to match phrases starting at each position
    for start_idx in range(n):
        matches = dictionary_index.find_matches(tokens_list, start_idx)
        
        for end_idx, entry_ids in matches:
            # Determine the range to annotate
            # Find first and last word tokens in the matched range
            first_word_idx = None
            last_word_idx = None
            
            for i in range(start_idx, end_idx):
                if dictionary_index._is_word_token(tokens_list[i]):
                    if first_word_idx is None:
                        first_word_idx = i
                    last_word_idx = i
            
            if first_word_idx is not None and last_word_idx is not None:
                # Annotate from first word to last word (inclusive)
                for i in range(first_word_idx, last_word_idx + 1):
                    annotations[i].update(entry_ids)
    
    # Create the result
    annotated_tokens = []
    for i, token in enumerate(tokens_list):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
```
