import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'words': {},     # normalized_word -> list of entry IDs
        'phrases': {},   # (word1, word2, ...) -> list of entry IDs
    }
    
    for key, value in dictionary.items():
        # Extract words using regex (split by non-word characters)
        words = [w for w in re.split(r'\W+', key) if w]
        
        if not words:
            continue
            
        # Check if this is a single word/compound word or a phrase
        # If the key contains only word characters (and possibly nothing between them),
        # it's a word; otherwise it's a phrase
        if re.match(r'^\w+$', key):
            # Single or compound word (no spaces or separators)
            normalized = key.lower()
            if normalized not in index['words']:
                index['words'][normalized] = []
            index['words'][normalized].append(value)
        else:
            # Phrase (contains non-word characters)
            normalized_tuple = tuple(w.lower() for w in words)
            if normalized_tuple not in index['phrases']:
                index['phrases'][normalized_tuple] = []
            index['phrases'][normalized_tuple].append(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        """Check if token contains only word characters."""
        return bool(re.match(r'^\w+$', token))
    
    # Process each starting position
    for start in range(n):
        # 1. Try single token as word
        if is_word_token(tokens[start]):
            normalized = tokens[start].lower()
            if normalized in dictionary_index['words']:
                annotations[start].update(dictionary_index['words'][normalized])
        
        # 2. Try concatenated tokens as compound word
        if is_word_token(tokens[start]):
            concatenated = tokens[start]
            for end in range(start + 1, n):
                if is_word_token(tokens[end]):
                    concatenated += tokens[end]
                    normalized = concatenated.lower()
                    if normalized in dictionary_index['words']:
                        # Annotate all tokens in this range
                        for i in range(start, end + 1):
                            annotations[i].update(dictionary_index['words'][normalized])
                else:
                    break  # Stop at first non-word token
        
        # 3. Try phrase matches
        # Skip if starting with a separator (leading separators don't count)
        if not is_word_token(tokens[start]):
            continue
            
        # Collect words and their positions
        words = []
        positions = []
        
        i = start
        while i < n:
            if is_word_token(tokens[i]):
                words.append(tokens[i].lower())
                positions.append(i)
                
                # Check if current word sequence matches any phrase
                words_tuple = tuple(words)
                if words_tuple in dictionary_index['phrases']:
