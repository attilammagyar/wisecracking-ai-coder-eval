```python
import collections.abc
import re


class _TrieNode:
    """A node in the dictionary Trie, a fine vessel for our treasures."""
    def __init__(self):
        # Children can be characters or a special marker for separators.
        self.children: dict[str | None, _TrieNode] = {}
        # A set of dictionary entry IDs for phrases ending at this node.
        self.entries: set = set()


def _is_word_token(token: str) -> bool:
    """A pirate's check: Is this token a word, or just flotsam?"""
    return any(c.isalnum() for c in token)


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases. Arrr!

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = _TrieNode()
    # A special key in our children map to represent a separator transition.
    _SEP = None

    for key, entry_id in dictionary.items():
        if not key:
            continue
        
        current_node = root
        
        # We split the key into word parts and separators.
        # For "AAA, BBB", this yields ['', 'AAA', ', ', 'BBB', ''].
        parts = re.split(r'([a-zA-Z0-9]+)', key)

        for part in parts:
            if not part:
                continue
            
            if _is_word_token(part):
                # Word parts are processed char by char, normalized to lowercase.
                for char in part.lower():
                    current_node = current_node.children.setdefault(char, _TrieNode())
            else:
                # Any sequence of separators becomes a single transition.
                current_node = current_node.children.setdefault(_SEP, _TrieNode())
        
        current_node.entries.add(entry_id)

    return root


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary. Let's find that treasure!

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    # Use the same special key for separator transitions.
    _SEP = None
    root: _TrieNode = dictionary_index
    
    tokens_list = list(tokens)
    n = len(tokens_list)
    if n == 0:
        return []

    annotated_tokens = [(token, set()) for token in tokens_list]

    for i in range(n):
        # A phrase must start with a word token, not a separator, says the captain.
        if not _is_word_token(tokens_list[i]):
            continue

        current_node = root
        last_word_token_idx = -1

        for j in range(i, n):
            token = tokens_list[j]
            next_node = current_node
            
            # If it's a word token, we traverse the Trie.
            if _is_word_token(token):
                # If there were separator tokens between this word and the last,
                # we must cross a separator edge in our Trie.
                if last_word_token_idx != -1 and j > last_word_token_idx + 1:
                    if _SEP not in next_node.children:
                        break  # Path has a separator, but dict key doesn't.
                    next_node = next_node.children[_SEP]

                # Now, traverse the characters of the token.
                path_found = True
                for char in token.lower():
                    if char not in next_node.children:
                        path_found = False
                        break
                    next_node = next_node.children[char]
                
                if not path_found:
                    break  # This token doesn't continue any phrase.
                
                current_node = next_node
                last_word_token_idx = j
            
            # If it's a separator, we do naught but continue. The transition
            # will be handled by the next word token.
            
            # After processing token j, check if we've landed on a complete entry.
            if current_node.entries:
                # A match! The annotation applies from the start of the search (i)
                # to the last word token we've seen in this match.
                for k in range(i, last_word_token_idx + 1):
                    annotated_tokens[k][1].update(current_node.entries)

    return annotated_tokens
```
