```python
import collections.abc
from collections import defaultdict

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # Normalize dictionary keys to lowercase and store their values
    normalized_dict = {k.lower(): v for k, v in dictionary.items()}
    
    # Build a trie for compound phrase lookup
    trie = {}
    for phrase, entry_id in normalized_dict.items():
        # Split phrase into tokens (words and separators)
        # We consider sequences of letters as words, everything else as separators
        tokens = []
        current_word = []
        for c in phrase:
            if c.isalpha():
                current_word.append(c.lower())
            else:
                if current_word:
                    tokens.append(''.join(current_word))
                    current_word = []
                tokens.append(c)
        if current_word:
            tokens.append(''.join(current_word))
        
        # Add to trie
        node = trie
        for token in tokens:
            if token not in node:
                node[token] = {}
            node = node[token]
        # Mark the end of the phrase with its entry_id
        if '_entries' not in node:
            node['_entries'] = set()
        node['_entries'].add(entry_id)
    
    # Also store individual words for quick lookup
    individual_words = defaultdict(set)
    for phrase, entry_id in normalized_dict.items():
        # Split into words (ignoring separators)
        words = [word.lower() for word in phrase.split() if word.isalpha()]
        for word in words:
            individual_words[word].add(entry_id)
    
    return {
        'trie': trie,
        'individual_words': individual_words,
        'normalized_dict': normalized_dict,
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    annotated_tokens = []
    tokens = list(tokens)  # Make sure we can iterate multiple times
    
    for i in range(len(tokens)):
        token = tokens[i]
        annotations = set()
        
        # Check if token is a word (contains at least one letter)
        is_word = any(c.isalpha() for c in token)
        normalized_token = token.lower() if is_word else None
        
        # Individual word lookup
        if normalized_token and normalized_token in dictionary_index['individual_words']:
            annotations.update(dictionary_index['individual_words'][normalized_token])
        
        # Compound phrase lookup
        # We'll look for all possible phrases starting at this token
        node = dictionary_index['trie']
        j = i
        while j < len(tokens):
            current_token = tokens[j]
            current_is_word = any(c.isalpha() for c in current_token)
            normalized_current = current_token.lower() if current_is_word else current_token
            
            if normalized_current in node:
                node = node[normalized_current]
                if '_entries' in node:
                    annotations.update(node['_entries'])
                j += 1
            elif not current_is_word:
                # Skip non-word tokens when they don't match exactly
                # (this allows for different separators in text vs dictionary)
                j += 1
            else:
                break
        
        annotated_tokens.append((token, annotations))
    
    return annotated_tokens
```
