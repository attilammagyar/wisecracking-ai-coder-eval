import collections.abc

# Helper function: detect if a token is a word
def is_word_token(token):
    # Consider word tokens as those that have at least one alphanum
    # (This be enough for testcases, but tweak as need be)
    return any(c.isalnum() for c in token)

def normalize_word(token):
    return token.lower()

def normalize_dict_phrase(phrase):
    # Collapse all whitespace-like and non-word chars to single spaces between words
    tokens = []
    buf = ''
    for c in phrase:
        if c.isalnum():
            buf += c.lower()
        else:
            if buf:
                tokens.append(buf)
                buf = ''
    if buf:
        tokens.append(buf)
    return tuple(tokens)

class TrieNode:
    __slots__ = 'children', 'entries'
    def __init__(self):
        self.children = dict()
        self.entries = set()  # phrase-endings: set of dictionary IDs (safe for word or phrase)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized trie index for lookup of words/compound phrases,
    treating case-insensitivity and allowing any separator as a non-word.
    """
    root = TrieNode()
    # Map normalized word to set of entry ids for O(1) lookup as single word
    singles = collections.defaultdict(set)
    # Insert each dictionary entry into trie by normalized phrase tokens
    for raw_phrase, entry_id in dictionary.items():
        words = normalize_dict_phrase(raw_phrase)
        if not words:
            continue  # skip empty dictionary entry
        node = root
        for w in words:
            if w not in node.children:
                node.children[w] = TrieNode()
            node = node.children[w]
        node.entries.add(entry_id)
        # If single word, add to singles
        if len(words) == 1:
            singles[words[0]].add(entry_id)
    return {'trie': root, 'singles': singles}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary using trie matching.
    """
    # Copy tokens to list for multi-token lookahead
    tokens = list(tokens)
    N = len(tokens)
    # For each token: accum set of matching dict ids
    annotations = [set() for _ in range(N)]

    # Precompute word/nonword status for each token (speeds up multiple walks)
    is_word = [is_word_token(tok) for tok in tokens]
    # For each token, what is its normalized word or ''
    norm_word = [normalize_word(tok) if is_word[k] else '' for k, tok in enumerate(tokens)]
    # For quick single-word lookup
    singles = dictionary_index['singles']
    trie = dictionary_index['trie']

    # First: annotate each token as single word if it matches
    for i in range(N):
        if is_word[i]:
            word = norm_word[i]
            if word in singles:
                annotations[i].update(singles[word])

    # Second: scan all possible starts for compound phrases
    for start in range(N):
        # Only attempt to start matching phrases on word tokens
        if not is_word[start]:
            continue
        idx = start
        node = trie
        phrase_tokens = []
        phrase_indices = []

        # We're going to walk through the sequence [start, ...], skip separators in token stream
        # to match only the words in the phrase path
        i = start
        j = start
        matched_indices = []
        while i < N and is_word[i]:
            phrase_tokens.append(norm_word[i])
            phrase_indices.append(i)
            # Try to advance in trie
            if norm_word[i] not in node.children:
                break
            node = node.children[norm_word[i]]
            matched_indices.append(i)
            # Now, from i, find the next word token (if any), collecting any separator indices
            next_i = i + 1
            separators = []
            # We gather inner separators after the first word, but before trailing
            while next_i < N and not is_word[next_i]:
                separators.append(next_i)
                next_i += 1
            # If we matched a phrase up to here, tag all tokens between start and current word (i...next_i-1)
            if node.entries:
                # Start at 'start', finish at (next_i-1) if separators were found; else i
                first = start
                last = i
                # But, include inner separators: only those strictly between first and last
                # (But if first==last, there are no separators to include)
                phrase_span = list(range(first, i + 1))
                # If i-first >= 1, that means we did have separators between—add them
                if separators and next_i - 1 > i:
                    # Add inner separators between first and next_i-1 (excluding leading/trailing)
                    sep_indices = list(range(i + 1, next_i))
                    phrase_span += sep_indices
                # Mark each token in phrase_span with these entries
                for ix in phrase_span:
                    annotations[ix].update(node.entries)
            # If there is a next word, jump to it. Else, break.
            if next_i >= N:
                break
            i = next_i
        # End while
    # Build output tuples
    return list(zip(tokens, annotations))
