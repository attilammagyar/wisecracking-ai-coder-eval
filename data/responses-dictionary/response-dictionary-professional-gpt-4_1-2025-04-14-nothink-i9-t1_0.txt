- **Goal:** Efficiently match dictionary entries (words and compounds/phrases), case-insensitively, across arbitrarily tokenized input, handling separator-insensitivity for compounds.

```python
import collections.abc
import re

def is_word_token(token):
    # Simple: treat as word token if alphanum (can adjust for locale if needed)
    return any(c.isalnum() for c in token)

def normalize_word(w):
    # Lowercase & strip
    return w.strip().lower()

def split_phrase(phrase):
    """
    Split a phrase string into (words, separators).
    Returns: [word0, sep0, word1, sep1, word2, ...]; separators are kept as a single space
    Since we are only interested in making a trie for words, we will just extract the words in order.
    """
    parts = []
    for m in re.finditer(r'\w+|\W+', phrase, flags=re.UNICODE):
        if m.group().strip() == '':
            continue
        text = m.group()
        if is_word_token(text):
            parts.append(('word', normalize_word(text)))
        else:
            parts.append(('sep', ' '))  # Normalize all separators to a single space placeholder
    # Coalesce multiple consecutive seps
    condensed = []
    last_type = None
    for typ, val in parts:
        if typ == 'sep' and last_type == 'sep':
            continue
        condensed.append((typ, val))
        last_type = typ
    # Remove leading/trailing separators for matching
    while condensed and condensed[0][0] == 'sep':
        condensed = condensed[1:]
    while condensed and condensed and condensed[-1][0] == 'sep':
        condensed = condensed[:-1]
    # Only keep words for trie steps
    wordlist = [val for typ, val in condensed if typ == 'word']
    return wordlist

class TrieNode:
    __slots__ = ('children', 'entries')
    def __init__(self):
        self.children = dict()
        self.entries = set()

    def insert(self, wordlist, entry):
        node = self
        for word in wordlist:
            if word not in node.children:
                node.children[word] = TrieNode()
            node = node.children[word]
        node.entries.add(entry)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    word_map = collections.defaultdict(set)
    trie_root = TrieNode()
    # Indexing
    for phrase, entry in dictionary.items():
        wordlist = split_phrase(phrase)
        if not wordlist:
            continue
        # Individual word (single word phrase)
        if len(wordlist) == 1:
            word_map[wordlist[0]].add(entry)
        # Compound phrase
        if len(wordlist) > 1:
            trie_root.insert(wordlist, entry)
            # Also allow for the possible annotation of phrases which are glued together (i.e., no separator tokens)

    return (word_map, trie_root)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    tokens = list(tokens)
    n = len(tokens)
    word_map, trie_root = dictionary_index
    # Precompute word or separator for each token
    token_types = [is_word_token(token) for token in tokens]  # True=word, False=sep
    normalized_tokens = [normalize_word(token) if is_word_token(token) else None for token in tokens]

    # For each token, we accumulate a set of annotation ids
    annotations = [set() for _ in range(n)]

    # 1. Word-level annotations
    for i, (is_word, norm) in enumerate(zip(token_types, normalized_tokens)):
        if is_word and norm in word_map:
            annotations[i].update(word_map[norm])

    # 2. Phrase/compound-level annotations
    i = 0
    while i < n:
        if not token_types[i]:
            i += 1
            continue
        # Start only from word tokens
        # For each trie path, walk skipping over separators, but not starting on or ending on one
        trie_node = trie_root
        match_positions = []  # indices of tokens which participate in the phrase (including inner separators)
        j = i
        phrase_steps = []  # for each phrase step we want to pair up: (token idx, normalized_word)
        while j < n:
            if not token_types[j]:
                # On a separator — for inner phrase, allow, but not before the first word
                match_positions.append(j)
                j += 1
                continue
            norm_word = normalized_tokens[j]
            if norm_word in trie_node.children:
                # Take a phrase step
                match_positions.append(j)
                trie_node = trie_node.children[norm_word]
                # If we reach an endpoint, annotate all tokens from i to j included, but prune leading/trailing separators
                if trie_node.entries:
                    # Determine the leftmost/rightmost word token position in match_positions, then include
                    l, r = 0, len(match_positions)-1
                    while l <= r and not token_types[match_positions[l]]:
                        l += 1
                    while l <= r and not token_types[match_positions[r]]:
                        r -= 1
                    # only consider if we matched at least one word (should always be true)
                    if l <= r:
                        for pos in match_positions[l:r+1]:
                            annotations[pos].update(trie_node.entries)
                j += 1
            else:
                break
        i += 1

    # Compose result
    return [(token, annotations[i]) for i, token in enumerate(tokens)]
```
