import collections.abc
import re

# A unique marker to signify the end of a phrase in the trie.
# Using object() ensures it won't conflict with any actual word part.
_END_MARKER = object()


def _get_word_parts(key: str) -> list[str]:
    """
    Splits a dictionary key into a list of its word components.
    Word components are sequences of alphanumeric characters.
    All other characters are treated as separators and are not included.
    
    Example: "once in a blue-moon" -> ["once", "in", "a", "blue", "moon"]
    """
    # Using re.findall is a concise way to extract all alphanumeric sequences.
    return [part.lower() for part in re.findall(r'[a-zA-Z0-9]+', key)]


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index is a dictionary containing two main structures:
    1. 'phrases': A trie (prefix tree) for matching sequences of word tokens.
       This handles single words and multi-word phrases where words are
       separated by non-word tokens.
    2. 'compounds': A dict for matching concatenated word tokens. This handles
       cases where multiple tokens form a single dictionary entry without
       separators (e.g., tokens ["AAA", "BBB"] matching key "AAABBB").

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    phrase_trie = {}
    compound_words = {}

    for key, value in dictionary.items():
        # Build the phrase trie. This handles any key that can be broken
        # down into one or more word parts.
        word_parts = _get_word_parts(key)
        if word_parts:
            node = phrase_trie
            for part in word_parts:
                node = node.setdefault(part, {})
            node.setdefault(_END_MARKER, set()).add(value)

        # Build the compound words dictionary. A key is considered for
        # this map if it's a single alphanumeric word, which could
        # potentially be formed by concatenating multiple tokens.
        if key.isalnum():
            compound_words.setdefault(key.lower(), set()).add(value)

    return {
        'phrases': phrase_trie,
        'compounds': compound_words,
        'end_marker': _END_MARKER,
    }


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    The function performs two passes over the tokens:
    1. Phrase Matching: It identifies sequences of "word tokens" that match
       phrases in the phrase trie. It correctly handles varying separators
       and annotates all tokens within a matched phrase (words and inner
       separators).
    2. Compound Word Matching: It checks for sequences of adjacent "word tokens"
       that, when concatenated, match an entry in the compound words map.

    Parameters:
        dictionary_index: A dictionary index created by build_dictionary_index().
        tokens:           The tokens to be annotated.

    Return:
        annotated_tokens: A list containing (token, annotations) pairs for each
                          token in tokens.
    """
    # Ensure tokens is a list for indexed access
    tokens = list(tokens)
    if not tokens:
        return []

    annotated_tokens = [(token, set()) for token in tokens]
    
    # Extract data from the index object
    phrase_trie = dictionary_index['phrases']
    compound_words = dictionary_index['compounds']
    _END = dictionary_index['end_marker']

    # A "word token" is defined as a token consisting of only alphanumeric
    # characters. This list stores their index and the token itself.
    word_data = [(i, token) for i, token in enumerate(tokens) if token.isalnum()]

    # Pass 1: Match phrases and single words using the phrase trie.
    for i in range(len(word_data)):
        start_token_idx, _ = word_data[i]
        node = phrase_trie
        for j in range(i, len(word_data)):
            current_token_idx, word = word_data[j]
            normalized_word = word.lower()

            if normalized_word not in node:
                break  # This path in the trie doesn't exist.

            node = node[normalized_word]
            if _END in node:
                # A match is found. The phrase spans from start_token_idx to current_token_idx.
                # Annotate all tokens within this span.
                ids_to_add = node[_END]
                for k in range(start_token_idx, current_token_idx + 1):
                    annotated_tokens[k][1].update(ids_to_add)

    # Pass 2: Match concatenated compound words.
    for i in range(len(word_data)):
        concatenated_word = ""
        matched_token_indices = []
        for j in range(i, len(word_data)):
            current_token_idx, word = word_data[j]
            
            # Ensure tokens are adjacent in the original token list.
            if j > i:
                prev_token_idx, _ = word_data[j - 1]
                if current_token_idx != prev_token_idx + 1:
                    break  # Sequence of word tokens is broken by a separator.

            concatenated_word += word.lower()
            matched_token_indices.append(current_token_idx)
            
            if concatenated_word in compound_words:
                ids_to_add = compound_words[concatenated_word]
                for k in matched_token_indices:
                    annotated_tokens[k][1].update(ids_to_add)

    return annotated_tokens
