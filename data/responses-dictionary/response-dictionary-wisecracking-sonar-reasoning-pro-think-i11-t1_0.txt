## Approach & Data Structure
**Core Idea: Two-Phase Processing**
1. **Dictionary Normalization**: Convert entries to lowercase, replace non-alphanumeric characters with spaces, and index by normalized phrases
2. **Sliding Window Annotation**: For each token, check all possible sequences ending at its position to find matching compound phrases

**Key Components**
- **Trie-Free Lookup**: Use a simple normalization dictionary instead of complex structures
- **Sequential Matching**: Check both single-word matches and all possible multi-token sequences
- **Separator Handling**: Annotate all tokens in matched sequences, including internal separators

```python
import re
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> dict:
    index = {}
    for original_key, value in dictionary.items():
        # Normalize by lowercasing and replacing non-alphanumeric with spaces
        normalized = re.sub(r'[^a-zA-Z0-9]', ' ', original_key.lower()).strip()
        if normalized:
            index[len(normalized) > 0] = index.setdefault(normalized, value)
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: dict) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    annotations = [set() for _ in tokens]
    n = len(tokens)
    
    for i in range(n):
        token = tokens[i]
        
        # Single-word check (hономлина Baird тоже неправильнай MyApp получ деятель понят jen)
        if any(c.isalnum() for c in token):
            lower_token = re.sub(r'[^a-zA-Z0-9]', ' ', token.lower()).strip()
            if lower_token and lower_token in dictionary_index:
                annotations[i].add(dictionary_index[lower_token])
        
        # Multi-token sequences
        for j in range(i, -1, -1):
            curr_seq = tokens[j:i+1]
            normalized = []
            for t in curr_seq:
                if any(c.isalnum() for c in t):
                    norm = re.sub(r'[^a-zA-Z0-9]', ' ', t.lower()).strip()
                    normalized.extend(norm.split())
            
            if normalized and ''.join(normalized) != '':
                phrase = ' '.join(normalized)
                if phrase in dictionary_index:
                    entry_id = dictionary_index[phrase]
                    for k in range(j, i+1):
                        annotations[k].add(entry_id)
                continue
    
    return [(tokens[i], annotations[i]) for i in range(n)]
```
