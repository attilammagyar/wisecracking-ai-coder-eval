import collections.abc

class TrieNode:
    def __init__(self):
        self.children = {}
        self.entry_ids = set()
        self.is_end = False

def is_word_token(token):
    """Check if a token contains any alphanumeric characters."""
    return any(c.isalnum() for c in token)

def normalize_token(token):
    """Normalize a token for case-insensitive lookup."""
    return token.lower()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Split phrase into words, ignoring separators
        words = []
        current_word = []
        
        for char in phrase:
            if char.isalnum():
                current_word.append(char)
            elif current_word:
                words.append(''.join(current_word).lower())
                current_word = []
        
        if current_word:
            words.append(''.join(current_word).lower())
        
        # Insert words into trie
        if words:
            node = root
            for word in words:
                if word not in node.children:
                    node.children[word] = TrieNode()
                node = node.children[word]
            node.is_end = True
            node.entry_ids.add(entry_id)
    
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens_list = list(tokens)
    n = len(tokens_list)
    annotations = [set() for _ in range(n)]
    
    # For each starting position
    for start in range(n):
        # Skip if starting with a separator
        if not is_word_token(tokens_list[start]):
            continue
        
        # Try to match phrases starting from this position
        node = dictionary_index
        word_positions = []
        
        for pos in range(start, n):
            token = tokens_list[pos]
            
            if is_word_token(token):
                # Normalize the token for lookup
                normalized = normalize_token(token)
                
                # Check if this normalized token exists in trie
                if normalized in node.children:
                    node = node.children[normalized]
                    word_positions.append(pos)
                    
                    # If this completes a phrase, annotate all involved tokens
                    if node.is_end:
                        # Determine the range to annotate
                        first_word_pos = word_positions[0]
                        last_word_pos = word_positions[-1]
                        
                        # Annotate all tokens from first word to last word
                        for i in range(first_word_pos, last_word_pos + 1):

                else:

                    break
            # If it's a separator and we're in the middle of a phrase, continue
