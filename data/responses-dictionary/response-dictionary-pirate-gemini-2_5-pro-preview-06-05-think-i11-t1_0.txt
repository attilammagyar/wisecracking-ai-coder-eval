import re


class _TrieNode:
    """A node in the Trie, representin' a part of a word or phrase."""
    __slots__ = ('children', 'meanings')

        self.children: dict[str, _TrieNode] = {}
        self.meanings: set = set()


class _DictionaryIndex:
    """Our index, built upon a mighty Trie."""
    __slots__ = ('root',)

    def __init__(self):
        self.root = _TrieNode()

    def add_entry(self, key: str, meaning: object):
        """Adds a dictionary entry to the Trie index."""
        # Normalize and tokenize the key from the dictionary.
        # We find all sequences o' letters and numbers, aye.
        word_chunks = re.findall(r'\w+', key.lower())
        if not word_chunks:
            return

        node = self.root
        for chunk in word_chunks:
            if chunk not in node.children:
                node.children[chunk] = _TrieNode()
            node = node.children[chunk]
        
        node.meanings.add(meaning)
