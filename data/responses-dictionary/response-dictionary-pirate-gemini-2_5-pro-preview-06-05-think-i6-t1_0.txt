import collections.abc
import re


class _TrieNode:
    """A helper class for the dictionary index Trie."""
    def __init__(self):
        self.children = {}
        self.entries = set()


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = _TrieNode()
    for key, value in dictionary.items():
        # Normalize the key by extracting alphanumeric parts. This handles
        # various separators and cases consistently.
        word_parts = re.findall(r'[a-zA-Z0-9]+', key.lower())
        if not word_parts:
            continue
        
        node = root
        for part in word_parts:
            node = node.children.setdefault(part, _TrieNode())
        node.entries.add(value)
    return root


def annotate(
    tokens: collections.abc.Iterable[str], 
    dictionary_index: object
) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each
                            token in tokens.
    """
    token_list = list(tokens)
    n = len(token_list)
    if n == 0:
        return []

    annotated_tokens = [(token, set()) for token in token_list]

    def _search_from(node: _TrieNode, token_cursor: int, phrase_first_word_idx: int):
        """
        Recursively search for matching phrases from a given token cursor and Trie node.

        Parameters:
            node:                   The current node in the Trie to search from.
            token_cursor:           The index in token_list to start consuming tokens from.
            phrase_first_word_idx:  The index of the first word in the entire match sequence
                                    being built.
        """
        # Find the start of the next potential word-chunk, skipping any separators.
        chunk_start_idx = token_cursor
        while chunk_start_idx < n and not token_list[chunk_start_idx].isalnum():
            chunk_start_idx += 1
        
        if chunk_start_idx == n:
            return  # No more words left to form a chunk.

        # Greedily build a word chunk token by token and check against the trie.
        current_chunk = ""
        for k in range(chunk_start_idx, n):
            token = token_list[k]
            
            if token.isalnum():
                current_chunk += token.lower()
                
                # If the chunk forms a valid next part of a phrase in the Trie...
                if current_chunk in node.children:
                    child_node = node.children[current_chunk]
                    last_word_idx = k
                    
                    # If this node corresponds to a full dictionary entry, annotate it.
                    if child_node.entries:
                        # The annotation spans from the first word of the phrase
                        # to the last word of the current match, inclusive of all
                        # tokens in between (words and separators).
                        for ann_idx in range(phrase_first_word_idx, last_word_idx + 1):
                            annotated_tokens[ann_idx][1].update(child_node.entries)
                    
                    # Continue search from this new state for longer phrases.
                    _search_from(child_node, last_word_idx + 1, phrase_first_word_idx)
            
            elif current_chunk:
                # A separator marks the end of a contiguous word chunk.
                break

    # Start a search for matches from every word-like token in the input.
    for i in range(n):
        if token_list[i].isalnum():
            _search_from(dictionary_index, i, i)

    return annotated_tokens
