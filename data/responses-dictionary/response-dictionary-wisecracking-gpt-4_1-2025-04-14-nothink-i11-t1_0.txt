import re

# Helper function for normalization: lowercase everything

    return _TOKEN_SPLIT_RE.findall(s)

def is_word(token):

    return any(c.isalnum() for c in token)

class TrieNode:
    __slots__ = ('children', 'values')
    def __init__(self):

    Words and separator tokens all live in the trie.
    Lookup, matching, etc. become fast and simple.

        tokens = [normalize(t) for t in tokens]
        node = root
        for token in tokens:
            if token not in node.children:
                node.children[token] = TrieNode()
            node = node.children[token]
        node.values.add(value)
    return root
