import collections.abc

def _is_word_token(token):
    # True if token has any alphanumeric character
    return any(c.isalnum() for c in token)

def _token_normalize(token):
    return token.lower()

# Trie node class for phrase dictionary
class _TrieNode:
    __slots__ = ['children', 'entries']
    def __init__(self):
        self.children = dict()    # key: word (str, normalized) -> node
        self.entries = set()      # set of dictionary entry ids (for word/phrase ending here)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    root = _TrieNode()

    for phrase, entry in dictionary.items():
        # Split the dictionary phrase into normalized word tokens
        # The "words" are sequences of alnum chars. Split only on those.
        # E.g., "AAA, BBB" splits into ["AAA", "BBB"]
        words = []
        word = []
        for c in phrase:
            if c.isalnum():
                word.append(c)
            else:
                if word:
                    words.append(''.join(word).lower())
                    word = []
        if word:
            words.append(''.join(word).lower())
        if not words:
            continue # skip key if no valid word tokens at all

        # Now insert into trie
        node = root
        for w in words:
            if w not in node.children:
                node.children[w] = _TrieNode()
            node = node.children[w]
        node.entries.add(entry) # Mark phrase end with entry id

    # Also, for quick single-token exact-word match, build per-word dict
    singles = dict()
    for phrase, entry in dictionary.items():
        phrase_norm = phrase.strip().lower()
        # If it's a single "word" (alnum)
        if all(c.isalnum() or c.isspace() for c in phrase_norm):
            # Split on spaces
            parts = phrase_norm.split()
            if len(parts) == 1:
                singles[parts[0]] = entry

    return (root, singles)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    root, singles = dictionary_index
    tokens = list(tokens)
    N = len(tokens)
    # Pre-compute properties
    normed = [_token_normalize(tok) for tok in tokens]
    isword = [_is_word_token(tok) for tok in tokens]

    # Annotate: entry_id_sets[i] = set of dictionary entry ids for token i
    results = [set() for _ in range(N)]

    # 1. Atom word matches
    for i, (tok, isw) in enumerate(zip(normed, isword)):
        if isw and tok in singles:
            results[i].add(singles[tok])

    # 2. Phrase matches (multi-token, include atom-word case)
    # Outer loop: for each start index in tokens
    for start in range(N):
        # Start only at word tokens
        if not isword[start]:
            continue

        # Attempt phrase match starting at start
        trie_node = root
        matched_token_indices = []
        i = start
        # Skip tokens to match dictionary phrase "words" only on word tokens, separators in between can be any.
        # At each trie step, advance to next word token.
        # We'll collect list of (token index, isword) for the span.
        word_indices = []
        curr = i
        while curr < N:
            # Find next word token >= curr
            while curr < N and not isword[curr]:
                curr += 1
            if curr >= N:
                break
            # Next word token at curr
            w = normed[curr]
            if w not in trie_node.children:
                break
            trie_node = trie_node.children[w]
            word_indices.append(curr)
            # For phrase like ["AAA", "BBB"], matched at tokens indexed by word_indices, between them can be any number of separators

            # If a phrase ends here, mark span from start to curr (inclusive) and annotate as per rules.
            # We need to find the span: tokens between start and curr, i.e. everything from tokens[start:curr+1]
            # But, between start and curr, some tokens are "word", some are separators.
            if trie_node.entries:
                # The phrase spans from first word token at start to this word token at curr.
                # Find indices of all tokens between word_indices[0] and word_indices[-1], including separators.
                span_start = word_indices[0]
                span_end = word_indices[-1]
                span = range(span_start, span_end+1)
                # Within this span, annotate:
                # - All word tokens
                # - All inner separator tokens (i.e. separators not at the boundaries of span)
                #   That means: for each idx in span:
                #       - If it's a word token, always annotate
                #       - If it's a separator and not span_start or span_end, annotate
                for idx in span:
                    if isword[idx]:
                        results[idx].update(trie_node.entries)
                    else:
                        if idx != span_start and idx != span_end:
                            results[idx].update(trie_node.entries)
            # Move to next potential word in phrase
            curr += 1

    # Output result in required format
    return [(tokens[i], results[i]) for i in range(N)]
