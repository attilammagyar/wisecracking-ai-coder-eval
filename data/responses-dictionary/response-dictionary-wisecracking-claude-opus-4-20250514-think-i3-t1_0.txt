    """Extract word tokens from text, preserving order."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

        'compound_entries': []  # list of (normalized_words, entry_id, is_phrase)
    }
