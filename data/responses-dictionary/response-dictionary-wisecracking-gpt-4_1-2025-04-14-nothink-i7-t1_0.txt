import collections.abc
from typing import Dict, Set, List, Any


def is_word_token(token: str) -> bool:
    # Any alphanumeric character means word token
    return any(c.isalnum() for c in token)


def normalize_token(token: str) -> str:
    # Lowercase normalization, assumes requirement is case-insensitivity.
    return token.lower()


################################################################################
# Trie-Based Index for Fast Phrase Lookup
################################################################################

class TrieNode:
    __slots__ = ['children', 'entry_ids']
    def __init__(self):
        self.children: Dict[str, TrieNode] = dict()  # Edges: normalized word token
        self.entry_ids: Set[Any] = set()             # Set of normalized entries

def tokenize_phrase(phrase: str) -> List[str]:
    """
    Split dictionary phrase into token 'words', lowercased. 
    Assume splitting on whitespace is enough—separators like commas are part of tokens.
    """
    return phrase.strip().split()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build trie, and single-word lookup for fast annotation.
    Both keyed on lower-case tokens.
    """
    # Single word/compound lookup
    root = TrieNode()
    # Single-word direct map: word_token -> set(entry_ids)
    single_word_map: Dict[str, Set[Any]] = collections.defaultdict(set)
    # Keep track of the word count for each phrase, for separator annotation
    phrase_word_lengths: Dict[Any, int] = dict()    # entry_id -> number of word tokens

    for phrase, entry_id in dictionary.items():
        words = [normalize_token(w) for w in tokenize_phrase(phrase)]
        if not words:
            continue
        # Trie insertion
        node = root
        for w in words:
            if w not in node.children:
                node.children[w] = TrieNode()
            node = node.children[w]
        node.entry_ids.add(entry_id)
        phrase_word_lengths[entry_id] = len(words)
        # If this is a single word, note in single word map too
        if len(words) == 1:
            single_word_map[words[0]].add(entry_id)

    return {
        "trie": root,
        "single_word": single_word_map,
        "phrase_word_lengths": phrase_word_lengths
        # We don't need the original dictionary anymore.
    }

################################################################################

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate every token, building up phrases using the trie.
    """
    tokens = list(tokens)  # May need random access. Sorry about your RAM, but it's 2024.
    n = len(tokens)
    trie = dictionary_index["trie"]
    single_word = dictionary_index["single_word"]
    phrase_word_lengths = dictionary_index["phrase_word_lengths"]

    # Pre-classify tokens as word/separator for fast access
    token_is_word = [is_word_token(token) for token in tokens]
    token_norm = [normalize_token(token) for token in tokens] # Lower-case for matching
    
    # Each token will accumulate its annotations here
    annotations: List[Set[Any]] = [set() for _ in range(n)]

    # First, annotate single words (needed for both single and compound words)
    for i, (norm, is_word) in enumerate(zip(token_norm, token_is_word)):
        if is_word and norm in single_word:
            annotations[i].update(single_word[norm])

    # Now, find all matching phrases, possibly multi-word, starting at every word
    for start in range(n):

        if not token_is_word[start]:

        node = trie
        idx = start
        match_len = 0  # Number of word tokens matched in current phrase
        positions = [] # List of indices of tokens in this phrase match (includes separators)

        word_token_positions = []  # List of indices of *word* tokens matched

        while idx < n:

            node = node.children[norm]
            positions.append(idx)
            word_token_positions.append(idx)
            match_len += 1
