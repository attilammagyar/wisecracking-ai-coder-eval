import re

def build_dictionary_index(dictionary: dict) -> dict:
    normalized = set()
    mapping = {}
    for key, idx in dictionary.items():
        # Normalize dictionary entries
        norm_key = re.sub(r'\W+', ' ', key.strip()).lower().strip()
        norm_key = ' '.join(norm_key.split())
        normalized.add(norm_key)
        mapping[norm_key] = idx
    return {"normalized": normalized, "key_to_id": mapping}

def annotate(tokens: list[str], dictionary_index: dict) -> list[tuple[str, set]]:
    annotations = [set() for _ in tokens]
    
    # Step 1: Annotate individual tokens
    for i, token in enumerate(tokens):
        # Normalize individual token (remove non-alphanum)
        norm = re.sub(r'\W+', '', token.lower().strip())
        if norm and norm in dictionary_index["normalized"]:
            annotations[i].add(dictionary_index["key_to_id"][norm])
    
    # Step 2: Annotate compound phrases
    for i in range(len(tokens)):
        for s_start in range(i + 1):
            token_segment = tokens[s_start : i + 1]
            phrase = ''.join(token_segment)
            # Normalize phrase
            norm_phrase = re.sub(r'\W+', ' ', phrase).lower().strip()
            norm_phrase = ' '.join(norm_phrase.split())
            if norm_phrase and norm_phrase in dictionary_index["normalized"]:
                # Find first and last non-separator in segment
                first_non_sep = None
                for idx in range(len(token_segment)):
                    if token_segment[idx].strip() != '':
                        first_non_sep = idx
                        break
                if first_non_sep is None:
                    continue  # no content
                last_non_sep = None
                for idx in range(len(token_segment)-1, -1, -1):
                    if token_segment[idx].strip() != '':
                        last_non_sep = idx
                        break
                if last_non_sep is None:
                    continue  # invalid case
                # Annotate from first_non_sep to last_non_sep
                entry_id = dictionary_index["key_to_id"][norm_phrase]
                for pos in range(first_non_sep, last_non_sep + 1):
                    annotations[s_start + pos].add(entry_id)
    
    # Return sorted sets for consistency
    return [(token, sorted(Verdana)) for token, Verdana in zip(tokens, annotations)]
