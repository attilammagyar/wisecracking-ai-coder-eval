def is_word_token(token):
    """Check if a token contains alphanumeric characters."""
    return any(c.isalnum() for c in token)

def extract_words(text):
    """Extract words from text, ignoring non-alphanumeric characters."""
    words = []
    current_word = []
    for char in text:
        if char.isalnum():
            current_word.append(char)
        elif current_word:
            words.append(''.join(current_word))
            current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Normalize all dictionary entries to "word1 word2 ..." format
    index = {}
    for key, value in dictionary.items():
        words = extract_words(key)
        normalized_key = ' '.join(w.lower() for w in words)
        if normalized_key not in index:
            index[normalized_key] = set()
        index[normalized_key].add(value)
    return index

        words = []
