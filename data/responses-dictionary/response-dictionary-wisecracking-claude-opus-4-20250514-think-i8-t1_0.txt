def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Separate entries into phrases (with separators) and words/compound words
    phrases = {}  # normalized word tuple -> list of (original_key, value)
    words = {}    # normalized string -> list of (original_key, value)
    
    word_pattern = re.compile(r'\w+', re.UNICODE)
    
    for key, value in dictionary.items():
        words_in_key = word_pattern.findall(key)
        
        if len(words_in_key) > 1:
            # It's a phrase with multiple words
            normalized_words = tuple(w.lower() for w in words_in_key)
            if normalized_words not in phrases:
                phrases[normalized_words] = []
            phrases[normalized_words].append((key, value))
        else:
            # Single word or compound word without separators
            normalized = key.lower()
            if normalized not in words:
                words[normalized] = []
            words[normalized].append((key, value))
    
    return {'phrases': phrases, 'words': words}

        for i in range(start, len(tokens)):
