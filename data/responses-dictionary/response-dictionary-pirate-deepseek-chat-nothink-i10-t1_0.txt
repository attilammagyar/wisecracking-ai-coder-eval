```python
import collections.abc
from typing import Dict, Set, List, Tuple

def build_dictionary_index(dictionary: collections.abc.Mapping) -> Dict[str, Set]:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    for key, value in dictionary.items():
        # Normalize key to lowercase and split into tokens
        normalized_key = key.lower()
        index[normalized_key] = index.get(normalized_key, set()) | {value}
    return index

def is_word_token(token: str) -> bool:
    """Check if a token is a word (as opposed to a separator)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: Dict[str, Set]) -> List[Tuple[str, Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Convert to list for random access
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # First pass: individual word tokens
    for i, token in enumerate(tokens):
        if is_word_token(token):
            lower_token = token.lower()
            if lower_token in dictionary_index:
                annotations[i].update(dictionary_index[lower_token])
    
    # Second pass: compound phrases
    for i in range(n):
        if not is_word_token(tokens[i]):
            continue  # Only start phrases from word tokens
            
        # Build possible phrases starting at position i
        current_phrase = []
        current_phrase_lower = []
        j = i
        
        while j < n:
            token = tokens[j]
            current_phrase.append(token)
            current_phrase_lower.append(token.lower())
            
            # Skip leading separators
            if j == i and not is_word_token(token):
                break
                
            # Check if we have a matching phrase
            phrase_key = ' '.join(current_phrase_lower)
            if phrase_key in dictionary_index:
                # Annotate all tokens in the phrase (except leading/trailing separators)
                start = i if is_word_token(tokens[i]) else i + 1
                end = j if is_word_token(tokens[j]) else j - 1
                
                if start <= end:  # Only annotate if we have valid tokens
                    for k in range(start, end + 1):
                        annotations[k].update(dictionary_index[phrase_key])
            
            # Move to next token if current is a word, or break if we hit a separator after start
            if is_word_token(token) or j == i:
                j += 1
            else:
                break
    
    return list(zip(tokens, annotations))
```
