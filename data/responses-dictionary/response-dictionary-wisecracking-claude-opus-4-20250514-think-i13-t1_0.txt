import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for key, entry_id in dictionary.items():
        # Normalize to lowercase for case-insensitive matching
        normalized = key.lower()
        
        # Extract words (alphanumeric sequences)
        words = re.findall(r'\w+', normalized)
        
        if not words:
            continue
            
        # Create a normalized key based on the words
        if len(words) == 1:
            # Single word
            norm_key = ('single', words[0])
        else:
            # Check if it's a compound word (no separators) or phrase (with separators)
            concatenated = ''.join(words)
            if concatenated == re.sub(r'\W+', '', normalized):
                # Compound word - no separators in original
                norm_key = ('compound_word', concatenated)
            else:
                # Compound phrase - has separators
                norm_key = ('compound_phrase', tuple(words))
        
        if norm_key not in index:
            index[norm_key] = set()
        index[norm_key].add(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        return bool(re.search(r'\w', token))
    
    # Try all possible starting positions
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
            
        # Track words found and their positions
        words_found = []
        word_positions = []
        concatenated = ""
        
        # Extend from start position
        for pos in range(start, n):
            token = tokens[pos]
            
            if is_word_token(token):
                words_found.append(token.lower())
                word_positions.append(pos)
                concatenated += token.lower()
                
                # Check single word match (only for first word)
                if len(words_found) == 1:
                    key = ('single', words_found[0])
                    if key in dictionary_index:
                        annotations[pos].update(dictionary_index[key])
                
                # Check compound word match (consecutive words without separators)
                if len(words_found) > 1 and all(word_positions[i] + 1 == word_positions[i + 1] 
                                                for i in range(len(word_positions) - 1)):
                    key = ('compound_word', concatenated)
                    if key in dictionary_index:
                        # Annotate all tokens in the compound word
                        for idx in range(start, pos + 1):
                            annotations[idx].update(dictionary_index[key])
                
                # Check compound phrase match
                if len(words_found) > 1:
                    key = ('compound_phrase', tuple(words_found))
                    if key in dictionary_index:
                        # Annotate from first word to last word (inclusive)
                        for idx in range(word_positions[0], word_positions[-1] + 1):
                            annotations[idx].update(dictionary_index[key])
            
            else:
                # Non-word token (separator)
                # Can't start a match with a separator
                if pos == start:
                    break
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
