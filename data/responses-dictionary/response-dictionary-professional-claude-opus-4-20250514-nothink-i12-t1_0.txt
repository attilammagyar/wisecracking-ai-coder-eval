import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'word_to_entries': {},  # word -> list of (entry_id, entry_key, position_in_phrase, phrase_length)
        'normalized_to_original': {},  # normalized_key -> original_key
        'entries': {}  # original_key -> entry_id
    }
    
    for key, entry_id in dictionary.items():
        index['entries'][key] = entry_id
        
        # Normalize: lowercase and split by non-word characters
        normalized_key = key.lower()
        words = []
        current_word = []
        
        for char in normalized_key:
            if char.isalnum():
                current_word.append(char)
            elif current_word:
                words.append(''.join(current_word))
                current_word = []
        if current_word:
            words.append(''.join(current_word))
        
        # Store normalized form
        normalized_form = ' '.join(words)
        index['normalized_to_original'][normalized_form] = key
        
        # Index each word
        for pos, word in enumerate(words):
            if word not in index['word_to_entries']:
                index['word_to_entries'][word] = []
            index['word_to_entries'][word].append((entry_id, key, pos, len(words)))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []
    
    # Identify word groups (sequences of word tokens potentially separated by non-word tokens)
    word_groups = []
    i = 0
    while i < n:
        # Skip leading separators
        while i < n and not any(c.isalnum() for c in tokens[i]):
            i += 1
        
        if i < n:
            start = i
            # Collect tokens until we hit a separator that's not followed by another word
            while i < n:
                if any(c.isalnum() for c in tokens[i]):
                    i += 1
                elif i + 1 < n and any(any(c.isalnum() for c in tokens[j]) for j in range(i + 1, n)):
                    # This is an inner separator
                    i += 1
                else:
                    # This is a trailing separator or end
                    break
            word_groups.append((start, i))
    
    # Process each word group
    for group_start, group_end in word_groups:
        # Try all possible subsequences of this group
        for start in range(group_start, group_end):
            # Skip if starting with a separator
            if not any(c.isalnum() for c in tokens[start]):
                continue
                
            for end in range(start + 1, group_end + 1):
                # Skip if ending with a separator
                if end > start and not any(c.isalnum() for c in tokens[end - 1]):
                    continue
                
                # Extract words from this subsequence
                words = []
                word_positions = []  # Token positions for each word
                
                for i in range(start, end):
                    if any(c.isalnum() for c in tokens[i]):
                        words.append(tokens[i].lower())
                        word_positions.append(i)
                
                if not words:
                    continue
                
                # Check single word case
                if len(words) == 1 and len(word_positions) == 1:
                    word = words[0]
                    pos = word_positions[0]
                    if word in dictionary_index['word_to_entries']:

                    entry_id = dictionary_index['entries'][original_key]
                    
                    # Annotate all tokens in this range
                    for i in range(start, end):
