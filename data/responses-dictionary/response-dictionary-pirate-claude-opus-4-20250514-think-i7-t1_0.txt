def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for key, value in dictionary.items():
        # Normalize the key to lowercase
        normalized_key = key.lower()
        
        # Check if it's a phrase (contains spaces) or a word/compound word
        if ' ' in key:
            # It's a phrase - extract words
            words = re.findall(r'\w+', normalized_key)
            index_key = ('phrase', tuple(words))
        else:
            # It's a single word or compound word
            index_key = ('word', normalized_key)
        
        # Store the entry ID
        if index_key not in index:
            index[index_key] = []
        index[index_key].append(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

        """Check if token contains word characters."""
        return bool(re.search(r'\w', token))
    
    # Convert to list for indexing
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Try all possible spans that start and end with word tokens
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
            
        for end in range(start, n):
            if not is_word_token(tokens[end]):
                continue
            
            span = tokens[start:end+1]

                index_key = ('word', token.lower())
                if index_key in dictionary_index:
                    for entry_id in dictionary_index[index_key]:
                        annotations[start].add(entry_id)
            
            # For consecutive word tokens, check compound word match
            if all(is_word_token(t) for t in span):
                concatenated = ''.join(span).lower()
                index_key = ('word', concatenated)
                if index_key in dictionary_index:
                    for entry_id in dictionary_index[index_key]:
                        for i in range(start, end+1):
                            annotations[i].add(entry_id)
            
            # For any span, check phrase match
            words_in_span = []
            for token in span:
                if is_word_token(token):
                    words_in_span.append(token.lower())
            
            index_key = ('phrase', tuple(words_in_span))
            if index_key in dictionary_index:
                for entry_id in dictionary_index[index_key]:
                    for i in range(start, end+1):
                        annotations[i].add(entry_id)
    
    # Build result
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
    return annotated_tokens
