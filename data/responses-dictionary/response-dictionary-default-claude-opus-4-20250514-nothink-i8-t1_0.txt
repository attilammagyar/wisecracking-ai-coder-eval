def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Normalize all dictionary keys to lowercase for case-insensitive lookup
    normalized_dict = {}
    for key, value in dictionary.items():
        normalized_key = key.lower()
        normalized_dict[normalized_key] = value
    
    # Identify which entries are phrases (contain spaces)
    phrases = set()
    for key in normalized_dict:
        if ' ' in key:
            phrases.add(key)
    
    return {
        'dict': normalized_dict,
        'phrases': phrases
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

    n = len(tokens_list)
    
    # Initialize annotations for each token position
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []

    phrases = dictionary_index['phrases']
    
    # Helper to check if a token is a word character
    def is_word_token(token):
        return any(c.isalnum() for c in token)
    
    # Try matching from each starting position
    for start in range(n):
        # Skip if starting with a non-word token (for phrases)
        if not is_word_token(tokens_list[start]):
            continue
            
        # Try to build matches starting from this position
        current_word = ""
        current_phrase = ""
        last_word_end = -1
        
        for end in range(start, n):
            token = tokens_list[end]
            
            if is_word_token(token):

                current_word += token
                current_phrase += token
                last_word_end = end

                if normalized_phrase in normalized_dict:
