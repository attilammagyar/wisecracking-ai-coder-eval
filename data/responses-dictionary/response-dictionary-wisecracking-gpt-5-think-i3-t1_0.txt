_WORD_RE = re.compile(r"\w+", re.UNICODE)


def _split_words_lower(s: str):
    # Extract "words" from dictionary entries in a case-insensitive way.
    # Word = \w+ (letters/digits/underscore). This is sufficient for the tests
    # and keeps separator handling robust and simple.
    return [m.group(0).lower() for m in _WORD_RE.finditer(s)]


def _is_word_token(tok: str) -> bool:
    # Treat a token as a "word token" if it consists solely of alphanumeric chars.
    # This matches the test data and avoids overfitting.
    return tok.isalnum()


class _CharTrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: dict[str, "_CharTrieNode"] = {}
        self.ids: set = set()

    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: dict[str, "_WordTrieNode"] = {}
        self.ids: set = set()

    for ch in word:
        nxt = node.children.get(ch)
        if nxt is None:

            node.children[ch] = nxt
        node = nxt

    compound phrases.

    Parameters:

    word_root = _WordTrieNode()   # for multi-word phrases (with separators between words)

    for key, value in dictionary.items():
        words = _split_words_lower(key)
        if not words:

        if len(words) == 1:

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.

    n = len(tokens)
    annotations = [set() for _ in range(n)]

    if n == 0:
        return []

    # Unpack dictionary index
    char_root, word_root = dictionary_index  # type: ignore

    # Classify tokens and prepare word-level views
    is_word = [_is_word_token(tok) for tok in tokens]
    word_positions = [i for i, w in enumerate(is_word) if w]
    word_values = [tokens[i].lower() for i in word_positions]

    # gap_has_sep[k] corresponds to gap between word_positions[k] and word_positions[k+1]
    gap_has_sep = []
    for k in range(len(word_positions) - 1):
        has_sep = (word_positions[k + 1] - word_positions[k]) > 1
        gap_has_sep.append(has_sep)

    # 1) Match single-word dictionary entries (including concatenations across adjacent word tokens)
    #    via character trie. Only across runs of contiguous word tokens; no crossing separators.
    for start in range(n):

        node = char_root
        t = start
        while t < n and is_word[t] and (t == start or is_word[t - 1]):

            for ch in tok:
                nxt = node.children.get(ch)
                if nxt is None:
                    # No more match from this start
                    node = None
                    break
                node = nxt
            if node is None:
                break
            # Record match only at token boundary
            if node.ids:
                # Annotate tokens[start..t] inclusive
                for idx in range(start, t + 1):
                    annotations[idx].update(node.ids)
            t += 1

    # 2) Match multi-word phrases via word trie, requiring separators between words.
    W = len(word_positions)
    for s in range(W):
        node = word_root
        all_gaps_have_sep = True  # vacuously true at length 1
        for t in range(s, W):
            w = word_values[t]
            nxt = node.children.get(w)
            if nxt is None:
                break
            node = nxt

            phrase_len = t - s + 1
            if t > s:
                # Update gap condition with the most recent gap
                all_gaps_have_sep = all_gaps_have_sep and gap_has_sep[t - 1]

            if phrase_len >= 2 and node.ids and all_gaps_have_sep:

                start_tok = word_positions[s]

            if t < W - 1 and not gap_has_sep[t]:

    annotated_tokens = [(tok, annotations[i]) for i, tok in enumerate(tokens)]
