import collections.abc
import re

class TrieNode:
    def __init__(self):
        self.children = {}
        self.entry_ids = set()  # Dictionary entry IDs that end at this node

class DictionaryIndex:
    def __init__(self, dictionary):
        self.trie = TrieNode()
        self.word_pattern = re.compile(r'\w+', re.UNICODE)
        
        # Build the trie from dictionary entries
        for key, entry_id in dictionary.items():
            self._add_to_trie(key, entry_id)
    
    def _add_to_trie(self, key, entry_id):
        # Normalize the key to lowercase
        normalized_key = key.lower()
        
        # Split into words, keeping track of their positions
        words = []
        for match in self.word_pattern.finditer(normalized_key):
            words.append(match.group())
        
        if not words:
            return
        
        # Add to trie
        node = self.trie
        for word in words:
            if word not in node.children:
                node.children[word] = TrieNode()
            node = node.children[word]
        node.entry_ids.add(entry_id)
    
    def find_matches(self, tokens, start_idx):
        """Find all dictionary matches starting at the given index."""
        matches = []
        
        # Track the current position in the trie
        node = self.trie
        current_idx = start_idx
        words_in_match = []
        token_indices = []
        
        while current_idx < len(tokens):
            token = tokens[current_idx]
            normalized_token = token.lower()
            
            # Check if this token is a word
            if self.word_pattern.fullmatch(normalized_token):
                # Try to continue the match with this word
                if normalized_token in node.children:
                    node = node.children[normalized_token]
                    words_in_match.append(normalized_token)
                    token_indices.append(current_idx)
                    
                    # Check if we have a complete match at this point
                    if node.entry_ids:
                        # Calculate the range of tokens involved
                        end_idx = current_idx
                        matches.append((start_idx, end_idx + 1, node.entry_ids.copy()))
                else:
                    # No further matches possible
                    break
            
            current_idx += 1
        
        # Also check for compound words (concatenated tokens)
        if start_idx < len(tokens):
            self._find_compound_word_matches(tokens, start_idx, matches)
        
        return matches
    
    def _find_compound_word_matches(self, tokens, start_idx, matches):
        """Find matches where consecutive tokens form a compound word."""
        # Build concatenations of consecutive word tokens
        current_idx = start_idx
        concatenated = ""
        token_count = 0
        
        while current_idx < len(tokens):
            token = tokens[current_idx]
            
            # Check if this token is a word

                    if node.entry_ids:
                        matches.append((start_idx, start_idx + token_count, node.entry_ids.copy()))
            else:
                # Non-word token breaks the concatenation
                break
            
            current_idx += 1

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Pattern to identify word tokens
    word_pattern = re.compile(r'\w+', re.UNICODE)

    for start_idx in range(n):
        matches = dictionary_index.find_matches(tokens, start_idx)
        
        for match_start, match_end, entry_ids in matches:
            # Determine which tokens to annotate
            # Find the first and last word tokens in the match range
            first_word_idx = None
            last_word_idx = None

                    if first_word_idx is None:

                    last_word_idx = idx
            
            if first_word_idx is not None and last_word_idx is not None:
                # Annotate all tokens from first_word to last_word (inclusive)
                for idx in range(first_word_idx, last_word_idx + 1):

    # Create the result
    annotated_tokens = []
