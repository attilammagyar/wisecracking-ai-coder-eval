import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Create normalized index: normalized_key -> set of (original_key, value) pairs
    normalized_index = {}
    
    for key, value in dictionary.items():
        # Normalize: lowercase and replace any sequence of non-word chars with single space
        normalized_key = re.sub(r'\W+', ' ', key.lower()).strip()
        
        if normalized_key not in normalized_index:
            normalized_index[normalized_key] = []
        normalized_index[normalized_key].append((key, value))
    
    return normalized_index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Convert to list for indexing
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Try to match phrases starting at each position
    for start in range(n):
        # Skip if starting with a separator (leading separators don't count)
        if re.match(r'^\W+$', tokens[start]):
            continue
            
        # Build phrases of increasing length
        current_phrase_tokens = []
        
        for end in range(start, n):
            current_phrase_tokens.append(tokens[end])
            
            # Check if we've gone past the last word token
            # (trailing separators don't count)
            last_word_pos = end
            for i in range(end, start - 1, -1):
                if not re.match(r'^\W+$', tokens[i]):
                    last_word_pos = i
                    break
            
            # Build the phrase up to the last word token
            phrase_tokens = current_phrase_tokens[:last_word_pos - start + 1]
            
            # Normalize the phrase
            phrase = ''.join(phrase_tokens)
            normalized_phrase = re.sub(r'\W+', ' ', phrase.lower()).strip()
            
            # Check if this normalized phrase exists in the dictionary
            if normalized_phrase in dictionary_index:
                # Add annotations to all tokens in the phrase
                for entry in dictionary_index[normalized_phrase]:
                    _, value = entry
                    for i in range(start, last_word_pos + 1):
                        annotations[i].add(value)
    
    # Return annotated tokens
    annotated_tokens = [(token, ann) for token, ann in zip(tokens, annotations)]
    return annotated_tokens
