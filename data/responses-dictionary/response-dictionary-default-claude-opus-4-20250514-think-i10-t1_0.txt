def is_word_token(token):
    """Check if a token contains alphabetic characters."""
    return any(c.isalpha() for c in token)

def extract_words(text):
    """Extract word tokens from text, preserving order."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalpha():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},      # normalized_word -> set of entry_ids
        'compound_words': {},    # normalized_concatenated -> set of (entry_id, num_tokens)
        'phrases': {}            # tuple of normalized_words -> set of entry_ids
    }
    
    for entry_text, entry_id in dictionary.items():
        words = extract_words(entry_text)
        normalized_words = [word.lower() for word in words]
        
        if len(words) == 1:
            # Single word entry
            normalized = normalized_words[0]
            if normalized not in index['single_words']:

            index['single_words'][normalized].add(entry_id)

            index['compound_words'][normalized].add((entry_id, 1))
