def is_word_token(token):
    """Check if token contains only word characters."""
    return bool(token) and bool(re.match(r'^\w+$', token))

    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'words': {},      # word (lowercase) -> set of entry IDs
        'phrases': {},    # tuple of words (lowercase) -> set of entry IDs
    }
    
    for key, entry_id in dictionary.items():
        # Extract words from the key
        words = re.findall(r'\w+', key)
        if not words:
            continue
        
        # Check if it's a pure word/compound word or a phrase
        is_pure_word = re.match(r'^\w+$', key) is not None
        
        if is_pure_word:
            # Store as word entry (for exact concatenation matching)
            normalized = key.lower()
            if normalized not in index['words']:
                index['words'][normalized] = set()
            index['words'][normalized].add(entry_id)
        else:
            # Store as phrase entry (for word sequence matching)
            normalized_words = tuple(w.lower() for w in words)
            if normalized_words not in index['phrases']:
                index['phrases'][normalized_words] = set()
            index['phrases'][normalized_words].add(entry_id)
