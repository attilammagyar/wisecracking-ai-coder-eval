    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for key, value in dictionary.items():
        # Determine if this is a phrase (contains non-word chars) or word
        has_non_word = bool(re.search(r'\W', key))
        
        if has_non_word:
            # Extract words from phrase, normalize to space-separated
            words = []
            for part in re.split(r'\W+', key):
                if part:
                    words.append(part.lower())
            if words:
                normalized_key = ' '.join(words)
                if normalized_key not in index:
                    index[normalized_key] = {'is_phrase': True, 'entries': set()}
                index[normalized_key]['entries'].add(value)
        else:
            # Pure word/compound word - just lowercase it
            normalized_key = key.lower()
            if normalized_key not in index:
                index[normalized_key] = {'is_phrase': False, 'entries': set()}
            index[normalized_key]['entries'].add(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Helper to check if token is a word
    is_word = lambda t: bool(re.match(r'^\w+$', t))
    
    # Try all starting positions
    for start in range(n):
        if not is_word(tokens[start]):
            continue
            
        # Build sequences from this position
        words = []
        word_positions = []
        
        i = start
        while i < n:
            if is_word(tokens[i]):
                words.append(tokens[i].lower())
                word_positions.append(i)
                
                # Check for word/compound match
                compound = ''.join(words)
                if compound in dictionary_index and not dictionary_index[compound]['is_phrase']:
                    for entry in dictionary_index[compound]['entries']:
                        for pos in word_positions:
                            annotations[pos].add(entry)
                
                # Check for phrase match (need 2+ words)
                if len(words) > 1:
                    phrase = ' '.join(words)
                    if phrase in dictionary_index and dictionary_index[phrase]['is_phrase']:
                        # Annotate all tokens from start to current (including separators)
                        for entry in dictionary_index[phrase]['entries']:
                            for pos in range(start, i + 1):
                                annotations[pos].add(entry)
                
                i += 1
            else:
                # Skip separators, find next word
                j = i
                while j < n and not is_word(tokens[j]):
                    j += 1
                
                if j < n:  # More words exist
                    i = j
                else:  # No more words
                    break
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
