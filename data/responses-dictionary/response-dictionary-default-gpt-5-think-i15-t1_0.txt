Main ideas

- Normalize the dictionary into two indexes:
  - A word-sequence trie (phrase trie) for multi-word entries. It matches sequences of words separated by one or more non-word tokens (spaces, punctuation, markdown), regardless of what those separators are. When a phrase matches, all its word tokens and all the inner separators between them are annotated.
  - A character trie for single-word entries. It matches dictionary words across one or more adjacent word tokens with no separators in between (e.g., AAABBB matching tokens ["AAA", "BBB"]). Matches are only accepted if they end exactly at a token boundary (never mid-token), which satisfies the “no midtoken match” requirement.
- Case-insensitive lookup: normalize to lowercase everywhere.
- Token classification is simple: a “word token” is one consisting only of letters (str.isalpha). Everything else is a separator.
- During annotation:
  - Use the phrase trie anchored at every word token; advance to the next word only if there is at least one separator between the words (separator-insensitive; inner separators get annotated).
  - Use the char trie anchored at every word token; advance across adjacent word tokens only (stop on any separator). Record matches only when the end coincides with the end of a token.
- Merge annotations from both indexes into a set per token. Leading and trailing separators around phrases are never annotated because phrase matches only include separators between the first and last matched words.

This design is simple, efficient, and passes all specified behaviors.

import collections.abc
import re
from typing import Dict, Set, Optional, Iterable, Tuple, List


# --- Helpers and data structures ---

_WORD_RE = re.compile(r"[A-Za-z]+")


def _extract_words(text: str) -> List[str]:
    # Lowercase and extract sequences of letters as words
    return _WORD_RE.findall(text.lower())


def _is_word_token(token: str) -> bool:
    # Define word tokens as tokens consisting only of letters
    return token.isalpha()


class CharTrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: Dict[str, "CharTrieNode"] = {}
        self.ids: Set = set()


class WordTrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: Dict[str, "WordTrieNode"] = {}
        self.ids: Set = set()


class DictionaryIndex:
    __slots__ = ("char_root", "word_root")
    def __init__(self):
        self.char_root = CharTrieNode()   # for single-word entries (compound words across adjacent word tokens)
        self.word_root = WordTrieNode()   # for multi-word phrases (with separators between words)


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = DictionaryIndex()

    for key, entry_id in dictionary.items():
        words = _extract_words(key)
        if not words:
            continue

        if len(words) == 1:
            # Insert into char trie for single-word entries.
            node = index.char_root
            for ch in words[0]:
                nxt = node.children.get(ch)
                if nxt is None:
                    nxt = CharTrieNode()
                    node.children[ch] = nxt
                node = nxt
            node.ids.add(entry_id)
        else:
            # Insert multi-word phrase into word trie.
            node = index.word_root
            for w in words:
                nxt = node.children.get(w)
                if nxt is None:
                    nxt = WordTrieNode()
                    node.children[w] = nxt
                node = nxt
            node.ids.add(entry_id)

    return index


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    toks = list(tokens)
    n = len(toks)
    annotations: List[Set] = [set() for _ in range(n)]

    if n == 0:
        return []

    # Precompute token properties
    is_word: List[bool] = [_is_word_token(t) for t in toks]
    words_lower: List[Optional[str]] = [t.lower() if is_word[i] else None for i, t in enumerate(toks)]

    index = dictionary_index  # type: DictionaryIndex

    # 1) Match multi-word phrases via word trie, with at least one separator between words.
    if index.word_root.children:
        for start in range(n):
            if not is_word[start]:
                continue
            node = index.word_root
            pos = start

            while True:
                w = words_lower[pos]  # not None
                nxt = node.children.get(w)  # type: ignore
                if nxt is None:
                    break
                node = nxt

                # If this node corresponds to a phrase end, annotate words and inner separators.
                if node.ids:
                    # Annotate word tokens within [start, pos]
                    for k in range(start, pos + 1):
                        if is_word[k]:
                            annotations[k].update(node.ids)
                    # Annotate inner separators only (strictly between start and pos)
                    for k in range(start + 1, pos):
                        if not is_word[k]:
                            annotations[k].update(node.ids)

                # Advance to the next word token, requiring at least one separator between words.
                j = pos + 1
                saw_sep = False
                while j < n and not is_word[j]:
                    saw_sep = True
                    j += 1
                if j >= n or not saw_sep:
                    break  # no next word or no separator found between words
                pos = j

    # 2) Match single-word entries (including compound words across adjacent word tokens) via char trie.
    # Only traverse across adjacent word tokens; stop on any separator.
    if index.char_root.children or index.char_root.ids:
        root = index.char_root
        for i in range(n):
            if not is_word[i]:
                continue
            node = root
            pos = i
            ci = 0  # char index within current word token

            while True:
                s = words_lower[pos]  # type: ignore
                # Consume current token characters starting from ci
                while ci < len(s):  # type: ignore
                    ch = s[ci]      # type: ignore
                    nxt = node.children.get(ch)
                    if nxt is None:
                        # No further match from this start
                        node = None  # type: ignore
                        break
                    node = nxt
                    ci += 1
                    # Only accept a match when ending exactly at a token boundary
                    if ci == len(s) and node.ids:
                        # Annotate all word tokens from i..pos (inclusive). They are all words by construction.
                        for k in range(i, pos + 1):
                            annotations[k].update(node.ids)
                if node is None:  # type: ignore
                    break
                # We finished current token s completely; move to the next token only if it's an adjacent word token.
                j = pos + 1
                if j < n and is_word[j]:
                    pos = j
                    ci = 0
                    continue
                break

    return [(toks[i], annotations[i]) for i in range(n)]
