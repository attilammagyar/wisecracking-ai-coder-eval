import collections.abc

def is_word_token(token):
    """Check if a token contains at least one alphabetic character."""
    return any(c.isalpha() for c in token)

def normalize_text(text):
    """Normalize text for case-insensitive lookup."""
    return text.lower()

def extract_words(text):
    """Extract words from text, ignoring non-alphabetic characters."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalpha():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # For words and compound words (entries with single word after extraction)
    word_lookup = {}  # normalized_text -> set of entries
    
    # For compound phrases (entries with multiple words)
    phrase_lookup = {}  # tuple of normalized words -> set of entries
    
    for key, entry in dictionary.items():
        words = extract_words(key)
        
        if len(words) == 1:
            # Single word entry (could be compound like "AAABBB")
            normalized = normalize_text(key)
            if normalized not in word_lookup:
                word_lookup[normalized] = set()
            word_lookup[normalized].add(entry)
        else:
            # Multi-word phrase
            normalized_words = tuple(normalize_text(w) for w in words)
            if normalized_words not in phrase_lookup:
                phrase_lookup[normalized_words] = set()
            phrase_lookup[normalized_words].add(entry)
    
    return (word_lookup, phrase_lookup)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    word_lookup, phrase_lookup = dictionary_index
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Process each starting position
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
        
        # Check single token
        normalized = normalize_text(tokens[start])
        if normalized in word_lookup:
            annotations[start].update(word_lookup[normalized])
        
        # Check compound words (consecutive word tokens)
        compound = tokens[start]
        for end in range(start + 1, n):
            if not is_word_token(tokens[end]):
                break
            compound += tokens[end]
            normalized = normalize_text(compound)
            if normalized in word_lookup:
                for i in range(start, end + 1):
                    annotations[i].update(word_lookup[normalized])
        
        # Check compound phrases
        words = []
        positions = []  # (token_index, is_word) pairs
        
        i = start
        while i < n:
            if is_word_token(tokens[i]):
                words.append(normalize_text(tokens[i]))
                positions.append((i, True))
                i += 1
                
                # Check if current word sequence matches a phrase
                if len(words) >= 2:
                    word_tuple = tuple(words)
                    if word_tuple in phrase_lookup:
                        # Mark all positions in the phrase
                        # Find the last word position
                        last_word_pos = positions[-1][0]
                        
                        for pos, is_word in positions:
                            if is_word:
                                # Always annotate words
                                annotations[pos].update(phrase_lookup[word_tuple])
                            else:
                                # Only annotate separators that are not trailing
                                if pos < last_word_pos:
                                    annotations[pos].update(phrase_lookup[word_tuple])
            else:
                # Separator
                if not words:
                    break
                positions.append((i, False))
                i += 1
    
    annotated_tokens = [(token, annotations[i]) for i, token in enumerate(tokens)]
    return annotated_tokens
