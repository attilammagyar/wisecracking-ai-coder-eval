```python
                        end_token_idx = logical_tokens[k][3]
                        
                        for token_idx in range(start_token_idx, end_token_idx + 1):
                            annotated_tokens[token_idx][1].update(match_values)
                else:
                    # Path in Trie is broken; no more matches from start `i`.
                    break
            # If the logical token is a separator group, we simply "consume" it
            # and continue, allowing for flexible phrase matching.
            
    return annotated_tokens
```
