import collections.abc
from collections import defaultdict

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # Build case-insensitive word index
    word_index = defaultdict(set)
    # Build trie for compound phrases
    trie = {}
    
    for phrase, meaning in dictionary.items():
        # Add to word index if single word
        normalized_phrase = phrase.lower()
        parts = normalized_phrase.split()
        if len(parts) == 1:
            word_index[parts[0]].add(meaning)
        else:
            # Add to trie for compound phrases
            current = trie
            for part in parts:
                if part not in current:
                    current[part] = {}
                current = current[part]
            # Mark end of phrase with its meaning
            current['__meaning__'] = current.get('__meaning__', set()) | {meaning}
    
    return {
        'word_index': word_index,
        'phrase_trie': trie,
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)  # Need random access
    annotated_tokens = []
    active_phrases = []
    
    for i, token in enumerate(tokens):
        current_annotations = set()
        normalized_token = token.lower()
        
        # Check for individual word match
        if normalized_token in dictionary_index['word_index']:
            current_annotations.update(dictionary_index['word_index'][normalized_token])
        
        # Check for new phrases starting at this position
        if normalized_token in dictionary_index['phrase_trie']:
            active_phrases.append({
                'trie_path': dictionary_index['phrase_trie'][normalized_token],
                'start_pos': i,
                'current_pos': i,
                'meaning': dictionary_index['phrase_trie'][normalized_token].get('__meaning__', set()),
            })
        
        # Update active phrases
        new_active_phrases = []
        for phrase in active_phrases:
            # Check if current token continues the phrase
            if (phrase['current_pos'] == i - 1 and 
                normalized_token in phrase['trie_path']):
                # Continue this phrase
                new_trie_path = phrase['trie_path'][normalized_token]
                new_meaning = new_trie_path.get('__meaning__', set())
                new_active_phrases.append({
                    'trie_path': new_trie_path,
                    'start_pos': phrase['start_pos'],
                    'current_pos': i,
                    'meaning': phrase['meaning'] | new_meaning,
                })
            elif phrase['current_pos'] >= i - 1:
                # Still valid (might be first token of new phrase)
                new_active_phrases.append(phrase)
        
        active_phrases = new_active_phrases
        
        # Add meanings from phrases that include this token
        for phrase in active_phrases:
            if phrase['start_pos'] <= i <= phrase['current_pos']:
                current_annotations.update(phrase['meaning'])
        
        annotated_tokens.append((token, current_annotations))
    
    return annotated_tokens
