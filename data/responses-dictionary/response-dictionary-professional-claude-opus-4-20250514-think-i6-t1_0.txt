                word_parts = []
                first_word_token = current_pos
                
                # Concatenate consecutive word tokens
                while current_pos < n and is_word_token(tokens[current_pos]):
                    word_parts.append(tokens[current_pos])
                    current_pos += 1
                
                # Add the complete word to our sequence
                word = normalize_word(''.join(word_parts))
                accumulated_words.append(word)
                last_word_token = current_pos - 1
                
                # Check if current word sequence matches any dictionary entry
                word_tuple = tuple(accumulated_words)
                if word_tuple in dictionary_index:
                    # Determine annotation span
                    if len(accumulated_words) == 1:
                        # Single word: annotate only its tokens
                        annotation_start = first_word_token
                        annotation_end = last_word_token
                    else:
                        # Phrase: annotate from start_pos to last_word_token
                        annotation_start = start_pos
                        annotation_end = last_word_token
                    
                    # Add annotations
                    for value in dictionary_index[word_tuple]:
                        for i in range(annotation_start, annotation_end + 1):
                            annotations[i].add(value)
            else:
                # Separator token
                if not accumulated_words:
                    # Can't start with a separator
                    break
                current_pos += 1
    
    # Build result
    annotated_tokens = []
    for i in range(n):
        annotated_tokens.append((tokens[i], annotations[i]))
    
    return annotated_tokens
