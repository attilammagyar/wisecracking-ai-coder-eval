import collections.abc
import re

def is_word_token(token):
    """Check if a token contains alphabetic characters."""
    return bool(re.search(r'[a-zA-Z]', token))

def extract_words(text):
    """Extract word tokens from text."""
    tokens = re.findall(r'\w+|\W+', text)
    return [t for t in tokens if is_word_token(t)]

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Index for exact matching (normalized)
    exact_index = {}
    
    # Index for phrase matching (by word sequence)
    phrase_index = {}
    
    for key, value in dictionary.items():
        # Add to exact index
        normalized_key = key.lower()
        if normalized_key not in exact_index:
            exact_index[normalized_key] = set()
        exact_index[normalized_key].add(value)
        
        # If it's a compound phrase, add to phrase index
        words = extract_words(key)
        if len(words) > 1:
            word_tuple = tuple(w.lower() for w in words)
            if word_tuple not in phrase_index:
                phrase_index[word_tuple] = set()
            phrase_index[word_tuple].add(value)
    
    return {
        'exact': exact_index,
        'phrases': phrase_index
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try all possible token subsequences
    for start in range(n):
        for end in range(start + 1, n + 1):
            subsequence = tokens[start:end]
            
            # Method 1: Exact match (concatenate and lookup)
            concatenated = ''.join(subsequence).lower()
