import re
import collections.abc

def is_word_token(token):
    """Check if a token consists entirely of word characters."""
    return bool(token) and bool(re.match(r'^\w+$', token))

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = []
    
    for key, value in dictionary.items():
        # Extract words from the key (normalized to lowercase)
        words = re.findall(r'\w+', key.lower())
        if not words:
            continue
        
        # Check if the key contains non-word characters (separators)
        has_separators = bool(re.search(r'\W', key))
        
        index.append({
            'words': tuple(words),
            'has_separators': has_separators,
            'entry_id': value
        })
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try all possible ranges of tokens
    for start in range(n):
        for end in range(start + 1, n + 1):
            # For phrases: extract words from token range
            word_tokens = []
            word_positions = []
            
            for i in range(start, end):
                if is_word_token(tokens[i]):
                    word_tokens.append(tokens[i].lower())
                    word_positions.append(i)
            
            # For compound words: concatenate all tokens in range
            concatenated = ''.join(tokens[start:end]).lower()
            
            # Check each dictionary entry
            for entry in dictionary_index:
                if entry['has_separators']:
                    # Phrase matching - compare word sequences
                    if tuple(word_tokens) == entry['words']:
                        # Annotate from first word to last word (inclusive)
                        if word_positions:
                            for i in range(word_positions[0], word_positions[-1] + 1):
                                annotations[i].add(entry['entry_id'])
                else:
                    # No separators - single word or compound word
                    # Check if concatenation matches
                    entry_concat = ''.join(entry['words'])
                    if concatenated == entry_concat:
                        # For compound words, verify no separators in range
                        has_separator = any(not is_word_token(tokens[i]) for i in range(start, end))
                        
                        if not has_separator:
                            # Annotate all tokens in range
                            for i in range(start, end):
                                annotations[i].add(entry['entry_id'])
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
