def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:

    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    # and another for phrases with separators (like 'shiver me timbers').

    for key, value in dictionary.items():

        key_lower = key.lower()

        # Find all the word-like treasures in the key.
        words = re.findall(r'[a-zA-Z0-9]+', key_lower)
        if not words:
            continue

        # A key is a phrase if it has separators between words.
        # We check for anythin' that ain't a letter or number.
        is_phrase = any(not char.isalnum() for char in key)

        if is_phrase:

            normalized_key = " ".join(words)
            if normalized_key:

        else:

            normalized_key = "".join(words)

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

    word_index, phrase_index = dictionary_index
    
    # We need to be able to look at any token, so we make a list o' them.
    token_list = list(tokens)
    num_tokens = len(token_list)

    annotated_tokens = [(token, set()) for token in token_list]
