class TrieNode:
    __slots__ = ('children', 'entry_ids')
    
    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.entry_ids: Set[int] = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a case-insensitive trie from the dictionary for efficient prefix matching.
    """
    root = TrieNode()
    
    for key, entry_id in dictionary.items():
        current = root
        normalized_key = key.lower()
        
        # Split into word tokens (case-insensitive) while preserving original case
        # This handles compound phrases with various separators
        word_tokens = re.findall(r'\w+|\W+', key)
        word_tokens_lower = [t.lower() for t in word_tokens]
        
        # Build the trie path for this phrase
        for token in word_tokens_lower:
            if token not in current.children:
                current.children[token] = TrieNode()
            current = current.children[token]
        
        # Store the entry ID at the end node
        current.entry_ids.add(entry_id)
    
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:

    Annotate tokens by finding all matching dictionary entries (individual words and phrases).
    """
    if not isinstance(dictionary_index, TrieNode):
        raise TypeError("Invalid dictionary index")
    
    tokens = list(tokens)  # Need random access
    annotated_tokens = []

    token_annotations = [set() for _ in range(len(tokens))]
    
    # First pass: find all individual word matches
    for i, token in enumerate(tokens):
        if not is_word_token(token):
            continue
            
        normalized_token = token.lower()
        current = dictionary_index
        
        if normalized_token in current.children:
            node = current.children[normalized_token]
            token_annotations[i].update(node.entry_ids)

    for i in range(len(tokens)):
        if not is_word_token(tokens[i]):

            token = tokens[j]
            normalized_token = token.lower()

                if normalized_token not in current.children:

            # If we have a complete phrase match, annotate all participating tokens

    for token, annotations in zip(tokens, token_annotations):
