import collections.abc
from collections import defaultdict
from typing import Dict, List, Set, Tuple

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    
    def __init__(self):
        self.children = defaultdict(TrieNode)
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie-based index from a dictionary for fast lookup of words and compound phrases.
    The trie is structured to handle sequences of words with arbitrary separators between them.
    """
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize case and split into components
        components = []
        current_word = []
        
        for c in phrase.lower():
            if c.isalpha():
                current_word.append(c)
            else:
                if current_word:
                    components.append(''.join(current_word))
                    current_word = []
                components.append(c)
        if current_word:
            components.append(''.join(current_word))
        
        # Build the trie path
        node = trie
        for comp in components:
            node = node.children[comp]
        node.entry_ids.add(entry_id)
    
    return trie

def is_word_token(token: str) -> bool:
    """Check if a token is a word (contains at least one letter)."""
    return any(c.isalpha() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary by:
    1. Checking individual word matches
    2. Finding all compound phrases starting at each position
    """
    if not dictionary_index or not isinstance(dictionary_index, TrieNode):
        return ((token, set()) for token in tokens)
    
    tokens = list(tokens)  # Need random access for lookahead
    annotated_tokens = []
    n = len(tokens)
    
    for i in range(n):
        current_annotations = set()
        token = tokens[i]
        
        # Check for individual word match (case-insensitive)
        if is_word_token(token):
            lower_word = token.lower()
            if lower_word in dictionary_index.children:
                current_annotations.update(dictionary_index.children[lower_word].entry_ids)
        
        # Check for compound phrases starting at this position
        active_matches = []
        if is_word_token(token):
            # Initialize with possible first word matches
            lower_word = token.lower()
            if lower_word in dictionary_index.children:
                node = dictionary_index.children[lower_word]
                active_matches.append((i, node))
        
        # Process active matches (compound phrases that might continue)
        new_active_matches = []
        for start_idx, node in active_matches:
            # If we're at the end of a phrase, add its annotations
            if node.entry_ids:
                # Annotate all tokens in this phrase
                for j in range(start_idx, i + 1):
                    if j >= len(annotated_tokens):
                        break
                    annotated_tokens[j][1].update(node.entry_ids)
            
            # Continue matching if possible
            if i + 1 < n:
                next_token = tokens[i + 1]
                # For compound phrases, we match either:
                # 1. The exact separator token, or
                # 2. Any separator when looking for the next word
                if next_token in node.children:
                    new_active_matches.append((start_idx, node.children[next_token]))
                elif is_word_token(next_token):
                    # Skip separators between words
                    for sep, child_node in node.children.items():
                        if not is_word_token(sep):
                            new_active_matches.append((start_idx, child_node))
        
        # For next iteration
        active_matches = new_active_matches
        
        # Also check for new phrases starting at next word
        if i + 1 < n and is_word_token(tokens[i + 1]):
            next_word = tokens[i + 1].lower()
            if next_word in dictionary_index.children:
                active_matches.append((i + 1, dictionary_index.children[next_word]))
        
        annotated_tokens.append((token, current_annotations))
    
    return annotated_tokens
