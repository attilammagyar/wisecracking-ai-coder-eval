#### 2. The Annotation Algorithm

The `annotate` function iterates through the input tokens to find all possible matches against the dictionary index. For each token in the input stream, it treats it as a potential starting point for a match and performs two types of lookups:

1.  **Concatenated Word Lookup**: The algorithm greedily consumes a sequence of *word* tokens (ignoring separators), concatenates their normalized forms, and checks for a match in the `word_lookup` dictionary at each step. import collections.abc
import re

# Helper classes for the dictionary index structure.
# A leading underscore indicates they are internal implementation details.

class _TrieNode:
    """A node in a Trie data structure for storing phrase parts."""
    def __init__(self):
        # Maps a normalized word part to the next node in the Trie.
        self.children: dict[str, _TrieNode] = {}
        # Stores dictionary entry IDs for phrases that end at this node.
        self.entry_ids: set = set()

class _DictionaryIndex:
    """
    A specialized index for efficient lookups of words and phrases.
    It contains two main components:
    - A dictionary for direct lookup of single words and concatenated compound words.
    - A Trie for sequential lookup of multi-word phrases.
    """
    def __init__(self):
        self.phrase_trie_root = _TrieNode()
        self.word_lookup: dict[str, set] = {}


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        An opaque index object to be used by the `annotate` function.
    """
    index = _DictionaryIndex()
    
    for key, entry_id in dictionary.items():
        # A key is treated as a "phrase" if it contains any non-alphanumeric
        # characters (separators like space, comma, etc.). Otherwise, it's
        # considered a single word or a concatenated compound word.
        if re.search(r'[^a-zA-Z0-9]', key):
            # This is a phrase (e.g., "black swan", "AAA, BBB").
            # We normalize it by extracting all alphanumeric parts and storing
            # them as a sequence in the Trie.
            parts = re.findall(r'[a-zA-Z0-9]+', key.lower())
            if not parts:
                continue
            
            node = index.phrase_trie_root
            for part in parts:
                # Traverse the Trie, creating nodes as needed.
                node = node.children.setdefault(part, _TrieNode())
            node.entry_ids.add(entry_id)
        else:
            # This is a single word ("swan") or a concatenated compound ("AAABBB").
            # We store it, lowercased, in the direct word lookup dictionary.
            normalized_word = key.lower()
            if normalized_word:
                index.word_lookup.setdefault(normalized_word, set()).add(entry_id)
            
    return index


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    # Ensure we have the correct index type passed from build_dictionary_index.
    if not isinstance(dictionary_index, _DictionaryIndex):
        # This indicates a misuse of the API. Return an empty list for safety.
        return []

    # Materialize the iterable of tokens into a list for indexed access.
    tokens_list = list(tokens)
    num_tokens = len(tokens_list)
    annotated_tokens = [(token, set()) for token in tokens_list]

    def is_word_token(token: str) -> bool:
        """A token is considered a 'word' if it contains any alphanumeric characters."""
        return any(c.isalnum() for c in token)

    def normalize_word_part(token: str) -> str:
        """Extracts and lowercases alphanumeric characters from a token."""
        return "".join(filter(str.isalnum, token.lower()))

    # Iterate through each token, treating it as a potential start of a match.
    for i in range(num_tokens):
        # --- 1. Concatenated Word Matches ---
        # Greedily build a string by concatenating subsequent word tokens
        # and look it up in the word_lookup dictionary.
        # This search stops at the first separator.
        current_concat_str = ""
        concat_indices: list[int] = []
        for j in range(i, num_tokens):
            token = tokens_list[j]
            if is_word_token(token):
                current_concat_str += normalize_word_part(token)
                concat_indices.append(j)
                if current_concat_str in dictionary_index.word_lookup:
                    entry_ids = dictionary_index.word_lookup[current_concat_str]
                    for token_idx in concat_indices:
                        annotated_tokens[token_idx][1].update(entry_ids)
            else:
                # Separators break concatenated word matches.
                break

        # --- 2. Phrase Matches ---
        # A phrase match must start with a word token.
        if not is_word_token(tokens_list[i]):
            continue
            
        # Traverse the phrase Trie, consuming tokens.
        node = dictionary_index.phrase_trie_root
        phrase_word_indices: list[int] = []
        phrase_separator_indices: list[int] = []
        
        for j in range(i, num_tokens):
            token = tokens_list[j]
            if is_word_token(token):
                normalized_part = normalize_word_part(token)
                if normalized_part not in node.children:
                    # This word does not continue any phrase from this point.
                    break
                
                # Advance in the Trie.
                node = node.children[normalized_part]
                phrase_word_indices.append(j)

                # If this node marks the end of a valid phrase, apply annotations.
                if node.entry_ids:
                    # Annotate all word tokens in the matched phrase.
                    for token_idx in phrase_word_indices:
                        annotated_tokens[token_idx][1].update(node.entry_ids)
                    # Annotate all inner separators within the phrase.
                    for token_idx in phrase_separator_indices:
                        annotated_tokens[token_idx][1].update(node.entry_ids)
            else:
                # This is a separator token.
                # It's only considered part of the phrase if it's not a leading one.
                if phrase_word_indices:
                    phrase_separator_indices.append(j)
    
    return annotated_tokens
