```python
import re
from typing import Iterable, Set, tuple

def build_dictionary_index(dictionary: dict) -> dict:
    """Build a trie-based index from the dictionary for efficient lookups."""
    trie_root = {'children': {}, 'entries': set()}
    
    for entry, entry_id in dictionary.items():
        word_sequence = re.findall(r'\w+', entry.lower())
        current_node = trie_root
        
        for word in word_sequence:
            if word not in current_node['children']:
                current_node['children'][word] = {'children': {}, 'entries': set()}
            current_node = current_node['children'][word]
        
        current_node['entries'].add(entry_id)
    
    return trie_root

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[tuple[str, Set]]:
    """Annotate tokens with dictionary entries, including compound phrases and individual words."""
    annotations = [set() for _ in tokens]
    
    for i in range(len(tokens)):
        current_node = dictionary_index
        entry_list = []  # Stores tuples of (end_position, entry_ids)
        j = i
        
        while j < len(tokens):
            token = tokens[j]
            if token.isalpha():  # Consider token as part of a word if alphabetic
                lower_token = token.lower()
                if lower_token in current_node['children']:
                    current_node = current_node['children'][lower_token]
                    if current_node['entries']:
                        entry_list.append((j, current_node['entries'].copy()))
                else:
                    break  # Break if invalid sequence
            j += 1  # Track j to include non-alphabetic tokens in sequence
            
        # Apply annotations from all valid sequences
        for end_pos, entries in entry_list:
            for k in range(i, end_pos + 1):
                annotations[k].update(entries)
    
    return [(token, annotations[idx]) for idx, token in enumerate(tokens)]
```

### Explanation

1. **Trie Construction**:
   - **Normalization**: Each dictionary entry is split into word segments using regex (`\w+` matches letters, digits, and underscores) and converted to lowercase.
   - **Path Building**: Words form paths in a trie structure. End nodes store dictionary entry IDs with `*'entries'` sets.

2. **Annotation Process**:
   - **Traversal**: For each token, the algorithm traverses the trie, collecting entries from all valid end nodes encountered.
   - **Dynamic Annotation**: Entries are applied to all tokens in the sequence from the starting index to the end of the matching phrase.

This approach ensures efficient lookups and correct handling of overlapping entries, as demonstrated by the unit tests.
