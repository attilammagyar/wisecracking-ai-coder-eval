Main ideas

- Normalize dictionary keys to lowercase and split them into “words” using the regex \w+ (letters/digits/underscore). This makes lookups case-insensitive and robust to different separators in the dictionary (spaces, commas, etc.).

- Two kinds of matches must be supported:
  1) Single-word entries (e.g., "aaabbb"). They may span multiple consecutive word tokens in the text with no separators between them (fine-grained tokenization), but matches must end on token boundaries (no mid-token matches).
  2) Multi-word entries (e.g., "aaa bbb ccc"). They must match words separated by at least one separator token in the text. The actual separator characters are ignored; any non-word tokens between words are accepted. Inner separators in the matched span are annotated; leading and trailing separators are not.

- Data structures:
  - A character trie for single-word entries. We feed characters from consecutive word tokens (with no intervening separators). We only record a hit after finishing whole tokens, which guarantees no mid-token matches.
  - A word trie for multi-word entries, keyed by normalized word strings. Matching walks over “word runs” (consecutive word tokens with no separators) as single units, and requires a separator block between consecutive words in a phrase. Inner separator blocks are annotated.

- Annotation algorithm:
  - Classify tokens into word tokens (fullmatch r'\w+') vs separators.
  - Build segments: alternating word runs (each run is the lowercase concatenation of its tokens) and separator blocks (one segment per consecutive group of separators).
  - Single-word matching: For each word token position, scan forward across consecutive word tokens through the character trie; whenever a terminal node is reached at a token boundary, annotate all tokens in that covered span.
  - Multi-word matching: For each word-run segment, walk the word trie through subsequent word-run segments separated by one separator block each. Record matches for every terminal node reached; annotate all word-run tokens in the span and all inner separator blocks between them.

import collections.abc
import re
from typing import Dict, Tuple, Iterable, Set, List, Any


# Internal helpers and trie structures

_WORD_RE = re.compile(r'\w+')
_WORD_FULL_RE = re.compile(r'^\w+$')


def _is_word_token(tok: str) -> bool:
    # Word tokens are made only of \w (letters/digits/_)
    return _WORD_FULL_RE.fullmatch(tok) is not None


def _split_words_lower(s: str) -> List[str]:
    # Extract lowercase word pieces from a string
    return _WORD_RE.findall(s.lower())


def _new_trie_node() -> Dict[str, Any]:
    # Generic trie node: 'n' -> children dict, 'v' -> tuple of ids (terminal values)
    return {'n': {}, 'v': ()}


def _trie_insert_chars(root: Dict[str, Any], word: str, value: Any) -> None:
    node = root
    for ch in word:
        nxt = node['n'].get(ch)
        if nxt is None:
            nxt = _new_trie_node()
            node['n'][ch] = nxt
        node = nxt
    # Store values as tuple; keep duplicates out by set conversion if needed
    if value not in node['v']:
        node['v'] = node['v'] + (value,)


def _trie_insert_words(root: Dict[str, Any], words: List[str], value: Any) -> None:
    node = root
    for w in words:
        nxt = node['n'].get(w)
        if nxt is None:
            nxt = _new_trie_node()
            node['n'][w] = nxt
        node = nxt
    if value not in node['v']:
        node['v'] = node['v'] + (value,)


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    char_trie = _new_trie_node()   # for single-word entries (character-level)
    word_trie = _new_trie_node()   # for multi-word entries (word-level)

    for key, val in dictionary.items():
        words = _split_words_lower(str(key))
        if not words:
            continue
        if len(words) == 1:
            _trie_insert_chars(char_trie, words[0], val)
        else:
            _trie_insert_words(word_trie, words, val)

    return {
        'char_trie': char_trie,
        'word_trie': word_trie,
    }


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    lower_tokens = [t.lower() for t in tokens]
    is_word = [_is_word_token(t) for t in tokens]
    annotations: List[Set] = [set() for _ in range(n)]

    char_trie = dictionary_index['char_trie']
    word_trie = dictionary_index['word_trie']

    # 1) Single-word matching via character trie across consecutive word tokens (no separators)
    i = 0
    while i < n:
        if not is_word[i]:
            i += 1
            continue
        node = char_trie
        j = i
        # Walk forward across consecutive word tokens
        while j < n and is_word[j]:
            token_str = lower_tokens[j]
            # Feed characters of this whole token; break if mismatch
            k = 0
            children = node['n']
            while k < len(token_str):
                nxt = children.get(token_str[k])
                if nxt is None:
                    # No further match possible starting at i
                    k = -1  # signal mismatch
                    break
                node = nxt
                children = node['n']
                k += 1
            if k == -1:
                break  # mismatch inside token j; stop extending from i
            # If current node is terminal, annotate tokens i..j
            if node['v']:
                for val in node['v']:
                    for tpos in range(i, j + 1):
                        annotations[tpos].add(val)
            j += 1
        i += 1

    # 2) Multi-word matching via word trie over word-run segments with separator blocks between
    # Build segments: alternate runs of word tokens and separator blocks
    segments: List[Dict[str, Any]] = []
    p = 0
    while p < n:
        kind_is_word = is_word[p]
        q = p + 1
        while q < n and is_word[q] == kind_is_word:
            q += 1
        if kind_is_word:
            # Concatenate the word tokens in this run (fine-grained tokens forming one "word")
            word_str = ''.join(lower_tokens[p:q])
            segments.append({'type': 'w', 'start': p, 'end': q - 1, 'word': word_str})
        else:
            segments.append({'type': 's', 'start': p, 'end': q - 1})
        p = q

    # Walk word trie from every word-run segment
    seg_len = len(segments)
    for si in range(seg_len):
        seg = segments[si]
        if seg['type'] != 'w':
            continue
        node = word_trie
        r_word_idx = si
        # Step over word segments; between them there must be one separator segment
        while True:
            # Current word segment
            if r_word_idx >= seg_len or segments[r_word_idx]['type'] != 'w':
                break
            word = segments[r_word_idx]['word']
            child = node['n'].get(word)
            if child is None:
                break
            node = child
            if node['v']:
                # Annotate from si to r_word_idx:
                # - all word segments' tokens
                # - all separator segments between them (inner separators only)
                for val in node['v']:
                    # annotate word segments
                    for sidx in range(si, r_word_idx + 1):
                        if segments[sidx]['type'] == 'w':
                            s = segments[sidx]['start']
                            e = segments[sidx]['end']
                            for tpos in range(s, e + 1):
                                annotations[tpos].add(val)
                    # annotate inner separator segments
                    for sidx in range(si + 1, r_word_idx):
                        if segments[sidx]['type'] == 's':
                            s = segments[sidx]['start']
                            e = segments[sidx]['end']
                            for tpos in range(s, e + 1):
                                annotations[tpos].add(val)
            # Move to next word segment: r_word_idx + 2 (skip the separator in between)
            next_idx = r_word_idx + 2
            if next_idx >= seg_len:
                break
            r_word_idx = next_idx

    # Build result
    annotated_tokens = [(tok, annotations[i]) for i, tok in enumerate(tokens)]
    return annotated_tokens
