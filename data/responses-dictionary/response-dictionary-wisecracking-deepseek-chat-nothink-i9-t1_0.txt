import collections.abc
from collections import defaultdict
from typing import Set, Dict, List, Tuple

class TrieNode:
    __slots__ = ('ids', 'children')
    
    def __init__(self):
        self.ids: Set[int] = set()
        self.children: Dict[str, 'TrieNode'] = defaultdict(TrieNode)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie-based index from the dictionary for efficient phrase lookup.
    Normalizes all keys to lowercase and splits them into tokens.
    """
    root = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize case and split into tokens
        normalized_tokens = phrase.lower().split()
        current = root
        
        for token in normalized_tokens:
            current = current.children[token]
        
        current.ids.add(entry_id)
    
    return root

def is_word_token(token: str) -> bool:
    """Check if token looks like a word (as opposed to separator/punctuation)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[Tuple[str, Set]]:
    """
    Annotate tokens by finding all matching dictionary entries.
    Handles both individual words and compound phrases.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    root: TrieNode = dictionary_index
    
    for i in range(len(tokens)):
        if not is_word_token(tokens[i]):
            continue
            
        current = root
        j = i
        
        while j < len(tokens):
            token = tokens[j]
            
            if is_word_token(token):
                normalized = token.lower()
                if normalized not in current.children:
                    break
                current = current.children[normalized]
                
                # If we found a complete phrase, annotate all its tokens
                if current.ids:
                    for k in range(i, j + 1):
                        annotations[k].update(current.ids)
            elif j == i or j == len(tokens) - 1:
                # Skip leading or trailing separators
                break
            # Else: separator in the middle of potential phrase - continue
            
            j += 1
    
    # Also check individual word matches
    for i, token in enumerate(tokens):
        if is_word_token(token):
            normalized = token.lower()
            node = dictionary_index.children.get(normalized)
            if node and node.ids:
                annotations[i].update(node.ids)
    
    return list(zip(tokens, annotations))
