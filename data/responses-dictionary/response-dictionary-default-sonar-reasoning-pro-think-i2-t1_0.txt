import re

def build_dictionary_index(dictionary):
    index = {}
    max_length = 0
    for entry, eid in dictionary.items():
        words = re.findall(r'\w+', entry)
        normalized = ' '.join(words).lower()
        index[normalized] = eid
        current_length = len(words)
        if current_length > max_length:
            max_length = current_length
    return {'index': index, 'max_length': max_length}

def annotate(tokens, dictionary_index):
    def is_word(token):
        return any(c.isalpha() for c in token)
    
    index = dictionary_index['index']
    max_length = dictionary_index['max_length']
    annotations = [(token, set()) for token in tokens]
    
    word_positions = [i for i, t in enumerate(tokens) if is_word(t)]
    
    for start in range(len(word_positions)):
        current_start = word_positions[start]
        token = tokens[current_start].lower()
        if token in index:
            annotations[current_start] = (annotations[current_start][0], {index[token]})
        
        max_phrase_length = min(
            max_length,
            len(word_positions) - start
        )
        
        for l in range(max_phrase_length, 0, -1):
            end = start + l
            if end >= len(word_positions):
                break
            end_pos = word_positions[end]
            
            words = [
                tokens[i].lower() 
                for i in word_positions[start:end+1]
            ]
            phrase = ' '.join(words)
            
            if phrase in index:
                eid = index[phrase]
                for k in range(current_start, end_pos + 1):
                    annotations[k] = (
                        annotations[k][0],
                        annotations[k][1].union({eid})
                    )
    
    return annotations
