import collections.abc

class TrieNode:
    def __init__(self):
        self.children = {}
        self.entries = set()  # Dictionary entries ending at this node

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize and split into words
        words = []
        current_word = []
        
        for char in phrase:
            if char.isalnum():
                current_word.append(char.lower())
            elif current_word:
                words.append(''.join(current_word))
                current_word = []
        
        if current_word:
            words.append(''.join(current_word))
        
        # Insert into Trie
        if words:
            node = root
            for word in words:
                if word not in node.children:
                    node.children[word] = TrieNode()
                node = node.children[word]
            node.entries.add(entry_id)
    
    return root

def is_separator(token: str) -> bool:
    """Check if a token is a separator (contains no alphanumeric characters)."""
    return not any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try to match phrases starting at each position
    for start_pos in range(n):
        # Skip if starting with a separator
        if is_separator(tokens[start_pos]):
            continue
        
        # Traverse the Trie
        node = dictionary_index
        word_positions = []  # Track positions of words in the current phrase
        
        for pos in range(start_pos, n):
            token = tokens[pos]
            
            if is_separator(token):
                # Skip separators but continue matching
                continue
            
            # Try to match the word
            normalized_word = token.lower()
            if normalized_word not in node.children:
                break
            
            node = node.children[normalized_word]
            word_positions.append(pos)
            
            # If we found entries at this node, annotate all tokens in the phrase
            if node.entries:
                # Determine the range to annotate
                first_word_pos = word_positions[0]
                last_word_pos = word_positions[-1]
                
                # Annotate all tokens from first word to last word (inclusive)
                for i in range(first_word_pos, last_word_pos + 1):
                    annotations[i].update(node.entries)
    
    # Create the result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
