import collections.abc
import re

# Helper function: classify a token as a word token - those that can match dictionary "words"
def is_word_token(token):
    # Definition: a word token is non-empty and contains at least one alphanum character
    # This is broad enough for language learners (to handle e.g. 'don't', 'él', etc.)
    return any(c.isalnum() for c in token)

def normalize_word(word):
    return word.lower()

def split_dict_phrase(phrase):
    # Split a dictionary phrase into its word tokens (ignore separators for trie)
    # e.g. "AAA, BBB" => ["aaa", "bbb"]
    # Use regex for robustness; treat consecutive runs of alphanums as words
    return [normalize_word(w) for w in re.findall(r"\w+", phrase)]

class TrieNode:
    __slots__ = ("children", "entry_ids")
    def __init__(self):
        self.children = dict()
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and compound phrases.
    """
    # Build a trie mapping sequences of words (lowercase) to entry ids
    root = TrieNode()
    single_word_dict = dict()  # normalized word -> set of entry ids

    for phrase, entry_id in dictionary.items():
        words = split_dict_phrase(phrase)
        if not words:
            continue  # skip empty key
        # Insert phrase into trie
        node = root
        for w in words:
            if w not in node.children:
                node.children[w] = TrieNode()
            node = node.children[w]
        node.entry_ids.add(entry_id)
        # Additionally, store single word entry ids
        if len(words) == 1:
            single_word_dict.setdefault(words[0], set()).add(entry_id)
    # Store single_word_dict on root for easy access
    root.single_word_dict = single_word_dict
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    if not tokens:
        return []
    tokens = list(tokens)  # for multiple passes
    n = len(tokens)
    # Classify token types and store word positions
    token_types = [is_word_token(tok) for tok in tokens]
    word_positions = [i for i, is_word in enumerate(token_types) if is_word]
    # Prepare: For each token, a set to accumulate annotations
    token_annotations = [set() for _ in tokens]
    trie = dictionary_index
    single_word_dict = getattr(trie, 'single_word_dict', {})

    ### 1. Single-word annotations
    for i, (tok, is_word) in enumerate(zip(tokens, token_types)):
        if is_word:
            norm_word = normalize_word(tok)
            # Accumulate all dictionary entries for this word token
            entry_ids = single_word_dict.get(norm_word)
            if entry_ids:
                token_annotations[i].update(entry_ids)

    ### 2. Compound phrase/phrase annotations
    # For all possible word starters
    for start_i in range(n):
        if not token_types[start_i]:
            continue  # Only start on word tokens
        trie_node = trie
        cur = start_i
        word_count = 0
        # We'll walk from start_i, maintaining indices of word tokens and spanning tokens
        indices_spanned = []
        end_i = start_i  # end token index to update for phrase ends
        wpos = start_i

        # Collects indices between start_i and end_i (inclusive, all tokens between first and last word)
        cur_word_idx = cur  # our current token index
        word_indices_in_span = []  # store positions of word tokens in the span

        while cur_word_idx < n:
            # Seek to next word token, skipping separators
            if not token_types[cur_word_idx]:
                cur_word_idx += 1
                continue
            word = normalize_word(tokens[cur_word_idx])
            if word not in trie_node.children:
                break  # No phrase starts here
            trie_node = trie_node.children[word]
            word_indices_in_span.append(cur_word_idx)
            # The span is from first to current word, including all tokens between
            start_span = start_i
            end_span = cur_word_idx
            # phrase_length is the number of words found
            phrase_length = len(word_indices_in_span)
            # Now if this trie node ends a dictionary phrase, annotate tokens in the span:
            if trie_node.entry_ids:
                # Find leading separator and trailing separator (for requirement #3)
                # - Do not annotate the leading separator (before first word)
                # - Do not annotate the trailing separator (after last word)
                # Annotate all tokens from start_span to end_span *inclusive* except leading/trailing seps
                for span_idx in range(start_span, end_span + 1):
                    # If at the boundaries, skip if not word token
                    if span_idx == start_span:
                        if not token_types[span_idx]:
                            continue  # skip leading sep
                    if span_idx == end_span:
                        if not token_types[span_idx]:
                            continue  # skip trailing sep
                    # Only annotate inner separators and word tokens
                    token_annotations[span_idx].update(trie_node.entry_ids)
            cur_word_idx += 1  # Move to next token for next trie traversal

    return [(tok, anns) for tok, anns in zip(tokens, token_annotations)]
