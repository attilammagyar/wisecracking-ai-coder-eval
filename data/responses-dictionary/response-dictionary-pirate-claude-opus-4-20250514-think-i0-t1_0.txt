   - **Phrases**: like "black swan" or "once in a blue moon" (with spaces or punctuation)

2. **The matching rules be:**
   - Single words match single tokens (case-insensitive, arr!)
   - Compound words match when ye concatenate consecutive word tokens (no separators allowed between 'em!)
   - Phrases match by their word sequence - the exact separators don't matter, just the words in order

3. **For annotations:**

   - For phrases, the separators between words also get marked

import re

def is_word_token(token):
    """Check if a token be containin' any alphanumeric characters."""
    return any(c.isalnum() for c in token)

def extract_words(text):
    """Extract all words from text, ignorin' punctuation and spaces."""
    return re.findall(r'\w+', text)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    entries = []
    
    for key, entry_id in dictionary.items():
        words = extract_words(key)
        
        entry = {
            'id': entry_id,
            'original_key': key,
            'normalized_key': key.lower(),
            'words': [w.lower() for w in words],
            'has_separators': any(not c.isalnum() for c in key)
        }
        entries.append(entry)
    
    return entries

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens_list = list(tokens)
    n = len(tokens_list)
    annotations = [set() for _ in range(n)]
    
    # Try to match starting at each position
    for start in range(n):
        # Skip if not starting with a word token
        if not is_word_token(tokens_list[start]):
            continue
        
        for entry in dictionary_index:
            # Skip empty entries
            if not entry['words']:
                continue
            
            # Try single word match
            if len(entry['words']) == 1 and not entry['has_separators']:
                if tokens_list[start].lower() == entry['normalized_key']:
                    annotations[start].add(entry['id'])
            
            # Try compound word match (concatenated consecutive word tokens)
            if not entry['has_separators']:
                concatenated = ''
                for end in range(start, n):
                    if not is_word_token(tokens_list[end]):
                        break

                    if concatenated.lower() == entry['normalized_key']:
                        # Annotate all tokens in the compound

                            annotations[i].add(entry['id'])

    annotated_tokens = [(tokens_list[i], annotations[i]) for i in range(n)]
    return annotated_tokens

def match_phrase_words(tokens, start_pos, target_words):
    """
    Try to match a sequence of target words starting at given position.
    Returns the index of the last token in the match, or None if no match.
    """
    if not target_words:
        return None
    
    word_index = 0
    last_matched_position = None
    
    for i in range(start_pos, len(tokens)):
        if is_word_token(tokens[i]):
            # Check if this word matches the expected word
            if word_index < len(target_words) and tokens[i].lower() == target_words[word_index]:
                word_index += 1
                last_matched_position = i
                
                # If we've matched all words, success!
                if word_index == len(target_words):
                    return last_matched_position
            else:
                # Word doesn't match - no match possible
                return None
    
    # Reached end without matching all words
