def annotate(tokens: list[str], dictionary_index: dict) -> list[tuple[str, set]]:
    norm_dict = dictionary_index['normalized']
    max_words = dictionary_index['max_words']
    annotated = [(token, set()) for token in tokens]

    for i, token in enumerate(tokens):
        # Check individual token (case-insensitive)
        lowercase_token = token.lower()
        if lowercase_token in norm_dict:
            annotated[i][1].add(norm_dict[lowercase_token])

        # Check compound phrases ending at current token
        for word_count in range(1, max_words + 1):
            # Collect specified number of words skipping separators
            words = []
            pos = i
            while len(words) < word_count and pos >= 0:
                current_token = tokens[pos]
                # Check if token is alphanumeric (word component)
                if re.fullmatch(r'^[a-z0-9]*$', current_token, flags=re.IGNORECASE):
                    words.append(current_token)
                pos -= 1
            if len(words) < word_count:
                break  # Not enough words available

            # Build normalized phrase
            normalized_phrase = ' '.join(words[::-1]).lower()  # Words are collected in reverse order
            if normalized_phrase in norm_dict:
                aid = norm_dict[normalized_phrase]
                start_idx = pos + 1  # pos was decremented beyond valid range
                for idx in range(start_idx, i + 1):
                    annotated[idx][1].add(aid)

    return annotated
