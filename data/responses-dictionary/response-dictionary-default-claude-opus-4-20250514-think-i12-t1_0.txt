        text_lookup[normalized].add(entry_id)
        
        # 2. Extract words and store phrases
        words = extract_words(text)
        if len(words) > 1:
            words_tuple = tuple(words)
            if words_tuple not in phrase_lookup:
                phrase_lookup[words_tuple] = set()
            phrase_lookup[words_tuple].add(entry_id)
    
    return {
        'text_lookup': text_lookup,
        'phrase_lookup': phrase_lookup
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Convert to list for indexing
    text_lookup = dictionary_index['text_lookup']
    phrase_lookup = dictionary_index['phrase_lookup']
    n = len(tokens)
    
    # Track annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Try all starting positions
    for start in range(n):
        # Skip if starting with a non-word token (can't start a match)
        if not is_word_token(tokens[start]):
            continue
            
        # 1. Try single token match
        normalized = tokens[start].lower()
        if normalized in text_lookup:
            annotations[start].update(text_lookup[normalized])
        
        # 2. Try compound word matches (concatenating adjacent word tokens)
        compound = normalized
        for i in range(start + 1, n):
            if not is_word_token(tokens[i]):
                break
            compound += tokens[i].lower()
            if compound in text_lookup:
                # Annotate all tokens in the compound word
                for j in range(start, i + 1):
                    annotations[j].update(text_lookup[compound])
        
        # 3. Try phrase matches
        words = [normalized]
        word_positions = [start]
        
        for i in range(start + 1, n):
            if is_word_token(tokens[i]):
                words.append(tokens[i].lower())
                word_positions.append(i)
                
                # Check if this word sequence matches a phrase
                words_tuple = tuple(words)
                if words_tuple in phrase_lookup:
                    # Annotate from first word to last word (inclusive)
                    for j in range(word_positions[0], word_positions[-1] + 1):
                        annotations[j].update(phrase_lookup[words_tuple])
    
    # Build result
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
    return annotated_tokens
