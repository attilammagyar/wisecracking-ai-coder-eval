import collections.abc

def _is_word_token(token: str) -> bool:
    """
    A token is considered a 'word token' if it's non-empty and consists
    solely of alphanumeric characters. These tokens can be part of dictionary
    words/phrases.
    """
    if not token:
        return False
    return token.isalnum()

class _TrieNode:
    """A node in the dictionary Trie."""
    def __init__(self):
        self.children: dict[str, _TrieNode] = {}
        self.entries: set = set()

class _DictionaryIndex:
    """
    A wrapper for the dictionary Trie to provide a clean interface.
    The Trie stores normalized words and phrases for efficient lookup.
    """
    def __init__(self):
        self.root = _TrieNode()

    def _split_phrase(self, phrase: str) -> list[str]:
        """
        Splits a phrase into word components based on non-alphanumeric characters.
        Example: "black-swan event" -> ["black", "swan", "event"]
        """
        words = []
        current_word = []
        for char in phrase:
            if char.isalnum():
                current_word.append(char)
            else:
                if current_word:
                    words.append("".join(current_word))
                    current_word = []
        if current_word:
            words.append("".join(current_word))
        return words

    def add_phrase(self, phrase: str, entry: object):
        """Adds a phrase and its corresponding entry to the Trie."""
        normalized_words = self._split_phrase(phrase.lower())
        
        node = self.root
        for word in normalized_words:
            if word not in node.children:
                node.children[word] = _TrieNode()
            node = node.children[word]
        node.entries.add(entry)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    This implementation uses a Trie (prefix tree) data structure. Each path
    from the root represents a sequence of normalized "word components".
    Normalization involves converting to lowercase and splitting by non-alphanumeric
    characters. This handles case-insensitivity and variations in separators.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    Returns:
        An index object for use with the annotate function.
    """
    index = _DictionaryIndex()
    for phrase, entry in dictionary.items():
        index.add_phrase(phrase, entry)
    return index

def annotate(
    tokens: collections.abc.Iterable[str], 
    dictionary_index: object
) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    The algorithm iterates through each token, and for each "word token",
    it initiates a search for all possible dictionary matches (single words,
    compound words, and phrases) starting from that token.

    The search is performed using a recursive helper function that traverses both
    the token stream and the dictionary Trie simultaneously. This allows it to handle:
    - Phrases with varying separators.
    - Compound words formed by concatenating multiple tokens.
    - Overlapping and nested phrases.

    Matches are recorded, and annotations are applied to all tokens that form
    part of a match, including inner separators.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each
                            token in tokens.
    """
    if not isinstance(dictionary_index, _DictionaryIndex):
        raise TypeError("dictionary_index must be an object created by build_dictionary_index.")

    token_list = list(tokens)
    n = len(token_list)
    if n == 0:
        return []

    annotations = [set() for _ in range(n)]

    def _find_matches_recursive(
        component_start_idx: int,
        current_trie_node: _TrieNode,
        phrase_start_token_idx: int,
    ):
        """
        Recursively search for matches in the token stream and Trie.

        Args:
            component_start_idx: The index in token_list from which to start forming
                                 the next word component of a phrase.
            current_trie_node: The current node in the Trie we are at.
            phrase_start_token_idx: The index of the first token of the entire
                                    phrase being matched.
        """
        word_buffer = []
        
        current_token_idx = component_start_idx
        while current_token_idx < n and not _is_word_token(token_list[current_token_idx]):
            current_token_idx += 1
        
        for j in range(current_token_idx, n):
            token = token_list[j]
            if not _is_word_token(token):
                break
            
            word_buffer.append(token.lower())
            word_component = "".join(word_buffer)

                next_trie_node = current_trie_node.children[word_component]

                        annotations[idx].update(next_trie_node.entries)

        if _is_word_token(token_list[i]):
            _find_matches_recursive(i, dictionary_index.root, i)
