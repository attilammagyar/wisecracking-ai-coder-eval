```python
import re
from collections import defaultdict

def build_dictionary_index(dictionary: dict) -> object:
    """Builds index from dictionary with phrase sequences and concatenated entries"""
    normalized = {k.lower(): v for k, v in dictionary.items()}
    
    # Phrase mapping: {(word1, word2, ...), ... : {id1, id2, ...}}
    phrase_map = defaultdict(set)
    # Max phrase length for phrase matching
    max_phrase_length = 0
    # Store phrase lengths to compute max
    phrase_lengths = []
    
    # Build phrase_map by splitting entries into words
    for entry, ids in normalized.items():
        words = re.split(r'\W+', entry.strip())
        words = tuple([w for w in words if w])
        if len(words) > 0:
            phrase_map[words].add(ids)
            phrase_lengths.append(len(words))
    max_phrase_length = max(phrase_lengths) if phrase_lengths else 0
    
    # Build concatenated string set: {normalized_string -> {ids}}
    concat_set = defaultdict(set)
    for entry, ids in normalized.items():
        concat_set[entry].add(ids)
        
    return {
        'phrase_map': phrase_map,
        'max_phrase_length': max_phrase_length,
        'concat_set': concat_set
    }

def annotate(tokens: list, dictionary_index: object) -> list:
    """Annotates tokens using the built dictionary index"""
    annotated = []
    # Preprocess: Track word positions and map word indices to token indices
    word_positions = []
    word_to_index = {}
    for i, t in enumerate(tokens):
        if re.fullmatch(r'\w+', t):
            word_positions.append(i)
            word_to_index[i] = len(word_positions) - 1  # Stores (token_idx → word_seq_idx)
    
    # Iterate through each token
    for i, t in enumerate(tokens):
        annotations = set()
        # Check phrase matches for all possible lengths
        for length in range(1, dictionary_index['max_phrase_length'] + 1):
            # Find starting word position
            start_idx = i - length + 1
            if start_idx < 0:
                continue
            # Get starting position in word list
            s = start_idx
            if s < len(word_positions):
                # Get end word position
                e = min(start_idx + length -1, len(word_positions)-1)
                # Check if all intermediate tokens are separators
                is_phrase = True
                phrase_start = word_positions[s]
                phrase_end = word_positions[e]
                # Check if all tokens between phrase_start and phrase_end are separators
                for pos in range(phrase_start +1, phrase_end):
                    if re.match(r'\w+', tokens[pos]):
                        is_phrase = False
                        break
                if is_phrase:
                    # Get phrase word sequence
                    words = [tokens[word_positions[j]].lower() for j in range(s, e+1)]
                    phrase_key = tuple(words)
                    if phrase_key in dictionary_index['phrase_map']:
                        annotations.update(dictionary_index['phrase_map'][phrase_key])
        # Check concatenated matches
        if i in word_to_index:
            j = word_to_index[i]  # Get word sequence index
            # Check all possible prefixes ending at j
            for s in range(0, j + 1):
                e = j
                # Get word positions from s to e
                word_pos_list = word_positions[s:e+1]
                # Concatenate their token values
                concatenated = ''.join(tokens[pos].lower() for pos in word_pos_list)
                if concatenated in dictionary_index['concat_set']:
                    annotations.update(dictionary_index['concat_set'][concatenated])
        # Append to results
        annotated.append( (t, annotations) )
    return annotated
```

**Explanation:**  
1. **Index Building:**  
   - **Phrase Handling:** Entries split into words (ignoring separators) are stored in `phrase_map` as tuples of words → set of IDs.  
   - **Concat Handling:** Raw entries are stored in `concat_set` for exact matches.  

2. **Annotation:**  
   - **Phrase Matching:** For each token, check backward across possible phrase lengths. Capture all phrase sequences where words exist with only separators between them.  
   - **Concat Matching:** For each word token, check all possible combined word sequences from beginning to current position, verifying exact matches.  

**Efficiency Note:**  
The solution uses sliding-window checks for phrases and sequential checks for concatenations. While not optimal for extremely large dictionaries or token lists, it handles typical cases efficiently using dictionary lookups and linear scans.
