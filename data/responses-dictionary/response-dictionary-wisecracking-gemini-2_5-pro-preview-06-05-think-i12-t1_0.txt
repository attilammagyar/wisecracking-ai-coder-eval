import collections.abc

class _DictionaryIndex:
    """
    A dictionary index using a Trie for efficient prefix and phrase matching.

    This internal class encapsulates the Trie structure and the logic for
    normalizing and adding dictionary entries. The normalization step is crucial
    for handling case-insensitivity and variations in word separators.
    """
    _VALUES_KEY = '_values'

    def __init__(self):
        self.root = {}

    def add(self, key: str, value: object):
        """
        Adds a key-value pair to the index, normalizing the key first.
        """
        normalized_key = self._normalize_string(key)
        if not normalized_key:
            return

        node = self.root
        for char in normalized_key:
            node = node.setdefault(char, {})
        
        values = node.setdefault(self._VALUES_KEY, set())
        values.add(value)

    @staticmethod
    def _normalize_string(s: str) -> str:
        """
        Normalizes a string for Trie insertion and lookup.

        Converts to lowercase and replaces any sequence of non-alphanumeric
        characters with a single space, then trims leading/trailing spaces.
        """
        result = []
        in_separator = False
        for char in s.lower():
            if char.isalnum():
                result.append(char)
                in_separator = False
            else:
                if not in_separator:
                    result.append(' ')
                in_separator = True
        
        return "".join(result).strip()

    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    index = _DictionaryIndex()
