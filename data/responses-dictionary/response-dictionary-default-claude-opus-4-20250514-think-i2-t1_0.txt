import collections.abc

def is_word_token(token):
    """Check if a token contains any alphabetic characters."""
    return any(c.isalpha() for c in token)

def normalize_word(word):
    """Normalize a word for case-insensitive lookup."""
    return word.lower()

def extract_words(text):
    """Extract words from text, ignoring separators."""
    words = []
    current_word = []
    for char in text:
        if char.isalpha():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    if current_word:
        words.append(''.join(current_word))
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    single_words = {}  # normalized_word -> set of entry_ids
    compound_entries = []  # list of (word_tuple, entry_id, is_compound_word)
    
    for key, entry_id in dictionary.items():
        words = extract_words(key)
        normalized_words = tuple(normalize_word(w) for w in words)
        
        if len(normalized_words) == 1:
            # Single word entry
            word = normalized_words[0]
            if word not in single_words:
                single_words[word] = set()
            single_words[word].add(entry_id)
        else:
            # Compound entry (phrase or compound word)
            # Check if it's a compound word (no separators in original)
            is_compound_word = ''.join(words).lower() == key.lower()
            compound_entries.append((normalized_words, entry_id, is_compound_word))
    
    return {
        'single_words': single_words,
        'compound_entries': compound_entries
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Convert to list for indexing
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    single_words = dictionary_index['single_words']
    compound_entries = dictionary_index['compound_entries']
    
    # Process each starting position
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
            
        # Check single word match
        normalized = normalize_word(tokens[start])
        if normalized in single_words:
            annotations[start].update(single_words[normalized])
        
        # Check compound entries
        for words, entry_id, is_compound_word in compound_entries:
            if is_compound_word:
                # Compound word: consecutive word tokens concatenated
                matched_tokens = []
                pos = start
                word_idx = 0
                
                while pos < n and word_idx < len(words):
                    if is_word_token(tokens[pos]):
                        # Try to match this word token
                        concat_word = ''
                        temp_pos = pos
                        
                        # Concatenate consecutive word tokens
                        while temp_pos < n and is_word_token(tokens[temp_pos]):
                            concat_word += tokens[temp_pos]
                            temp_pos += 1
                            
                            # Check if we've matched the current word
                            if normalize_word(concat_word) == words[word_idx]:
                                matched_tokens.extend(range(pos, temp_pos))
                                pos = temp_pos
                                word_idx += 1
                                break
                        else:
                            # Couldn't match this word
                            break
                    else:
                        # Hit a separator, can't continue compound word
                        break
                
                # Check if we matched all words
                if word_idx == len(words):
                    for idx in matched_tokens:
                        annotations[idx].add(entry_id)
            
            else:
                # Compound phrase: words with any separators between
                matched_tokens = []
                pos = start
                word_idx = 0
                
                while pos < n and word_idx < len(words):
                    if is_word_token(tokens[pos]):
                        if normalize_word(tokens[pos]) == words[word_idx]:
                            matched_tokens.append(pos)
                            word_idx += 1
                            pos += 1
                        else:
                            break
                    else:
                        # Separator - include if we're between words in the phrase
                        if word_idx > 0 and word_idx < len(words):

                        pos += 1
                
                # Check if we matched all words
                if word_idx == len(words):
                    for idx in matched_tokens:
                        annotations[idx].add(entry_id)
    
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
