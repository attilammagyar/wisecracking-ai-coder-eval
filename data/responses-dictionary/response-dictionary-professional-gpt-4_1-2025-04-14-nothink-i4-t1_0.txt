import collections.abc
import string

def _is_word_token(token):
    # Heuristic: a word token has at least one alphanumeric char.
    return any(c.isalnum() for c in token)

def _normalize_token(token):
    # Lowercase and strip for normalization
    return token.lower()

def _normalize_dict_entry(entry):
    # Lowercase, and split on whitespace and common separators for index building.
    # We'll keep contiguous runs of word characters as tokens, and everything else as separators.
    # This simulates a generic tokenizer.
    tokens = []
    current = []
    for c in entry:
        if c.isalnum():
            current.append(c.lower())
        else:
            if current:
                tokens.append(''.join(current))
                current = []
            if not c.isspace():  # don't keep spaces as tokens at index build time
                tokens.append(c)
    if current:
        tokens.append(''.join(current))
    # Remove tokens that are only whitespace
    tokens = [t for t in tokens if t.strip()]
    return tokens

class _TrieNode:
    __slots__ = ('children', 'entry_ids')
    def __init__(self):
        self.children = dict()
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # 1. Build single-token word index: {normalized token: set(entries)}
    word_map = dict()
    # 2. Build multi-token phrase trie
    trie_root = _TrieNode()
    max_phrase_len = 1
    
    for entry, entry_id in dictionary.items():
        entry_tokens = _normalize_dict_entry(entry)
        if not entry_tokens:
            continue
        # 1. Single-word entries
        if len(entry_tokens) == 1:
            k = entry_tokens[0]
            word_map.setdefault(k, set()).add(entry_id)
        # 2. Compound entries (including multi-word and pure symbol "words")
        node = trie_root
        for t in entry_tokens:
            if t not in node.children:
                node.children[t] = _TrieNode()
            node = node.children[t]
        node.entry_ids.add(entry_id)
        if len(entry_tokens) > max_phrase_len:
            max_phrase_len = len(entry_tokens)
    return {
        'word_map': word_map,
        'trie_root': trie_root,
        'max_phrase_len': max_phrase_len,
    }

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    # Convert tokens to a list so we can access by index
    tokens = list(tokens)
    n = len(tokens)
    # Prepare normalized view
    normalized_tokens = [_normalize_token(tok) for tok in tokens]
    # Prepare classification (word/separator)
    is_word = [_is_word_token(tok) for tok in tokens]

    word_map = dictionary_index['word_map']
    trie_root = dictionary_index['trie_root']
    max_phrase_len = dictionary_index['max_phrase_len']

    # Output: one set of entry_ids per token position
    ann = [set() for _ in range(n)]
    
    # 1. Compound phrase annotation
    for i in range(n):
        node = trie_root
        last_match = None     # (end_idx, entry_ids, first_word_idx, last_word_idx)
        j = i
        first_word_j = None   # index of first word token in window
        last_word_j = None    # index of last word token in window
        # We'll build the candidate window up to max_phrase_len tokens
        phrase_token_idxs = []
        for k in range(max_phrase_len):
            if j >= n:
                break
            norm_tok = normalized_tokens[j]
            if norm_tok not in node.children:
                break
            node = node.children[norm_tok]
            phrase_token_idxs.append(j)
            # update first_word/last_word indices
            if is_word[j]:
                if first_word_j is None:
                    first_word_j = j
                last_word_j = j
            if node.entry_ids:
                # Only treat as match if phrase starts and ends with word tokens (per test rules).
                # Leading/trailing separators outside them are not to be included in annotation.
                # But if phrase is e.g. [" ", "aaa", " ", "bbb", " "], we only annotate the subset ["aaa", " ", "bbb"] (the "inner" part).
                # So only if there is at least one word token, we match
                if first_word_j is not None and last_word_j is not None:
                    # Find extent from first_word_j to last_word_j (inclusive)
                    last_match = (first_word_j, last_word_j, set(node.entry_ids))
            j += 1
        if last_match:
            # Annotate the window from first_word_j to last_word_j inclusive
            start, end, entry_ids = last_match
            for k in range(start, end + 1):
                # Annotate all tokens in window.
                # Only annotate separators that are *within* window (that is, not at start/end of phrase as per rules)
                if is_word[k]:
                    ann[k].update(entry_ids)
                else:
                    # Only annotate if not at window boundary
                    if k != start and k != end:
                        ann[k].update(entry_ids)

    # 2. Single token words
    for i in range(n):
        if is_word[i]:
            entry_ids = word_map.get(normalized_tokens[i])
            if entry_ids:
                ann[i].update(entry_ids)

    return [(tokens[i], ann[i]) for i in range(n)]
