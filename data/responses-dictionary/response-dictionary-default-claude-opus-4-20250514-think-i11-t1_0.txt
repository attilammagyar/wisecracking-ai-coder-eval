def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations
    annotations = [set() for _ in range(n)]
    
    # Try matching each dictionary entry at each position
    for entry_words, is_compound_word, entry_id in dictionary_index['entries']:
        # Try starting from each position
        for start in range(n):
            if len(entry_words) == 1:
                # Single word match
                if is_word_token(tokens[start]) and tokens[start].lower() == entry_words[0]:
                    annotations[start].add(entry_id)
            
            elif is_compound_word:
                # Compound word - concatenate consecutive word tokens
                if is_word_token(tokens[start]):
                    concatenated = ""
                    pos = start
                    
                    while pos < n and is_word_token(tokens[pos]):
                        concatenated += tokens[pos].lower()
                        pos += 1
                        
                        # Check if we have a match
                        if concatenated == ''.join(entry_words):
                            # Annotate all tokens from start to pos-1
                            for i in range(start, pos):
                                annotations[i].add(entry_id)
                            break
            
            else:
                # Phrase - match words in sequence
                word_idx = 0
                first_word_pos = None
                last_word_pos = None
                
                for i in range(start, n):
                    if is_word_token(tokens[i]):
                        if word_idx < len(entry_words) and tokens[i].lower() == entry_words[word_idx]:
                            if first_word_pos is None:
                                first_word_pos = i
                            last_word_pos = i
                            word_idx += 1
                            
                            if word_idx == len(entry_words):
                                # Found all words - annotate from first to last word position
                                for j in range(first_word_pos, last_word_pos + 1):
                                    annotations[j].add(entry_id)
                                break
                        else:
                            # Word doesn't match, stop
                            break
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
