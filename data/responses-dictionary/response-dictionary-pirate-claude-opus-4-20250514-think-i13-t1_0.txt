    matched_positions = []
    pos = start_pos
    word_idx = 0
    
    # Skip leading separators
    while pos < len(tokens) and not is_word_token(tokens[pos]):
        pos += 1
    
    if pos >= len(tokens):
        return None
    
    # Try to match each word in the entry
    while pos < len(tokens) and word_idx < len(entry_words):
        if is_word_token(tokens[pos]):
            # Accumulate consecutive word tokens
            accumulated = ""
            word_start_pos = pos
            
            while pos < len(tokens) and is_word_token(tokens[pos]):
                accumulated += tokens[pos]
                pos += 1
                
                if accumulated.lower() == entry_words[word_idx]:
                    # Exact match found
                    for i in range(word_start_pos, pos):
                        matched_positions.append(i)
                    word_idx += 1
                    break
                elif not entry_words[word_idx].startswith(accumulated.lower()):
                    # No possible match
                    return None
            else:
                # Reached end without exact match
                if accumulated.lower() != entry_words[word_idx]:
                    return None
            
            # If not the last word, consume separators
            if word_idx < len(entry_words):
                separator_start = pos
                while pos < len(tokens) and not is_word_token(tokens[pos]):
                    pos += 1
                
                # Include separators between words
                if pos > separator_start:
                    for i in range(separator_start, pos):
                        matched_positions.append(i)
        else:
            # Should not happen if logic is correct
            return None
    
    # Check if all words were matched
    if word_idx == len(entry_words):
        return matched_positions
    else:
        return None

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    # Try to match each dictionary entry at each position
    for start_pos in range(len(tokens)):
        for entry in dictionary_index:
            matched_positions = try_match_entry_at_position(tokens, start_pos, entry['words'])
            if matched_positions is not None:
                # Add this entry's ID to all matched positions
                for pos in matched_positions:
                    annotations[pos].add(entry['id'])
    
    # Build result
    annotated_tokens = []
    for token, annotation_set in zip(tokens, annotations):
        annotated_tokens.append((token, annotation_set))
    
    return annotated_tokens
