import collections.abc

class _TrieNode:
    __slots__ = ("children", "outputs")
    def __init__(self):
        self.children: dict[str, _TrieNode] = {}
        self.outputs: set = set()

def _split_words(text: str) -> list[str]:
    """Extract alphanumeric runs (words) only, in order."""
    words = []
    i = 0
    n = len(text)
    while i < n:
        if text[i].isalnum():
            j = i + 1
            while j < n and text[j].isalnum():
                j += 1
            words.append(text[i:j])
            i = j
        else:
            i += 1
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized trie over dictionary words (alnum runs), case-insensitive.
    Also keep a fast map for exact single-word lookups.
    """
    root = _TrieNode()
    single_word_map: dict[str, set] = {}

    for key, entry_id in dictionary.items():
        # Normalize to word-sequence (ignore separators), case-insensitive.
        words = _split_words(str(key).casefold())
        if not words:
            # Ignore entries with no word content.
            continue

        # Insert into trie.
        node = root
        for w in words:
            node = node.children.setdefault(w, _TrieNode())
        node.outputs.add(entry_id)

        # For exact single-token matches.
        if len(words) == 1:
            single_word_map.setdefault(words[0], set()).add(entry_id)

    return {"root": root, "single": single_word_map}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with single-word entries and all matching phrases.
    """
    tokens = list(tokens)
    n = len(tokens)
    anns = [set() for _ in range(n)]
    root: _TrieNode = dictionary_index["root"]
    single: dict[str, set] = dictionary_index["single"]

    # Classify tokens and normalize words.
    is_word = [tok.isalnum() for tok in tokens]
    word_fold = [tok.casefold() if is_word[i] else None for i, tok in enumerate(tokens)]

    # Fast exact single-word tagging.
    for i in range(n):
        if is_word[i]:
            ids = single.get(word_fold[i])
            if ids:
                anns[i].update(ids)

    # Helper: try match a dictionary word (string W) starting at token index pos
    # by concatenating consecutive word tokens (no separators allowed between).
    # Return token index j_end (exclusive) if matched; else None.
    def match_word_at(pos: int, W: str):
        j = pos
        idx = 0
        Wlen = len(W)
        while j < n and is_word[j] and idx < Wlen:
            t = word_fold[j]
            tlen = len(t)
            if idx + tlen > Wlen:
                return None  # would split a token: forbidden
            if W[idx:idx + tlen] != t:
                return None  # mismatch
            idx += tlen
            j += 1
        if idx == Wlen:
            return j
        return None

    # Phrase scanning with a DFS from every word-token start.
    # State: (node, pos, depth, word_indices_list, inner_seps_set, pending_seps_list)
    # - depth = number of dictionary words matched so far.
    # - pending_seps = separators after the last matched word; they become inner_seps only
    #   if we successfully match the next dictionary word (enforcing at least one separator between dict words).
    for s in range(n):
        if not is_word[s]:
            continue

        stack = [(root, s, 0, [], set(), [])]
        while stack:
            node, pos, depth, word_idxs, inner_seps, pending_seps = stack.pop()

            for w, child in node.children.items():
                # Enforce at least one separator between consecutive dictionary words in a phrase.
                if depth >= 1 and not pending_seps:
                    continue

                j_end = match_word_at(pos, w)
                if j_end is None:

                seps_after = []
                k = j_end
                while k < n and not is_word[k]:
                    seps_after.append(k)
                    k += 1

                # Extend matched token indices.
                new_word_idxs = word_idxs + list(range(pos, j_end))

                # Move pending separators (between previous and this word) into inner separators.
                if depth >= 1 and pending_seps:
                    new_inner_seps = inner_seps | set(pending_seps)
                else:
                    new_inner_seps = inner_seps

                new_depth = depth + 1

                # Record any phrase(s) that end here.
                if child.outputs:
                    outs = child.outputs
                    for ti in new_word_idxs:
                        anns[ti].update(outs)
                    if new_inner_seps:
                        for si in new_inner_seps:
                            anns[si].update(outs)

                next_pos = k  # after any separators that follow the just-matched word
                stack.append((child, next_pos, new_depth, new_word_idxs, new_inner_seps, seps_after))
