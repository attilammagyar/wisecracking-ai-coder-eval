def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},  # Lowercase word -> set of entries
        'compound_words': {},  # Lowercase concatenated words -> set of entries
        'compound_phrases': [],  # List of (normalized_words, entry) tuples
    }
    
    for key, value in dictionary.items():
        # Split into tokens (words and separators)
        tokens = re.findall(r'\w+|\W+', key)
        word_tokens = [t for t in tokens if re.match(r'\w+', t)]
        
        if len(word_tokens) == 1:
            # Single word
            word_lower = word_tokens[0].lower()
            if word_lower not in index['single_words']:
                index['single_words'][word_lower] = set()
            index['single_words'][word_lower].add(value)
        else:
            # Multiple words
            # Check if it's a compound word (no separators) or phrase (with separators)
            if len(tokens) == len(word_tokens):
                # Compound word (no separators between words)
                compound_lower = ''.join(word_tokens).lower()
                if compound_lower not in index['compound_words']:
                    index['compound_words'][compound_lower] = set()
                index['compound_words'][compound_lower].add(value)
            else:
                # Compound phrase (has separators)
                words_lower = [w.lower() for w in word_tokens]
                index['compound_phrases'].append((words_lower, value))
    
    return index

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []
    
    # Helper to check if a token is a word
    def is_word_token(token):
        return bool(re.match(r'\w+', token))
