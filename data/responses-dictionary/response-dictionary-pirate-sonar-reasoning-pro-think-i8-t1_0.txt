```python
import re
from typing import Iterable, Tuple, Set, Mapping, Object

def build_dictionary_index(dictionary: Mapping[str, Object]) -> Object:
    # Preprocess dictionary entries
    entries = []
    for entry_key, entry_id in dictionary.items():
        # Split words, exclude separators, concat to normalized form
        words = re.split(r'\W+', entry_key.strip())
        words = [w for w in words if w]
        normalized = ''.join(words).lower()
        entries.append((normalized, entry_id))
    
    # Sort by length descending, then lex order
    entries.sort(key=lambda x: (-len(x[0]), x[0]))
    
    # Build configurable sorted list and max_length
    normalized_dict = {k: v for k, v in entries}
    max_length = max(len(k) for k in normalized_dict.keys()) if normalized_dict else 0
    
    return (max_length, normalized_dict)

def annotate(tokens: Iterable[str], dictionary_index: Object) -> Iterable[Tuple[str, Set]]:
    max_length, normalized_dict = dictionary_index
    annotations = [set() for _ in tokens]

    for i, token in enumerate(tokens):
        # Individual token check
        if token.isalnum():
            normalized = token.lower()
            if normalized in normalized_dict:
                annotations[i].add(normalized_dict[normalized])
        
        # Compound phrase checks
        for length in range(1, max_length + 1):
            j = i + length - 1
            if j >= len(tokens):
                continue
            window = tokens[i:j+1]
            
            # Validate window starts/ends with non-separator tokens
            start = window[0]
            end = window[-1]
            if not (start.isalnum() and end.isalnum()):
                continue
            
            # Generate normalized string from window tokens
            concat = ''.join(filter(str.isalnum, window)).lower()
            if concat in normalized_dict:
                entry_id = normalized_dict[concat]
                for idx in range(i, j+1):
                    annotations[idx].add(entry_id)

    return [(token, s) for token, s in zip(tokens, annotations)]
```
