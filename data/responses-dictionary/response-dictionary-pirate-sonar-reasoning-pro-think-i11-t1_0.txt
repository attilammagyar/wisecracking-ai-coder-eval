   - **Normalize Phrases**: Treat different separators uniformly (e.g., spaces, commas, etc.)
   - **Multi-Phrase Support**: Tokens in overlapping phrases get all matching entries

- **Efficient Normalization**: Uses regex to preprocess tokens once
- **Trim Logic**: Reduces unnecessary checks for phrases with extra separators

### Solution Code
```python
import re
from collections.abc import Iterable

def build_dictionary_index(dictionary: Iterable) -> dict:
    """Build a normalized index from a dictionary with fast lookups"""
    normalized_dict = {}
    for key, value in dictionary.items():
        # Normalize keys to lowercase with non-words replaced by spaces
        normalized = re.sub(r'\W+', ' ', key)
        normalized = re.sub(r'\s+', ' ', normalized).strip().lower()
        if normalized:
            normalized_dict[normalized] = value
    return normalized_dict

def is_word(token: str) -> bool:
    """Helper to check if token contains valid word characters"""
    return any(c.isalnum() for c in token)

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[tuple[str, set]]:
    """Annotate tokens with dictionary entries"""
    annotations = [set() for _ in tokens]
    
    for i, start_token in enumerate(tokens):
        for j in range(i, len(tokens)):
            # Extract substring from current token to token j
            substring = tokens[i:j+1]
            
            # Trim leading/trailing non-word separators
            left = 0
            while left < len(substring) and not is_word(substring[left]):
                left += 1
                if left >= len(substring):  # Break if no valid tokens
                    break
                    
            right = len(substring) - 1
            while right >= left and not is_word(substring[right]):
                right -= 1
                if right < left:  # Break if no valid tokens
                    break
                    
            # Skip if no valid substring found after trimming
            if left >= len(substring) or right < left:
                continue
            
            # Normalize the trimmed substring
            processed = [
                re.sub(r'\W+', ' ', token) for token in substring[left:right+1]
            ]
            joined = ' '.join(processed)
            normalized = re.sub(r'\s+', ' ', joined).strip().lower()
            
            # Add entry to all tokens in the valid phrase range
            if normalized in dictionary_index:
                entry = dictionary_index[normalized]
                for k in range(left, right + 1):
                    original_index = i + k
                    if original_index < len(tokens):
                        annotations[original_index].add(entry)
    
    return annotations
```

**2. Annotation Process**  
For every possible phrase:

2. **Normalize**: Makes different separators (commas, spaces, etc.) equivalent
3. **Match Check**: Uses precomputed dictionary map for O(1) lookups

**Optimization**: By checking all possible phrases programmatically, we handle complex cases like overlapping phrases and nested compounds efficiently.
