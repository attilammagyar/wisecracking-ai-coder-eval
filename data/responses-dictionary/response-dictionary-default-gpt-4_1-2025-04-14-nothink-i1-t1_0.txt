    last_type = None
    for m in re.finditer(r'(\w+)|(\W+)', phrase):
        if m.group(1):
            parts.append( ("word", m.group(1).lower()) )
            last_type = "word"
        else:
            parts.append( ("sep", m.group(2)) )
            last_type = "sep"
    return parts

            if key not in node.children:
                node.children[key] = TrieNode()
            node = node.children[key]
        node.endings.add(entry)
    return trie
