        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]

    for start in range(n):

            end = start + length
            subsequence = tokens[start:end]
            
            # Strategy 1: Direct match (single tokens and compound words)
            concatenated = ''.join(subsequence)
            normalized = normalize(concatenated)
            if normalized in dictionary_index['direct']:
                for entry_id in dictionary_index['direct'][normalized]:
                    for i in range(start, end):
                        annotations[i].add(entry_id)
            
            # Strategy 2: Phrase match (multi-word phrases with flexible separators)
            # Find word positions in the subsequence
            word_positions = []
            for i, token in enumerate(subsequence):
                if is_word_token(token):
                    word_positions.append(i)
            
            if len(word_positions) >= 2:  # Need at least 2 words for a phrase
                # Extract the words
                words = []
                for pos in word_positions:
                    words.append(normalize(subsequence[pos]))
                
                words_tuple = tuple(words)
                if words_tuple in dictionary_index['phrases']:
                    # Found a phrase match
                    # Annotate from first word to last word (inclusive)
                    first_word_pos = word_positions[0] + start
                    last_word_pos = word_positions[-1] + start
                    
                    for entry_id in dictionary_index['phrases'][words_tuple]:
                        for i in range(first_word_pos, last_word_pos + 1):
