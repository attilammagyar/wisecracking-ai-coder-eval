def build_dictionary_index(dictionary: dict) -> object:
    normalized_entries = defaultdict(set)
    word_set = set()
    max_len = 0
    # Preprocess dictionary: normalize and track max length
    for key, value in dictionary.items():
        normalized = ' '.join(re.findall(r'\w+', key.lower())).strip()
        normalized_entries[normalized].add(value)
        word_set.add(normalized)
        max_len = max(max_len, len(normalized), len(key.lower()))
    return normalized_entries, max_len

def annotate(tokens: list, dictionary_index: object) -> list:
    normalized_entries, max_len = dictionary_index
    n = len(tokens)
    word_tokens = []
    word_indices = []
    for idx, token in enumerate(tokens):
        if re.search(r'\w', token):
            word_tokens.append(token.lower())
            word_indices.append(idx)
    # Precompute word sequences
    word_sequences = []
    for i in range(len(word_tokens)):
        for j in range(i, min(i + max_len, len(word_tokens))):
            seq = ' '.join(word_tokens[i:j+1])
            if seq in normalized_entries:
                word_sequences.append( (i, j, seq) )
    # Annotate each token
    annotations = [set() for _ in tokens]
    # Handle individual words
    for i, word in enumerate(word_tokens):
        if word in normalized_entries:

    # Handle compound phrases
    for i, j, seq in word_sequences:
        entry_ids = normalized_entries[seq]
        start_idx = word_indices[i]

        for k in range(start_idx, end_idx + 1):
