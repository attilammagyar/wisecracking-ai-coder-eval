```python
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """Preprocess dictionary for fast lookups."""
    entries_set = set()
    entry_ids = {}
    for key, value in dictionary.items():
        lower_key = key.lower()
        entries_set.add(lower_key)
        entry_ids[lower_key] = value
    return {"entries": entries_set, "entry_ids": entry_ids}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """Annotate tokens using brute-force substring matching."""
    annotated = [(token, set()) for token in tokens]
    
    # Check all possible spans for dictionary matches
    for start in range(len(tokens)):
        for end in range(start, len(tokens)):
            substring = ''.join(tokens[start:end+1]).lower()
            if substring in dictionary_index["entries"]:
                entry_id = dictionary_index["entry_ids"][substring]
                for pos in range(start, end + 1):
                    annotated[pos][1].add(entry_id)
    
    return annotated
```

- **Case Insensitivity**: All lookups use lowercase for consistency.
- **Global Span Checks**: Every possible token combination is checked, ensuring all entries (including concatenated tokens) are considered.
- **Efficiency**: Uses Python’s built-in data structures for fast lookups despite the O(n²) complexity, which is manageable for reasonable text lengths.
