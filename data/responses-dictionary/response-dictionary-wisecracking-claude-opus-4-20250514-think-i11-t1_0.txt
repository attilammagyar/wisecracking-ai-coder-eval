def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for entry, entry_id in dictionary.items():
        # Normalize to lowercase for case-insensitive matching
        normalized = entry.lower()
        
        # Extract word tokens (letters and possibly numbers)
        word_tokens = re.findall(r'\w+', normalized)
        
        if not word_tokens:
            continue
            
        # Check if this is a compound word (no separators) or phrase (has separators)
        is_phrase = bool(re.search(r'\W', entry))
        
        # Create key from word tokens
        key = tuple(word_tokens)
        
        if key not in index:
            index[key] = {'words': [], 'phrases': []}
            
        if is_phrase:
            index[key]['phrases'].append(entry_id)
        else:
            index[key]['words'].append(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Helper to check if a token is a word token
    def is_word_token(token):
        return bool(re.match(r'\w', token))
    
    # Try all starting positions
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
            
        # Track word tokens for matching
        word_sequence = []
        word_positions = []
        
        # Try progressively longer sequences
        pos = start
        while pos < n:
            if is_word_token(tokens[pos]):
                word_sequence.append(tokens[pos].lower())
                word_positions.append(pos)
                
                # Check for matches
                key = tuple(word_sequence)
                if key in dictionary_index:
                    entries = dictionary_index[key]
                    
                    # Check compound words (only if tokens are adjacent)
                    if entries['words']:
                        # All positions must be contiguous

                            word_positions[i] + 1 == word_positions[i + 1]
