import re
from collections.abc import Iterable, Mapping

def normalize(key: str) -> str:
    """Normalize a string by replacing non-alphanumeric with spaces and converting to lowercase."""
    cleaned = re.sub(r'[^a-z0-9]', ' ', key.lower())
    return ' '.join(cleaned.strip().split())

def build_dictionary_index(dictionary: Mapping) -> dict:
    """Build an index from normalized dictionary entries to their IDs."""
    index = {}
    for entry, value in dictionary.items():
        norm = normalize(entry)
        if norm not in index:
            index[norm] = set()
        index[norm].add(value)
    return index

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[tuple[str, set]]:
    """Annotate tokens with relevant dictionary entries."""
    annotations = [set() for _ in tokens]
    
    # Annotate individual tokens
    for i, token in enumerate(tokens):
        norm = normalize(token)
        if norm in dictionary_index:
            annotations[i].update(dictionary_index[norm])
    
    # Annotate compound phrases
    n = len(tokens)
    for i in range(n):
        for j in range(i, n):
            substring = ''.join(tokens[i:j+1])
            norm = normalize(substring)
            if norm in dictionary_index:
                for k in range(i, j+1):
                    annotations[k].update(dictionary_index[norm])
    
    return [(tokens[i], ann) for i, ann in enumerate(annotations)]
