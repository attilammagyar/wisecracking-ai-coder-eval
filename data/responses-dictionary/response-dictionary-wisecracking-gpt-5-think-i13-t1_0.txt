class _TrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children = {}  # word -> _TrieNode
        self.ids = set()    # set of dictionary entry ids that end here


def _split_words(s: str):
    """
    Split a string into lowercase alphanumeric 'words' (runs of .isalnum()).
    Separators are anything else and are ignored by design.
    """
    out = []
    current = []
    for ch in s:
        if ch.isalnum():
            current.append(ch.lower())
        else:
            if current:
                out.append("".join(current))
                current = []
    if current:
        out.append("".join(current))
    return out

      - single_prefixes: set of prefixes of all single_words, for concat matching
