```python
import re
from collections import defaultdict

def build_dictionary_index(dictionary: dict) -> object:
    """Builds a dictionary index for efficient annotation"""
    index = defaultdict(set)
    for key, value in dictionary.items():
        words = re.findall(r"\w+", key)  # Extract word-like sequences
        if not words:
            continue
        words_lower = [w.lower() for w in words]
        for substr_len in range(len(words_lower), 0, -1):
            substr = tuple(words_lower[:substr_len])
            index[substr].add(value)
    return dict(index)

def annotate(tokens: list, dictionary_index: dict) -> list:
    """Annotates tokens with dictionary entries using the prebuilt index"""
    annotated = [set() for _ in tokens]
    word_info = []
    
    # Preprocess tokens for word detection
    for token in tokens:
        is_word = bool(re.fullmatch(r"\w+", token))  # Word tokens (letters only)
        lower_word = token.lower() if is_word else None
        word_info.append((is_word, lower_word))
    
    # Apply annotations
    for i, (is_word_i, lower_word_i) in enumerate(word_info):
        if not is_word_i:
            continue
            
        # 1. Annotate standalone words
        if lower_word_i:
            key = (lower_word_i,)
            if key in dictionary_index:
                annotated[i].update(dictionary_index[key])
        
        # 2. Check compound phrases
        for j in range(i, len(tokens)):
            non_sep_indicies = []
            non_sep_words = []
            for k in range(i, j + 1):
                if word_info[k][0]:
                    non_sep_indicies.append(k)
                    non_sep_words.append(word_info[k][1])
            
            # Check longest substrings first
            for substr_len in range(len(non_sep_words), 0, -1):
                substr = tuple(non_sep_words[:substr_len])
                if substr not in dictionary_index:
                    continue
                
                # Annotate ALL tokens between first and last word in substring
                start = min(non_sep_indicies[:substr_len])
                end = max(non_sep_indicies[:substr_len])
                for pos in range(start, end + 1):
                    annotated[pos].update(dictionary_index[substr])
    
    return [(token, s) for token, s in zip(tokens, annotated)]
```
