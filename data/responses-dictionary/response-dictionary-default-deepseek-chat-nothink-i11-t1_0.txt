```python
import collections.abc
from collections import defaultdict

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # Normalize dictionary keys (case-insensitive) and store in a trie and a word set
    trie = {}
    word_set = set()

    for key, value in dictionary.items():
        # Split the key into tokens (words and separators)
        # We consider sequences of alphanumeric characters as words, rest as separators
        tokens = []
        current_token = []
        for c in key:
            if c.isalnum():
                current_token.append(c.lower())
            else:
                if current_token:
                    tokens.append(''.join(current_token))
                    current_token = []
                tokens.append(c.lower())
        if current_token:
            tokens.append(''.join(current_token))

        # Add to word set if it's a single word
        if len(tokens) == 1 and tokens[0].isalnum():
            word_set.add((tokens[0], value))

        # Add to trie for compound phrases
        current = trie
        for token in tokens:
            if token not in current:
                current[token] = {}
            current = current[token]
        if 'values' not in current:
            current['values'] = set()
        current['values'].add(value)

    return {'trie': trie, 'word_set': word_set}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    trie = dictionary_index['trie']
    word_set = dictionary_index['word_set']
    annotated_tokens = []
    tokens = list(tokens)  # To allow multiple passes

    # First pass: find all individual word matches
    individual_annotations = [set() for _ in tokens]
    for i, token in enumerate(tokens):
        if not token.isalnum():
            continue
        normalized = token.lower()
        for (word, value) in word_set:
            if word == normalized:
                individual_annotations[i].add(value)

    # Second pass: find all compound phrase matches
    compound_annotations = [set() for _ in tokens]
    n = len(tokens)
    for i in range(n):
        if not tokens[i].isalnum():
            continue  # Skip non-word tokens at start
        current = trie
        j = i
        while j < n:
            token = tokens[j]
            normalized = token.lower()
            # Check if current token matches in the trie
            if normalized in current:
                current = current[normalized]
                j += 1
                if 'values' in current:
                    # Found a match, annotate all tokens in [i..j-1]
                    for k in range(i, j):
                        compound_annotations[k].update(current['values'])
            else:
                break  # No further matches possible

    # Combine individual and compound annotations
    for i, token in enumerate(tokens):
        combined = individual_annotations[i].union(compound_annotations[i])
        annotated_tokens.append((token, combined))

    return annotated_tokens
```
