import collections.abc
import re
from typing import Set, Dict, List, Tuple, Any

# Helper to check if a token is a word (as opposed to separator)
def is_word_token(token: str) -> bool:
    return bool(token.strip()) and not token.isspace()

class DictionaryIndex:
    def __init__(self):
        self.words: Dict[str, Set[Any]] = {}
        self.phrases: Dict[Tuple[str, ...], Set[Any]] = {}
        self.max_phrase_length = 0

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    index = DictionaryIndex()
    
    for phrase, meaning in dictionary.items():
        # Normalize case
        normalized_phrase = phrase.lower()
        
        # Split into words (handling various separators)
        words = [w for w in re.split(r'(\W+)', normalized_phrase) if w.strip()]
        
        if not words:
            continue
            
        if len(words) == 1:
            # Single word entry
            index.words.setdefault(words[0], set()).add(meaning)
        else:
            # Compound phrase entry
            word_sequence = tuple(words)
            index.phrases.setdefault(word_sequence, set()).add(meaning)
            index.max_phrase_length = max(index.max_phrase_length, len(word_sequence))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[Tuple[str, Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    if not isinstance(dictionary_index, DictionaryIndex):
        raise TypeError("Invalid dictionary index type")
        
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    # First annotate individual words
    for i, token in enumerate(tokens):
        if is_word_token(token):
            lower_token = token.lower()
            if lower_token in dictionary_index.words:
                annotations[i].update(dictionary_index.words[lower_token])
    
    # Then look for compound phrases
    for i in range(len(tokens)):
        if not is_word_token(tokens[i]):
            continue
            
        # Try all possible phrase lengths starting at this position
        max_length = min(
            dictionary_index.max_phrase_length,
            len(tokens) - i
        )
        
        for length in range(2, max_length + 1):
            # Check if we have a sequence of word tokens with separators in between
            phrase_words = []
            valid = True
            for j in range(i, i + length):
                if (j - i) % 2 == 0:  # Should be word token
                    if not is_word_token(tokens[j]):
                        valid = False
                        break
                    phrase_words.append(tokens[j].lower())
                else:  # Should be separator
                    if is_word_token(tokens[j]):
                        valid = False
                        break
            
            if not valid:
                continue
                
            # Check if this word sequence exists in our phrases
            word_sequence = tuple(phrase_words)
            if word_sequence in dictionary_index.phrases:
                # Annotate all participating tokens
                for j in range(i, i + length):
                    annotations[j].update(dictionary_index.phrases[word_sequence])
    
    return list(zip(tokens, annotations))
