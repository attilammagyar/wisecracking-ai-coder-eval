    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index is a Trie (prefix tree) where paths represent normalized,
    concatenated dictionary keys (lowercase, alphanumeric characters only). Each
    node that marks the end of a key stores a set of the corresponding
    dictionary values.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        The root of the Trie.
    """
    root = {'children': {}, 'values': set()}
    for key, value in dictionary.items():
        # Normalize the key by converting to lowercase and keeping only alphanumeric chars.
        # This makes lookups insensitive to case and separators.
        normalized_key = "".join(c.lower() for c in key if c.isalnum())
        if not normalized_key:
            continue

        node = root
        for char in normalized_key:
            if char not in node['children']:
                node['children'][char] = {'children': {}, 'values': set()}
            node = node['children'][char]
        
        node['values'].add(value)
