```python
import collections.abc
import re

def _make_trie_node():
    """Factory for a Trie node."""
    return {'children': {}, 'values': set()}

def _is_word_token(token: str) -> bool:
    """
    Determines if a token is a "word" token.
    A token is considered a word token if it contains at least one alphanumeric
    character. Otherwise, it's considered a separator.
    """
    return any(c.isalnum() for c in token)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index consists of two Trie data structures:
    1. A phrase_trie: for matching sequences of word tokens. Dictionary keys are
       split into word parts, and these sequences are stored in the trie. This
       handles phrases with different separators robustly.
    2. A word_trie: for matching concatenated tokens that form a single compound
       word from the dictionary. Keys are character sequences.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        A tuple containing the roots of the word_trie and phrase_trie.
    """
    word_trie_root = _make_trie_node()
    phrase_trie_root = _make_trie_node()

    for key, value in dictionary.items():
        normalized_key = key.lower()
        
        # Extract word parts from the key. A "word" is a sequence of alphanumeric characters.
        word_parts = re.findall(r'\w+', normalized_key)

        if not word_parts:
            continue

        # Add to phrase_trie for phrase matching.
        node = phrase_trie_root
        for part in word_parts:
            node = node['children'].setdefault(part, _make_trie_node())
        node['values'].add(value)

        # Add to word_trie if key is a single word (no separators).
        # This is for matching compound words formed by concatenating tokens.
        if re.fullmatch(r'\w+', normalized_key):
            node = word_trie_root
            for char in normalized_key:
                node = node['children'].setdefault(char, _make_trie_node())
            node['values'].add(value)
            
    return word_trie_root, phrase_trie_root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    The function performs two types of matching:
    1. Phrase matching: It identifies sequences of "word" tokens in the input
       that match phrases in the dictionary. This is insensitive to the specific
       separators used.
    2. Compound word matching: It identifies sequences of consecutive "word"
       tokens that, when concatenated, match a single compound word entry in
       the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for
                            each token in tokens.
    """
    token_list = list(tokens)
    if not token_list:
        return []

    annotated_tokens = [(token, set()) for token in token_list]
    word_trie_root, phrase_trie_root = dictionary_index

    # --- Phrase Matching ---
    # First, identify all word tokens and their original indices.
    word_tokens_with_indices = [
        (i, token) for i, token in enumerate(token_list) if _is_word_token(token)
    ]

    # Iterate through each word token as a potential start of a phrase.
    for i in range(len(word_tokens_with_indices)):
        start_token_original_idx = word_tokens_with_indices[i][0]
        
        node = phrase_trie_root
        # Try to match a phrase by consuming subsequent word tokens.
        for j in range(i, len(word_tokens_with_indices)):
            word = word_tokens_with_indices[j][1].lower()
            
            if word not in node['children']:
                break  # No such phrase continuation.
            
            node = node['children'][word]
            if node['values']:
                # A match is found. Annotate all original tokens spanning the phrase.
                end_token_original_idx = word_tokens_with_indices[j][0]
                for k in range(start_token_original_idx, end_token_original_idx + 1):
                    annotated_tokens[k][1].update(node['values'])

    # --- Compound Word Matching ---
    # Iterate through each token as a potential start of a compound word.
    for i in range(len(token_list)):
        if not _is_word_token(token_list[i]):
            continue

        node = word_trie_root
        # Try to match by consuming subsequent consecutive word tokens.
        for j in range(i, len(token_list)):
            if not _is_word_token(token_list[j]):
                break  # Separators break compound words.
            
            token_part = token_list[j].lower()
            
            # Traverse the character-based word_trie.
            possible_path = True
            for char in token_part:
                if char not in node['children']:
                    possible_path = False
                    break
                node = node['children'][char]
            
            if not possible_path:
                break

            # A match is registered only at the end of a token.
            if node['values']:
                # The tokens from i to j form a valid compound word.
                for k in range(i, j + 1):
                    annotated_tokens[k][1].update(node['values'])
    
    return annotated_tokens
```
