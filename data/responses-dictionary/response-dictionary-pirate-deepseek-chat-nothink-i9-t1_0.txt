import collections.abc
import re
from typing import Set, Dict, List, Tuple

def _is_word_token(token: str) -> bool:
    """Check if token looks like a word (contains at least one letter/digit)."""
    return bool(re.search(r'[\w]', token))

def _normalize_phrase(phrase: str) -> str:
    """Normalize phrase by lowercasing and normalizing separators."""
    # Convert to lowercase and replace sequences of non-word chars with single space
    normalized = re.sub(r'[^\w]+', ' ', phrase.lower()).strip()
    return normalized

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """Build a search-optimized index from the dictionary."""
    index = {
        'words': {},          # Single word entries (normalized -> set of ids)
        'phrases': {},        # Phrase entries (normalized -> set of ids)
        'phrase_words': set(), # All words that appear in phrases
        'max_phrase_length': 0 # Longest phrase in word count
    }
    
    for phrase, entry_id in dictionary.items():
        normalized = _normalize_phrase(phrase)
        words = normalized.split()
        
        if not words:
            continue
            
        if len(words) == 1:
            # Single word entry
            word = words[0]
            if word not in index['words']:
                index['words'][word] = set()
            index['words'][word].add(entry_id)
        else:
            # Multi-word phrase
            if normalized not in index['phrases']:
                index['phrases'][normalized] = set()
                index['max_phrase_length'] = max(index['max_phrase_length'], len(words))
            index['phrases'][normalized].add(entry_id)
            index['phrase_words'].update(words)
    
    # Pre-sort phrases by length (longest first) for greedy matching
    index['sorted_phrases'] = sorted(index['phrases'].keys(), key=lambda x: -len(x.split()))
    
    return index

def _find_matching_phrases(tokens: List[str], start: int, index: Dict) -> List[Tuple[int, Set]]:
    """Find all phrases in the dictionary that start at the given token position."""
    matches = []
    max_length = min(index['max_phrase_length'], len(tokens) - start)
    
    # Try to build normalized phrases starting at this position
    for length in range(1, max_length + 1):
        # Collect tokens that form the phrase (skip leading/trailing non-word tokens)
        phrase_tokens = tokens[start:start+length]
        
        # Skip if first or last token isn't a word
        if not _is_word_token(phrase_tokens[0]) or not _is_word_token(phrase_tokens[-1]):
            continue
            
        # Build normalized phrase
        normalized_parts = []
        for token in phrase_tokens:
            if _is_word_token(token):
                normalized_parts.append(token.lower())
            else:
                normalized_parts.append(' ')
        normalized_phrase = ' '.join(' '.join(normalized_parts).split())
        
        # Check if this normalized phrase exists in our dictionary
        if normalized_phrase in index['phrases']:
            matches.append((length, index['phrases'][normalized_phrase]))
    
    return matches

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """Annotate tokens with dictionary entries."""
    tokens = list(tokens)  # We need random access
    annotations = [set() for _ in tokens]  # Initialize empty annotations
    
    # First pass: single word annotations
    for i, token in enumerate(tokens):
        if _is_word_token(token):
            lower_word = token.lower()
            if lower_word in dictionary_index['words']:
                annotations[i].update(dictionary_index['words'][lower_word])
    
    # Second pass: phrase annotations
    i = 0
    while i < len(tokens):
        if _is_word_token(tokens[i]):
            # Find all phrases starting at this position
            for length, entry_ids in _find_matching_phrases(tokens, i, dictionary_index):
                # Annotate all tokens in the phrase (including separators)
                for j in range(i, i + length):
                    annotations[j].update(entry_ids)
            i += 1
        else:
            i += 1
    
    return list(zip(tokens, annotations))
