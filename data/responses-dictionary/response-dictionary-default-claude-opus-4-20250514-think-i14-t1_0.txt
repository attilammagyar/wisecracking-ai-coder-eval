def is_word_char(c):
    """Check if character is alphabetic."""
    return c.isalpha()

def is_word_token(token):
    """Check if token contains alphabetic characters."""
    return any(is_word_char(c) for c in token)

def extract_words(text):
    """Extract word tokens from text, ignoring separators."""
    words = []
    current = []
    for c in text:
        if is_word_char(c):
            current.append(c)
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    entries = []
    for key, value in dictionary.items():
        words = extract_words(key)
        # Check if entry has separators by comparing total word length with key length
        has_separators = len(''.join(words)) < len(key.replace(' ', ''))
        entries.append({
            'key': key,
            'value': value,
            'words': words,
            'has_separators': has_separators
        })
    return entries

def try_match_entry(tokens, start, entry):
    """Try to match dictionary entry starting at given position. Returns matched positions or None."""
    words = entry['words']
    if not words:
        return None
    
    if len(words) == 1:
        # Single word - simple case-insensitive match
        if start < len(tokens) and is_word_token(tokens[start]):
            if tokens[start].lower() == words[0].lower():
                return [start]
        return None
    
    # Multiple words
    if entry['has_separators']:
        # Phrase with separators - match words allowing any separators between them
        return match_phrase(tokens, start, words)
    else:
        # Compound word - match by concatenating consecutive tokens
        return match_compound_word(tokens, start, words)

def match_phrase(tokens, start, target_words):
    """Match phrase with flexible separators between words."""
    # Skip leading separators
    while start < len(tokens) and not is_word_token(tokens[start]):
        start += 1
    
    if start >= len(tokens):
        return None
    
    positions = []
    word_idx = 0
    i = start
    
    while i < len(tokens) and word_idx < len(target_words):
        if is_word_token(tokens[i]):
            # Check if this word matches the next target word
            if tokens[i].lower() == target_words[word_idx].lower():
                positions.append(i)
                word_idx += 1
            else:
                return None  # Word mismatch
        else:
            # Non-word token (separator)
            # Include it only if we're between matched words
            if word_idx > 0 and word_idx < len(target_words):
                positions.append(i)
        i += 1
    
    if word_idx == len(target_words):
        # Successfully matched all words - remove any trailing separators
        while positions and not is_word_token(tokens[positions[-1]]):
            positions.pop()
        return positions
    
    return None

def match_compound_word(tokens, start, target_words):
    """Match compound word by concatenating consecutive word tokens."""
    target = ''.join(target_words).lower()
    concat = ''
    positions = []
    i = start
    
    while i < len(tokens) and len(concat) < len(target):
        if is_word_token(tokens[i]):
            concat += tokens[i].lower()
            positions.append(i)
            if concat == target:
                return positions
            elif not target.startswith(concat):
                return None  # No longer a prefix match
        else:
            # Hit a separator - compound words can't have separators
            return None
        i += 1
    
    return None

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try to match dictionary entries starting at each position
    for start in range(n):
        # Try each dictionary entry
        for entry in dictionary_index:
            positions = try_match_entry(tokens, start, entry)
            if positions:
                # Add this entry's ID to all matched positions
                for pos in positions:
                    annotations[pos].add(entry['value'])
    
    # Build result list
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
    return annotated_tokens
