**Approach:**
1. **Trie Structure**: Normalize dictionary entries to lowercase and store in a trie where each node tracks phrase completions.
2. **Active Phrase Tracking**: Maintain ongoing phrase matches across tokens using dynamic programming principles.
3. **Case Insensitivity**: Automatically convert tokens to lowercase during processing.
4. **Compound Phrase Handling**: Include non-word tokens in annotated spans when they appear between valid phrase components.

**Implementation:**

```python
class TrieNode:
    def __init__(self):
        self.children = {}
        self.phrase_ids = set()

def build_dictionary_index(dictionary):
    """Normalize dictionary entries and build a trie structure."""
    root = TrieNode()
    for key, value in dictionary.items():
        # Normalize: lowercase, replace non-alphanumeric with spaces, collapse
        normalized = [' ' if not c.isalnum() else c.lower() for c in key]
        normalized = ' '.join(normalized).split()
        normalized = ' '.join(normalized).strip()
        if not normalized:
            continue
        
        # Insert into trie
        current = root
        for word in normalized.split():
            if word not in current.children:
                current.children[word] = TrieNode()
            current = current.children[word]
        current.phrase_ids.add(value)
    return root

def annotate(tokens, dictionary_index):
    """Annotate tokens using the dictionary trie, handling words and phrases."""
    root = dictionary_index
    active_phrases = []
    annotated = []
    
    for idx, token in enumerate(tokens):
        # Initialize entry if not exists
        while len(annotated) <= idx:
            annotated.append( (token, set()) )
        
        # Check if current token is a word
        original_token = token
        is_word = token and token.isalnum()
        normalized = token.lower() if is_word else None
        
        new_active = []
        
        # Process existing active phrases
        for node, start in active_phrases:
            if is_word:
                if normalized in node.children:
                    next_node = node.children[normalized]
                    new_active.append( (next_node, start) )
                    if next_node.phrase_ids:
                        # Annotate all tokens from start to current
                        for i in range(start, idx+1):
                            annotated[i][1].update(next_node.phrase_ids)
                # Else drop this phrase
            else:
                new_active.append( (node, start) )  # Keep non-word in phrase span
        
        # Check for new phrases starting with this token
        if is_word and normalized in root.children:
            new_node = root.children[normalized]
            new_active.append( (new_node, idx) )
            if new_node.phrase_ids:
                annotated[idx][1].update(new_node.phrase_ids)
        
        active_phrases = new_active
    
    return annotated
```

3. **Non-Word Handling**: Preserves context for phrases containing separators

5. **Case-Insensitive Search**: Automatic normalization to lowercase

- **Annotation Phase**: O(N) time where N is token count, with trie operations being O(1) amortized
- **Space Complexity**: O(M) for trie storage
