    """Normalize dictionary entry to lowercase and replace non-alnum with spaces"""
    norm = []
    word = []
    for c in entry.lower():
        if c.isalnum():
            word.append(c)
        else:
            if word:
                norm.append(''.join(word))
                word = []
    if word:
        norm.append(''.join(word))
    return ' '.join(norm).strip()

def build_dictionary_index(dictionary):
    """Build mapping from normalized entries to original values"""
    index = {}
    for entry, value in dictionary.items():
        normalized = normalize_entry(entry)
        if normalized not in index:
            index[normalized] = set()
        index[normalized].add(value)
    return index

def normalize_span(span):
    """Normalize a sequence of tokens to lowercase with spaces only"""
    cleaned = []
    for tok in span:
        cleaned.append(''.join([c.lower() for c in tok if c.isalnum()]))
    return ' '.join(cleaned).strip()

def annotate(tokens, dictionary_index):
    """Annotate tokens based on dictionary entries"""
    result = [set() for _ in tokens]
    is_sep = [any(not c.isalnum() for c in tok) for tok in tokens]

    for i in range(len(tokens)):
        for length in range(1, len(tokens) - i + 1):
            end = i + length - 1
            if end >= len(tokens):
                break
            if not is_sep[i] and not is_sep[end]:
                span = tokens[i:end+1]
                normalized = normalize_span(span)
                if normalized in dictionary_index:
                    for pos in range(i, end+1):
                        result[pos].update(dictionary_index[normalized])

    return [(tok, s.copy()) for tok, s in zip(tokens, result)]
