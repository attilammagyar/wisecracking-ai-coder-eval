```python
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries': {},  # normalized form -> list of (original_form, entry_id)
        'word_sequences': {}  # tuple of words -> list of (original_form, entry_id, has_separators)
    }
    
    def extract_words(text):
        """Extract words from text, treating non-alphanumeric as separators."""
        words = []
        current_word = []
        for char in text:
            if char.isalnum():
                current_word.append(char)
            else:
                if current_word:
                    words.append(''.join(current_word).lower())
                    current_word = []
        if current_word:
            words.append(''.join(current_word).lower())
        return words
    
    for entry, entry_id in dictionary.items():
        # Store normalized single entry
        normalized = entry.lower()
        if normalized not in index['entries']:
            index['entries'][normalized] = []
        index['entries'][normalized].append((entry, entry_id))
        
        # Extract and store word sequence
        words = extract_words(entry)
        if words:
            has_separators = any(not c.isalnum() for c in entry)
            word_tuple = tuple(words)
            if word_tuple not in index['word_sequences']:
                index['word_sequences'][word_tuple] = []
            index['word_sequences'][word_tuple].append((entry, entry_id, has_separators))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Ensure we can index
    annotations = [set() for _ in tokens]
    
    def is_word_token(token):
        """Check if token contains any alphanumeric characters."""
        return any(c.isalnum() for c in token)
    
    def extract_words_from_tokens(token_slice):
        """Extract word tokens and their positions from a slice of tokens."""
        words = []
        positions = []
        for i, token in enumerate(token_slice):
            if is_word_token(token):
                words.append(token.lower())
                positions.append(i)
        return words, positions
    
    # Try all possible token sequences
    for start in range(len(tokens)):
        for end in range(start + 1, len(tokens) + 1):
            token_slice = tokens[start:end]
            
            # Try direct match (for single tokens)
            if end - start == 1:
                normalized = tokens[start].lower()
                if normalized in dictionary_index['entries']:
                    for _, entry_id in dictionary_index['entries'][normalized]:
                        annotations[start].add(entry_id)
            
            # Try compound word match (concatenated adjacent word tokens)
            words, positions = extract_words_from_tokens(token_slice)
            if len(words) > 1 and len(positions) == len(token_slice):
                # All tokens are words and adjacent
                concatenated = ''.join(words)
                if concatenated in dictionary_index['entries']:
                    for _, entry_id in dictionary_index['entries'][concatenated]:
                        for i in range(start, end):
                            annotations[i].add(entry_id)
            
            # Try phrase match (word sequence)
            if words:
                word_tuple = tuple(words)
                if word_tuple in dictionary_index['word_sequences']:
                    for _, entry_id, has_separators in dictionary_index['word_sequences'][word_tuple]:
                        if len(words) == 1:
                            # Single word match
                            annotations[start].add(entry_id)
                        elif has_separators or len(words) > 1:
                            # Phrase match - annotate from first word to last word
                            first_word_pos = start + positions[0]
                            last_word_pos = start + positions[-1]
                            for i in range(first_word_pos, last_word_pos + 1):
                                annotations[i].add(entry_id)
    
    annotated_tokens = [(token, annot) for token, annot in zip(tokens, annotations)]
    return annotated_tokens
```
