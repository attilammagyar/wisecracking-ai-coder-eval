import collections.abc

# ---------- Helpers ----------

def _is_word_token(tok: str) -> bool:
    # Word tokens are purely alphanumeric (ASCII/Unicode aware via isalnum()).
    # Separators (spaces, punctuation, Markdown, etc.) are not.
    return tok.isalnum()


def _split_words_lower(s: str) -> list[str]:
    # Split dictionary keys into "word units" by any non-alphanumeric chars.
    # Lowercased with casefold() to be fully case-insensitive.
    words = []
    buf = []
    for ch in s:
        if ch.isalnum():
            buf.append(ch)
        else:
            if buf:
                words.append("".join(buf).casefold())
                buf = []
    if buf:
        words.append("".join(buf).casefold())
    return words


class _CharTrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: dict[str, _CharTrieNode] = {}
        self.ids: set = set()


class _WordTrieNode:
    __slots__ = ("children", "ids")
    def __init__(self):
        self.children: dict[str, _WordTrieNode] = {}
        self.ids: set = set()


class _DictIndex:
    __slots__ = ("char_root", "phrase_root", "has_single_words", "has_phrases")
    def __init__(self, char_root: _CharTrieNode, phrase_root: _WordTrieNode,
                 has_single_words: bool, has_phrases: bool):
        self.char_root = char_root
        self.phrase_root = phrase_root
        self.has_single_words = has_single_words
        self.has_phrases = has_phrases


# ---------- Public API ----------

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Gather normalized entries.
    # For single-word entries we aggregate IDs by normalized word,
    # to keep multiple source spellings unified.
    single_words: dict[str, set] = {}
    phrase_root = _WordTrieNode()

    has_phrases = False
    has_single_words = False

    for key, entry_id in dictionary.items():
        words = _split_words_lower(key)
        if not words:
            continue
        if len(words) == 1:
            has_single_words = True
            w = words[0]
            bucket = single_words.setdefault(w, set())
            bucket.add(entry_id)
        else:
            has_phrases = True
            node = phrase_root
            for w in words:
                nxt = node.children.get(w)
                if nxt is None:
                    nxt = _WordTrieNode()
                    node.children[w] = nxt
                node = nxt
            node.ids.add(entry_id)

    # Build character trie for single words.
    char_root = _CharTrieNode()
    if has_single_words:
        for w, ids in single_words.items():
            node = char_root
            for ch in w:
                nxt = node.children.get(ch)
                if nxt is None:
                    nxt = _CharTrieNode()
                    node.children[ch] = nxt
                node = nxt
            node.ids.update(ids)

    return _DictIndex(char_root=char_root, phrase_root=phrase_root,
                      has_single_words=has_single_words, has_phrases=has_phrases)


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    idx: _DictIndex = dictionary_index  # type: ignore[assignment]
    toks = list(tokens)
    n = len(toks)
    ann = [set() for _ in range(n)]

    # 1) Single-word entries (and concatenations of adjacent word tokens) via char trie.
    if idx.has_single_words:
        i = 0
        while i < n:
            if not _is_word_token(toks[i]):
                i += 1
                continue
            # Find run of consecutive word tokens [i..j]
            j = i
            while j + 1 < n and _is_word_token(toks[j + 1]):
                j += 1

            # For each start position in the run, walk the char trie across tokens
            # and only accept matches at token boundaries.
            for start in range(i, j + 1):
                node = idx.char_root
                # If the trie is empty, skip quickly
                if not node.children and not node.ids:
                    break
                # Extend through end tokens
                end = start
                while end <= j:
                    # Try to consume the whole token at 'end'
                    t = toks[end].casefold()
                    ok = True
                    for ch in t:
                        child = node.children.get(ch)
                        if child is None:
                            ok = False
                            break
                        node = child
                    if not ok:
                        break  # no further extension possible from this start
                    # We are exactly at a token boundary; accept if terminal
                    if node.ids:
                        ids = node.ids
                        for k in range(start, end + 1):
                            ann[k].update(ids)
                    end += 1
            i = j + 1

    # 2) Multi-word phrases via word-trie, ignoring separators between words.
    if idx.has_phrases:
        # Collect indices and normalized forms of word tokens
        word_indices = []
        word_norms = []
        for pos, tok in enumerate(toks):
            if _is_word_token(tok):
                word_indices.append(pos)
                word_norms.append(tok.casefold())

        W = len(word_indices)
        root = idx.phrase_root
        for p in range(W):
            node = root
            # Early skip if no children at root
            if not node.children and not node.ids:
                break
            for q in range(p, W):
                w = word_norms[q]
                node = node.children.get(w)
                if node is None:
                    break
                if node.ids:
                    ids = node.ids
                    # Annotate the words in the span
                    for k in range(p, q + 1):
                        ann[word_indices[k]].update(ids)
                    # Annotate only inner separators between first and last word
                    a = word_indices[p]
                    b = word_indices[q]
                    # Inner tokens strictly between a and b that are non-words
                    for t in range(a + 1, b):
                        if not _is_word_token(toks[t]):
                            ann[t].update(ids)

    return [(toks[i], ann[i]) for i in range(n)]
