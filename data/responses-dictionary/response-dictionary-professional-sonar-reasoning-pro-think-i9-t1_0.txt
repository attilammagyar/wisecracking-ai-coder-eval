   - **Normalization**: Convert all dictionary entries to lowercase, replace non-alphanumeric separators with spaces, and split into word lists.

   - **Single-Word Check**: For individual tokens, directly check against single-word mappings.

def build_dictionary_index(dictionary: dict) -> dict:
    """Preprocess dictionary into normalized word tuples and single-word mappings."""
    word_map = {}    # Map: {tuple(words), set(entry_ids)}
    single_word_map = {}  # Map: {word_lower: set(entry_ids)}
    max_words = 0
    
    for entry, entry_id in dictionary.items():
        # Normalize: replace separators, lowercase, split into words
        normalized = ''.join([c if c.isalnum() else ' ' for c in entry]).lower().strip()
        words = normalized.split()
        if not words:
            continue
        
        # Compound phrase mapping
        word_tuple = tuple(words)
        if word_tuple not in word_map:
            word_map[word_tuple] = set()
        word_map[word_tuple].add(entry_id)
        max_words = max(max_words, len(words))
        
        # Single word mapping
        if len(words) == 1:
            word = words[0]
            if word not in single_word_map:
                single_word_map[word] = set()
            single_word_map[word].add(entry_id)
    
    return {
        'word_map': word_map,
        'single_word_map': single_word_map,
        'max_phrase_length': max_words
    }

def annotate(tokens: list[str], index: dict) -> list[tuple[str, set]]:
    """Annotate tokens against preprocessed dictionary index."""
    annotations = []
    max_length = index['max_phrase_length']
    
    for i in range(len(tokens)):
        entry_ids = set()
        
        # Check single-word case
        token_lower = tokens[i].lower()
        if token_lower in index['single_word_map']:
            entry_ids.update(index['single_word_map'][token_lower])
        
        # Check compound phrase cases
        for j in range(1, max_length + 1):
            end = i + j
            if end > len(tokens):
                break
            seq = tokens[i:end]
            
            # Generate normalized compound phrase
            normalized = ''.join([t.lower() if t.isalnum() else ' ' for t in seq]).strip().replace('  ', ' ')
            words = normalized.split()
            word_tuple = tuple(words)
            if word_tuple in index['word_map']:
                entry_ids.update(index['word_map'][word_tuple])
        
        annotations.append((tokens[i], entry_ids))
    
    return annotations

     - `single_word_map`: Direct mapping from single words to entry IDs for quick lookups.

2. **Annotation Logic**:
   - **Single-Word Check**: For each token, we check if its lowercase version exists in `single_word_map`.
   - **Compound Phrase Check**: For sequences of tokens up to the maximum phrase length, we generate normalized versions (with/without separators) and check against `word_map`. This ensures we handle both spaced and concatenated phrases.
