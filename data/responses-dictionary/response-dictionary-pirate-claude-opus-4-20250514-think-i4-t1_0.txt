import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for key, value in dictionary.items():
        # Extract words from the key, ignoring non-word characters
        words = []
        current_word = []
        
        for char in key:
            if char.isalnum():
                current_word.append(char)
            elif current_word:
                words.append(''.join(current_word).lower())
                current_word = []
        
        if current_word:
            words.append(''.join(current_word).lower())
        
        # Store as tuple for hashability
        word_tuple = tuple(words)
        if word_tuple not in index:
            index[word_tuple] = []
        index[word_tuple].append(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        return any(c.isalnum() for c in token)
    
    # Try all possible starting positions
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
        
        # Build word sequences starting from this position
        words = []
        word_positions = []
        
        for i in range(start, n):
            if is_word_token(tokens[i]):
                words.append(tokens[i].lower())
                word_positions.append(i)
                
                # Check if this word sequence exists in dictionary
                word_tuple = tuple(words)
                if word_tuple in dictionary_index:
                    for value in dictionary_index[word_tuple]:
                        if len(word_positions) == 1:
                            # Single word match
                            annotations[word_positions[0]].add(value)
                        else:
                            # Multi-word match
                            # Check if it's a compound word (consecutive) or phrase
                            first_pos = word_positions[0]
                            last_pos = word_positions[-1]
                            
                            if last_pos - first_pos == len(word_positions) - 1:
                                # Compound word: consecutive word tokens
                                for pos in word_positions:
                                    annotations[pos].add(value)
                            else:
                                # Phrase: includes separators between words
                                for j in range(first_pos, last_pos + 1):
                                    annotations[j].add(value)
    
    annotated_tokens = [(token, annotations[i]) for i, token in enumerate(tokens)]
    return annotated_tokens
