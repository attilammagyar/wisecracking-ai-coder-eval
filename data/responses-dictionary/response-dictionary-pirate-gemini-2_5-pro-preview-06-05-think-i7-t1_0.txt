import collections.abc
import re

class _TrieNode:
    """A node in the Trie, me hearty."""
    def __init__(self):
        self.children = {}
        self.values = set()

def _is_word_token(token: str) -> bool:
    """Check if a token be a word or just some gibberish."""
    return any(c.isalnum() for c in token)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    phrase_trie = _TrieNode()
    compound_trie = _TrieNode()

    for key, value in dictionary.items():
        norm_key = key.lower()
        words = re.findall(r'\w+', norm_key)

        if not words:
            continue

        # A key is considered a phrase if it contains internal non-word characters.
        # Otherwise, it's a single word or a compound word.
        is_phrase = len(words) > 1 or (len(words) == 1 and words[0] != re.sub(r'[\W_]+', '', norm_key))

        if is_phrase:
            # Add to the phrase trie (word-based)
            node = phrase_trie
            for word in words:
                if word not in node.children:
                    node.children[word] = _TrieNode()
                node = node.children[word]
            node.values.add(value)
        else: # It's a compound word or a single word
            # Add to the compound trie (char-based)
            compound_word = words[0]
            node = compound_trie
            for char in compound_word:
                if char not in node.children:
                    node.children[char] = _TrieNode()
                node = node.children[char]
            node.values.add(value)

    return phrase_trie, compound_trie

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    token_list = list(tokens)
    if not token_list:
        return []

    annotated_tokens = [(token, set()) for token in token_list]
    phrase_trie, compound_trie = dictionary_index
    n = len(token_list)

    # 1. Phrase matching using the word-based trie
    word_tokens = [(i, token.lower()) for i, token in enumerate(token_list) if _is_word_token(token)]

    for i in range(len(word_tokens)):
        node = phrase_trie
        for j in range(i, len(word_tokens)):
            original_idx, word = word_tokens[j]
            if word not in node.children:
                break
            node = node.children[word]
            if node.values:
                start_token_idx = word_tokens[i][0]
                end_token_idx = original_idx
                for k in range(start_token_idx, end_token_idx + 1):
                    annotated_tokens[k][1].update(node.values)

    # 2. Compound word matching using the character-based trie
    for i in range(n):

        match_start_idx = -1

                for k in range(match_start_idx, j + 1):
                    annotated_tokens[k][1].update(node.values)
