I'll normalize dictionary entries by extracting their words (ignoring separators and case) and store them as tuples. This allows matching phrases regardless of the specific separators used.

### Lookup Algorithm
1. Try all possible token subsequences
2. For each subsequence, extract the words and check if they match a dictionary entry
3. Handle two cases:
   - **Compound phrases**: Match word sequences ignoring separators
   - **Compound words**: Match concatenated consecutive word tokens
4. Only annotate tokens from the first word to the last word in a match (excluding leading/trailing separators)

### Implementation

```python
import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for key, value in dictionary.items():
        # Extract words from the key (case-insensitive)
        words = []
        current_word = []
        
        for char in key.lower():
            if char.isalnum():
                current_word.append(char)
            else:
                if current_word:
                    words.append(''.join(current_word))
                    current_word = []
        if current_word:
            words.append(''.join(current_word))
        
        if words:
            # Store as tuple of words
            word_tuple = tuple(words)
            if word_tuple not in index:
                index[word_tuple] = set()
            index[word_tuple].add(value)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Helper function to check if a token is a word token
    def is_word_token(token):
        return any(c.isalnum() for c in token)
    
    # Try all possible token sequences
    for start in range(n):
        for end in range(start + 1, n + 1):
            token_slice = tokens[start:end]
            
            # Find first and last word tokens in the slice
            first_word_idx = None
            last_word_idx = None
            word_indices = []
            
            for i, token in enumerate(token_slice):
                if is_word_token(token):
                    if first_word_idx is None:
                        first_word_idx = i
                    last_word_idx = i
                    word_indices.append(i)
            
            # Skip if no word tokens
            if first_word_idx is None:
                continue
            
            # Extract words
            words = []
            for token in token_slice:
                if is_word_token(token):
                    words.append(token.lower())
            
            # Check for matches
            matched_values = set()
            
            # 1. Check as word sequence (handles phrases)
            word_tuple = tuple(words)
            if word_tuple in dictionary_index:
                matched_values.update(dictionary_index[word_tuple])
            
            # 2. If all word tokens are consecutive, also check as concatenated word
            if len(word_indices) > 1:
                consecutive = all(word_indices[i] == word_indices[i-1] + 1 
                                for i in range(1, len(word_indices)))
                
                if consecutive:
                    # Concatenate and check
                    concat_word = ''.join(words)
                    concat_tuple = (concat_word,)
                    if concat_tuple in dictionary_index:
                        matched_values.update(dictionary_index[concat_tuple])
            
            # Annotate tokens if we found matches
            if matched_values:
                for i in range(first_word_idx, last_word_idx + 1):
                    annotations[start + i].update(matched_values)
    
    # Create result
    annotated_tokens = []
    for token, annotation_set in zip(tokens, annotations):
        annotated_tokens.append((token, annotation_set))
    
    return annotated_tokens
```
