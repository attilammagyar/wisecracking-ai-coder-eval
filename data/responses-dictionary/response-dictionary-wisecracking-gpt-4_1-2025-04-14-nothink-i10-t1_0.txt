import collections.abc

# ------ Helper Class: Trie Node ------
class TrieNode:
    __slots__ = ('children', 'separators', 'entries')
    def __init__(self):
        # Children: {normalized_word: TrieNode}
        self.children = {}
        # For moving over a separator token: {separator_marker: TrieNode}
        self.separators = {}  # (future extension, not strictly needed)
        # Set of dictionary entry IDs if this node represents a word/phrase end
        self.entries = set()

def is_word_token(token):
    # Token with at least an alphanumeric character is considered a word.
    return any(c.isalnum() for c in token)

def normalize_token(token):
    return token.casefold()  # case-insensitive compare

def tokenize_phrase(phrase):
    # Split a phrase into a sequence of word/separator tokens.
    # e.g. "AAA BBB" -> ["AAA", " ", "BBB"]
    tokens = []
    current = ''
    prev_is_word = None
    for c in phrase:
        is_word = c.isalnum()
        if prev_is_word is None:
            current = c
            prev_is_word = is_word
            continue
        if is_word == prev_is_word:
            current += c
        else:
            tokens.append(current)
            current = c
            prev_is_word = is_word
    if current:
        tokens.append(current)
    return tokens

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and compound phrases.
    """
    root = TrieNode()
    # Store the longest phrase length (in tokens) for optimization.
    max_phrase_len = 1

    for phrase, entry_id in dictionary.items():
        # Tokenize phrase by natural language rules
        tokens = tokenize_phrase(phrase)
        types = [is_word_token(t) for t in tokens]
        node = root
        for i, token in enumerate(tokens):
            normalized = normalize_token(token)
            is_word = types[i]
            key = ("W", normalized) if is_word else ("S", None)
            if is_word:
                if key not in node.children:
                    node.children[key] = TrieNode()
                node = node.children[key]
            else:
                # All separators are muxed into a single path (since any sequence of separators is acceptable)
                if key not in node.children:
                    node.children[key] = TrieNode()
                node = node.children[key]
        node.entries.add(entry_id)
        if len(tokens) > max_phrase_len:
            max_phrase_len = len(tokens)
    # Index is just root node and the max length
    return {"root": root, "max_phrase_len": max_phrase_len}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotated = [set() for _ in range(n)]
    # For optimization: classify tokens as word or separator up front
    token_types = [is_word_token(tok) for tok in tokens]

    root = dictionary_index["root"]
    max_phrase_len = dictionary_index["max_phrase_len"]

    # We search for all dictionary entries that match at each word position
    i = 0
    while i < n:
        # We only begin searches at word tokens!
        if not token_types[i]:
            i += 1
            continue

        # For each window starting at position i, try to match in trie.
        node = root
        phrase_start = i
        phrase_end = i

        j = i
        first_word_idx = i  # Actual start of phrase (first word)
        last_word_idx = i   # Will move as we progress
        while j < n:
            tok = tokens[j]
            typ = token_types[j]
            if typ:
                key = ("W", normalize_token(tok))
            else:
                key = ("S", None)  # any separator

            if key not in node.children:
                break
            node = node.children[key]
            phrase_end = j

            # Update last_word_idx
            if typ:
                last_word_idx = j

            # If a phrase ends here, annotate all participating tokens (inner separators too)
            if node.entries:
                # Find phrase token range (do not include leading/trailing separators)
                start = first_word_idx
                end = last_word_idx
                # Annotate the full range: from start to end, including inner separators
                for k in range(start, end + 1):
                    # Annotate word tokens and inner separators
                    if token_types[k]:
                        annotated[k].update(node.entries)
                    else:
                        # Annotate only if not a leading/trailing separator
                        if k != start and k != end:
                            annotated[k].update(node.entries)
            j += 1
            # If the phrase window exceeds some max reasonable length, stop.
            # (In practice, since tokens are short and phrase index is tight, this keeps us fast.)
            if j - i > max_phrase_len:
                break
        i += 1

    # Now, single-word annotations. Go over tokens again, annotate from trie.
    node = root
    for idx, tok in enumerate(tokens):
        if not token_types[idx]:
            # Separators can't be single-word dictionary entries
            continue
        key = ("W", normalize_token(tok))
        node1 = node.children.get(key)
        if node1 and node1.entries:
            annotated[idx].update(node1.entries)

    # Final answer
    result = []
    for tok, anns in zip(tokens, annotated):
        result.append((tok, set(anns)))
    return result
