```python
import collections.abc


def build_dictionary_index(dictionary: collections.abc.Mapping) -> dict:
    index = {}
    for key, value in dictionary.items():
        normalized = ''.join(c if c.isalnum() else ' ' for c in key.lower())
        words = normalized.split()
        if not words:
            continue
        tuple_key = tuple(words)
        index[tuple_key] = value
    return index


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    token_types = []
    token_normals = []

    for t in tokens:
        normalized = ''.join(c if c.isalnum() else ' ' for c in t.lower())
        token_normals.append(normalized)
        words = normalized.split()
        token_types.append(len(words) > 0)

    annotations = [set() for _ in tokens]

    # Single word checks
    for i in range(len(tokens)):
        if not token_types[i]:
            continue

        words = token_normals[i].split()
        if len(words) != 1:
            continue

        single_word = tuple(words)
        if single_word in dictionary_index:
            annotations[i].add(dictionary_index[single_word])

    # Compound checks
    for i in range(len(tokens)):
        max_len = i + 1
        for length in range(max_len, 0, -1):
            start = i - length + 1
            if start < 0:
                continue

            sub_types = token_types[start:i+1]
            first = sub_types[0]
            last = sub_types[-1]

            if first and not last or not first and last:
                continue

            word_list = []
            for t in range(start, i+1):
                if token_types[t]:
                    words = token_normals[t].split()
                    if words:
                        word_list.append(words[0])

            if len(word_list) < 1:
                continue

            tuple_key = tuple(word_list)
            if tuple_key in dictionary_index:
                entry_id = dictionary_index[tuple_key]
                for t in range(start, i+1):
                    annotations[t].add(entry_id)

    return [(t, s) for t, s in zip(tokens, annotations)]
```
