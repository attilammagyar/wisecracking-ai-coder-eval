def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = []
    
    for key, value in dictionary.items():
        # Extract all words from the key
        words = re.findall(r'\w+', key)
        words_lower = [w.lower() for w in words]
        
        # Check if key contains separators
        has_separators = bool(re.search(r'\W', key))
        
        index.append({
            'words': words_lower,
            'has_separators': has_separators,
            'entry_id': value
        })
    
    return index

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()

    n = len(tokens)

        return bool(re.match(r'^\w+$', token))

                if is_word(tokens[i]) and tokens[i].lower() == word:
                    annotations[i].add(entry_id)

            target = ''.join(words)

                if not is_word(tokens[start]):
                    continue
                
                # Try concatenating from this position
                concat = ""
                pos = start
                
                while pos < n and is_word(tokens[pos]):
                    concat += tokens[pos].lower()
                    
                    if concat == target:
                        # Match found
                        for i in range(start, pos + 1):
                            annotations[i].add(entry_id)
                        break
                    elif len(concat) >= len(target):
                        break
                    
                    pos += 1
        
        else:
            # Compound phrase (with separators)
            for start in range(n):
                if not is_word(tokens[start]) or tokens[start].lower() != words[0]:
                    continue
                
                # Match phrase from here
                word_idx = 0
                pos = start
                match_positions = []
                
                while pos < n and word_idx < len(words):
                    if is_word(tokens[pos]):
                        if tokens[pos].lower() == words[word_idx]:
                            match_positions.append(pos)
                            word_idx += 1
                        else:
                            break
                    else:
                        # Separator
                        if 0 < word_idx < len(words):
                            match_positions.append(pos)
                    
                    pos += 1
                
                if word_idx == len(words):
                    for i in match_positions:
                        annotations[i].add(entry_id)
    
    annotated_tokens = [(token, annotations[i]) for i, token in enumerate(tokens)]
    return annotated_tokens
