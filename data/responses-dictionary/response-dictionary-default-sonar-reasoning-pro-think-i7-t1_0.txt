import collections.abc
from typing import Iterable, Tuple, Set

def build_dictionary_index(dictionary: collections.abc.Mapping) -> dict:
    """Build a normalized index for fast dictionary lookups."""
    # Store entries as lowercase tokens (exact sequence matching)
    # Case insensitivity is handled via lowercase conversion
    # Maximum entry length is tracked for optimization
    entries = []
    max_length = 0
    for key, value in dictionary.items():
        key_lower = key.lower()
        tokens = key_lower.split(' ')  # Split on spaces (assumes words are space-separated)
        entries.append((tuple(tokens), value))
        max_length = max(max_length, len(tokens))
    
    return {
        'entries': entries,
        'max_length': max_length
    }

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[Tuple[str, Set]]:
    """Annotate tokens using the prebuilt dictionary index."""
    entries = dictionary_index['entries']
    max_length = dictionary_index['max_length']
    token_list = list(tokens)  # Convert to list for indexing
    
    # Annotations: {index: set of entry values}
    annotation_map = [[] for _ in range(len(token_list))]
    
    # Check single-token entries
    for i in range(len(token_list)):
        token_lower = token_list[i].lower()
        # Match against entries with single token
        for (entry_tokens, entry_val) in entries:
            if len(entry_tokens) == 1 and entry_tokens[0] == token_lower:
                annotation_map[i].append(entry_val)
    
    # Check multi-token entries using sliding window
    for i in range(len(token_list)):
        # Max window size backward from current position
        for entry in entries:
            entry_tokens, entry_val = entry
            entry_len = len(entry_tokens)
            if entry_len > max_length:
                continue  # Skip entries longer than max tracked
            
            # Check all possible windows ending at position i
            start = i - entry_len + 1
            if start < 0:
                continue
            
            # Compare tokens in the window
            match = True
            for pos in range(entry_len):
                token = token_list[start + pos].lower()
                if token != entry_tokens[pos]:
                    match = False
                    break
            
            if match:
                # Annotate all tokens in this window (entry_tokens length)
                for pos in range(start, i + 1):
                    annotation_map[pos].append(entry_val)
    
    # Convert annotation lists to sets
    return [(token, set(ids)) for token, ids in zip(token_list, annotation_map)]
