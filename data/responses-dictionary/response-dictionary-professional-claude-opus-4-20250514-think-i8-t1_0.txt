def is_word_token(token):
    """Check if a token contains any alphanumeric character."""
    return any(c.isalnum() for c in token)

def normalize_word(word):
    """Normalize a word for case-insensitive matching."""
    return word.lower()

def extract_words(text):
    """Extract word tokens from text, ignoring separators."""
    words = []
    current = []
    for c in text:
        if c.isalnum():
            current.append(c)
        elif current:
            words.append(''.join(current))
            current = []
    if current:
        words.append(''.join(current))
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'exact': {},      # For single word and compound word matches
        'phrases': {},    # For phrase matches (word sequences)
        'max_words': 0    # Maximum number of words in any phrase
    }
    
    for key, value in dictionary.items():
        words = extract_words(key)
        normalized_words = [normalize_word(w) for w in words]

            normalized = ''.join(normalized_words)
            if normalized not in index['exact']:

        else:

            concat_words = [normalize_word(tokens[i])]
            j = i + 1
            while j < n and is_word_token(tokens[j]):
                concat_words.append(normalize_word(tokens[j]))
                concatenated = ''.join(concat_words)
                if concatenated in dictionary_index['exact']:
                    # Found compound word match
                    for k in range(i, j + 1):
                        annotations[k].update(dictionary_index['exact'][concatenated])
                j += 1
        
        # Phrase matching
        if is_word_token(tokens[i]):
            words = [normalize_word(tokens[i])]
            first_word_pos = i
            last_word_pos = i
            
            for j in range(i + 1, n):
                if is_word_token(tokens[j]):
                    words.append(normalize_word(tokens[j]))
                    last_word_pos = j
                    
                    if len(words) > dictionary_index['max_words']:
                        break
                    
                    word_tuple = tuple(words)
                    if word_tuple in dictionary_index['phrases']:
                        # Found phrase match - annotate from first to last word
                        for k in range(first_word_pos, last_word_pos + 1):
                            annotations[k].update(dictionary_index['phrases'][word_tuple])
    
    return [(tokens[i], annotations[i]) for i in range(n)]
