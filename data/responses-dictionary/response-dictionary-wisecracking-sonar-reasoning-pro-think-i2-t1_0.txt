import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """Build a trie-based index optimized for phrase matching."""
    trie = {"children": {}, "entries": set()}
    for entry, entry_id in dictionary.items():
        words = [word.lower() for word in re.split(r'\W+', entry.lower()) if word]
        node = trie
        for word in words:
            if word not in node["children"]:
                node["children"][word] = {"children": {}, "entries": set()}
            node = node["children"][word]
        node["entries"].add(entry_id)
    return trie

def annotate(tokens: collections.abc.Iterable[str], dictionary_index) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """Annotate tokens using the dictionary index."""
    word_info = []  # List of (original_index, lowercase_word)
    for original_index, token in enumerate(tokens):
        if token.isalnum():  # Treat non-alphanumeric as separators
            word_info.append((original_index, token.lower()))
    
    active_nodes = []
    annotations = [set() for _ in tokens]
    
    for i, (original_pos, word) in enumerate(word_info):

            if word in node["children"]:
