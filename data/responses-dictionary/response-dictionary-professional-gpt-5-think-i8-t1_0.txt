Main ideas

- Normalize the dictionary to be case-insensitive and based on words, not separators. Each entry key is split into a sequence of alphanumeric “words” (separators and punctuation are ignored). For example, "AAA, BBB" becomes ("aaa", "bbb"), and "AAABBB" becomes ("aaabbb").
- Build two structures:
  - A trie over multi-word phrases (sequences of length ≥ 2). This lets us match phrases over the token stream while being insensitive to the separators that appear between words in the text.
  - A hash map for single-word entries (length = 1), and a set of all its prefixes. The prefix set allows fast pruning when trying to match single-word dictionary entries by concatenating adjacent word tokens (only when there are no separators between them).
- Annotation algorithm:
  - Classify each token as a “word token” (token.isalnum()) or a separator.
  - Annotate direct single-word matches for word tokens.
  - Annotate single-word matches formed by concatenating adjacent word tokens with no separators in between (e.g., ["AAA", "BBB"] → "aaabbb").
  - Annotate multi-word phrase matches by scanning the sequence of word tokens with the trie, ignoring separators between words. If a phrase matches spanning word token positions s..e, annotate all tokens in the inclusive range s..e (thus annotating all inner separators, but not any leading or trailing separators outside the phrase).
- Overlapping and nested phrases are naturally supported: we record all matches found during scanning and union all annotations into per-token sets.
- Mid-token matches are impossible because we only ever match at token boundaries (tokens are never split; concatenation uses whole adjacent word tokens only).

import collections.abc

# ----------------------------- Helpers and Index -----------------------------

class _TrieNode:
    __slots__ = ("children", "out")
    def __init__(self):
        self.children = {}  # word -> _TrieNode
        self.out = set()    # set of dictionary ids ending at this node


def _split_words(s: str) -> tuple[str, ...]:
    """
    Split a dictionary key into alphanumeric 'words' (lowercased).
    Non-alphanumeric characters are treated as separators and discarded.
    """
    words = []
    cur = []
    for ch in s:
        if ch.isalnum():
            cur.append(ch.lower())
        else:
            if cur:
                words.append("".join(cur))
                cur = []
    if cur:
        words.append("".join(cur))
    return tuple(words)


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index:
      - phrase_trie: trie of multi-word phrases (len(words) >= 2)
      - single_map:  dict word -> set(ids) for single-word entries
      - single_prefixes: set of all prefixes of single_map keys (for pruning)
    """
    root = _TrieNode()
    single_map = {}       # word -> set(ids)
    single_prefixes = set()

    for key, val in dictionary.items():
        words = _split_words(key)
        if not words:
            continue
        if len(words) == 1:
            w = words[0]
            s = single_map.get(w)
            if s is None:
                s = set()
                single_map[w] = s
                # add prefixes once per new word
                for i in range(1, len(w) + 1):
                    single_prefixes.add(w[:i])
            s.add(val)
        else:
            node = root
            for w in words:
                child = node.children.get(w)
                if child is None:
                    child = _TrieNode()
                    node.children[w] = child
                node = child
            node.out.add(val)

    return (root, single_map, single_prefixes)


# -------------------------------- Annotation ---------------------------------

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with dictionary entries (individual words and compound phrases).
    """
    tokens = list(tokens)
    n = len(tokens)
    annotated_tokens = [set() for _ in range(n)]
    if n == 0:
        return []

    phrase_root, single_map, single_prefixes = dictionary_index

    # Precompute classifications and lowercase forms
    is_word = [t.isalnum() for t in tokens]
    lower_tokens = [t.lower() for t in tokens]

    # 1) Direct single-word matches on individual tokens
    for i, (t_is_word, lt) in enumerate(zip(is_word, lower_tokens)):
        if t_is_word:
            vals = single_map.get(lt)
            if vals:
                annotated_tokens[i].update(vals)

    # 2) Single-word matches by concatenating adjacent word tokens with no separators
    i = 0
    while i < n:
        if not is_word[i]:
            i += 1
            continue
        j = i
        while j < n and is_word[j]:
            j += 1
        # Now [i, j) is a maximal run of contiguous word tokens with no separators
        run = lower_tokens[i:j]
        run_len = len(run)
        # Try all starts in this run
        for s in range(run_len):
            cur = ""
            # Extend to the right while prefixes match
            for k in range(s, run_len):
                # Build incrementally; break early if no prefix
                cur += run[k]
                if cur not in single_prefixes:
                    break
                vals = single_map.get(cur)
                if vals:
                    # Annotate tokens from global positions [i+s .. i+k]
                    for p in range(i + s, i + k + 1):
                        annotated_tokens[p].update(vals)
        i = j

    # 3) Multi-word phrase matches via trie over the sequence of word tokens
    word_positions = [idx for idx, w in enumerate(is_word) if w]
    word_values = [lower_tokens[idx] for idx in word_positions]
    W = len(word_positions)

    for start in range(W):
        node = phrase_root
        # Try to extend as long as edges exist
        for end in range(start, W):
            w = word_values[end]
            child = node.children.get(w)
            if child is None:
                break
            node = child
            if node.out:
                # Map word indices [start .. end] to token indices [s .. e]
                s_tok = word_positions[start]
                e_tok = word_positions[end]
                # Annotate the entire inclusive span, which includes only inner separators
                for p in range(s_tok, e_tok + 1):
                    annotated_tokens[p].update(node.out)

    # Prepare output
    return [(tok, anns) for tok, anns in zip(tokens, annotated_tokens)]
