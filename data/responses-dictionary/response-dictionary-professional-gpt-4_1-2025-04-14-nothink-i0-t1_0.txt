```python
import collections.abc
import string

def is_word(token: str) -> bool:
    """
    Returns True if the token is considered a 'word'.
    We'll treat a token as a word if it contains any alphanumeric character.
    (More sophisticated rules can be applied if needed.)
    """
    return any(char.isalnum() for char in token)

def normalize_phrase_words(phrase: str):
    """
    Split a phrase into its word components (casefolded, and ignoring separator patterns).
    """
    # We split by whitespace for phrases, but "words" in the dictionary may contain internal punctuation.
    # So, for simplicity, we'll treat any runs of whitespace as separators.
    return [w.casefold() for w in phrase.split() if w]

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # For single tokens, direct mapping: word_casefolded -> set(ids)
    single_word = collections.defaultdict(set)
    # For phrase matching: list of (word_sequence, id)
    phrases = []

    max_phrase_words = 1

    for key, entry_id in dictionary.items():
        words = normalize_phrase_words(key)
        if not words:
            continue
        if len(words) == 1:
            single_word[words[0]].add(entry_id)
        else:
            phrases.append((words, entry_id))
            max_phrase_words = max(max_phrase_words, len(words))
    # The index is a tuple of (single_word, phrases, max_phrase_words)
    return (single_word, phrases, max_phrase_words)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    # Unpack the dictionary index
    single_word, phrases, max_phrase_words = dictionary_index
    tokens = list(tokens)  # We'll need to index into the token list
    N = len(tokens)

    # Precompute token casefolds & "is word?" for efficiency
    token_cased = tokens
    token_cf = [t.casefold() for t in tokens]
    is_word_token = [is_word(t) for t in tokens]

    # Prepare result: list of sets, one per token
    annotations = [set() for _ in tokens]

    # 1. Annotate single-word entries
    for idx, (token_cf_i, is_word_i) in enumerate(zip(token_cf, is_word_token)):
        if is_word_i and token_cf_i in single_word:
            annotations[idx].update(single_word[token_cf_i])

    # 2. Annotate phrase entries
    # For each possible phrase entry, scan through the tokens, using a sliding window
    # Since tokens may be longer/finer than phrase words, we implement a greedy matching:
    for words_seq, entry_id in phrases:
        num_words = len(words_seq)
        # Sliding window of tokens; look for positions of the first word
        i = 0
        while i < N:
            # Skip until we find a word token matching the first phrase word
            if not is_word_token[i]:
                i += 1
                continue
            if token_cf[i] != words_seq[0]:
                i += 1
                continue
            # Possible start of phrase match at i
            word_indices = [i]  # indices of tokens matching phrase words
            j = i + 1
            k = 1  # index in words_seq
            # Accumulate indices of phrase words and internal separators
            while j < N and k < num_words:
                # If we see a word token matching the next phrase word
                if is_word_token[j] and token_cf[j] == words_seq[k]:
                    word_indices.append(j)
                    k += 1
                    j += 1
                else:
                    j += 1  # skip non-matching token (i.e., possible separator)
            if k == num_words:
                # Matched all phrase words: words at positions word_indices, possibly with separators in between
                # Now expand from word_indices[0] to word_indices[-1], include all tokens in slice [start:end+1]
                start = word_indices[0]
                end = word_indices[-1]
                # Annotate: for each token from start to end:
                # - If word, always annotate
                # - If separator, annotate only if not leading or trailing: i.e., not at start or end
                for pos in range(start, end + 1):
                    if is_word_token[pos]:
                        annotations[pos].add(entry_id)
                    else:
                        # Only annotate if not at start or end of phrase span
                        if pos != start and pos != end:
                            annotations[pos].add(entry_id)
                # Continue searching: allow overlapping, so advance i by 1
                i = start + 1
            else:
                # Not enough phrase words left; break
                break

    # Return paired annotation per token
    for token, ann in zip(token_cased, annotations):
        yield (token, ann)
```
