def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:

    Build a normalized index from a dictionary for fast lookup of words and

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    # Index for compound phrases: first_word -> list of (word_sequence, entry_id)
    phrase_index = {}

        normalized = entry.lower()
        
        # Tokenize into words (split by non-alphanumeric)
        words = []
        current_word = []
        for char in normalized:
            if char.isalnum():
                current_word.append(char)
            elif current_word:
                words.append(''.join(current_word))
                current_word = []
        if current_word:
            words.append(''.join(current_word))
        
        if len(words) == 1:
            # Single word entry
            word = words[0]
            if word not in word_index:
                word_index[word] = set()
            word_index[word].add(entry_id)
        elif len(words) > 1:
            # Compound phrase
            # Also add individual words if they're not already in dictionary
            for word in words:
                if word not in word_index:
                    word_index[word] = set()

            if first_word not in phrase_index:
                phrase_index[first_word] = []
            phrase_index[first_word].append((words, entry_id))

    annotations = [set() for _ in range(n)]

        return any(c.isalnum() for c in token)

        return token.lower()
