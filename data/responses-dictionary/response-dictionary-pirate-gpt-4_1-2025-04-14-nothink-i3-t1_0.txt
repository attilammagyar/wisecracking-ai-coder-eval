def _is_word_token(token):

    return any(c in _WORD_CHARS for c in token)

def _normalize_word(word):
    return word.lower()

class TrieNode:
    __slots__ = ['children', 'end_values']
    def __init__(self):
        self.children = dict()
        self.end_values = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie index with support for compound phrases and single words,
    normalizing all entries to lowercase and collapsing separator sequences.
    """
    root = TrieNode()

    for key, value in dictionary.items():
        # Break key into words, skipping separators
        words = []
        idx = 0
        N = len(key)
        while idx < N:
            # Skip non-word chars (treat as separator)
            while idx < N and not (key[idx].isalnum() or key[idx] in "_'-"):
                idx += 1
            if idx == N:
                break
            j = idx
            while j < N and (key[j].isalnum() or key[j] in "_'-"):
                j += 1
            word = key[idx:j]
            words.append(_normalize_word(word))
            idx = j
        if not words:
            continue # Empty key
        # Insert into trie
        node = root
        for w in words:
            if w not in node.children:
                node.children[w] = TrieNode()
            node = node.children[w]
        node.end_values.add(value)
    return root
