import collections.abc
from collections import defaultdict

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    def __init__(self):
        self.children = defaultdict(TrieNode)
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Builds a case-insensitive trie from the dictionary for efficient phrase lookup.
    """
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        current = trie
        # Normalize and split into tokens (words and separators)
        tokens = []
        current_word = []
        for c in phrase.lower():
            if c.isalpha():
                current_word.append(c)
            else:
                if current_word:
                    tokens.append(''.join(current_word))
                    current_word = []
                tokens.append(c)
        if current_word:
            tokens.append(''.join(current_word))
        
        # Build the trie path
        for token in tokens:
            current = current.children[token]
        current.entry_ids.add(entry_id)
    
    return trie

def is_word_token(token: str) -> bool:
    """Returns True if the token is a word (contains at least one letter)."""
    return any(c.isalpha() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotates tokens by finding all dictionary entries that include each token,
    either as individual words or as part of compound phrases.
    """
    if not dictionary_index or not isinstance(dictionary_index, TrieNode):
        return ((token, set()) for token in tokens)
    
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # First pass: find all individual word matches
    for i, token in enumerate(tokens):
        if is_word_token(token):
            normalized = token.lower()
            # Check if this exact token exists in the trie
            if normalized in dictionary_index.children:
                node = dictionary_index.children[normalized]
                annotations[i].update(node.entry_ids)
    
    # Second pass: find all compound phrases
    for i in range(n):
        if not is_word_token(tokens[i]):
            continue
            
        current_node = dictionary_index
        j = i
        while j < n:
            token = tokens[j]
            normalized = token.lower() if is_word_token(token) else token
            
            if normalized not in current_node.children:
                break
                
            current_node = current_node.children[normalized]
            if current_node.entry_ids:
                # Found a matching phrase from i to j
                for k in range(i, j + 1):
                    annotations[k].update(current_node.entry_ids)
            j += 1
    
    # Handle leading/trailing separators not being annotated
    for i in range(n):
        token = tokens[i]
        if not is_word_token(token):
            # Check if it's between two words that form a compound
            if i > 0 and i < n - 1 and is_word_token(tokens[i-1]) and is_word_token(tokens[i+1]):
                pass  # keep annotations
            else:
                annotations[i] = set()
    
    return zip(tokens, annotations)

2. We do two passes - one for individual words, one for phrases
