import collections.abc

class _Node:
    __slots__ = ("next", "ids")
    def __init__(self):
        self.next: dict[str, _Node] = {}
        self.ids: set = set()

def _split_components(s: str) -> list[str]:
    # Split into runs of alphanumeric chars; everything else be separators.
    comps = []
    cur = []
    for ch in s:
        if ch.isalnum():
            cur.append(ch)
        else:
            if cur:
                comps.append("".join(cur))
                cur = []
    if cur:
        comps.append("".join(cur))
    return comps

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized trie index over dictionary phrases, case-insensitive,
    where edges are alnum-only components, and non-alnum chars are separators.
    """
    root = _Node()
    for key, value in dictionary.items():
        comps = _split_components(key.casefold())
        if not comps:
            # Ignore entries that contain no alphanumeric component.
            continue
        node = root
        for comp in comps:
            node = node.next.setdefault(comp, _Node())
        node.ids.add(value)
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary index.
    """
    root = dictionary_index  # type: ignore
    toks = list(tokens)
    n = len(toks)
    lowers = [t.casefold() for t in toks]
    is_word = [bool(t) and all(c.isalnum() for c in t) for t in toks]
    annos = [set() for _ in range(n)]

    # Helper: attempt to match a component starting at token index 'start',
    # consuming whole word tokens with no separators; return end index or None.
    def match_component_from(start: int, comp: str) -> int | None:
        if start >= n or not is_word[start]:
            return None
        acc = ""
        j = start
        while j < n and is_word[j]:
            acc += lowers[j]
            # Early prune if prefix mismatches.
            if not comp.startswith(acc):
                return None
            if acc == comp:
                return j + 1  # end-exclusive
            j += 1
        return None

    for i in range(n):
        if not is_word[i]:
            continue

        # Try all possible ways to glue consecutive word tokens to form the first component.
        stack = []
        acc = ""
        j = i
        while j < n and is_word[j]:
            acc += lowers[j]
            node = root.next.get(acc)
            if node is not None:
                # Initial state: matched first component over [i, j+1)
                # Keep track of word spans and inner separators seen so far.
                stack.append((node, j + 1, [(i, j + 1)], []))  # (node, pos, word_spans, sep_idxs)
            j += 1

        # DFS over phrase continuations.
        while stack:
            node, pos, spans, sep_idxs = stack.pop()

            # If current node ends a dictionary entry, annotate matched word spans and inner separators.
            if node.ids:
                for entry_id in node.ids:
                    for a, b in spans:
                        for k in range(a, b):
                            annos[k].add(entry_id)
                    for k in sep_idxs:
                        annos[k].add(entry_id)

            # Try to extend with a next component; must have at least one separator between components.
            for comp2, child in node.next.items():
                s = pos
                seps = []
                while s < n and not is_word[s]:
                    seps.append(s)
                    s += 1
                if not seps:
                    continue  # phrases require at least one separator between components
                end2 = match_component_from(s, comp2)
                if end2 is None:
                    continue
                # Successful extension: record inner separators only now (they're inner only if continuation exists).
                stack.append((child, end2, spans + [(s, end2)], sep_idxs + seps))

    return [(toks[i], annos[i]) for i in range(n)]
