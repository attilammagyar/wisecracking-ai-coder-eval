1. **Dictionary Normalization**  
   Convert all phrases to lowercase and remove non-alphanumeric characters for unified lookup.

```python
from typing import Iterable, Union, Dict, Set

def build_dictionary_index(dictionary: Dict[str, Union[int, str]]) -> Dict:
    """Build a normalized dictionary index from original word mappings."""
    index: Dict[str, Set] = {}
    for phrase, entry in dictionary.items():
        # Normalize phrase: lowercase + remove non-alphanumeric chars
        normalized = ''.join(filter(str.isalnum, phrase.lower()))
        if normalized in index:
            index[normalized].add(entry)
        else:
            index[normalized] = {entry}
    return index

def annotate(tokens: Iterable[str], dictionary_index: Dict) -> Iterable[tuple[str, Set]]:
    """Annotate tokens with dictionary entries for words and compound phrases."""
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    for i in range(n):
        if not tokens[i].isalnum():  # Only consider word tokens as start points
            continue
        current_normalized = ''.join(filter(str.isalnum, tokens[i].lower()))
        
        # Handle single-word matches first
        if current_normalized in dictionary_index:
            annotations[i].update(dictionary_index[current_normalized])
        
        # Now try extending to multi-word phrases
        for j in range(i + 1, n):
            if not tokens[j].isalnum():  # Only check word token positions for phrase endings
                continue
            current_normalized += ''.join(filter(str.isalnum, tokens[j].lower()))
            
            if current_normalized in dictionary_index:
                # Annotate entire range [i, j]
                for k in range(i, j + 1):
                    annotations[k].update(dictionary_index[current_normalized])
    
    return [(token, annot.copy()) for token, annot in zip(tokens, annotations)]
```
