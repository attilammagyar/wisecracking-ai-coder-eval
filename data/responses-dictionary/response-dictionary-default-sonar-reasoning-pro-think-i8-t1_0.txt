The solution involves preprocessing the dictionary into a normalized format and using dynamic checking to identify all phrase matches. Here's the implementation:

```python
import re
from collections.abc import Mapping

def build_dictionary_index(dictionary: Mapping) -> object:
    """Build normalized phrase list from dictionary entries"""
    normalized = {}
    for phrase, value in dictionary.items():
        words = re.findall(r'\w+', phrase.lower())
        normalized_phrase = ' '.join(words)
        if not normalized_phrase:
            continue
        normalized[normalized_phrase] = value
    return (normalized,)

def annotate(tokens: list[str], dictionary_index: tuple) -> list[tuple[str, set]]:
    """Annotate tokens using dynamic phrase checking"""
    normalized_phrases, = dictionary_index
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    for i in range(n):
        for j in range(i, n):
            sub_tokens = tokens[i:j+1]
            words = [re.sub(r'[^a-zA-Z0-9]', '', t).lower() for t in sub_tokens]
            words = [w for w in words if w]  # Skip empty entries
            if not words:
                continue
            phrase = ' '.join(words)
            if phrase in normalized_phrases:
                identifier = normalized_phrases[phrase]
                for k in range(i, j+1):
                    annotations[k].add(identifier)
    
    return [(tokens[i], annotations[i]) for i in range(n)]
```
