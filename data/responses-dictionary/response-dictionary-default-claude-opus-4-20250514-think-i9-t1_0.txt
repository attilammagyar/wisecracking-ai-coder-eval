    """Extract words from text, ignoring separators."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:

    """

        # Single word: match single token
        if is_word_token(tokens[pos]) and tokens[pos].lower() == entry_words[0].lower():
            return (pos, pos)
        return None
    
    elif entry_type == 'compound_phrase':
        # Compound phrase: match word tokens with any separators
        if not is_word_token(tokens[pos]):
            return None  # Must start with a word
        
        word_count = 0
        current_pos = pos
        
        while current_pos < len(tokens) and word_count < len(entry_words):
            if is_word_token(tokens[current_pos]):
                if tokens[current_pos].lower() != entry_words[word_count].lower():
                    return None
                word_count += 1
            current_pos += 1
        
        if word_count == len(entry_words):
            return (pos, current_pos - 1)
        return None
    
    else:  # compound_word
        # Compound word: try to match concatenated tokens
        # First, collect consecutive word tokens
        word_tokens = []
        end_pos = pos
        
        while end_pos < len(tokens) and is_word_token(tokens[end_pos]):
            word_tokens.append(tokens[end_pos])
            end_pos += 1
            
            # Check if concatenation matches
            concatenated = ''.join(word_tokens).lower()
            entry_key_lower = entry['key'].lower()
            
            if concatenated == entry_key_lower:
                return (pos, end_pos - 1)
        
        return None

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try to match each dictionary entry at each position
    for entry in dictionary_index:
        for start_pos in range(n):
            match_range = try_match_at_position(tokens, start_pos, entry)
            if match_range:
                start, end = match_range
                for i in range(start, end + 1):
                    annotations[i].add(entry['id'])
    
    # Create result
    annotated_tokens = []
    for i in range(n):
        annotated_tokens.append((tokens[i], annotations[i]))
    
    return annotated_tokens
