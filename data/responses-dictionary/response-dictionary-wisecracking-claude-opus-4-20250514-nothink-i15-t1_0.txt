import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Normalize entries: lowercase keys, preserve original IDs
    normalized = {}
    for key, value in dictionary.items():
        normalized[key.lower()] = value
    
    # Pre-process compound entries to identify word vs phrase patterns
    compound_data = {}
    for key, value in dictionary.items():
        tokens = re.split(r'(\W+)', key)
        tokens = [t for t in tokens if t]  # Remove empty strings
        if len(tokens) > 1:
            # Store: normalized tokens, is it a phrase (has separators), original ID
            word_tokens = [t.lower() for t in tokens if re.match(r'\w+', t)]
            has_separators = any(re.match(r'\W+', t) for t in tokens)
            compound_data[key.lower()] = {
                'tokens': tokens,
                'word_tokens': word_tokens,
                'has_separators': has_separators,
                'id': value
            }
    
    return {
        'normalized': normalized,
        'compound_data': compound_data
    }

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.

    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []

    compound_data = dictionary_index['compound_data']
    
    # Helper to check if a token is a word token
    def is_word_token(token):
        return bool(re.match(r'\w+', token))
    
    # Check each starting position
    for start in range(n):
        # 1. Check single token match
        token_lower = tokens[start].lower()
        if token_lower in normalized:
            annotations[start].add(normalized[token_lower])
        
        # 2. Check compound matches starting at this position
        # Try to match compound words (consecutive word tokens)
        if is_word_token(tokens[start]):
            word_buffer = [tokens[start]]
            for end in range(start + 1, n + 1):
                # Check if we have a match with current buffer
                compound_key = ''.join(word_buffer).lower()
                if compound_key in normalized:
