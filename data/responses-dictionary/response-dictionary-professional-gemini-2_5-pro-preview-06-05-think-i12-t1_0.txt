```python
import collections.abc
import re

# A special key to store dictionary entry values in the Trie.
_VALUES_KEY = "__values__"

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index is a Trie (prefix tree) implemented with nested dictionaries.
    Keys are normalized to lowercase, alphanumeric-only parts.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        A Trie-like dictionary representing the index.
    """
    trie_root = {}
    for phrase, entry_id in dictionary.items():
        # Normalize the phrase into a sequence of word-like parts.
        # This handles case-insensitivity and variations in separators.
        # e.g., "Black Swan" -> ("black", "swan")
        #       "black-swan" -> ("black", "swan")
        #       "BlackSwan"  -> ("blackswan",)
        normalized_parts = tuple(re.findall(r'[a-zA-Z0-9]+', phrase.lower()))

        if not normalized_parts:
            continue

        node = trie_root
        for part in normalized_parts:
            node = node.setdefault(part, {})
        
        # Store the entry ID at the end of the phrase path.
        node.setdefault(_VALUES_KEY, set()).add(entry_id)

    return trie_root


def annotate(
    tokens: collections.abc.Iterable[str], dictionary_index: object
) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    This function iterates through all possible sub-sequences of tokens,
    normalizes them, and looks them up in the provided Trie-based index.
    Annotations are applied to all tokens that form a matched phrase.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Returns:
        annotated_tokens:   A list containing (token, annotations) pairs for each token.
    """
    tokens = list(tokens)
    num_tokens = len(tokens)
    if not num_tokens:
        return []

    annotated_tokens = [(token, set()) for token in tokens]
    
    # Pre-calculate which tokens are word-like to avoid repeated checks.
    # A token is considered word-like if it contains at least one alphanumeric character.
    is_word = [any(c.isalnum() for c in token) for token in tokens]
    
    # Memoization cache for phrase lookups to avoid redundant computations.
    memo = {}

    def get_entries_for_phrase(start_idx: int, end_idx: int) -> collections.abc.Set:
        """
        Looks up a phrase defined by a slice of tokens in the dictionary index.
        Uses memoization to cache results.
        """
        if (start_idx, end_idx) in memo:
            return memo[(start_idx, end_idx)]

        phrase_tokens = tokens[start_idx : end_idx + 1]
        
        # Normalize the token slice into the canonical representation.
        # This mirrors the normalization done in build_dictionary_index.
        canonical_parts = []
        current_chunk = ""
        for i, token in enumerate(phrase_tokens):
            if is_word[start_idx + i]:
                current_chunk += token.lower()
            else: # Separator token
                if current_chunk:
                    canonical_parts.append(current_chunk)
                    current_chunk = ""
        if current_chunk:
            canonical_parts.append(current_chunk)

        # Traverse the Trie with the canonical parts.
        node = dictionary_index
        for part in canonical_parts:
            if part not in node:
                node = None
                break
            node = node[part]
        
        result = node.get(_VALUES_KEY, set()) if node else set()
        memo[(start_idx, end_idx)] = result
        return result

    # Iterate through all possible sub-sequences of tokens.
    for i in range(num_tokens):
        for j in range(i, num_tokens):
            # For each sub-sequence, find the "core phrase" by trimming
            # leading/trailing non-word tokens.
            first_word_idx, last_word_idx = -1, -1
            for k in range(i, j + 1):
                if is_word[k]:
                    if first_word_idx == -1:
                        first_word_idx = k
                    last_word_idx = k

            # If the sub-sequence contains no word-like tokens, skip it.
            if first_word_idx == -1:
                continue
            
            # Look up the core phrase in the dictionary index.
            entries = get_entries_for_phrase(first_word_idx, last_word_idx)
            if entries:
                # If a match is found, apply the annotations to all tokens
                # within the core phrase (including inner separators).
                for k in range(first_word_idx, last_word_idx + 1):
                    annotated_tokens[k][1].update(entries)

    return annotated_tokens
```
