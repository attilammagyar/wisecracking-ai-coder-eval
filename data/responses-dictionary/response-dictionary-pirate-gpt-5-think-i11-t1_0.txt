import re
import collections.abc

# ---------- Helpers (private to this module) ----------

def _new_node():
    # Trie node: children maps word->node, ends is set of entry IDs
    return {"children": {}, "ends": set()}

def _normalize_dict_words(entry_key: str):
    # Lowercase and extract alphanumeric word chunks
    # This tolerates any separators inside the dictionary key.
    return re.findall(r"[0-9A-Za-z]+", entry_key.lower())

def _is_word_token(tok: str) -> bool:
    # Word token iff entirely alphanumeric (and non-empty)
    return tok.isalnum()

# ---------- Public API ----------

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index (a trie over dictionary words) for fast lookup
    of words and compound phrases. Dictionary lookup is case-insensitive and
    separator-insensitive at phrase level.
    """
    root = _new_node()
    for key, entry_id in dictionary.items():
        words = _normalize_dict_words(key)
        if not words:
            continue
        node = root
        for w in words:
            node = node["children"].setdefault(w, _new_node())
        node["ends"].add(entry_id)
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary (single words and phrases).
    Inner separators between phrase words are annotated; leading/trailing
    separators are not. Matches are case-insensitive, separator-insensitive
    between words, and never end mid-token.
    """
    toks = list(tokens)
    n = len(toks)
    if n == 0:
        return []

    # Precompute token kinds and lowercase
    is_word = [_is_word_token(t) for t in toks]
    lower = [t.lower() for t in toks]

    # Result containers
    ann = [set() for _ in range(n)]

    root = dictionary_index  # trie root

    # Matches a single dictionary word string starting at word-token index wi.
    # The dictionary word may span multiple adjacent word tokens but may not
    # cross any separator; must end exactly at a token end.
    # Returns last token index (ej) if matched, else None.
    def match_dict_word_at(wi: int, dict_word: str):
        pos = 0
        j = wi
        L = len(dict_word)
        while True:
            tj = lower[j]
            lj = len(tj)
            # Can't overrun the target word length, must fit exactly
            if pos + lj > L:
                return None
            if dict_word[pos:pos + lj] != tj:
                return None
            pos += lj
            if pos == L:
                # Must end exactly at the end of token j
                return j
            # Need another adjacent word token
            j += 1
            if j >= n or not is_word[j]:
                return None

    # Record a phrase match by annotating covered word tokens and inner separators
    def add_match(start_word_idx: int, end_word_idx: int, inner_seps: list[int], ids):
        # Annotate word tokens from start_word_idx..end_word_idx inclusive
        for k in range(start_word_idx, end_word_idx + 1):
            if is_word[k]:
                ann[k].update(ids)
        # Annotate inner separators specifically collected between words
        for k in inner_seps:
            ann[k].update(ids)

    # For each word-token as a potential phrase start, walk the trie
    for i in range(n):
        if not is_word[i]:
            continue

        # DFS stack items:
        # (node, wi, inner_seps_list, start_idx)
        stack = [(root, i, [], i)]
        while stack:
            node, wi, inner_seps, start_idx = stack.pop()

            # Try each outgoing dictionary word from this node
            children = node["children"]
            if not children:
                continue

            first_tok = lower[wi]
            for dict_word, child in children.items():
                # Quick prune: first word token must be a prefix of the dict word
                if not dict_word.startswith(first_tok):
                    continue

                ej = match_dict_word_at(wi, dict_word)
                if ej is None:
                    continue

                # We matched one more dictionary "word" (which can span multiple tokens)
                if child["ends"]:
                    add_match(start_idx, ej, inner_seps, child["ends"])

                # To continue the phrase, we must have at least one separator
                # after ej and then another word token to start the next dict word.
                j = ej + 1
                if j < n and not is_word[j]:
                    s_start = j
                    while j < n and not is_word[j]:
                        j += 1
                    # j is now either n or the next word-token index
                    if j < n and is_word[j]:

                        new_inner = inner_seps + list(range(s_start, j))
                        stack.append((child, j, new_inner, start_idx))
                # If there is no separator after ej (or no following word), we cannot go deeper.

    # Build final annotated pairs
    annotated_tokens = [(toks[i], ann[i]) for i in range(n)]
