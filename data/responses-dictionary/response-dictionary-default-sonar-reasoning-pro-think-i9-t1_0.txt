```python
import re
from collections import defaultdict

def build_dictionary_index(dictionary):
    index = {
        'single_word': {},
        'compound_phrases': defaultdict(set),
        'max_entry_length': 0
    }
    
    word_pattern = re.compile(r"\b[\w']+\b")
    
    for entry, entry_id in dictionary.items():
        words = [word.lower() for word in word_pattern.findall(entry)]
        if len(words) == 1:
            index['single_word'][words[0]] = entry_id
        else:
            phrase = tuple(words)
            index['compound_phrases'][phrase].add(entry_id)
            if len(phrase) > index['max_entry_length']:
                index['max_entry_length'] = len(phrase)
    
    return index

def is_word(token):
    return bool(re.match(r"\b[\w']+\b", token))

def annotate(tokens, dictionary_index):
    def match_word(token):
        return token.lower() in dictionary_index.get('single_word', {})
    
    annotated = [(token, set()) for token in tokens]
    max_length = dictionary_index.get('max_entry_length', 0)
    single_entries = dictionary_index.get('single_word', {})
    compound_entries = dictionary_index.get('compound_phrases', {})
    word_sequence = []
    word_indices = []
    
    for i, token in enumerate(tokens):
        # Check individual word matches
        lower_token = token.lower()
        if lower_token in single_entries:
            annotated[i] = (token, {single_entries[lower_token]})
        else:
            annotated[i] = (token, set())
        
        # Update word tracking
        if is_word(token):
            word_sequence.append(token.lower())
            word_indices.append(i)
        
        # Check compound phrases ending at current token
        for l in range(1, max_length + 1):
            if len(word_sequence) < l:
                continue
            phrase = tuple(word_sequence[-l:])
            if phrase in compound_entries:
                start_idx = word_indices[-l]
                entry_ids = compound_entries[phrase]
                for idx in range(start_idx, i + 1):
                    if idx < len(annotated):
                        existing_ids = annotated[idx][1].copy()
                        existing_ids.update(entry_ids)
                        annotated[idx] = (tokens[idx], existing_ids)
    
    return annotated
```
