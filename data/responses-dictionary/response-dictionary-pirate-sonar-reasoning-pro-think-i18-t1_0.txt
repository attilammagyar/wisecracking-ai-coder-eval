from collections import defaultdict

def build_dictionary_index(dictionary):
    # Normalize dictionary entries into words and create indexes
    compounds = defaultdict(set)  # key: normalized string, value: set of identifiers
    individual_words = defaultdict(set)
    max_length = 0

    word_pattern = re.compile(r'[\w]+')
    for entry_str, identifier in dictionary.items():
        # Split entry_str into words, ignoring separators
        words = word_pattern.findall(entry_str)
        if not words:
            continue  # no words in the entry
        normalized = ''.join(words).lower()
        compounds[normalized].add(identifier)
        max_length = max(max_length, len(normalized))

        # Add individual words to the individual_words index
        for word in words:
            individual_words[word.lower()].add(identifier)

    return {
        'compounds': compounds,
        'individual_words': individual_words,
        'max_length': max_length
    }

### **Implementation Code**
```python
import collections
import re

def build_dictionary_index(dictionary):
    """Builds a normalized index from a dictionary for fast lookup."""
    compounds = collections.defaultdict(set)
    individual_words = collections.defaultdict(set)
    max_length = 0

    # Regex to split into word characters
    word_pattern = re.compile(r'[\w]+')
    
    for entry_str, entry_id in dictionary.items():
        words = word_pattern.findall(entry_str)
        if not words:
            continue
        # Normalize entry by joining words and lowercasing
        normalized = ''.join(words).lower()
        compounds[normalized].add(entry_id)
        max_length = max(max_length, len(normalized))
        # Register individual words
        for word in words:
            individual_words[word.lower()].add(entry_id)

    return {'compounds': dict(compounds),
            'individual_words': dict(individual_words),
            'max_length': max_length}

def annotate(tokens, dictionary_index):
    """Annotates tokens with entries from the dictionary."""
    compounds = dictionary_index['compounds']
    individual_words = dictionary_index['individual_words']
    max_length = dictionary_index['max_length']
    num_tokens = len(tokens)
    annotations = [set() for _ in range(num_tokens)]
    
    word_pattern = re.compile(r'[\w]+')
    
    # 1. Check individual words
    for i in range(num_tokens):
        token = tokens[i]
        if any(c.isalnum() for c in token):
            lower_token = token.lower()
            annotations[i].update(individual_words.get(lower_token, set()))
    
    # 2. Check all possible compound sequences
    for i in range(num_tokens):
        for l in range(1, max_length + 1):
            j = i + l - 1
            if j >= num_tokens:
                break
            subsequence = tokens[i:j+1]
            # Generate normalized string for comparison
            normalized = []
            for t in subsequence:
                words = word_pattern.findall(t)
                normalized.extend(words)
            normalized_str = ''.join(normalized).lower()
            if normalized_str in compounds and len(normalized_str) > 0:
                # Update all tokens in the sequence
                for k in range(i, j + 1):
                    annotations[k].update(compounds[normalized_str])
    
    return [(tokens[i], annotations[i]) for i in range(num_tokens)]
```

---

### **Key Explanations**
1. **Normalization**  
   - Converts dictionary entries to a uniform format (e.g., "AAA BBB" → "aaabbb") to handle varying word separators.  

2. **Efficient Lookup**  
   - Uses normalized strings in `compounds` and `individual_words` dictionaries for O(1) lookups.  

3. **Sequence Checking**  
   - Iterates over all possible token sequences (up to the max compound length) to detect overlaps and nested compounds. Normalizes each sequence and checks against the compounds index.  

4. **Annotation Propagation**  
   - Individual word annotations are added first. Compound annotations override or supplement these, ensuring all relevant tokens (including inner separators) are marked.
