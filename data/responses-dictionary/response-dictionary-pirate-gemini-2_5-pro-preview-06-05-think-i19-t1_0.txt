import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    This function prepares a dictionary for the annotation process by splitting
    it into two parts: one for single/compound words (alphanumeric keys) and
    one for phrases (keys containing non-alphanumeric separators).
    All keys are lowercased for case-insensitive matching.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        A dictionary index object for use with the annotate function.
    """
    # This be our treasure map, with two chests.
    words = {}
    phrases = {}

    for key, value in dictionary.items():
        # A key is a simple word if it only has letters and numbers.
        # Otherwise, it's a phrase held together by separators.
        if key.isalnum():
            words[key.lower()] = value
        else:
            # For phrases, we pluck out the words, casting them to lowercase.
            # The specific separators don't matter, only the words themselves.
            word_parts = tuple(p.lower() for p in re.findall(r'[a-zA-Z0-9]+', key))
            if word_parts:
                phrases[word_parts] = value
    
    return {'words': words, 'phrases': phrases}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    This function iterates through the tokens and, for each token, finds all
    dictionary entries that begin at that token. It handles single words,
    compound words (concatenated tokens), and phrases (sequences of words
    separated by non-word tokens).

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each
                            token in tokens.
    """
    # We'll need to jump around the tokens, so a list be better than an iterator.
    tokens_list = list(tokens)
    n = len(tokens_list)
    if n == 0:
        return []

    # Each token gets its own set to store its booty (annotations).
    annotations = [set() for _ in range(n)]
    
    # We need to know which tokens be words and which be mere separators.
    is_word_token = [t.isalnum() for t in tokens_list]

    # We sail through the tokens, starting our search at each one.
    for i in range(n):
        # We only start a search from a word token, not from a separator.
        if not is_word_token[i]:
            continue

        # --- Search for compound words startin' at token i ---
        current_concatenated_word = ""
        for j in range(i, n):
            # A compound word is a sequence of word tokens with no separators.
            if not is_word_token[j]:
                break
            
            current_concatenated_word += tokens_list[j]
            entry_id = dictionary_index['words'].get(current_concatenated_word.lower())
            
            if entry_id is not None:
                # We found a match! All tokens that form this word get the annotation.
                for k in range(i, j + 1):
                    annotations[k].add(entry_id)

        # --- Search for phrases startin' at token i ---
        current_phrase_parts = []
        for j in range(i, n):
            if is_word_token[j]:
                current_phrase_parts.append(tokens_list[j].lower())
                entry_id = dictionary_index['phrases'].get(tuple(current_phrase_parts))

                if entry_id is not None:
                    # A phrase match! Annotate all tokens from the start of the phrase
                    # to the current token, including the separators between 'em.
                    for k in range(i, j + 1):
                        annotations[k].add(entry_id)

    return list(zip(tokens_list, annotations))
