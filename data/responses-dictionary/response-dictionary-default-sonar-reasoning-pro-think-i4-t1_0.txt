    """Build data structures for efficient word/phrase lookup"""
    words = {}  # Word: {lower} -> id
    phrases = []  # (lower_word_seq, orig_phrase, orig_tokens, phrase_id)

        original_tokens = key.lower().split(' ')

def annotate(tokens: list, dictionary_index: dict) -> list:
