1. Individual words (case-insensitive)
2. Compound phrases (multiple words with flexible separators)
3. Compound words (multiple tokens concatenated together)

## Main Ideas

The solution uses a two-phase approach:

**1. Index Building Phase:**
- Store all dictionary entries in lowercase for case-insensitive exact matching
- Extract word sequences from multi-word entries for phrase matching
- Differentiate between entries that contain separators (phrases) vs pure alphanumeric entries

**2. Annotation Phase:**
For each position in the token list, try three types of matches:
- **Single token match**: Direct lookup of the token in the dictionary
- **Compound word match**: Concatenate consecutive word tokens (no separators between them) and check if the result exists in the dictionary
- **Phrase match**: Match sequences of words ignoring the specific separators between them

The key insight is that phrases in the dictionary can have different separators than in the text (e.g., "AAA, BBB" can match "AAA * BBB"), while compound words require exact concatenation without separators.

## Implementation

```python
import collections.abc

def is_word_token(token):
    """Check if a token is a word (only alphanumeric characters)."""
    return bool(token) and token.isalnum()

def extract_words(s):
    """Extract words from a string."""
    words = []
    current = ""
    for c in s:
        if c.isalnum():
            current += c
        else:
            if current:
                words.append(current.lower())
                current = ""
    if current:
        words.append(current.lower())
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries': {},  # normalized_string -> entry_id
        'phrases': []   # (word_list, entry_id) for multi-word entries
    }
    
    for key, entry_id in dictionary.items():
        # Store for exact match lookup
        index['entries'][key.lower()] = entry_id
        
        # Extract words for phrase matching
        words = extract_words(key)
        if len(words) > 1:
            index['phrases'].append((words, entry_id))
    
    return index

def find_phrase_match(tokens, start, phrase_words):
    """
    Try to match a phrase starting from position start.
    Returns (match_found, first_word_pos, last_word_pos)
    """
    n = len(tokens)
    word_positions = []
    word_idx = 0
    
    for i in range(start, n):
        if is_word_token(tokens[i]):
            if word_idx < len(phrase_words) and tokens[i].lower() == phrase_words[word_idx]:
                word_positions.append(i)
                word_idx += 1
                if word_idx == len(phrase_words):
                    # Found all words
                    return True, word_positions[0], word_positions[-1]
            elif word_idx > 0:
                # We were in the middle of matching but this word doesn't match
                return False, -1, -1
            # else: we haven't started matching yet, keep looking
    
    return False, -1, -1

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # For each starting position
    for start in range(n):
        # 1. Single token exact match
        if tokens[start].lower() in dictionary_index['entries']:
            annotations[start].add(dictionary_index['entries'][tokens[start].lower()])
        
        # 2. Compound word matching (concatenate consecutive word tokens)
        if is_word_token(tokens[start]):
            concat = tokens[start]
            for end in range(start + 1, n):
                if not is_word_token(tokens[end]):
                    break
                concat += tokens[end]
                if concat.lower() in dictionary_index['entries']:
                    entry_id = dictionary_index['entries'][concat.lower()]
                    for i in range(start, end + 1):
                        annotations[i].add(entry_id)
        
        # 3. Phrase matching
        for phrase_words, entry_id in dictionary_index['phrases']:
            match_found, first_pos, last_pos = find_phrase_match(tokens, start, phrase_words)
            if match_found:
                # Annotate from first to last word position (inclusive)
                for i in range(first_pos, last_pos + 1):
                    annotations[i].add(entry_id)
    
    return [(tokens[i], annotations[i]) for i in range(n)]
```
