```python
import collections.abc
import re

# Helper to classify a token
def classify_token(token):
    # 'W' for word, 'S' for separator
    # (Alphanumeric per Unicode)
    return 'W' if any(c.isalnum() for c in token) else 'S'

# Helper to normalize tokens: lower and strip whitespace/punc
def normalize_token(token):
    return token.lower()

# Split dictionary phrase into sequence of (kind, value)
def split_phrase(phrase):
    # Use regex to extract alternating word / nonword runs
    # We preserve the actual separator values for normalization
    parts = []
    idx = 0
    while idx < len(phrase):
        if phrase[idx].isalnum():
            # collect a word
            start = idx
            while idx < len(phrase) and phrase[idx].isalnum():
                idx += 1
            parts.append(('W', phrase[start:idx].lower()))
        else:
            # collect a separator
            start = idx
            while idx < len(phrase) and not phrase[idx].isalnum():
                idx += 1
            parts.append(('S', phrase[start:idx]))
    return parts

# Trie node class
class TrieNode:
    __slots__ = ['children', 'entries']
    def __init__(self):
        self.children = dict()      # (kind, value) -> TrieNode
        self.entries = set()        # set of entry_ids for phrases ending here

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index (trie) from the dictionary.
    """
    trie = TrieNode()
    maxlen = 0

    for phrase, entry_id in dictionary.items():
        parts = split_phrase(phrase)
        node = trie
        for kind, value in parts:
            if kind == 'W':
                value = value.lower()
            # For separators, leave as-is (to permit fine-grained matching in phrase)
            key = (kind, value)
            if key not in node.children:
                node.children[key] = TrieNode()
            node = node.children[key]
        node.entries.add(entry_id)
        maxlen = max(maxlen, len(parts))

    return {'trie': trie, 'maxlen': maxlen}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]

    trie = dictionary_index['trie']
    maxlen = dictionary_index['maxlen']

    # Preprocess: classify / normalize all tokens
    token_kinds = []
    token_norms = []
    for token in tokens:
        kind = classify_token(token)
        val = normalize_token(token) if kind == 'W' else token
        token_kinds.append(kind)
        token_norms.append(val)

    # Sliding window match
    for i in range(n):
        node = trie
        pos = i
        matched = []
        for span in range(0, maxlen):
            if pos >= n:
                break
            kind = token_kinds[pos]
            val = token_norms[pos]
            key = (kind, val)
            if key not in node.children:
                break
            node = node.children[key]
            matched.append(pos)
            if node.entries:
                # Only annotate inner tokens for leading/trailing separators
                match_indices = list(range(i, pos+1))
                # Exclude leading separator
                if len(match_indices) > 1 and token_kinds[match_indices[0]] == 'S':
                    match_indices = match_indices[1:]
                # Exclude trailing separator
                if len(match_indices) > 1 and token_kinds[match_indices[-1]] == 'S':
                    match_indices = match_indices[:-1]
                for mi in match_indices:
                    annotations[mi].update(node.entries)
            pos += 1

    # Return the list
    return [(tokens[i], annotations[i]) for i in range(n)]
```
