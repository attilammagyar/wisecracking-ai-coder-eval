import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries': {},  # normalized_key -> entry_id
        'word_positions': {},  # normalized_word -> list of (normalized_key, position)
        'phrase_lengths': {},  # normalized_key -> number of words in phrase
    }
    
    for key, value in dictionary.items():
        normalized_key = key.lower()
        index['entries'][normalized_key] = value
        
        # Split into words, keeping track of positions
        words = []
        current_word = []
        for char in normalized_key:
            if char.isalnum():
                current_word.append(char)
            else:
                if current_word:
                    words.append(''.join(current_word))
                    current_word = []
        if current_word:
            words.append(''.join(current_word))
        
        index['phrase_lengths'][normalized_key] = len(words)
        
        # Store word positions for compound phrase detection
        for i, word in enumerate(words):
            if word not in index['word_positions']:
                index['word_positions'][word] = []
            index['word_positions'][word].append((normalized_key, i))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []
    
    # Helper function to check if a token contains word characters
    def is_word_token(token):
        return any(c.isalnum() for c in token)
    
    # Helper function to normalize a token
    def normalize_token(token):
        return token.lower()
    
    # Find all matches starting at each position
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
            
        # Try individual word match
        normalized = normalize_token(tokens[start])
        if normalized in dictionary_index['entries']:
            annotations[start].add(dictionary_index['entries'][normalized])
        
        # Try compound words (consecutive word tokens without separators)
        compound = normalized
        for end in range(start + 1, n):
            if not is_word_token(tokens[end]):
                break
            compound += normalize_token(tokens[end])
            if compound in dictionary_index['entries']:
                entry_id = dictionary_index['entries'][compound]
                for i in range(start, end + 1):
                    annotations[i].add(entry_id)

        words = []
        word_positions = []  # (token_index, word_index)

                word_positions.append((i, word_idx))
                word_idx += 1
            elif i == start:

                break

        for phrase_len in range(1, len(words) + 1):
            phrase_words = words[:phrase_len]
            
            # Check all dictionary entries that could match
            if phrase_words[0] in dictionary_index['word_positions']:
                for dict_key, pos_in_dict in dictionary_index['word_positions'][phrase_words[0]]:
                    # Check if this could be a match
                    if dict_key not in dictionary_index['phrase_lengths']:
                        continue
                    
                    expected_len = dictionary_index['phrase_lengths'][dict_key]
                    if pos_in_dict + phrase_len > expected_len:

                            break
                    
                    if match and pos_in_dict + phrase_len == len(dict_words):
