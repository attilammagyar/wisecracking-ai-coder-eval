def build_dictionary_index(dictionary: Mapping) -> object:
    """Builds an efficient index fer fast term/phrase lookup."""
    singles = {}
    phrases = {}

    for key, value in dictionary.items():
        lower_key = key.lower()
        tokens = re.split(r'\W+', lower_key.strip())
        tokens = [t for t in tokens if t]
        
        if not tokens:
            continue
        
        if len(tokens) == 1:
            singles[tokens[0]] = value
        else:
            phrases_key = "_".join(tokens)
            phrases[phrases_key] = value

    return (singles, phrases)

def annotate(tokens: Iterable[str], dictionary_index: object) -> Iterable[tuple[str, Iterable]]:
    """Annotates tokens with dictionary entries, handlin' phrases an' single words."""
    singles, phrases = dictionary_index
    n = len(tokens)
    annotated = [set() for _ in range(n)]

    # Handle single-word matches first
    for i, token in enumerate(tokens):
        lower_token = token.lower()
        if lower_token in singles:
            annotated[i].add(singles[lower_token])
