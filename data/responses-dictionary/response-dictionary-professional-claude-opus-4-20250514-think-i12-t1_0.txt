import collections.abc

def is_word_token(token):
    """Check if token contains alphanumeric characters"""
    return any(c.isalnum() for c in token)

def extract_words(text):
    """Extract words from text, return list of lowercase words"""
    words = []
    current_word = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char.lower())
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries': []  # List of (words, entry_id) tuples
    }
    
    for text, entry_id in dictionary.items():
        words = extract_words(text)
        if words:
            index['entries'].append((words, entry_id))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try to match dictionary entries starting at each position
    for start_pos in range(n):
        if not is_word_token(tokens[start_pos]):
            continue
            
        # Try each dictionary entry
        for words, entry_id in dictionary_index['entries']:
            # Try to match this entry starting at start_pos
            matched_positions = try_match_entry(tokens, start_pos, words)
            if matched_positions:
                # Annotate all matched positions
                for pos in matched_positions:
                    annotations[pos].add(entry_id)
    
    # Create result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens

    Try to match a dictionary entry starting at start_pos.
    Returns list of token positions that are part of the match, or None if no match.
    """

        if tokens[start_pos].lower() == entry_words[0]:

        positions = []

                return positions
            elif len(concat) >= len(entry_words[0]):
                break

        return None
