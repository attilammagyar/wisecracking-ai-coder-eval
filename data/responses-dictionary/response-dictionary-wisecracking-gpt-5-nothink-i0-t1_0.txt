Main idea (aka “how this thing actually works”):
- Normalize the dictionary:
  - Split keys into lowercase “word runs” by alphanumeric vs non-alphanumeric.
  - If a key has exactly 1 word run (e.g., "AAABBB"), index it in a character trie. This lets us match across multiple adjacent word tokens by concatenation (e.g., tokens ["AAA", "BBB"] match "AAABBB").
  - If a key has 2+ words (e.g., "AAA, BBB"), index it in a word-trie (sequence of words). This makes phrase lookup insensitive to the separators in the text (spaces, *, commas, markdown fluff), while ensuring only inner separators are annotated.
- Token classification:
  - A token is a “word token” if all its characters are alphanumeric. Everything else is a “separator”.
- Matching strategy:
  - Phrases: Walk the word-trie over the sequence of word tokens (ignoring separators). For each match, annotate all word tokens in the phrase and all separators between them (inner only). Leading/trailing separators don’t get a free ticket.
  - Single words (including big compound words): Scan each contiguous run of word tokens with the character trie, character-by-character across token boundaries, but only count matches that end at token boundaries. This enforces “no mid-token matches” (so "CC" will not match inside "BBBCCC", and "AA" won’t match inside "AAA").
- Overlaps are fine: unions of all matches at all layers.
- Case-insensitive, built-in only, and designed to be fast enough without summoning the entire standard library.

# ---------- Helpers ----------

def _is_word_token(tok: str) -> bool:
    # Word token = non-empty and all characters are alphanumeric.
    # This keeps separators (spaces, punctuation, markdown junk) out.
    return bool(tok) and all(ch.isalnum() for ch in tok)

def _split_words_lower(s: str) -> list[str]:
    # Split a dictionary key into lowercase word-runs (alnum-only chunks).
    words = []
    cur = []
    for ch in s:
        if ch.isalnum():
            cur.append(ch.lower())
        else:
            if cur:
                words.append("".join(cur))
                cur = []
    if cur:
        words.append("".join(cur))
    return words


class _CharTrieNode:
    __slots__ = ("children", "values")
    def __init__(self):
        self.children: dict[str, _CharTrieNode] = {}
        self.values: set = set()


class _WordTrieNode:
    __slots__ = ("children", "values")
    def __init__(self):
        self.children: dict[str, _WordTrieNode] = {}
        self.values: set = set()


def _chartrie_insert(root: _CharTrieNode, word: str, val):
    node = root
    for ch in word:
        child = node.children.get(ch)
        if child is None:
            child = _CharTrieNode()
            node.children[ch] = child
        node = child
    node.values.add(val)


def _wordtrie_insert(root: _WordTrieNode, words: list[str], val):
    node = root
    for w in words:
        child = node.children.get(w)
        if child is None:
            child = _WordTrieNode()
            node.children[w] = child
        node = child
    node.values.add(val)

    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    #  - char_trie for single-word entries (enables cross-token concatenation, no mid-token matches)
    #  - word_trie for multi-word phrases (insensitive to separators in text)
    char_root = _CharTrieNode()
    phrase_root = _WordTrieNode()

    for k, v in dictionary.items():
        words = _split_words_lower(k)
        if not words:
            continue  # ignore weird empty keys like " ,,, "
        if len(words) == 1:
            _chartrie_insert(char_root, words[0], v)
        else:
            _wordtrie_insert(phrase_root, words, v)

    return {"char_root": char_root, "phrase_root": phrase_root}

    if n == 0:

    phrase_root: _WordTrieNode = dictionary_index["phrase_root"]

    # Build sequence of word-token indices to walk the word-trie over.
    word_positions = [i for i, w in enumerate(is_word) if w]
    word_lowers = [lower_tok[i] for i in word_positions]

    if char_root.children:

        while i < n:
            if not is_word[i]:
                i += 1
                continue

            j = i
