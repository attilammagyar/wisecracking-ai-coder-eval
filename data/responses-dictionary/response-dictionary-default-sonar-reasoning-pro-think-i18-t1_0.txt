def build_dictionary_index(dictionary: collections.abc.Mapping) -> dict:
    index = {}
    for entry, id in dictionary.items():
        normalized = ''.join(c.lower() for c in entry if c.isalnum())
        index.setdefault(normalized, set()).add(id)
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: dict) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    annotated = []
    n = len(tokens)
    
    # Individual word annotations
    for i, token in enumerate(tokens):
        individual = set()
        normalized = ''.join(c.lower() for c in token if c.isalnum())
        if normalized in dictionary_index:
            individual.update(dictionary_index[normalized])
        annotated.append((token, individual.copy()))
    
    # Compound phrase annotations
    for i in range(n):
        if not any(c.isalnum() for c in tokens[i]):
            continue  # Skip non-word starts
        for j in range(i, n):
            if not any(c.isalnum() for c in tokens[j]):
                continue  # Skip non-word ends
            word_tokens = []
            for k in range(i, j + 1):
                if any(c.isalnum() for c in tokens[k]):
                    word_tokens.append(tokens[k])
            normalized = ''.join(c.lower() for c in ''.join(word_tokens) if c.isalnum())
            if normalized in dictionary_index:
                for k in range(i, j + 1):
                    annotated[k][1].update(dictionary_index[normalized])
    
    return annotated
