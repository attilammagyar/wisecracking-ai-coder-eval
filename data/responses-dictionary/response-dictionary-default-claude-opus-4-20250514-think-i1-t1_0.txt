def is_word_token(token):
    """Check if token contains alphabetic characters."""
    return any(c.isalpha() for c in token)

def normalize(text):
    """Normalize text for case-insensitive comparison."""
    return text.lower()

def extract_words(text):
    """Extract normalized words from text, ignoring non-alphabetic characters."""
    words = []
    current = []
    for char in text:
        if char.isalpha():
            current.append(char.lower())
        else:
            if current:
                words.append(''.join(current))
                current = []
    if current:
        words.append(''.join(current))
    return words

def build_dictionary_index(dictionary):
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    entries = []
    for key, value in dictionary.items():
        entries.append({
            'key': key,
            'normalized': normalize(key),
            'words': extract_words(key),
            'value': value,
            'is_phrase': any(not c.isalpha() for c in key)
        })
    return entries

def annotate(tokens, dictionary_index):
    """
    Annotate tokens with entries from the dictionary.

    Parameters:

    """
    tokens = list(tokens)
    n = len(tokens)
