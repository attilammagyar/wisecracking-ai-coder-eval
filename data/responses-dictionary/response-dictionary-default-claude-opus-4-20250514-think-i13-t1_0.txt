def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},      # lowercase_word -> set of ids
        'compound_words': {},    # lowercase_compound -> set of ids  
        'phrases': {}            # tuple of lowercase words -> set of ids
    }
    
    def is_word_token(token):
        return any(c.isalpha() for c in token)
    
    def tokenize(text):
        # Simple tokenization: alternate between word and non-word characters
        tokens = []
        current = []
        in_word = None
        
        for char in text:
            char_is_word = char.isalnum() or char == "'"
            if in_word is None:
                in_word = char_is_word
                current.append(char)
            elif in_word == char_is_word:
                current.append(char)
            else:
                tokens.append(''.join(current))
                current = [char]
                in_word = char_is_word
        
        if current:
            tokens.append(''.join(current))
        
        return tokens
    
    for entry, entry_id in dictionary.items():
        tokens = tokenize(entry)
        word_tokens = [t for t in tokens if is_word_token(t)]

            continue
        elif len(word_tokens) == 1 and len(tokens) == 1:
            # Single word
            word_lower = word_tokens[0].lower()
            if word_lower not in index['single_words']:
                index['single_words'][word_lower] = set()
            index['single_words'][word_lower].add(entry_id)

            if compound_lower not in index['compound_words']:

            index['compound_words'][compound_lower].add(entry_id)

    Annotate tokens with entries from the dictionary.

    Parameters:

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

    n = len(tokens)
    annotations = [set() for _ in range(n)]

    for start in range(n):
