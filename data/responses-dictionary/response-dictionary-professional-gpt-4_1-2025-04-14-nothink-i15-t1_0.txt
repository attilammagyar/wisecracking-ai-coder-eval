import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    def is_word_token(token):
        # Accept tokens that have at least one alphanumeric character
        return any(c.isalnum() for c in token)
    def normalize_token(token):
        return token.lower()

    trie = {'children': {}, 'entries': set()}
    word_lookup = collections.defaultdict(set)
    maxlen = 1

    for phrase, entry in dictionary.items():
        # Split by words and separators (preserve original tokens with their positions)
        tokens = []
        buf = ''
        for c in phrase:
            if c.isalnum():
                buf += c
            else:
                if buf:
                    tokens.append(buf)
                    buf = ''
                # treat each separator char as a separator token
                tokens.append(c)
        if buf:
            tokens.append(buf)

        # Only keep "word" tokens for trie structure
        words = [normalize_token(tok) for tok in tokens if is_word_token(tok)]
        if not words:
            continue  # No word in phrase, skip (should not happen)
        # For single-word dictionary entries, save to flat map as well
        if len(words) == 1:
            word_lookup[words[0]].add(entry)
        # Build trie for all (multi-)word compounds
        node = trie
        for word in words:
            if word not in node['children']:
                node['children'][word] = {'children': {}, 'entries': set()}
            node = node['children'][word]
        node['entries'].add(entry)
        maxlen = max(maxlen, len(words))

    # Add remaining single-word entries to word_lookup
    # (if not already handled above)
    for key, entry in dictionary.items():
        word = key.strip()
        if word and all(c.isalnum() for c in word):
            word_lookup[normalize_token(word)].add(entry)

    return {
        'trie': trie,
        'word_lookup': word_lookup,
        'maxlen': maxlen,
        'normalize_token': normalize_token,
        'is_word_token': is_word_token,
    }

    Annotate tokens with entries from the dictionary index.

    tokens = list(tokens)  # In case not a list
    n = len(tokens)
    annotations = [set() for _ in tokens]

    is_word_token = dictionary_index['is_word_token']
    trie = dictionary_index['trie']
    word_lookup = dictionary_index['word_lookup']
    maxlen = dictionary_index['maxlen']

    # Precompute normalized tokens and word/separator marks
    norm_tokens = [norm(t) for t in tokens]
    is_word = [is_word_token(t) for t in tokens]

    # Annotate single-word matches
    for i, (nt, iw) in enumerate(zip(norm_tokens, is_word)):
        if iw and nt in word_lookup:
            annotations[i].update(word_lookup[nt])

    # For each position, try all possible compound phrase matches starting here
    for start in range(n):
        # Only try compound match if at word token (dict compound can't start at separator)
        if not is_word[start]:
            continue
        node = trie
        span_indices = []
        word_pos = start
        token_pos = start

        # To find all possible matches in the trie, scan forward skipping separators in tokens, but not in the trie.
        cur_pos = start
        matched_positions = []
        matched_word_indices = []

        # Implementation: for each lookahead, walk through word tokens, but keep track of the token indices.
        cur_node = node
        positions = []
        word_indices = []
        num_words = 0
        j = cur_pos
