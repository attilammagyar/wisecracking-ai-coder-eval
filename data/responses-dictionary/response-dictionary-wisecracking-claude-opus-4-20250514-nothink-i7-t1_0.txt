import collections.abc

class TrieNode:
    def __init__(self):
        self.children = {}
        self.word_children = {}  # For continuing words without separators
        self.entries = set()     # Dictionary entries ending at this node
        self.is_end = False

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    
    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = TrieNode()
    
    for key, value in dictionary.items():
        # Normalize the key to lowercase
        normalized_key = key.lower()
        
        # Split into tokens (words and separators)
        tokens = []
        current_token = []
        in_word = None
        
        for char in normalized_key:
            is_word_char = char.isalnum() or char in "-'"
            if in_word is None:
                in_word = is_word_char
            elif in_word != is_word_char:
                if current_token:
                    tokens.append(''.join(current_token))
                current_token = []
                in_word = is_word_char
            current_token.append(char)
        
        if current_token:
            tokens.append(''.join(current_token))
        
        # Build trie path
        node = root
        for i, token in enumerate(tokens):
            is_word = token[0].isalnum() or token[0] in "-'"
            
            if is_word:
                # Word token - add to word_children
                if token not in node.word_children:
                    node.word_children[token] = TrieNode()
                node = node.word_children[token]
            else:
                # Separator - just mark that we can continue through separators
                if '__SEP__' not in node.children:
                    node.children['__SEP__'] = TrieNode()
                node = node.children['__SEP__']
        
        node.entries.add(value)
        node.is_end = True
    
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    
    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.
        
    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        return token and (token[0].isalnum() or token[0] in "-'")
    
    # Try to match starting from each position
    for start_pos in range(n):
        # Skip if starting with a non-word token (for phrase matching)
        if not is_word_token(tokens[start_pos]):
            continue
            
        # Track all possible matches from this position
        matches = []
        
        # DFS to find all matches
        def explore(pos, node, path_indices, in_phrase=False):
            if pos > n:
                return
                
            # Record current match if it's valid
            if node.is_end and path_indices:
                # For phrases, don't include leading/trailing separators
                indices_to_annotate = path_indices[:]
                if in_phrase and len(indices_to_annotate) > 1:
                    # Trim leading separators
                    while indices_to_annotate and not is_word_token(tokens[indices_to_annotate[0]]):
                        indices_to_annotate.pop(0)
                    # Trim trailing separators
                    while indices_to_annotate and not is_word_token(tokens[indices_to_annotate[-1]]):
                        indices_to_annotate.pop()
                
                if indices_to_annotate:
                    matches.append((indices_to_annotate, node.entries))
            
            if pos >= n:
                return
                
            current_token = tokens[pos].lower()
            
            # Try continuing as a word (no separator)
            if is_word_token(current_token):
                # Direct word continuation
                if current_token in node.word_children:
                    explore(pos + 1, node.word_children[current_token], 
                           path_indices + [pos], in_phrase)
                
                # Try through separator (phrase continuation)
                if '__SEP__' in node.children:
                    sep_node = node.children['__SEP__']
                    if current_token in sep_node.word_children:
                        explore(pos + 1, sep_node.word_children[current_token], 
                               path_indices + [pos], True)
            else:
                # It's a separator - try to continue the phrase
                if '__SEP__' in node.children and pos + 1 < n:
                    # Look ahead to see if there's a word after the separator
                    next_pos = pos + 1
                    while next_pos < n and not is_word_token(tokens[next_pos]):
                        next_pos += 1

        # Apply all found matches
        for indices, entries in matches:
            for idx in indices:
                annotations[idx].update(entries)
    
    # Return annotated tokens
    annotated_tokens = [(token, ann) for token, ann in zip(tokens, annotations)]
