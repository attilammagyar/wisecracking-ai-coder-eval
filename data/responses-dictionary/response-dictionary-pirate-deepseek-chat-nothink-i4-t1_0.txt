import collections.abc
from typing import Dict, Set, List, Tuple

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    
    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.entry_ids: Set[int] = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie index from a dictionary for fast lookup of words and compound phrases.
    """
    trie = TrieNode()
    single_word_map = {}
    
    for phrase, entry_id in dictionary.items():
        # Normalize case for matching
        normalized_tokens = [token.lower() for token in phrase.split()]
        
        if len(normalized_tokens) == 1:
            # Single word entry - add to both trie and direct map
            token = normalized_tokens[0]
            single_word_map[token] = single_word_map.get(token, set()) | {entry_id}
            # Add to trie as well for consistency in compound phrases
            node = trie
            for token in normalized_tokens:
                if token not in node.children:
                    node.children[token] = TrieNode()
                node = node.children[token]
            node.entry_ids.add(entry_id)
        else:
            # Multi-word phrase - add to trie
            node = trie
            for token in normalized_tokens:
                if token not in node.children:
                    node.children[token] = TrieNode()
                node = node.children[token]
            node.entry_ids.add(entry_id)
    
    return (trie, single_word_map)

def is_word_token(token: str) -> bool:
    """Check if a token looks like a word (not punctuation/space)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    trie, single_word_map = dictionary_index
    tokens = list(tokens)
    annotated_tokens = []
    
    for i in range(len(tokens)):
        current_token = tokens[i]
        annotations = set()
        
        # Check single word match (case-insensitive)
        normalized_token = current_token.lower()
        if is_word_token(current_token) and normalized_token in single_word_map:
            annotations.update(single_word_map[normalized_token])
        
        # Check for compound phrases starting at this position
        if is_word_token(current_token):
            node = trie
            j = i
            while j < len(tokens):
                token = tokens[j]
                if is_word_token(token):
                    normalized_token = token.lower()
                    if normalized_token not in node.children:
                        break
                    node = node.children[normalized_token]
                    annotations.update(node.entry_ids)
                    j += 1
                else:
                    # Skip non-word tokens when building phrases
                    j += 1
        
        annotated_tokens.append((current_token, annotations))
    
    return annotated_tokens
