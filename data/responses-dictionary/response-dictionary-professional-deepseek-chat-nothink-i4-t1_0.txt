import collections.abc
from typing import Dict, Set, List, Tuple

class TrieNode:
    __slots__ = ('children', 'word_entries', 'phrase_entries')
    
    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.word_entries: Set = set()
        self.phrase_entries: Set = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a case-insensitive trie index from the dictionary.
    Each node stores both individual word matches and phrase matches.
    """
    root = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize phrase to lowercase and split into words
        normalized_phrase = phrase.lower()
        words = normalized_phrase.split()
        
        current_node = root
        
        # For single-word entries
        if len(words) == 1:
            current_node.word_entries.add(entry_id)
            continue
            
        # For multi-word phrases
        for word in words:
            if word not in current_node.children:
                current_node.children[word] = TrieNode()
            current_node = current_node.children[word]
        current_node.phrase_entries.add(entry_id)
    
    return root

def is_word_token(token: str) -> bool:
    """Check if a token is a word (not punctuation/whitespace)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens by finding all individual word matches and phrase matches.
    """
    tokens = list(tokens)  # Convert to list for random access
    annotated_tokens = []
    n = len(tokens)
    
    # For each token, we'll track all active phrase matches
    active_matches: List[Dict[TrieNode, Set]] = [{} for _ in range(n + 1)]
    root = dictionary_index
    
    # Initialize with root node
    active_matches[0][root] = set()
    
    for i in range(n):
        token = tokens[i]
        token_lower = token.lower()
        is_word = is_word_token(token)
        annotations = set()
        
        # Check for individual word match
        if is_word and hasattr(root, 'word_entries'):
            annotations.update(root.word_entries)
        
        # Process all active phrase matches
        new_active_matches = {}
        
        for node, phrase_ids in active_matches[i].items():
            if is_word:
                # Try to continue phrase matches
                if token_lower in node.children:
                    child_node = node.children[token_lower]
                    new_phrase_ids = phrase_ids.copy()
                    new_phrase_ids.update(child_node.phrase_entries)
                    new_active_matches[child_node] = new_phrase_ids
            else:
                # Non-word token - continue existing phrase matches
                new_active_matches[node] = phrase_ids.copy()
                
            # Add phrase annotations for this token
            annotations.update(phrase_ids)
        
        # Start new phrase matches from root if this is a word
        if is_word and token_lower in root.children:
            child_node = root.children[token_lower]
            new_phrase_ids = set(child_node.phrase_entries)
            if child_node in new_active_matches:
                new_active_matches[child_node].update(new_phrase_ids)
            else:
                new_active_matches[child_node] = new_phrase_ids
        
        active_matches[i + 1] = new_active_matches
        annotated_tokens.append((token, annotations))
    
    return annotated_tokens
