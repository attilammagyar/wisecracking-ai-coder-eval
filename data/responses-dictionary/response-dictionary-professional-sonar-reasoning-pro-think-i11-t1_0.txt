To implement the text annotation algorithm, we need an efficient solution that handles both individual words and compound phrases while accounting for case insensitivity, varied separators, and overlapping matches. The approach combines trie-like prefix checking with string normalization to handle multi-token lookups quickly.

---

### **Core Design Approach**

1. **Dictionary Normalization**:
   - Convert all dictionary entries to lowercase.
   - Replace non-alphanumeric separators with a placeholder (`\x01`).
   - Strip leading and trailing separators to focus on core word sequences.

2. **Lookup Mechanism**:
   - Use a **trie-like structure** to group phrases by their normalized forms.
   - For each token sequence, generate all possible normalized substring cores and check against stored entries.

3. **Annotation Process**:
   - For each token position, check all valid substring cores starting at that position.
   - Track entries using a set to handle multiple matches (e.g., nested compounds).

import sys
import re
from collections.abc import Mapping, Iterable

def build_dictionary_index(dictionary: Mapping) -> object:
    """
    Preprocesses the dictionary into a normalized trie structure for efficient lookups.
    Returns a tuple of:
    - A dict mapping normalized core tuples to entry IDs
    - Max phrase length in the dictionary
    """
    entry_map = {}
    max_length = 0

    # Helper regex to split strings into word/non-word tokens
    tokenizer = re.compile(r'(\w+|\W+)')

    for key, value in dictionary.items():
        # Split the key into tokens (words and non-words)
        tokens = tokenizer.findall(key)

        # Normalize the tokens (lowercase alphanumerics, placeholder for separators)
        normalized = []
        for token in tokens:
            if sys.getdefaultencoding() == 'ascii':
                token = token.lower()
            else:
                token = token.lower()  # Python's Unicode case normalization
            normalized.append(token if token.isalnum() else '\x01')

        # Remove leading/trailing placeholders
        leading = 0
        while leading < len(normalized) and normalized[leading] == '\x01':
            leading += 1
        trailing = len(normalized) - 1
        while trailing >= 0 and normalized[trailing] == '\x01':
            trailing -= 1
        if leading > trailing:
            continue

        core = normalized[leading:trailing + 1]
        if core:
            core_tuple = tuple(core)
            entry_map.setdefault(core_tuple, set()).add(value)
            max_length = max(max_length, len(core))

    return {
        'entry_map': entry_map,
        'max_length': max_length
    }

def annotate(tokens: Iterable[str], dictionary_index: object) -> Iterable[tuple[str, set]]:
    """
    Annotates tokens by checking all possible substring cores against the normalized dictionary.
    Returns a list of tuples with original tokens and their matching entry IDs.
    """
    entry_map = dictionary_index['entry_map']
    max_length = dictionary_index['max_length']
    token_list = list(tokens)
    n = len(token_list)
    annotations = [set() for _ in range(n)]

    # Precompute token-wise properties for faster access
    alnum = [t.isalnum() for t in token_list]

    for i in range(n):
        for j in range(i, min(i + max_length, n)):
            # Extract tokens from i to j (inclusive)
            substokens = token_list[i:j+1]
            leading = 0
            while leading < len(substokens) and not substokens[leading].isalnum():
                leading += 1
            trailing = len(substokens) - 1
            while trailing >= 0 and not substokens[trailing].isalnum():
                trailing -= 1
            if leading > trailing:
                continue

            # Extract core (stripping leading/trailing non-alnum)
            core = substokens[leading:trailing + 1]
            if not core:
                continue

            # Normalize core tokens
            normalized = []
            for token in core:
                if token.isalnum():
                    normalized.append(token.lower())
                else:
                    normalized.append('\x01')
            normalized_core = tuple(normalized)

            if normalized_core in entry_map:
                # Annotate all tokens in the original substokens (including separators in the middle)
                for k in range(i, j + 1):
                    annotations[k].update(entry_map[normalized_core])

    return [(token_list[i], set(ans)) for i, ans in enumerate(annotations)]
