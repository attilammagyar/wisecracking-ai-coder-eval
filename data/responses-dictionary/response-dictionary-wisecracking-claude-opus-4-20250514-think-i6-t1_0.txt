    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).

    index = {
        'single_words': {},  # word -> set of entry IDs
        'compound_entries': [],  # list of (word_sequence, entry_id, original_key)
    }
    
    word_pattern = re.compile(r'[a-zA-Z0-9]+')
    
    for key, entry_id in dictionary.items():
        # Extract all words from the key
        words = word_pattern.findall(key)
        
        if not words:
            continue
            
        # Normalize to lowercase for matching
        normalized_words = [w.lower() for w in words]
        
        if len(words) == 1:

            word = normalized_words[0]
            if word not in index['single_words']:
                index['single_words'][word] = set()
            index['single_words'][word].add(entry_id)
        
        # Also add compound entries (including single words for consistency)
        index['compound_entries'].append((normalized_words, entry_id, key.lower()))
