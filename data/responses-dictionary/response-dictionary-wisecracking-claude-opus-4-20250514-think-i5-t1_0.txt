import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries': {},  # All entries normalized
        'phrase_starters': {}  # Phrases indexed by first word
    }
    
    for key, value in dictionary.items():
        normalized = key.lower()
        index['entries'][normalized] = value
        
        # If it's a phrase, also index by first word for efficient lookup
        words = normalized.split()
        if len(words) > 1:
            first_word = words[0]
            if first_word not in index['phrase_starters']:
                index['phrase_starters'][first_word] = []
            index['phrase_starters'][first_word].append((words, value))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)  # Ensure we can index
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        """Check if token contains alphanumeric characters."""
        return any(c.isalnum() for c in token)
    
    def try_match_phrase(start_pos, phrase_words):
        """Try to match a phrase starting at start_pos."""
        positions = []
        word_idx = 0
        i = start_pos
        
        while i < n and word_idx < len(phrase_words):
            if is_word_token(tokens[i]):
                if tokens[i].lower() == phrase_words[word_idx]:
                    positions.append(i)
                    word_idx += 1
                else:
                    return None  # Word mismatch
            else:
                # Non-word token (separator)
                if 0 < word_idx < len(phrase_words):
                    # Inner separator - include it
                    positions.append(i)
            i += 1
        
        return positions if word_idx == len(phrase_words) else None
    
    # Process each position
    for i in range(n):
        if not is_word_token(tokens[i]):
            continue
        
        token_lower = tokens[i].lower()
        
        # Check single token match
        if token_lower in dictionary_index['entries']:
            annotations[i].add(dictionary_index['entries'][token_lower])
        
        # Check phrases starting with this token
        if token_lower in dictionary_index['phrase_starters']:
            for phrase_words, entry_id in dictionary_index['phrase_starters'][token_lower]:
                matched_positions = try_match_phrase(i, phrase_words)
                if matched_positions:
                    for pos in matched_positions:
                        annotations[pos].add(entry_id)
        
        # Check compound words (concatenated tokens)
        compound = tokens[i]
        for j in range(i + 1, n):
            if not is_word_token(tokens[j]):
                break
            compound += tokens[j]
            compound_lower = compound.lower()
            
            if compound_lower in dictionary_index['entries']:
                # Found a compound word match
                entry_id = dictionary_index['entries'][compound_lower]
                for k in range(i, j + 1):
                    annotations[k].add(entry_id)
    
    # Build result
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
