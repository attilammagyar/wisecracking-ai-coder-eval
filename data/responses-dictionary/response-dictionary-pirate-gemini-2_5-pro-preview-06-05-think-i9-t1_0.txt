import collections.abc
import re

class _TrieNode:
    """A node in the Trie structure for dictionary indexing."""
    def __init__(self):
        self.children: dict[str, '_TrieNode'] = {}
        # A list of values for dictionary entries ending at this node.
        self.values: list = []

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = _TrieNode()
    for phrase, value in dictionary.items():
        # Normalize by lowercasing and splitting into word parts by any
        # non-alphanumeric characters. Empty parts are discarded.
        word_parts = [part for part in re.split(r'\W+', phrase.lower()) if part]
        
        if not word_parts:
            continue

        node = root
        for part in word_parts:
            if part not in node.children:
                node.children[part] = _TrieNode()
            node = node.children[part]
        node.values.append(value)
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    if n == 0:
        return []
        
    annotated_tokens = [(token, set()) for token in tokens]
    trie_root = dictionary_index

    for i in range(n):
        # 1. Search for phrases (sequences of separated word tokens) starting from index i.
        # This search can start at any token, including separators.
        node = trie_root
        first_word_idx = -1
        for j in range(i, n):
            token = tokens[j]
            if token.isalnum():
                if first_word_idx == -1:
                    first_word_idx = j
                
                word = token.lower()
                if word in node.children:
                    node = node.children[word]
                    if node.values:
                        # Found a match. Annotate all tokens from the first word
                        # of the phrase to the current token.
                        for value in node.values:
                            for k in range(first_word_idx, j + 1):
                                annotated_tokens[k][1].add(value)
                else:
                    # No further match possible for this phrase.
                    break
        
        # 2. Search for compound words (concatenated word tokens) starting from index i.
        # This search must start with a word-like token.
        if not tokens[i].isalnum():
            continue
            
        node = trie_root
        chunk = ""
