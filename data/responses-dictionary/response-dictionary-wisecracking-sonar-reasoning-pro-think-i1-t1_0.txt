```python
import re
from collections.abc import Iterable, Mapping, Set
from typing import List, Tuple

def build_dictionary_index(dictionary: Mapping) -> dict:
    """Builds a normalized dictionary index using two representations."""
    normalized = {}
    
    def add_version(version, identifier):
        if version in normalized:
            normalized[version].add(identifier)
        else:
            normalized[version] = {identifier}
    
    for entry, identifier in dictionary.items():
        # Version A: Split words by non-alnum and join with spaces
        words = re.findall(r'\w+', entry.lower())
        version_a = ' '.join(words) if words else ''
        # Version B: Remove all non-alnum characters
        version_b = ''.join(words).lower()
        
        add_version(version_a, identifier)
        add_version(version_b, identifier)
    
    return normalized

def annotate(tokens: Iterable[str], dictionary_index: dict) -> Iterable[Tuple[str, Set]]:
    """Annotes tokens using prepared dictionary index."""
    n = len(tokens)
    annotations = [Set() for _ in range(n)]
    
    for i in range(n):
        token = tokens[i]
        
        # Check individual token matches
        words = re.findall(r'\w+', token.lower())
        if words:
            ver_a = ' '.join(words)
            ver_b = ''.join(words)
            if ver_a in dictionary_index:
                annotations[i].update(dictionary_index[ver_a])
            if ver_b in dictionary_index:
                annotations[i].update(dictionary_index[ver_b])
        
        # Check compound phrases ending at this token
        for j in range(i-1, -1, -1):
            # Skip substrings with leading/trailing non-word tokens
            if not re.search(r'\w', tokens[j]) or not re.search(r'\w', tokens[i]):
                continue
            
            # Extract and process substring content
            substring = tokens[j:i+1]
            combined = ''.join(substring)
            words_sub = re.findall(r'\w+', combined.lower())
            if len(words_sub) < 1:
                continue
            
            # Generate both versions for the substring
            ver_a_sub = ' '.join(words_sub)
            ver_b_sub = ''.join(words_sub)
            
            if ver_a_sub in dictionary_index:
                for k in range(j, i+1):
                    annotations[k].add(next(iter(dictionary_index[ver_a_sub])))
            if ver_b_sub in dictionary_index:
                for k in range(j, i+1):
                    annotations[k].add(next(iter(dictionary_index[ver_b_sub])))
    
    return [(tokens[i], annotations[i]) for i in range(n)]
```

1. **Dual-Normalization System**  
   - **Version A**: Handles typical compound phrases with any separators ("AAA BBB", "AAA-BBB")  
   - **Version B**: Addresses concatenated compounds ("AAABBB") without separators.  

2. **Leading/Trailing Separator Handling**  
   Substrings starting or ending with non-word characters are skipped during annotation checks to comply with problem constraints.  

3. **Efficient Lookups**  
   Built-in dictionary structures allow O(1) lookups for normalized keys, while regular expressions quickly process text sequences.  

This implementation balances efficiency with functionality, handling both individual words and complex compound phrases while adhering to annotation constraints.
