Looking at the requirements and test cases, I need to build an efficient annotation system that can handle:

1. Case-insensitive matching
2. Single words and compound phrases/words
3. Different separators between dictionary and text
4. Overlapping and nested phrases
5. Leading/trailing separator exclusion for phrases

## Main Ideas

**Data Structure**: I'll build an index that groups dictionary entries by their first word (normalized to lowercase). This allows efficient lookup - instead of checking every dictionary entry for each token position, I only check entries that could potentially match.

**Matching Algorithm**: 
- For entries with separators (compound phrases): Match words in sequence, allowing any separators between them. Only annotate the actual phrase components (words + inner separators).
- For entries without separators (compound words or single words): Match by concatenating adjacent word tokens.

**Annotation Process**: For each position in the token list, find all dictionary entries that could start there and attempt to match them. A token can belong to multiple dictionary entries.

Here's the implementation:

```python
import collections.abc

def is_word_token(token):
    """Check if a token is a word (contains alphanumeric characters)."""
    return any(c.isalnum() for c in token)

def normalize_word(word):
    """Normalize a word for case-insensitive comparison."""
    return word.lower()

def extract_words(text):
    """Extract words from text."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def has_separators(text):
    """Check if text contains any non-alphanumeric characters."""
    return any(not c.isalnum() for c in text)

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {}
    
    for entry, value in dictionary.items():
        words = extract_words(entry)
        if not words:
            continue
        
        normalized_words = [normalize_word(w) for w in words]
        first_word = normalized_words[0]
        
        if first_word not in index:
            index[first_word] = []
        
        index[first_word].append({
            'value': value,
            'words': normalized_words,
            'has_separators': has_separators(entry)
        })
    
    return index

def match_compound_word(tokens, start_pos, target_words):
    """Try to match a compound word (no separators) starting from start_pos."""
    # For compound words, we start from the first word token at or after start_pos
    pos = start_pos
    while pos < len(tokens) and not is_word_token(tokens[pos]):
        pos += 1
    
    if pos >= len(tokens):
        return None
    
    # Concatenate adjacent word tokens
    concatenated = ""
    positions = []
    
    while pos < len(tokens) and is_word_token(tokens[pos]):
        concatenated += tokens[pos]
        positions.append(pos)
        pos += 1
    
    # Check if it matches
    target = ''.join(target_words)
    if normalize_word(concatenated) == normalize_word(target):
        return positions
    
    return None

def match_compound_phrase(tokens, start_pos, target_words):
    """Try to match a compound phrase (with separators) starting from start_pos."""
    # Find the first word token at or after start_pos
    pos = start_pos
    while pos < len(tokens) and not is_word_token(tokens[pos]):
        pos += 1
    
    if pos >= len(tokens):
        return None
    
    matched_positions = []
    word_idx = 0
    
    while pos < len(tokens) and word_idx < len(target_words):
        token = tokens[pos]
        
        if is_word_token(token):
            if normalize_word(token) == target_words[word_idx]:
                matched_positions.append(pos)
                word_idx += 1
            else:
                # Word doesn't match
                return None
        else:
            # Separator between matched words
            if word_idx > 0 and word_idx < len(target_words):
                matched_positions.append(pos)
        
        pos += 1
    
    if word_idx == len(target_words):
        return matched_positions
    
    return None

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    # Process each position
    for start_pos in range(len(tokens)):
        # Find the first word token at or after this position
        first_word_pos = start_pos
        while first_word_pos < len(tokens) and not is_word_token(tokens[first_word_pos]):
            first_word_pos += 1
        
        if first_word_pos >= len(tokens):
            continue
        
        first_word = normalize_word(tokens[first_word_pos])
        
        # Get all dictionary entries that start with this word
        if first_word in dictionary_index:
            for entry in dictionary_index[first_word]:
                if entry['has_separators']:
                    # Compound phrase - can have different separators
                    matched_positions = match_compound_phrase(tokens, start_pos, entry['words'])
                else:
                    # Compound word or single word - must be adjacent tokens
                    matched_positions = match_compound_word(tokens, start_pos, entry['words'])
                
                if matched_positions:
                    for pos in matched_positions:
                        annotations[pos].add(entry['value'])
    
    return [(tokens[i], annotations[i]) for i in range(len(tokens))]
```
