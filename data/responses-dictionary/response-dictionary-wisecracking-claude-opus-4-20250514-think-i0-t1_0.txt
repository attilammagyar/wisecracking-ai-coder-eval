import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    
    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'normalized': {},  # normalized_key -> set of entry IDs
        'word_sequences': {}  # tuple of words -> set of (original_key, entry_id) pairs
    }
    
    for key, entry_id in dictionary.items():
        # Store by normalized key for exact matching
        normalized_key = key.lower()
        if normalized_key not in index['normalized']:
            index['normalized'][normalized_key] = set()
        index['normalized'][normalized_key].add(entry_id)
        
        # Extract words for phrase matching
        words = re.findall(r'\w+', key.lower())
        if len(words) > 1:
            word_tuple = tuple(words)
            if word_tuple not in index['word_sequences']:
                index['word_sequences'][word_tuple] = set()
            index['word_sequences'][word_tuple].add((key, entry_id))
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    
    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.
    
    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    def is_word_token(token):
        """Check if token contains alphanumeric characters."""
        return bool(re.search(r'\w', token))

    for start in range(n):
        # Single token lookup

        if token_lower in dictionary_index['normalized']:
            annotations[start].update(dictionary_index['normalized'][token_lower])

            sequence = tokens[start:end]

                    word_tuple = tuple(words)
                    if word_tuple in dictionary_index['word_sequences']:
                        # Annotate from first word to last word (inclusive)
                        # This excludes leading/trailing separators
