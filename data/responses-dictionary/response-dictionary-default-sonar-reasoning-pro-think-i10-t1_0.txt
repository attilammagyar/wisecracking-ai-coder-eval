**Core Approach**

The algorithm solves the annotation problem through two key strategies:

1. **Phrase Matching**  
   Handle words and compounds as contiguous sequences (ignoring separators between). Uses a Trie-like structure for word sequences and checks all possible substrings.

2. **Concatenation Matching**  
   Handle merged tokens (e.g., "AAA"+"BBB" → "AAABBB") by normalizing tokens to alphanumeric forms and checking all possible consecutive spans.

**Efficiency**  
We use precomputed dictionaries and heuristics to limit computational overhead, focusing on maximum phrase lengths derived from the input dictionary.

import re
from typing import Mapping, Iterable, Tuple, Set

def build_dictionary_index(dictionary: Mapping) -> object:
    entry_map_a = {}  # Word sequences (e.g., "aaa bbb")
    entry_map_b = {}  # Concatenated tokens (e.g., "aaabbb")
    max_words = 0
    max_chars = 0

    for phrase, entry_id in dictionary.items():
        # Process for word sequences (type A)
        normalized_a = re.sub(r'\W+', ' ', phrase.lower()).strip()
        words_a = normalized_a.split()
        key_a = ' '.join(words_a)
        if key_a:
            entry_map_a.setdefault(key_a, set()).add(entry_id)
            if len(words_a) > max_words:
                max_words = len(words_a)
        
        # Process for concatenated tokens (type B)
        normalized_b = re.sub(r'\W+', '', phrase.lower())
        if normalized_b:
            entry_map_b.setdefault(normalized_b, set()).add(entry_id)
            if len(normalized_b) > max_chars:
                max_chars = len(normalized_b)

    return {
        'entry_map_a': entry_map_a,
        'entry_map_b': entry_map_b,
        'max_words': max_words,
        'max_chars': max_chars
    }

def annotate(tokens: Iterable[str], dictionary_index: object) -> Iterable[Tuple[str, Set]]:
    # Preprocess tokens
    normalized_a = [re.sub(r'\W+', '', tok).lower() for tok in tokens]  # For word positions
    normalized_b = normalized_a.copy()  # For concatenation checks
    word_positions = [pos for pos, val in enumerate(normalized_a) if val]

    # Initialize annotations
    ann = [set() for _ in tokens]

    # Check type A matches (word sequences with separators)
    entry_map_a = dictionary_index['entry_map_a']
    max_words_a = dictionary_index['max_words']

    for word_idx in range(len(word_positions)):
        start_token = word_positions[word_idx]
        for k in range(1, max_words_a + 1):
            end_word_idx = word_idx + k - 1
            if end_word_idx >= len(word_positions):
                break
            end_token = word_positions[end_word_idx]
            
            # Build phrase from word positions
            phrase_words = [
                normalized_a[word_positions[i]] 
                for i in range(word_idx, end_word_idx + 1)
            ]
            phrase_key = ' '.join(phrase_words)
            
            if phrase_key in entry_map_a:
                current_ids = entry_map_a[phrase_key]
                for pos in range(start_token, end_token + 1):
                    ann[pos].update(current_ids)

    # Check type B matches (concatenated tokens)
    entry_map_b = dictionary_index['entry_map_b']
    max_chars = dictionary_index['max_chars']

    for i in range(len(normalized_b)):
        for k in range(1, max_chars + 1):
            if i + k > len(normalized_b):
                break
            concat = ''.join(normalized_b[i:i + k])
            if concat in entry_map_b:
                current_ids = entry_map_b[concat]
                for pos in range(i, i + k):
                    if pos < len(ann):
                        ann[pos].update(current_ids)

    return [(tokens[i], ann[i]) for i in range(len(tokens))]
