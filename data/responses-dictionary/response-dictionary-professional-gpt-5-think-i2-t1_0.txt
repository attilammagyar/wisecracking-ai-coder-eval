Main ideas

- Normalize dictionary keys by casefolding and splitting them into sequences of word pieces (runs of alphanumeric characters). Non-alphanumeric characters are treated as separators.
- Build two tries:
  - single_word_trie over characters for entries that are a single word (no separators in the dictionary key after normalization). This supports matching dictionary words that may span multiple adjacent word tokens with no separators between them, e.g., tokens ["AAA","BBB"] matching "AAABBB".
  - phrase_trie over normalized words (alphanumeric runs) for entries that have two or more words (i.e., at least one separator). Between words, any positive number of non-word tokens is allowed in the text. We annotate all tokens from the first word to the last word, including the inner separators.
- During annotation:
  - Precompute which tokens are “word tokens” (all characters alphanumeric) and their casefolded form.
  - Single-word matches: for each start at a word token, walk the character trie across subsequent consecutive word tokens. Only consider matches that end exactly at token boundaries (never mid-token).
  - Phrase matches: for each start at a word token matching the first word in the phrase trie, repeatedly jump to the next word token after at least one separator and extend the match. For every terminal node reached, annotate the full span from the first to the current word token (including any inner separator tokens). Leading/trailing separators outside the span are never annotated.

This satisfies:
- Case-insensitive lookups.
- Separator-insensitivity for phrases (any non-word tokens allowed between words).
- Inner separators only are annotated for phrases.
- Concatenation across adjacent word tokens for single-word entries.
- No mid-token matches.

import collections.abc
from typing import Dict, Set, Any, Iterable, Tuple, List, Optional


_TERMINAL = "_ids"  # key used in trie nodes for terminal identifier sets


def _split_words(s: str) -> List[str]:
    """Split a string into alphanumeric word runs, lowercased via casefold()."""
    s = s.casefold()
    words = []
    start = -1
    for i, ch in enumerate(s):
        if ch.isalnum():
            if start < 0:
                start = i
        else:
            if start >= 0:
                words.append(s[start:i])
                start = -1
    if start >= 0:
        words.append(s[start:])
    return words


def _is_word_token(token: str) -> bool:
    """A token is a 'word token' if it consists only of alphanumeric chars and is non-empty."""
    if not token:
        return False
    return all(c.isalnum() for c in token)


def _trie_insert_char(trie: Dict[str, Any], word: str, entry_id: Any) -> None:
    node = trie
    for ch in word:
        nxt = node.get(ch)
        if nxt is None:
            nxt = {}
            node[ch] = nxt
        node = nxt
    ids = node.get(_TERMINAL)
    if ids is None:
        ids = set()
        node[_TERMINAL] = ids
    ids.add(entry_id)


def _trie_insert_phrase(trie: Dict[str, Any], words: List[str], entry_id: Any) -> None:
    node = trie
    for w in words:
        nxt = node.get(w)
        if nxt is None:
            nxt = {}
            node[w] = nxt
        node = nxt
    ids = node.get(_TERMINAL)
    if ids is None:
        ids = set()
        node[_TERMINAL] = ids
    ids.add(entry_id)


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and compound phrases.
    """
    single_word_trie: Dict[str, Any] = {}
    phrase_trie: Dict[str, Any] = {}

    for key, entry_id in dictionary.items():
        words = _split_words(str(key))
        if not words:
            continue
        if len(words) == 1:
            _trie_insert_char(single_word_trie, words[0], entry_id)
        else:
            _trie_insert_phrase(phrase_trie, words, entry_id)

    return {
        "single_word_trie": single_word_trie,
        "phrase_trie": phrase_trie,
    }


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    tokens_list = list(tokens)
    n = len(tokens_list)

    # Precompute token types and normalized forms
    is_word: List[bool] = [_is_word_token(t) for t in tokens_list]
    norm_word: List[Optional[str]] = [t.casefold() if is_word[i] else None for i, t in enumerate(tokens_list)]

    # Prepare result sets
    annotations: List[Set] = [set() for _ in range(n)]

    single_word_trie: Dict[str, Any] = dictionary_index.get("single_word_trie", {})
    phrase_trie: Dict[str, Any] = dictionary_index.get("phrase_trie", {})

    # 1) Single-word matches, possibly spanning multiple adjacent word tokens (no separators).
    for i in range(n):
        if not is_word[i]:
            continue
        node = single_word_trie
        dead = False
        # Traverse over consecutive word tokens starting at i
        j = i
        while j < n and is_word[j] and not dead:
            w = norm_word[j]
            # advance over characters of token j
            for ch in w:  # type: ignore
                nxt = node.get(ch)
                if nxt is None:
                    dead = True
                    break
                node = nxt
            if dead:
                break
            # Only consider matches ending exactly at token boundaries
            ids = node.get(_TERMINAL)
            if ids:
                for k in range(i, j + 1):
                    annotations[k].update(ids)
            j += 1

    # 2) Multi-word phrase matches (two or more words), allowing any positive number of separators between words.
    for i in range(n):
        if not is_word[i]:
            continue
        first = norm_word[i]  # type: ignore
        node = phrase_trie.get(first)
        if not node:
            continue

        # Attempt to extend through successive words separated by >=1 non-word token(s)
        pos = i
        while True:
            # Find next word token after at least one separator
            j = pos + 1
            seen_sep = False
            while j < n and not is_word[j]:
                seen_sep = True
                j += 1
            if not seen_sep or j >= n:
                break  # need at least one separator, and a next word token

            w = norm_word[j]  # type: ignore
            nxt = node.get(w)
            if not nxt:
                break
            node = nxt
            pos = j

            # Every time we reach a terminal node, we have a phrase match from i..pos
            ids = node.get(_TERMINAL)
            if ids:
                for k in range(i, pos + 1):
                    annotations[k].update(ids)

    # Build output
    annotated_tokens: List[Tuple[str, Set]] = [(tokens_list[i], annotations[i]) for i in range(n)]
    return annotated_tokens
