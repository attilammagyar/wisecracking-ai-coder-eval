from collections.abc import Iterable, Mapping, Set
import sys

class DictionaryIndex:
    def __init__(self, normalized: dict, max_length: int):
        self.normalized = normalized  # {normalized_string: set(ids)}
        self.max_length = max_length

def build_dictionary_index(dictionary: Mapping) -> object:
    """Preprocess dictionary entries for efficient lookups."""
    normalized = {}
    max_length = 0
    
    for original, value in dictionary.items():
        # Normalize: lowercase + remove non-alphanumeric characters
        norm = ''.join(c.lower() for c in original if c.isalnum())
        if norm:  # Handle empty strings from all-separators entries
            normalized.setdefault(norm, set()).add(value)
            max_length = max(max_length, len(norm))
    
    return DictionaryIndex(normalized, max_length)

def annotate(tokens: Iterable[str], dictionary_index: object) -> Iterable[tuple[str, Set]]:
    """Annotate tokens with dictionary entries handling case and compounded phrasing."""
    # Initialize annotations
    annotated = [(t, set()) for t in tokens]
    normalized, max_len = dictionary_index.normalized, dictionary_index.max_length
    
    for i in range(len(tokens)):
        # Find starting word index, skipping leading separators
        start = i
        while start < len(tokens) and not tokens[start].isalnum():
            start += 1
            
        if start >= len(tokens):
            continue  # No word to start from
            
        word_idx = [start]
        word_buff = [tokens[start].lower()]
        next_idx = start + 1
        
        for j in range(next_idx, len(tokens)):
            if tokens[j].isalnum():
                word_idx.append(j)
                word_buff.append(tokens[j].lower())
            
            # Check all possible prefixes up to max length
            for k in range(1, len(word_buff) + 1):

                if current_norm in normalized:
