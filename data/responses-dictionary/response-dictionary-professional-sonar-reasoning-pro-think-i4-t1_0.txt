## **Algorithm Design**
**Core Ideas:**
1. **Normalization & Indexing**  
   Convert all dictionary entries to lowercase, split into non-alphanumeric "words", then concatenate.  
   Example: `"Black Swan"` → `"blackswan"`.

2. **Efficient Phrase Matching**  
   Precompute indices of non-separator tokens for each position and dynamically build phrases to check against the normalized index.

3. **Window-Based_annotation**  
   For each token, check all possible phrase lengths (up to max dictionary phrase length) to detect compounds.

## **Implementation Code**
```python
import re
import collections

def build_dictionary_index(dictionary: collections.abc.Mapping) -> dict:
    """Preprocess dictionary entries into normalized keys and track maximum phrase length."""
    index = {}
    max_length = 0
    split_regex = re.compile(r'[^a-zA-Z0-9]+')  # Split on non-alphanumeric characters

    for entry, value in dictionary.items():
        # Normalize entry: split, lowercase, join
        tokens = split_regex.split(entry)
        tokens = [t.lower() for t in tokens if t and any(c.isalnum() for c in t)]  # Filter empty/space tokens
        
        normalized = ''.join(tokens)
        if normalized in index:
            index[normalized].add(value)
        else:
            index[normalized] = {value}
        
        # Update max phrase length
        if len(tokens) > max_length:
            max_length = len(tokens)
    
    return {'index': index, 'max_length': max_length}

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: dict) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """Annotate tokens using precomputed dictionary index and maximum phrase length."""
    index = dictionary_index['index']
    max_len = dictionary_index['max_length']
    
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]  # Initialize empty sets for annotations

    # Precompute non-separator indices for each starting position
    non_sep_indices = []
    for i in range(n):
        indices = []
        j = i
        while j < n and len(indices) < max_len:
            if any(c.isalnum() for c in tokens[j]):  # Check alphanumeric presence
                indices.append(j)
            j += 1
        non_sep_indices.append(indices)

    # Check all possible phrases
    for i in range(n):
        possible_indices = non_sep_indices[i]
        for k in range(1, len(possible_indices) + 1):  # Phrase length 1 to max_len
            if k > max_len:
                continue
            # Extract phrase components
            phrase_parts = [tokens[j].lower() for j in possible_indices[:k]]
            phrase = ''.join(phrase_parts)
            # Check index for phrase match
            if phrase in index:
                ids = index[phrase]
                # Annotate all tokens in this phrase
                end_idx = possible_indices[k-1]
                for j in range(i, end_idx + 1):
                    annotations[j].update(ids)
    
    return [(tokens[i], annotations[i]) for i in range(n)]
```
