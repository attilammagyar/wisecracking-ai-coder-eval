import collections.abc
from collections import defaultdict

class TrieNode:
    __slots__ = ('children', 'entry_ids')
    
    def __init__(self):
        self.children = defaultdict(TrieNode)
        self.entry_ids = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a trie-based index from a dictionary for fast lookup.
    Normalizes all keys to lowercase and treats all separators equally.
    """
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Normalize phrase: lowercase and treat all separators as single space
        normalized = ' '.join(word.lower() for word in phrase.split())
        words = normalized.split(' ')
        
        # Insert into trie
        node = trie
        for word in words:
            node = node.children[word]
        node.entry_ids.add(entry_id)
    
    return trie

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens by finding all individual words and compound phrases.
    """
    tokens = list(tokens)  # We need random access
    annotated_tokens = []
    n = len(tokens)
    
    # First pass: find all phrase matches and their spans
    phrase_matches = defaultdict(set)  # token index -> set of entry_ids
    
    for i in range(n):
        node = dictionary_index
        current_phrase_length = 0
        
        # Skip non-word tokens at start (they can't be part of a phrase)
        if not tokens[i].isalpha():
            continue
            
        # Look ahead for possible phrases starting at this position
        j = i
        while j < n:
            token = tokens[j]
            
            # Skip separators (but count them as part of phrase)
            if token.isalpha():
                word = token.lower()
                if word not in node.children:
                    break
                node = node.children[word]
                current_phrase_length += 1
            elif j > i:  # Only count inner separators
                current_phrase_length += 1
            else:
                break  # Leading separator - not part of phrase
                
            # If we found a phrase, mark all its tokens
            if node.entry_ids and current_phrase_length >= 1:
                for k in range(i, j + 1):
                    phrase_matches[k].update(node.entry_ids)
                    
            j += 1
    
    # Second pass: combine individual word matches with phrase matches
    for i, token in enumerate(tokens):
        annotations = set()
        
        # Add individual word match (if any)
        if token.isalpha():
            word = token.lower()
            node = dictionary_index.children.get(word)
            if node and node.entry_ids:
                annotations.update(node.entry_ids)
        
        # Add any phrase matches this token participates in
        annotations.update(phrase_matches.get(i, set()))
        
        annotated_tokens.append((token, annotations))
    
    return annotated_tokens
