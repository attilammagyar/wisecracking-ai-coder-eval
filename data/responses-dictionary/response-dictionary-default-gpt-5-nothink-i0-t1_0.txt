Main ideas

- Normalize dictionary keys and tokens with casefold() so lookups are case-insensitive.
- Treat tokens as either word tokens (contain at least one alphanumeric character) or separators (no alphanumerics). This lets us:
  - Match multi-word phrases over sequences Word Sep+ Word Sep+ ... Word and annotate all inner separators.
  - Match “compound words” (no separators in the dictionary key) by concatenating adjacent word tokens with no separators between them.
- Build two indexes from the dictionary:
  - single_words: map of normalized single-word entries to their IDs. This is used for exact token matches and for concatenated compound-word matches across adjacent tokens.
  - phrase_trie: a trie over normalized words for multi-word phrases (length ≥ 2). Dictionary separators are ignored when building it.
- Annotation is done in three passes:
  1) Per-token single-word lookups.
  2) Traverse the phrase trie starting at each word token, requiring at least one separator between successive words; annotate all words and inner separators for each terminal node encountered (to capture nested and overlapping phrases).
  3) For each maximal run of adjacent word tokens (no separators), try all concatenations and annotate spans that match a single-word dictionary entry (to capture AAABBB over ["AAA","BBB"]).

import collections.abc

class _TrieNode:
    __slots__ = ("children", "end_ids")
    def __init__(self):
        self.children = {}  # dict[str, _TrieNode]
        self.end_ids = set()  # set of dictionary entry ids


class _DictionaryIndex:
    __slots__ = ("single_words", "phrase_trie_root")
    def __init__(self, single_words, phrase_trie_root):
        self.single_words = single_words  # dict[str, set]
        self.phrase_trie_root = phrase_trie_root  # _TrieNode


def _extract_words_norm(s: str) -> list[str]:
    """
    Split a dictionary key into words by grouping consecutive isalnum() chars.
    Normalize by casefold().
    """
    words = []
    buf = []
    for ch in s:
        if ch.isalnum():
            buf.append(ch)
        else:
            if buf:
                words.append(''.join(buf).casefold())
                buf = []
    if buf:
        words.append(''.join(buf).casefold())
    return words


def _is_word_token(token: str) -> bool:
    """
    A token is considered a word if it contains at least one alphanumeric char.
    """
    for ch in token:
        if ch.isalnum():
            return True
    return False


def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    single_words = {}  # str -> set(ids)
    root = _TrieNode()

    for key, entry_id in dictionary.items():
        words = _extract_words_norm(key)
        if not words:
            continue
        if len(words) == 1:
            w = words[0]
            s = single_words.get(w)
            if s is None:
                s = set()
                single_words[w] = s
            s.add(entry_id)
        else:
            # Insert into phrase trie
            node = root
            for w in words:
                child = node.children.get(w)
                if child is None:
                    child = _TrieNode()
                    node.children[w] = child
                node = child
            node.end_ids.add(entry_id)

    return _DictionaryIndex(single_words, root)


def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    # Materialize tokens for random access
    tokens = list(tokens)
    n = len(tokens)
    if n == 0:
        return []

    idx = dictionary_index  # type: _DictionaryIndex
    single = idx.single_words
    trie_root = idx.phrase_trie_root

    # Precompute token properties
    is_word = [_is_word_token(t) for t in tokens]
    norm = [tokens[i].casefold() if is_word[i] else None for i in range(n)]

    anns = [set() for _ in range(n)]

    # Pass 1: direct single-word matches (exact token equality)
    if single:
        for i in range(n):
            if is_word[i]:
                ids = single.get(norm[i])
                if ids:
                    anns[i].update(ids)

    # Pass 2: multi-word phrases (require at least one separator between words)
    # Traverse trie starting at each word token.
    # Annotate words and inner separators for every terminal encountered.
    if trie_root.children:
        for i in range(n):
            if not is_word[i]:
                continue
            first_w = norm[i]
            node = trie_root.children.get(first_w)
            if node is None:
                continue

            words_pos = [i]
            inner_seps = []

            # If phrase ends at first word
            if node.end_ids:
                ids = node.end_ids
                for pos in words_pos:
                    anns[pos].update(ids)
                # no inner separators yet

            last_word_pos = i
            while True:
                j = last_word_pos + 1
                if j >= n:
                    break
                # Require at least one separator between words
                if is_word[j]:
                    break

                # Collect one or more separators
                sep_indices = []
                while j < n and not is_word[j]:
                    sep_indices.append(j)
                    j += 1
                if j >= n:
                    break  # no next word

                next_w = norm[j]  # j is a word
                child = node.children.get(next_w)
                if child is None:
                    break

                # Extend match
                node = child
                words_pos.append(j)
                inner_seps.extend(sep_indices)

                if node.end_ids:
                    ids = node.end_ids
                    for pos in words_pos:
                        anns[pos].update(ids)
                    for pos in inner_seps:
                        anns[pos].update(ids)

                last_word_pos = j

    # Pass 3: compound words formed by concatenating adjacent word tokens (no separators)
    # For each maximal run of consecutive word tokens, try all substrings by concatenation.
    if single:
        i = 0
        while i < n:
            if not is_word[i]:
                i += 1
                continue
            # Find end of run [start..end] of consecutive word tokens
            start = i
            while i + 1 < n and is_word[i + 1]:
                i += 1
            end = i

            # Try all spans within the run
            for s in range(start, end + 1):
                concat = ""
                for e in range(s, end + 1):
                    concat += norm[e]
                    ids = single.get(concat)
                    if ids:
                        # Annotate all tokens in [s..e]
                        for k in range(s, e + 1):
                            anns[k].update(ids)
            i = end + 1

    # Build result
    return [(tokens[i], anns[i]) for i in range(n)]
