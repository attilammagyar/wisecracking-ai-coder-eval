1. **Dictionary entries fall into two categories:**
   - **Phrases**: Contain non-word characters (spaces, commas, etc.)
   - **Words**: Either single words or compound words (no separators)

2. **Matching rules differ by category:**
   - **Phrases**: Match word sequences with any separators between them
   - **Words**: Match exact tokens or consecutive word tokens that concatenate

3. **Annotation scope:**
   - **Phrases**: Annotate all tokens from first to last word (including inner separators)
   - **Words**: Annotate only the word tokens involved

### Data Structure

The index uses separate storage for different entry types to optimize lookups:
- Single words: Direct hash map for O(1) lookup
- Phrases: List of word sequences to match flexibly
- Compound words: Hash map of concatenated forms

### Algorithm

For each starting position, we attempt to match:
1. Single words directly
2. Phrases by matching word sequences with flexible separators
3. Compound words by concatenating consecutive word tokens

```python
import re
import collections.abc

def is_word_token(token):
    """Check if token contains only word characters."""
    return bool(re.match(r'^\w+$', token))

def normalize(text):
    """Normalize text for case-insensitive comparison."""
    return text.lower()

def extract_words(text):
    """Extract words from text in order."""
    return re.findall(r'\w+', text)

def contains_separator(text):
    """Check if text contains non-word characters."""
    return bool(re.search(r'\W', text))

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},  # word -> [entry_ids]
        'phrases': [],       # [{id, words}]
        'compound_words': {} # concatenated -> [entry_ids]
    }
    
    for entry_text, entry_id in dictionary.items():
        normalized = normalize(entry_text)
        words = extract_words(normalized)
        
        if len(words) == 0:
            continue
        elif len(words) == 1:
            # Single word entry
            word = words[0]
            if word not in index['single_words']:
                index['single_words'][word] = []
            index['single_words'][word].append(entry_id)
        elif contains_separator(entry_text):
            # Phrase (has separators like spaces, commas, etc.)
            index['phrases'].append({
                'id': entry_id,
                'words': words
            })
        else:
            # Compound word (no separators, like "AAABBB")
            if normalized not in index['compound_words']:
                index['compound_words'][normalized] = []
            index['compound_words'][normalized].append(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try matching from each position
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
        
        start_word = normalize(tokens[start])
        
        # 1. Check single word matches
        if start_word in dictionary_index['single_words']:
            for entry_id in dictionary_index['single_words'][start_word]:
                annotations[start].add(entry_id)
        
        # 2. Check phrase matches
        for phrase_info in dictionary_index['phrases']:
            phrase_words = phrase_info['words']
            if phrase_words[0] != start_word:
                continue
            
            # Try to match all words in the phrase
            word_idx = 1  # We already matched the first word
            last_pos = start
            
            if len(phrase_words) == 1:
                # Single-word phrase (edge case)
                annotations[start].add(phrase_info['id'])
                continue
            
            # Match remaining words
            for i in range(start + 1, n):
                if is_word_token(tokens[i]):
                    if word_idx < len(phrase_words) and normalize(tokens[i]) == phrase_words[word_idx]:
                        word_idx += 1
                        last_pos = i
                        if word_idx == len(phrase_words):
                            # Found complete phrase - annotate all tokens in range
                            for j in range(start, last_pos + 1):
                                annotations[j].add(phrase_info['id'])
                            break
                    else:
                        # Word doesn't match - phrase match failed
                        break
        
        # 3. Check compound word matches
        word_parts = []
        word_positions = []
        
        for i in range(start, n):
            if is_word_token(tokens[i]):
                word_parts.append(normalize(tokens[i]))
                word_positions.append(i)
                
                # Check if concatenated words form a compound
                concatenated = ''.join(word_parts)
                if concatenated in dictionary_index['compound_words']:
                    for entry_id in dictionary_index['compound_words'][concatenated]:
                        # Annotate only the word tokens
                        for pos in word_positions:
                            annotations[pos].add(entry_id)
            else:
                # Hit a separator - stop looking for compound words
                break
    
    annotated_tokens = []
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
```
