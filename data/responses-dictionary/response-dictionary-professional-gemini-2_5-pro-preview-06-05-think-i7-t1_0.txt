import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index is a dictionary mapping a normalized string to a set of dictionary
    entry IDs. Normalization involves lowercasing and standardizing separators
    to a single space, which allows for flexible phrase matching. For example,
    "black-swan" and "black swan" both normalize to "black swan".

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        A dictionary-based index for use with the annotate function.
    """
    # This pattern finds all sequences of one or more alphanumeric characters.
    word_pattern = re.compile(r'[a-zA-Z0-9]+')
    
    index = {}
    for key, value in dictionary.items():
        # Normalize the dictionary key by extracting all word parts,
        # lowercasing them, and joining with a single space.
        # e.g., "AAA, BBB" -> ["aaa", "bbb"] -> "aaa bbb"
        # e.g., "AAABBB" -> ["aaabbb"] -> "aaabbb"
        lower_key = key.lower()
        words = word_pattern.findall(lower_key)

        if not words:
            continue

        normalized_key = " ".join(words)
        
        if normalized_key not in index:
            index[normalized_key] = set()
        index[normalized_key].add(value)
        
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    This function iterates through the token stream, and for each potential
    starting token, it attempts to match phrases and compound words from the
    dictionary. It handles overlapping and nested phrases by checking all
    possible match lengths starting from each token.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for
                            each token in tokens.
    """
    if not isinstance(dictionary_index, dict):
        raise TypeError("dictionary_index must be a dict created by build_dictionary_index.")
        
    token_list = list(tokens)
    n = len(token_list)
    annotated_tokens = [(token, set()) for token in token_list]

    if n == 0:
        return []

    # The outer loop iterates through each token, considering it as a potential
    # start of a match.
    for i in range(n):
        # A match must begin with a word-like token. This check correctly
        # handles the rule for leading/trailing separators.
        if not token_list[i].isalnum():
            continue

        # These lists will collect the word parts for the current potential match.
        phrase_parts = []
        compound_parts = []
        
        # The inner loop extends the potential match from `i` to `j`.
        for j in range(i, n):
            token = token_list[j]

            # We only consider word-like tokens for forming lookup keys.
            if token.isalnum():
                lower_token = token.lower()
                phrase_parts.append(lower_token)
                compound_parts.append(lower_token)
                
                # Form the lookup keys from the collected word parts.
                # 1. Phrase key (words joined by spaces)
                phrase_key = " ".join(phrase_parts)
                if phrase_key in dictionary_index:
                    values = dictionary_index[phrase_key]
                    # Annotate all tokens from the start of the match `i`
                    # up to the current token `j`.
                    for k in range(i, j + 1):
                        annotated_tokens[k][1].update(values)

                # 2. Compound word key (words concatenated)
                compound_key = "".join(compound_parts)
                # Avoid redundant lookups if the compound and phrase keys are identical
                # (which happens for single-word matches).
                if compound_key != phrase_key and compound_key in dictionary_index:
                    values = dictionary_index[compound_key]
                    for k in range(i, j + 1):
                        annotated_tokens[k][1].update(values)
                        
    return annotated_tokens
