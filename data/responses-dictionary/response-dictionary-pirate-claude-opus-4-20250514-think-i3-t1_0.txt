def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = []
    
    for text, value in dictionary.items():
        # Extract word tokens from the entry
        words = re.findall(r'\w+', text.lower())
        if not words:
            continue
            
        # Check if it's a phrase (contains non-word characters)
        # A phrase has non-word chars between or around words
        is_phrase = len(re.findall(r'\W', text)) > 0
        
        index.append({
            'text': text,
            'value': value,
            'words': words,
            'is_phrase': is_phrase,
            'normalized': text.lower()
        })
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    annotations = [set() for _ in tokens]
    
    # Helper to check if token contains only word characters
    def is_word_token(token):
        return bool(re.match(r'^\w+$', token))
    
    # Try to match each dictionary entry
    for entry in dictionary_index:
        if entry['is_phrase']:
            # Match phrase with flexible separators
            target_words = entry['words']
            
            # Try starting at each position
            for start_pos in range(len(tokens)):
                # Must start with a word token
                if not is_word_token(tokens[start_pos]):
                    continue
                    
                # First word must match
                if tokens[start_pos].lower() != target_words[0]:
                    continue
                
                # Try to match the rest of the phrase
                matched_positions = [start_pos]
                word_idx = 1
                pos = start_pos + 1
                
                while pos < len(tokens) and word_idx < len(target_words):
                    if is_word_token(tokens[pos]):
                        if tokens[pos].lower() == target_words[word_idx]:
                            matched_positions.append(pos)
                            word_idx += 1
                        else:
                            break
                    else:
                        # Non-word token - separator within phrase
                        matched_positions.append(pos)
                    pos += 1
                
                # Check if we matched all words
                if word_idx == len(target_words):
                    # Remove trailing non-word tokens from matched positions
                    while len(matched_positions) > 1 and not is_word_token(tokens[matched_positions[-1]]):
                        matched_positions.pop()
