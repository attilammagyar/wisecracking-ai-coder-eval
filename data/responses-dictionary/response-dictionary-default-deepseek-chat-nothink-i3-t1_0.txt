class TrieNode:
    __slots__ = ('children', 'values')
    def __init__(self):

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:

    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # Build word index (case-insensitive)
    word_index = defaultdict(set)
    trie = TrieNode()
    
    for key, value in dictionary.items():
        # Normalize key to lowercase
        normalized_key = key.lower()
        
        # Add to word index
        word_index[normalized_key].add(value)
        
        # Split into words (treating any non-alphanumeric as separator)
        words = []
        current_word = []
        for c in normalized_key:
            if c.isalnum():
                current_word.append(c)
            else:
                if current_word:
                    words.append(''.join(current_word))
                    current_word = []
        if current_word:
            words.append(''.join(current_word))
        
        # Add to trie if it's a multi-word phrase
        if len(words) > 1:
            node = trie
            for word in words:
                node = node.children[word]
            node.values.add(value)
    
    return {
        'word_index': word_index,
        'phrase_trie': trie,
    }

def is_word_token(token: str) -> bool:

    return any(c.isalnum() for c in token)

    Annotate tokens with entries from the dictionary.

    word_index = dictionary_index['word_index']
    phrase_trie = dictionary_index['phrase_trie']

    annotations = [set() for _ in tokens]

    for i, token in enumerate(tokens):
        if is_word_token(token):

            if normalized in word_index:

    for start in range(n):

        while end < n:

            if not is_word_token(token):
