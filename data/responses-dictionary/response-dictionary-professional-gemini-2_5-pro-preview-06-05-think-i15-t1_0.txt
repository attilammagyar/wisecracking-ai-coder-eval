import collections.abc
import re

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    The index is a Trie data structure, implemented using nested dictionaries.
    Each node in the Trie is a dictionary with two keys:
    - 'children': A dictionary mapping a normalized word to the next node.
    - 'values': A set of dictionary entry IDs for phrases ending at this node.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Returns:
        A Trie-like object representing the indexed dictionary.
    """
    root = {'children': {}, 'values': set()}
    for phrase, value in dictionary.items():
        # Normalize the phrase by converting to lowercase and splitting into
        # alphanumeric "word" tokens. This makes matching insensitive to
        # case and original separators.
        words = re.findall(r'[a-z0-9]+', phrase.lower())
        if not words:
            continue

        node = root
        for word in words:
            if word not in node['children']:
                node['children'][word] = {'children': {}, 'values': set()}
            node = node['children'][word]
        node['values'].add(value)
    return root

def annotate(
    tokens: collections.abc.Iterable[str],
    dictionary_index: object
) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    This function iterates through the tokens and, for each starting position,
    finds all matching dictionary entries (both phrases and compound words).

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index().
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each
                            token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotated_tokens = [(token, set()) for token in tokens]

    for i in range(n):
        # A match can only start with a word-like token.
        if not tokens[i].isalnum():
            continue

        # --- 1. Phrase Matching (e.g., "black swan") ---
        # Traverse the Trie and tokens to find multi-word phrases.
        node = dictionary_index
        word_indices = []
        for j in range(i, n):
            token = tokens[j]
            if token.isalnum():
                token_lower = token.lower()
                if token_lower in node.get('children', {}):
                    node = node['children'][token_lower]
                    word_indices.append(j)
                    if node['values']:
                        # Found a valid phrase. Annotate the relevant tokens.
                        # The phrase spans from the first to the last word,
                        # including any inner separators.
                        first_word_idx = word_indices[0]
                        last_word_idx = word_indices[-1]
                        for k in range(first_word_idx, last_word_idx + 1):
                            annotated_tokens[k][1].update(node['values'])
                else:
                    # This path is not in the dictionary, so no longer phrases
                    # can be found starting from this point.
                    break
        
        # --- 2. Compound Word Matching (e.g., "blackswan") ---
        # Concatenate adjacent word tokens and look them up in the dictionary.
        concatenated_word = ""
        word_indices_compound = []
        for j in range(i, n):
            token = tokens[j]
            if token.isalnum():
                concatenated_word += token.lower()
                word_indices_compound.append(j)
                
                # Check if the concatenated string is a single-word entry in the Trie.
                if concatenated_word in dictionary_index.get('children', {}):
                    match_node = dictionary_index['children'][concatenated_word]
                    if match_node['values']:
                        # Found a compound word. Annotate its constituent tokens.
                        for k in word_indices_compound:
                            annotated_tokens[k][1].update(match_node['values'])
            else:
                # A separator breaks a potential compound word.
                break

    return annotated_tokens
