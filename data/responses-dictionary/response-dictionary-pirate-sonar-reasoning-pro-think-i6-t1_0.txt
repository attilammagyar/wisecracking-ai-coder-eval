def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    import re
    individual_words = collections.defaultdict(collections.abc.Set)
    phrase_map = {}
    for key, entry_id in dictionary.items():
        lower_key = key.lower()
        phrase_map[lower_key] = entry_id
        words = re.findall(r'\w+', lower_key)  # Extract sequences of word characters
        for word in words:
            individual_words[word].add(entry_id)
    return {
        "individual_words": individual_words,
