import collections.abc
from typing import Dict, Set, List, Tuple

class TrieNode:
    __slots__ = ('children', 'entries', 'is_separator')
    
    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.entries: Set[int] = set()
        self.is_separator = False

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a case-insensitive trie from the dictionary keys.
    Also identifies separator nodes in the trie.
    """
    root = TrieNode()
    
    for key, entry_id in dictionary.items():
        node = root
        normalized_key = key.lower()
        
        for char in normalized_key:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
            
            # Mark nodes that represent separators (non-word chars)
            if not char.isalnum():
                node.is_separator = True
        
        node.entries.add(entry_id)
    
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with dictionary entries, including compound phrases.
    """
    if not isinstance(dictionary_index, TrieNode):
        return [(token, set()) for token in tokens]
    
    tokens = list(tokens)
    annotated_tokens = []
    n = len(tokens)
    
    # First pass: find all possible phrase matches
    phrase_matches = []
    for i in range(n):
        node = dictionary_index
        current_phrase = []
        entries = set()
        
        for j in range(i, n):
            token = tokens[j]
            normalized_token = token.lower()
            matched = False
            
            for char in normalized_token:
                if char in node.children:
                    node = node.children[char]
                    matched = True
                else:
                    matched = False
                    break
            
            if not matched:
                break
            
            current_phrase.append(j)
            entries.update(node.entries)
            
            # Only add if we have actual entries (not just partial match)
            if node.entries:
                phrase_matches.append((i, j, entries.copy()))
    
    # Second pass: collect all annotations
    annotations = [set() for _ in range(n)]
    
    # Add individual word matches
    for i, token in enumerate(tokens):
        normalized_token = token.lower()
        node = dictionary_index
        matched = True
        
        for char in normalized_token:
            if char in node.children:
                node = node.children[char]
            else:
                matched = False
                break
        
        if matched and node.entries:
            annotations[i].update(node.entries)
    
    # Add compound phrase matches
    for start, end, entries in phrase_matches:
        # Skip leading and trailing separators that aren't part of any phrase
        if (start > 0 and end < n - 1) or (start == 0 and end < n - 1) or (start > 0 and end == n - 1):
            for i in range(start, end + 1):
                annotations[i].update(entries)
    
    # Build final annotated tokens
    for i, token in enumerate(tokens):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
