import collections.abc
from collections import defaultdict
from typing import Dict, Set, List, Tuple

class TrieNode:
    __slots__ = ('children', 'ids')
    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.ids: Set[int] = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.
    """
    # Case-insensitive word dictionary
    word_dict = defaultdict(set)
    # Trie for compound phrases
    trie = TrieNode()
    
    for phrase, entry_id in dictionary.items():
        # Handle individual words
        words = phrase.lower().split()
        if len(words) == 1:
            word_dict[words[0]].add(entry_id)
        # Handle compound phrases
        if len(words) > 1:
            current = trie
            for word in words:
                if word not in current.children:
                    current.children[word] = TrieNode()
                current = current.children[word]
            current.ids.add(entry_id)
    
    return {
        'word_dict': word_dict,
        'phrase_trie': trie,
        'max_phrase_length': max(len(phrase.split()) for phrase in dictionary) if dictionary else 0
    }

def _is_word_token(token: str) -> bool:
    """Check if token is a word (not separator or punctuation)."""
    return any(c.isalnum() for c in token)

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.
    """
    if not dictionary_index:
        return [(token, set()) for token in tokens]
    
    tokens = list(tokens)
    word_dict = dictionary_index['word_dict']
    phrase_trie = dictionary_index['phrase_trie']
    max_phrase_length = dictionary_index['max_phrase_length']
    
    # Initialize annotations
    annotations = [set() for _ in tokens]
    
    # First pass: individual words
    for i, token in enumerate(tokens):
        if _is_word_token(token):
            lower_token = token.lower()
            if lower_token in word_dict:
                annotations[i].update(word_dict[lower_token])
    
    # Second pass: compound phrases
    n = len(tokens)
    for i in range(n):
        if not _is_word_token(tokens[i]):
            continue
            
        current = phrase_trie
        j = i
        matched = True
        
        while matched and j < n:
            token = tokens[j]
            if not _is_word_token(token):
                # Skip non-word tokens in the middle of phrase matching
                j += 1
                continue
                
            lower_token = token.lower()
            if lower_token in current.children:
                current = current.children[lower_token]
                if current.ids:
                    # Found a matching phrase, annotate all participating tokens
                    for k in range(i, j + 1):
                        if _is_word_token(tokens[k]):
                            annotations[k].update(current.ids)
                j += 1
            else:
                matched = False
            
            # Early exit if we've reached max phrase length
            if j - i >= max_phrase_length:
                break
    
    return list(zip(tokens, annotations))
