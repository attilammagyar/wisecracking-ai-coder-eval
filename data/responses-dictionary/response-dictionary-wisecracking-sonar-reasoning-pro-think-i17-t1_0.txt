import re
from collections.abc import Iterable, Mapping, Set

def build_dictionary_index(dictionary: Mapping) -> object:
    """Builds a normalized index from dictionary entries for efficient lookup."""
    normalized_phrases = {}
    max_length = 0
    
    for phrase, entry_id in dictionary.items():
        # Normalize phrase: lowercase, replace non-alphanumeric with space
        normalized = phrase.lower()
        normalized = re.sub(r'[^\w]', ' ', normalized)  # Replace non-alnum chars with space
        normalized = normalized.strip()
        
        # Split into ordered tokens while preserving required structure
        tokens = tuple(normalized.split())
        normalized_phrases[tokens] = entry_id
        
        # Update max phrase tokens length
        max_length = max(max_length, len(tokens))
    
    return {'normalized_phrases': normalized_phrases, 'max_length': max_length}

def annotate(tokens: Iterable[str], dictionary_index: object) -> Iterable[tuple[str, Set]]:
    """Annotate tokens with dictionary entries."""
    annotations = [{} for _ in tokens]
    max_length = dictionary_index['max_length']
    
    # Precompute normalized tokens for quick access
    normalized_tokens = [
        (t.lower(), re.sub(r'[^\w]', ' ', t), t)
        for t in tokens
    ]
    
    for i in range(len(tokens)):
        # Check all possible phrase lengths starting at current token
        for k in range(1, max_length + 1):
            if i + k > len(tokens):
                break
            
            # Collect tokens in current window (including separators)
            window = []
            for j in range(i, i + k):
                # Normalize individual token in this window
                original, normalized, raw = normalized_tokens[j]
                window.append(normalized)
            
            # Join tokens with spaces to form normalized phrase string
            full_str = ' '.join(window).strip()
            if not full_str:
                continue
            
            # Re-split into tokens to handle canonical format
            canonical_tokens = full_str.split()
            phrase_key = tuple(canonical_tokens)
            
            if phrase_key in dictionary_index['normalized_phrases']:
                entry_id = dictionary_index['normalized_phrases'][phrase_key]
                for pos in range(i, i + k):
                    annotations[pos][entry_id] = True
    
    # Convert to sets of ints as per interface
    result = []
    for idx, token in enumerate(tokens):
        keys = list(annotations[idx].keys())
        result.append((token, set(keys)) if keys else (token, set()))
    
    return result
