```python
import re
from collections import defaultdict

def build_dictionary_index(dictionary):
    trie = {'nodes': {}, 'max_length': 0}
    max_len = 0
    
    for key, entry_id in dictionary.items():
        lower_key = key.lower()
        words = re.findall(r'[a-zA-Z]+', lower_key)
        if len(words) > max_len:
            max_len = len(words)
        
        current = trie['nodes']
        for word in words:
            if word not in current:
                current[word] = {'nodes': {}, 'entries': set()}
            current = current[word]['nodes']
        current['entries'].add(entry_id)
    
    return {'trie': trie, 'max_length': max_len}

def traverse_trie(trie, words):
    current = trie['nodes']
    for word in words:
        if word not in current:
            return set()
        current = current[word]['nodes']
    return current['entries']

def annotate(tokens, dictionary_index):
    trie = dictionary_index['trie']
    max_length = dictionary_index['max_length']
    
    # Preprocess tokens into word info
    tokens_info = []
    word_list = []
    for idx, token in enumerate(tokens):
        is_word = any(c.isalpha() for c in token)
        lower_token = token.lower() if is_word else None
        tokens_info.append({
            'original_token': token,
            'is_word': is_word,
            'lower_token': lower_token,
            'original_index': idx
        })
        if is_word:
            word_list.append({
                'lower_word': lower_token,
                'original_index': idx
            })
    
    # Collect ranges for every entry ID
    index_to_ids = defaultdict(set)
    
    # Process single-word matches first
    for word in word_list:
        entry_ids = traverse_trie(trie, [word['lower_word']])
        if entry_ids:
            start = word['original_index']
            end = start
            for idx in range(start, end + 1):
                index_to_ids[idx].update(entry_ids)
    
    # Process multi-word sequences
    for pos in range(len(word_list)):
        for length in range(2, max_length + 1):
            if pos + length > len(word_list):
                break
            subsequence = [word_list[pos + i]['lower_word'] for i in range(length)]
            entry_ids = traverse_trie(trie, subsequence)
            if entry_ids:
                start = word_list[pos]['original_index']
                end = word_list[pos + length - 1]['original_index']
                for idx in range(start, end + 1):
                    index_to_ids[idx].update(entry_ids)
    
    # Generate result
    return [
        (token_info['original_token'], index_to_ids.get(token_info['original_index'], set()))
        for token_info in tokens_info
    ]
```

3. **Range Annotation**  
   For each matched sequence, all tokens in its range (incl. separators) get ID annotated.
