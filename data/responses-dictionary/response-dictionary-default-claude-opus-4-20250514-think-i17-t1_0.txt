```python
    """Extract normalized words from text."""
    words = []
    current_word = []

        else:

        words = extract_words(entry_text)
        if not words:
            continue
        
        # Normalize the full entry for exact matching
        entry_lower = entry_text.lower()
        
        # Single word entries
        if len(words) == 1:
            if entry_lower not in index['exact_tokens']:
                index['exact_tokens'][entry_lower] = set()
            index['exact_tokens'][entry_lower].add(entry_id)
        else:
            # Multi-word entries
            # Check if it's a compound word (no separators between letters)
            letters_only = ''.join(c for c in entry_text if c.isalnum()).lower()
            concatenated_words = ''.join(words)
            
            if letters_only == concatenated_words:
                # Pure compound word like "AAABBB"
                if concatenated_words not in index['compound_words']:
                    index['compound_words'][concatenated_words] = set()
                index['compound_words'][concatenated_words].add(entry_id)
            
            # Always store as phrase for flexible separator matching
            word_tuple = tuple(words)
            if word_tuple not in index['phrases']:
                index['phrases'][word_tuple] = set()
            index['phrases'][word_tuple].add(entry_id)
    
    return index

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Process all possible starting positions
    for start in range(n):
        # Single token exact match
        token_lower = tokens[start].lower()
        if token_lower in dictionary_index['exact_tokens']:
            annotations[start].update(dictionary_index['exact_tokens'][token_lower])
        
        # Multi-token patterns starting from this position
        if is_word_token(tokens[start]):
            # Track words for phrase matching
            words_found = []
            word_positions = []
            
            # Track concatenation for compound word matching
            concatenated = ""
            last_pos = start - 1
            
            for pos in range(start, n):
                if is_word_token(tokens[pos]):
                    word_lower = tokens[pos].lower()
                    words_found.append(word_lower)
                    word_positions.append(pos)
                    
                    # Update concatenated string (only if adjacent to last word token)
                    if pos == last_pos + 1:
                        concatenated += word_lower
                    else:
                        concatenated = None  # Gap breaks compound word
                    last_pos = pos
                    
                    # Check for compound word match (adjacent tokens only)
                    if concatenated and len(words_found) > 1:
                        if concatenated in dictionary_index['compound_words']:
                            # Annotate all tokens from start to current position
                            for i in range(start, pos + 1):
                                annotations[i].update(dictionary_index['compound_words'][concatenated])
                    
                    # Check for phrase match
                    word_tuple = tuple(words_found)
                    if word_tuple in dictionary_index['phrases']:
                        # Annotate from first word to last word (inclusive of separators)
                        for i in range(word_positions[0], word_positions[-1] + 1):
                            annotations[i].update(dictionary_index['phrases'][word_tuple])
    
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
    return annotated_tokens
```

The algorithm works as follows:

**Dictionary Index Building:**
1. Extract normalized words from each dictionary entry
2. Classify entries as single words, compound words (no separators), or phrases (with separators)
3. Store them in appropriate lookup structures for efficient matching

**Annotation Algorithm:**
1. For each starting position, try to match dictionary entries
2. Single tokens are matched exactly against the dictionary
3. For compound words, concatenate adjacent word tokens and check for matches
4. For phrases, collect sequences of words (ignoring specific separators) and match against phrase patterns
5. All overlapping matches are found and combined in the annotation sets

The time complexity is O(n²m) where n is the number of tokens and m is the average number of words in a phrase, which should be efficient for typical text annotation tasks.
