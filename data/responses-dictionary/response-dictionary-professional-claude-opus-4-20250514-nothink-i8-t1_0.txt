import collections.abc

class TrieNode:
    def __init__(self):
        self.children = {}
        self.values = set()  # Dictionary entry IDs that end at this node

class Trie:
    def __init__(self):
        self.root = TrieNode()
    
    def insert(self, key, value):
        """Insert a normalized key with its value into the trie."""
        node = self.root
        for char in key:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.values.add(value)
    
    def search_prefix(self, text, start=0):
        """
        Find all matches starting at position 'start' in 'text'.
        Returns list of (end_position, values) tuples.
        """
        matches = []
        node = self.root
        
        for i in range(start, len(text)):
            char = text[i]
            if char not in node.children:
                break
            node = node.children[char]
            if node.values:
                matches.append((i + 1, node.values))
        
        return matches

def is_word_token(token):
    """Check if a token contains any alphabetic characters."""
    return any(c.isalpha() for c in token)

def normalize_text(text):
    """Convert text to lowercase for case-insensitive matching."""
    return text.lower()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:

    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    trie = Trie()
    
    # Insert all dictionary entries into the trie with normalized keys
    for key, value in dictionary.items():
        normalized_key = normalize_text(key)
        trie.insert(normalized_key, value)
    
    return trie

    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    if n == 0:
        return []
    
    # Try matching from each position
    for start_idx in range(n):
        # Skip if starting with a non-word token (for phrases)
        if not is_word_token(tokens[start_idx]):
            continue
        
        # Build normalized text starting from this position
        normalized_parts = []
        token_ranges = []  # Track which tokens are included
        
        for idx in range(start_idx, n):
            token = tokens[idx]

                token_ranges.append(idx)

                if not normalized_parts:
