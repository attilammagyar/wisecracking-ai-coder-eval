Main ideas

- Normalize the dictionary into a compact trie indexed case-insensitively over characters of words. Between words in a phrase, we insert a special “separator” edge (SEP). The trie nodes store:
  - children: transitions by character
  - sep: a transition that requires at least one non-word token in the text before the next word
  - ids: set of dictionary entry identifiers that end at this node

- What is a “word” vs. “separator” token:
  - A token is a word token if it contains only “word” characters (letters, digits, underscore). Everything else is a separator token (space, punctuation, Markdown, etc.).
  - Dictionary words are contiguous runs of “word” characters; runs of non-word characters in dictionary keys are normalized to SEP edges. Leading/trailing non-word runs are ignored.

- Matching on tokens:
  - We scan the token list and start matching at every word token.
  - We consume whole tokens only (never partial tokens). That guarantees “no midtoken match.”
  - A single dictionary word can span multiple adjacent word tokens (no separators between them), by concatenating these tokens’ strings and matching character-by-character in the trie.
  - A dictionary SEP edge matches one or more separator tokens in the input. We annotate only these inner separator tokens (between words), never leading/trailing separators around the whole phrase.
  - Nested and overlapping phrases are supported naturally by continuing via SEP and recording all terminal nodes (ids) encountered at word-token boundaries.

- Efficiency notes:
  - Case-insensitive matching via str.casefold().
  - A small cache of trie transitions per (node, token_string) avoids re-walking characters repeatedly.
  - We only branch at SEP edges; advancing within a word segment is deterministic with the given next word tokens.

import collections.abc

# ---------------- Utilities ----------------

def _is_word_char(ch: str) -> bool:
    # Define "word" characters: ASCII/Unicode letters, digits, underscore.
    return ch.isalnum() or ch == "_"

def _is_word_token(tok: str) -> bool:
    # A token is a "word token" iff it consists entirely of word chars.
    if not tok:
        return False
    for ch in tok:
        if not _is_word_char(ch):
            return False
    return True

def _extract_words_casefolded(s: str) -> list[str]:
    # Split the dictionary key into runs of word chars; casefold for CI.
    s_cf = s.casefold()
    words = []
    i = 0
    n = len(s_cf)
    while i < n:
        # skip non-word chars
        while i < n and not _is_word_char(s_cf[i]):
            i += 1
        if i >= n:
            break
        j = i
        while j < n and _is_word_char(s_cf[j]):
            j += 1
        words.append(s_cf[i:j])
        i = j
    return words

# ---------------- Trie ----------------

class _TrieNode:
    __slots__ = ("children", "sep", "ids")
    def __init__(self):
        self.children: dict[str, _TrieNode] = {}
        self.sep: _TrieNode | None = None
        self.ids: set = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    root = _TrieNode()
    for key, value in dictionary.items():
        words = _extract_words_casefolded(key)
        if not words:
            # Ignore entries that do not contain any word characters.
            continue
        node = root
        last_idx = len(words) - 1
        for idx, w in enumerate(words):
            # add characters of this word
            for ch in w:
                nxt = node.children.get(ch)
                if nxt is None:
                    nxt = _TrieNode()
                    node.children[ch] = nxt
                node = nxt
            # if not the last word, add a separator edge
            if idx != last_idx:
                if node.sep is None:
                    node.sep = _TrieNode()
                node = node.sep
        node.ids.add(value)
    return root

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    # Materialize tokens for repeated access
    toks = list(tokens)
    n = len(toks)
    anns = [set() for _ in range(n)]
    if n == 0:
        return []

    # Classify and normalize tokens
    is_word = [_is_word_token(t) for t in toks]
    toks_cf = [t.casefold() if is_word[i] else t for i, t in enumerate(toks)]

    root = dictionary_index
    if not isinstance(root, _TrieNode):
        # Defensive: empty index / unexpected object
        return [(t, set()) for t in toks]

    # Cache transitions: (node_id, token_str) -> _TrieNode | None
    advance_cache: dict[tuple[int, str], _TrieNode | None] = {}

    def advance(node: _TrieNode, token_str: str) -> _TrieNode | None:
        """
        Move through the trie by consuming an entire word token (all its chars).
        Returns the resulting node or None if the token doesn't fit from 'node'.
        """
        key = (id(node), token_str)
        if key in advance_cache:
            return advance_cache[key]
        cur = node
        for ch in token_str:
            nxt = cur.children.get(ch)
            if nxt is None:
                advance_cache[key] = None
                return None
            cur = nxt
        advance_cache[key] = cur
        return cur

    def add_match(start_word_idx: int, end_word_idx: int, sep_indices: list[int], ids: collections.abc.Iterable):
        # Annotate all word tokens in span and the inner separators.
        for did in ids:
            for k in range(start_word_idx, end_word_idx + 1):
                anns[k].add(did)
            for si in sep_indices:
                anns[si].add(did)

    # Core matcher: consume a run of one or more word tokens as a dictionary "word segment".
    # At each boundary, emit matches and branch via SEP edges (skipping >=1 separator tokens).
    def consume_word_run(node: _TrieNode, idx: int, start_idx: int, sep_so_far: list[int]):
        if idx >= n or not is_word[idx]:
            return
        # First token of this segment
        cur_node = advance(node, toks_cf[idx])
        if cur_node is None:
            return

        i = idx
        while True:
            # Any dictionary entries ending exactly here? Record them.
            if cur_node.ids:
                add_match(start_idx, i, sep_so_far, cur_node.ids)

            # Branch via SEP: must skip >=1 separator tokens before next word segment.
            if cur_node.sep is not None:
                j = i + 1
                seps = []
                while j < n and not is_word[j]:
                    seps.append(j)
                    j += 1
                if seps and j < n and is_word[j]:
                    consume_word_run(cur_node.sep, j, start_idx, sep_so_far + seps)

            # Try to extend the current word segment with the next word token (no separators).
            next_i = i + 1
            if next_i < n and is_word[next_i]:
                next_node = advance(cur_node, toks_cf[next_i])
                if next_node is not None:
                    i = next_i
                    cur_node = next_node
                    continue
            break

    # Start matching at every word token
    for s in range(n):
        if not is_word[s]:
            continue
        consume_word_run(root, s, s, [])

    # Produce output pairs (token, set_of_ids)
    return [(toks[i], anns[i]) for i in range(n)]
