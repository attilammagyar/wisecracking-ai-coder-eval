def is_word_token(token):
    """Check if token contains alphanumeric characters."""
    return any(c.isalnum() for c in token)

def normalize_word(word):
    """Normalize word for case-insensitive matching."""
    return word.lower()

def extract_words_from_text(text):
    """Extract words from text, ignoring separators."""
    words = []
    current_word = []
    
    for char in text:
        if char.isalnum():
            current_word.append(char)
        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))
    
    return words

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'entries': {}  # Maps (entry_type, normalized_key) to set of entry IDs
    }
    
    for text, entry_id in dictionary.items():
        words = extract_words_from_text(text)
        
        if not words:
            continue
        
        if len(words) == 1:
            # Single word entry (including compound words without spaces)
            normalized = normalize_word(words[0])
            key = ('word', normalized)
            if key not in index['entries']:
                index['entries'][key] = set()
            index['entries'][key].add(entry_id)
        else:
            # Multi-word phrase
            normalized_words = tuple(normalize_word(w) for w in words)
            key = ('phrase', normalized_words)
            if key not in index['entries']:
                index['entries'][key] = set()

    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    annotations = [set() for _ in range(n)]
    
    # Try to find matches starting at each position
    for start in range(n):
        if not is_word_token(tokens[start]):
            continue
        
        # 1. Try single word match
        normalized = normalize_word(tokens[start])
        key = ('word', normalized)
        if key in dictionary_index['entries']:
            annotations[start].update(dictionary_index['entries'][key])
        
        # 2. Try phrase matches (sequences of words with flexible separators)
        words_found = [normalize_word(tokens[start])]
        word_end_positions = [start]
        
        for i in range(start + 1, n):
            if is_word_token(tokens[i]):
                words_found.append(normalize_word(tokens[i]))
                word_end_positions.append(i)
                
                # Check if this word sequence matches a phrase
                words_tuple = tuple(words_found)
                key = ('phrase', words_tuple)
                
                if key in dictionary_index['entries']:
                    # Annotate all tokens from start to current position
                    for entry_id in dictionary_index['entries'][key]:
                        for j in range(start, i + 1):
                            annotations[j].add(entry_id)
        
        # 3. Try compound word matches (concatenated adjacent word tokens)
        concatenated = ""
        for i in range(start, n):
            if not is_word_token(tokens[i]):
                break
            concatenated += tokens[i]
            normalized_concat = normalize_word(concatenated)
            key = ('word', normalized_concat)
            
            if key in dictionary_index['entries']:
                # Annotate all tokens that form this compound word
                for entry_id in dictionary_index['entries'][key]:
                    for j in range(start, i + 1):
                        annotations[j].add(entry_id)
    
    annotated_tokens = [(tokens[i], annotations[i]) for i in range(n)]
    return annotated_tokens
