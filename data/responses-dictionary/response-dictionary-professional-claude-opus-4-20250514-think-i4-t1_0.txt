    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    entries = []
    
    for key, value in dictionary.items():
        parsed = parse_dictionary_key(key)
        if parsed:
            entries.append({
                'key': key,
                'value': value,
                'words': parsed['words'],
                'has_separators': parsed['has_separators'],
                'concatenated': parsed['concatenated']
            })
    
    return {'entries': entries}

def match_with_separators(tokens, start_pos, target_words):
    """Match a compound phrase with flexible separators between words."""
    matched_positions = []
    word_index = 0
    pos = start_pos
    
    # Skip any leading non-word tokens
    while pos < len(tokens) and not is_word_token(tokens[pos]):
        pos += 1
    
    # Match words with flexible separators
    while pos < len(tokens) and word_index < len(target_words):
        if is_word_token(tokens[pos]):
            # Check if this word matches the next expected word
            if normalize_token(tokens[pos]) == target_words[word_index]:
                matched_positions.append(pos)
                word_index += 1
                pos += 1
            else:
                # Word doesn't match
                return None
        else:
            # Non-word token (separator)
            # Include it if we're between matched words (not at the end)
            if word_index > 0 and word_index < len(target_words):
                matched_positions.append(pos)
            pos += 1
    
    # Check if we matched all required words
    if word_index == len(target_words):
        return matched_positions
    
    return None

def match_without_separators(tokens, start_pos, target_concatenated):
    """Match a word or compound word (no separators)."""
    # First try: single token exact match
    if start_pos < len(tokens) and is_word_token(tokens[start_pos]):
        if normalize_token(tokens[start_pos]) == target_concatenated:
            return [start_pos]
    
    # Second try: concatenate adjacent word tokens
    current_concat = ''
    positions = []
    pos = start_pos
    
    while pos < len(tokens) and is_word_token(tokens[pos]):
        current_concat += normalize_token(tokens[pos])
        positions.append(pos)
        
        if current_concat == target_concatenated:
            return positions
        elif not target_concatenated.startswith(current_concat):
            # No match possible
            return None
        
        pos += 1
    
    return None

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens_list = list(tokens)
    n = len(tokens_list)
    
    # Initialize annotations for each token position
    annotations = [set() for _ in range(n)]
    
    # Try to match dictionary entries starting from each position
    for start_pos in range(n):
        for entry in dictionary_index['entries']:
            if entry['has_separators']:
                # Match with flexible separators
                matched_positions = match_with_separators(tokens_list, start_pos, entry['words'])
            else:
                # Match without separators (single word or compound word)
                matched_positions = match_without_separators(tokens_list, start_pos, entry['concatenated'])
            
            if matched_positions:
                # Add this entry's value to all matched positions
                for pos in matched_positions:
                    annotations[pos].add(entry['value'])
    
    # Build the result
    annotated_tokens = []
    for i, token in enumerate(tokens_list):
        annotated_tokens.append((token, annotations[i]))
    
    return annotated_tokens
