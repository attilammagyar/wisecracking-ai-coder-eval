class _TrieNode:

        self.values: set = set()

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    
    Return:
        The root of the dictionary Trie.
    """
    root = _TrieNode()
    for phrase, value in dictionary.items():

        # "black swan" -> ['black', 'swan']
        # "AAABBB" -> ['aaabbb']

            node = node.children[word]
        node.values.add(value)
    return root

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """

        return [(token, set()) for token in tokens]

    tokens_list = list(tokens)

    annotated_tokens = [(token, set()) for token in tokens_list]

        """Helper to apply annotations to a range of tokens, skipping leading/trailing separators."""
        
        # Find the first token of the phrase that is a word.
        start_idx = phrase_start_idx
        while start_idx <= phrase_end_idx and not tokens_list[start_idx].isalnum():
            start_idx += 1

        # Find the last token of the phrase that is a word.
        end_idx = phrase_end_idx
        while end_idx >= start_idx and not tokens_list[end_idx].isalnum():
            end_idx -= 1
