        else:
            if current_word:
                words.append(''.join(current_word))
                current_word = []
    
    if current_word:
        words.append(''.join(current_word))

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    return dict(dictionary)

def try_match_compound_phrase(tokens, start_pos, dict_words):
    """
    Try to match a compound phrase starting at start_pos.
    Returns (first_word_pos, last_word_pos) if successful, None otherwise.
    """
    word_idx = 0
    pos = start_pos
    first_word_pos = None
    last_word_pos = None
    
    while pos < len(tokens) and word_idx < len(dict_words):
        if is_word_token(tokens[pos]):
            # This token contains words
            token_words = extract_words_from_text(tokens[pos])
            
            # Try to match words from this token
            for token_word in token_words:
                if word_idx < len(dict_words) and normalize_word(token_word) == normalize_word(dict_words[word_idx]):
                    if first_word_pos is None:
                        first_word_pos = pos
                    last_word_pos = pos
                    word_idx += 1
                else:
                    # Word mismatch
                    return None
        
        pos += 1
    
    # Check if we matched all words
    if word_idx == len(dict_words) and first_word_pos is not None:
        return (first_word_pos, last_word_pos)
    
    return None

def try_match_compound_word(tokens, start_pos, dict_text):
    """
    Try to match a compound word (no separators) starting at start_pos.
    Returns end_pos if successful, None otherwise.
    """
    if not is_word_token(tokens[start_pos]):
        return None
    
    concat = ""
    pos = start_pos
    
    while pos < len(tokens) and is_word_token(tokens[pos]):
        concat += tokens[pos]
        if normalize_word(concat) == normalize_word(dict_text):
            return pos
        pos += 1
    
    return None

def annotate(tokens: collections.abc.Iterable[str], dictionary_index: object) -> collections.abc.Iterable[tuple[str, collections.abc.Set]]:
    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    token_list = list(tokens)
    n = len(token_list)
    annotations = [set() for _ in range(n)]
    
    # Process each dictionary entry
    for dict_key, dict_id in dictionary_index.items():
        dict_words = extract_words_from_text(dict_key)
        
        if not dict_words:
            continue
        
        # Determine entry type
        has_separators = len(dict_words) > 1 and any(not c.isalnum() for c in dict_key)
        
        # Try to match at each position
        for start_pos in range(n):
            if len(dict_words) == 1 and not has_separators:
                # Single word entry
                if is_word_token(token_list[start_pos]):
                    if normalize_word(token_list[start_pos]) == normalize_word(dict_key):
                        annotations[start_pos].add(dict_id)
            
            elif has_separators:
                # Compound phrase (with separators)
                match = try_match_compound_phrase(token_list, start_pos, dict_words)
                if match:
                    first_word_pos, last_word_pos = match
                    # Annotate all tokens from first word to last word (inclusive)
                    for i in range(first_word_pos, last_word_pos + 1):
                        annotations[i].add(dict_id)
            
            else:
                # Compound word (no separators)
                end_pos = try_match_compound_word(token_list, start_pos, dict_key)
                if end_pos is not None:
                    # Annotate all tokens from start to end (inclusive)
                    for i in range(start_pos, end_pos + 1):
                        annotations[i].add(dict_id)
    
    # Build result
    annotated_tokens = []
    for token, annot in zip(token_list, annotations):
        annotated_tokens.append((token, annot))
    
    return annotated_tokens
