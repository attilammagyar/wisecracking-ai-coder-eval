import collections.abc

def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    # Normalize dictionary keys to lowercase and split into words
    normalized_dict = {}
    phrase_index = {}

    for key, value in dictionary.items():
        # Normalize the key to lowercase for case-insensitive lookup
        normalized_key = key.lower()
        normalized_dict[normalized_key] = value

        # Split the key into words (ignoring non-word characters)
        words = re.split(r'[^a-zA-Z0-9]+', normalized_key)
        words = [word for word in words if word]  # Remove empty strings

        if len(words) > 1:
            # Build a trie-like structure for phrases
            current_level = phrase_index
            for word in words[:-1]:
                if word not in current_level:
                    current_level[word] = {}
                current_level = current_level[word]
            last_word = words[-1]
            if last_word not in current_level:
                current_level[last_word] = {}
            # Store the original key and value in the last node
            if '_entries' not in current_level[last_word]:
                current_level[last_word]['_entries'] = []
            current_level[last_word]['_entries'].append((key, value))

    return {
        'normalized_dict': normalized_dict,
        'phrase_index': phrase_index,
    }
