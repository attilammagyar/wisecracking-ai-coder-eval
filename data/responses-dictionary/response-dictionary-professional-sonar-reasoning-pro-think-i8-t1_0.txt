import sys
def build_dictionary_index(dictionary: dict) -> dict:
    import re
    from collections import defaultdict

    index = defaultdict(set)
    max_length = 0

    for key, value in dictionary.items():
        # Split into alphanumeric parts (case/separator insensitive)
        words = re.findall(r'\w+', key.lower())
        normalized = tuple(words)
        index[normalized].add(value)
        max_length = max(max_length, len(normalized))

    return {'index': index, 'max_length': max_length}

def annotate(tokens: list[str], index_info: dict) -> list[tuple[str, set]]:
    index, max_length = index_info['index'], index_info['max_length']
    annotated = [(token, set()) for token in tokens]

    for i in range(len(tokens)):
        for l in range(1, max_length + 1):
            if i + l > len(tokens): break
            sub = tokens[i:i+l]

            # Extract alphanumeric parts for both checks
            alphanumeric = []
            for token in sub:
                parts = re.findall(r'\w+', token.lower())
                alphanumeric.extend(parts)

            # Word sequence match
            if alphanumeric:
                parts_tuple = tuple(alphanumeric)
                if parts_tuple in index:
                    for j in range(i, i+l):
                        annotated[j] = (annotated[j][0], annotated[j][1].union(index[parts_tuple]))

            # Compound word match
            if alphanumeric:
                compound_word = ''.join(alphanumeric)
                compound_tuple = (compound_word,)
                if compound_tuple in index:
                    for j in range(i, i+l):
                        annotated[j] = (annotated[j][0], annotated[j][1].union(index[compound_tuple]))

    return annotated
