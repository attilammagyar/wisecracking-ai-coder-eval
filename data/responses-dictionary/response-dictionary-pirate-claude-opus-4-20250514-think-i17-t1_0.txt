def build_dictionary_index(dictionary: collections.abc.Mapping) -> object:
    """
    Build a normalized index from a dictionary for fast lookup of words and
    compound phrases.

    Parameters:
        dictionary: Mapping strings (keys) to meanings (values).
    """
    index = {
        'single_words': {},      # normalized word -> set of IDs
        'compound_words': {},    # normalized concatenated form -> set of IDs
        'compound_phrases': {}   # normalized word tuple -> set of IDs
    }
    
    for key, value in dictionary.items():
        normalized_key = key.lower()
        
        # Split into tokens (words and separators)
        tokens = re.findall(r'\w+|\W+', normalized_key)
        if not tokens:
            continue
            
        # Extract only word tokens
        word_tokens = [t for t in tokens if re.search(r'\w', t)]
        
        if len(word_tokens) == 0:
            continue
        elif len(word_tokens) == 1:
            # Single word
            word = word_tokens[0]
            if word not in index['single_words']:
                index['single_words'][word] = set()
            index['single_words'][word].add(value)
        else:
            # Multiple words - could be compound word or phrase
            # Check if there are separators between words
            has_separators = any(not re.search(r'\w', t) for t in tokens[1:-1])
            
            if has_separators or len(tokens) > len(word_tokens):
                # Compound phrase
                word_tuple = tuple(word_tokens)
                if word_tuple not in index['compound_phrases']:
                    index['compound_phrases'][word_tuple] = set()
                index['compound_phrases'][word_tuple].add(value)
            else:
                # Compound word (no separators)
                concatenated = ''.join(word_tokens)
                if concatenated not in index['compound_words']:
                    index['compound_words'][concatenated] = set()
                index['compound_words'][concatenated].add(value)
    
    return index

    """
    Annotate tokens with entries from the dictionary.

    Parameters:
        dictionary_index:   A dictionary index created by build_dictionary_index()
        tokens:             The tokens to be annotated.

    Return:
        annotated_tokens:   A list containing (token, annotations) pairs for each token in tokens.
    """
    tokens = list(tokens)
    n = len(tokens)
    
    # Initialize annotations for each token
    annotations = [set() for _ in range(n)]
    
    # Helper function to check if a token is a word
    def is_word(token):
        return bool(re.search(r'\w', token))
    
    # Try to find matches starting at each position
    for start in range(n):
        if not is_word(tokens[start]):

        # Try single word match
