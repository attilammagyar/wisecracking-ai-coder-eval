_RE_WORD = re.compile(r"\w+", re.UNICODE)

    """

        m = _RE_WORD.match(phrase, idx)
        if m:
            tokens.append(m.group(0))
            idx = m.end()
        else:
            tokens.append(phrase[idx])
            idx += 1
    return tokens

    return bool(_RE_WORD.fullmatch(token))
